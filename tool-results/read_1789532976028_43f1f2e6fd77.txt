    68→    /**
    69→     * 加载表单运行时上下文（表名 + 列 + 子表元数据）。
    70→     */
    71→    public BizDataContext loadContext(String formKey) {
    72→        if (formKey == null || !FORM_KEY_PATTERN.matcher(formKey).matches()) {
    73→            throw new BusinessException(400, "非法表单 key: " + formKey);
    74→        }
    75→        String tableName = "wf_biz_" + formKey;
    76→        if (!tableManager.tableExists(tableName)) {
    77→            throw new BusinessException(404, "业务表单数据表不存在: " + formKey);
    78→        }
    79→        List<ColumnConfig> columns = formDefService.getBusinessColumnsByKey(formKey);
    80→        // 主表列：排除子表字段（子表字段映射独立物理表，非主表列）
    81→        List<String> keys = columns.stream()
    82→                .filter(c -> c.getSubColumns() == null || c.getSubColumns().isEmpty())
    83→                .map(ColumnConfig::getKey)
    84→                .toList();
    85→        Map<String, BizDataContext.SubTableDef> subTables = new LinkedHashMap<>();
    86→        for (ColumnConfig c : columns) {
    87→            if (c.getSubColumns() != null && !c.getSubColumns().isEmpty()) {
    88→                String mode = c.getSubMode() == null || c.getSubMode().isBlank() ? "embedded" : c.getSubMode();
    89→                List<String> subKeys = c.getSubColumns().stream().map(ColumnConfig::getKey).toList();
    90→                subTables.put(c.getKey(), new BizDataContext.SubTableDef(
    91→                        "wf_biz_" + formKey + "_" + c.getKey(), mode, c.getSubColumns(), subKeys));
    92→            }
    93→        }
    94→        return new BizDataContext(tableName, formKey, columns, keys, subTables);
    95→    }

   112→    public void validateRequired(List<ColumnConfig> columns, Map<String, Object> data) {
   113→        Map<String, Object> safe = data == null ? Map.of() : data;
   114→        for (ColumnConfig c : columns) {
   115→            if (c.isRequired()) {
   116→                Object v = safe.get(c.getKey());
   117→                if (v == null || (v instanceof String s && s.isBlank())) {
   118→                    throw new BusinessException(400, "必填字段不能为空: " + c.getLabel());
   119→                }
   120→            }
   121→        }
   122→    }

   124→    /**
   125→     * 遍历 data-picker 引用列：校验 id 存在并生成 {@code <key>_text} 展示缓存文本。
   126→     * 不修改原 data，返回附加字段（{@code <key>_text} → 文本）；引用值为空时返回空文本。
   127→     */
   128→    @SuppressWarnings("unchecked")
   129→    public Map<String, Object> resolvePickerValues(BizDataContext ctx, Map<String, Object> data) {
   130→        Map<String, Object> extra = new LinkedHashMap<>();
   131→        for (ColumnConfig col : ctx.columns()) {
   132→            String pickerConfig = col.getPickerConfig();
   133→            if (pickerConfig == null || pickerConfig.isBlank()) {
   134→                continue;
   135→            }
   136→            // 仅 data-picker 引用列生成冗余文本；LookupPicker 列跳过
   137→            if (!isDataPickerColumn(ctx, col)) {
   138→                continue;
   139→            }
   140→            String key = col.getKey();
   141→            Object raw = data.get(key);
   142→            String text = resolvePickerText(ctx, col, raw);
   143→            extra.put(key + "_text", text);
   144→        }
   145→        return extra;
   146→    }

   149→     * 按表单 key 批量解析显示文本（resolve API 入口）。

   151→    public Map<String, String> resolveByFormKey(String formKey, List<String> ids, String displayField) {
   152→        BizDataContext ctx = loadContext(formKey);
   153→        String field = displayField;
   154→        if (field == null || field.isBlank()) {
   155→            field = ctx.columns().stream()
   156→                    .filter(c -> !c.isHidden() && c.getPickerConfig() == null)
   157→                    .map(ColumnConfig::getKey)
   158→                    .findFirst()
   159→                    .orElseThrow(() -> new BusinessException(400, "目标表单无可解析的显示字段"));
   160→        }
   161→        return resolveDisplayTexts(formKey, ids, field);
   162→    }

   164→    /**
   165→     * 批量解析被引用记录的显示文本（id → displayField 值）。
   166→     */
   167→    public Map<String, String> resolveDisplayTexts(String sourceFormKey, List<String> ids, String displayField) {
   168→        if (sourceFormKey == null || !FORM_KEY_PATTERN.matcher(sourceFormKey).matches()) {
   169→            throw new BusinessException(400, "非法目标表单 key: " + sourceFormKey);
   170→        }
   171→        if (ids == null || ids.isEmpty()) {
   172→            return Map.of();
   173→        }
   174→        if (displayField == null || !displayField.matches("^[a-zA-Z][a-zA-Z0-9_]{0,63}$")) {
   175→            throw new BusinessException(400, "非法显示字段: " + displayField);
   176→        }
   177→        String table = "wf_biz_" + sourceFormKey;
   178→        String placeholders = String.join(",", java.util.Collections.nCopies(ids.size(), "?"));
   179→        String sql = "SELECT id, " + displayField + " FROM " + table
   180→                + " WHERE tenant_id = ? AND id IN (" + placeholders + ")";
   181→        List<Object> params = new ArrayList<>();
   182→        params.add(tenantProvider.getTenantId());
   183→        params.addAll(ids);
   184→
   185→        Map<String, String> result = new LinkedHashMap<>();
   186→        for (Map<String, Object> row : jdbcTemplate.queryForList(sql, params.toArray())) {
   187→            Object v = row.get(displayField);
   188→            result.put(String.valueOf(row.get("id")), v == null ? "" : String.valueOf(v));
   189→        }
   190→        return result;
   191→    }

   216→    /**
   217→     * 通用新增实现（不含装饰钩子链；覆盖 handler 委托此方法执行通用插入）。
   218→     */
   219→    @Transactional
   220→    public BizDataVO createGeneric(String formKey, Map<String, Object> data) {
   221→        String tenantId = tenantProvider.getTenantId();
   222→        BizDataContext ctx = loadContext(formKey);
   223→
   224→        validateRequired(ctx.columns(), data);
   225→
   226→        // data-picker 引用校验与冗余文本生成（不改原 data，返回附加字段）
   227→        Map<String, Object> merged = serializeJsonColumns(data);
   228→        merged.putAll(resolvePickerValues(ctx, merged));
   229→
   230→        BizDataQueryBuilder.SqlAndParams insert = BizDataQueryBuilder.buildInsert(
   231→                ctx.tableName(), ctx.columnKeys(), merged, tenantId);
   232→        jdbcTemplate.update(insert.sql(), insert.params().toArray());
   233→
   234→        // 子表行写入（随主表创建批量插入）
   235→        String bizId = insertedId(insert);
   236→        for (Map.Entry<String, BizDataContext.SubTableDef> e : ctx.subTables().entrySet()) {
   237→            Object raw = data.get(e.getKey());
   238→            if (raw instanceof List<?> rows) {
   239→                writeSubRows(e.getValue(), bizId, rows);
   240→            }
   241→        }
   242→
   243→        // 新行 id 由 buildInsert 内部生成，查询返回
   244→        return findById(ctx.tableName(), tenantId, ctx, bizId);
   245→    }

   247→    /**
   248→     * 通用分页查询实现。
   249→     */
   250→    public BizDataPageVO queryGeneric(String formKey, BizDataQueryRequest req) {
   251→        String tenantId = tenantProvider.getTenantId();
   252→        BizDataContext ctx = loadContext(formKey);
   253→
   254→        Map<String, Object> filters = parseFilter(req.getFilter());
   255→        // 列类型映射（key → columnType）：JSON 数组列筛选走 JSON 函数（JSON_CONTAINS/JSON_OVERLAPS）
   256→        Map<String, String> columnTypeOf = ctx.columns().stream()
   257→                .collect(Collectors.toMap(ColumnConfig::getKey,
   258→                        c -> c.getColumnType() == null ? "" : c.getColumnType().toUpperCase(), (a, b) -> a));
   259→        int page = Math.max(req.getPage(), 1);
   260→        // size <= 0 表示不分页取全部（buildSelect 跳过 LIMIT/OFFSET）；正数沿用原钳制上限
   261→        int size = req.getSize() <= 0 ? req.getSize() : Math.min(Math.max(req.getSize(), 1), 100);
   262→
   263→        try {
   264→            BizDataQueryBuilder.SqlAndParams count = BizDataQueryBuilder.buildCount(
   265→                    ctx.tableName(), ctx.columnKeys(), columnTypeOf, tenantId, filters, req.getKeyword(), req.getKeywordColumn());
   266→
   267→            BizDataQueryBuilder.SqlAndParams select = BizDataQueryBuilder.buildSelect(
   268→                    ctx.tableName(), ctx.columnKeys(), columnTypeOf, tenantId, filters,
   269→                    req.getKeyword(), req.getKeywordColumn(), req.getSort(), req.getOrder(), page - 1, size);
   270→
   271→            return sqlQueryEngine.execPage(page, size, count, select, row -> toVO(ctx, row));
   272→        } catch (IllegalArgumentException e) {
   273→            throw new BusinessException(400, e.getMessage());
   274→        }
   275→    }

   305→    /**
   306→     * sql 模式分页查询实现（管理员 SQL 模板包裹，运行时参数白名单透传）。
   307→     * <p>仅校验 formKey 合法性；不校验主表单物理表（管理员 SQL 独立定义，可跨表/聚合）。
   308→     */
   309→    public BizDataPageVO querySqlTemplate(String formKey, BizDataQueryRequest req, FormQueryConfig cfg) {
   310→        // SQL 类型无 formKey 时跳过校验（管理员 SQL 独立定义，不依赖表单）
   311→        if (formKey != null && !FORM_KEY_PATTERN.matcher(formKey).matches()) {
   312→            throw new BusinessException(400, "非法表单 key: " + formKey);
   313→        }
   314→        String tenantId = tenantProvider.getTenantId();
   315→        List<JoinSqlGenerator.QueryColumn> columns = toQueryColumns(cfg.columns());
   316→        Map<String, Object> filters = parseFilter(req.getFilter());
   317→        Map<String, Object> runtimeParams = parseRuntimeParams(req.getParams());
   318→        int page = Math.max(req.getPage(), 1);
   319→        // size <= 0 表示不分页取全部；正数沿用原钳制上限
   320→        int size = req.getSize() <= 0 ? req.getSize() : Math.min(Math.max(req.getSize(), 1), 100);
   321→
   322→        try {
   323→            SqlQueryEngine.WrappedQuery wq = SqlTemplateEngine.wrap(
   324→                    cfg.query(), tenantId, columns, filters, req.getKeyword(), req.getKeywordColumn(),
   325→                    req.getSort(), req.getOrder(), page, size, cfg.declaredParams(), runtimeParams);
   326→            return sqlQueryEngine.execPage(page, size, wq.count(), wq.select(),
   327→                    row -> toSqlVO(cfg.columns(), row));
   328→        } catch (IllegalArgumentException e) {
   329→            throw new BusinessException(400, e.getMessage());
   330→        }
   331→    }

   333→    /** 构建查询列映射：主表列（ref="m."+key，默认全可排可筛）+ 虚拟列（ref=alias+"."+joinField，能力取 join 声明） */
   334→    private List<JoinSqlGenerator.QueryColumn> buildJoinColumns(BizDataContext ctx,
   335→                                                                List<JoinSqlGenerator.JoinConfig> joins) {
   336→        Map<String, String> typeOf = new HashMap<>();
   337→        for (ColumnConfig c : ctx.columns()) {
   338→            typeOf.put(c.getKey(), c.getColumnType() == null ? "" : c.getColumnType().toUpperCase());
   339→        }
   340→        List<JoinSqlGenerator.QueryColumn> columns = new ArrayList<>();
   341→        for (String key : ctx.columnKeys()) {
   342→            columns.add(new JoinSqlGenerator.QueryColumn(key, "m." + key, typeOf.getOrDefault(key, ""), true, true));
   343→        }
   344→        for (JoinSqlGenerator.JoinConfig j : joins) {
   345→            columns.add(new JoinSqlGenerator.QueryColumn(j.virtualKey(), j.alias() + "." + j.joinField(),
   346→                    resolveJoinColumnType(j), j.sortable(), j.filterable()));
   347→        }
   348→        return columns;
   349→    }

   351→    /** 虚拟列类型：目标表单 joinField 的列类型，找不到 fallback "VARCHAR"（查询与 metadata 两处一致） */
   352→    private String resolveJoinColumnType(JoinSqlGenerator.JoinConfig j) {
   353→        try {
   354→            List<ColumnConfig> targetCols = formDefService.getBusinessColumnsByKey(j.targetFormKey());
   355→            if (targetCols != null) {
   356→                for (ColumnConfig c : targetCols) {
   357→                    if (j.joinField().equals(c.getKey())) {
   358→                        return c.getColumnType() == null ? "VARCHAR" : c.getColumnType().toUpperCase();
   359→                    }
   360→                }
   361→            }
   362→        } catch (BusinessException ignored) {
   363→            // 目标表单不可解析时 fallback 类型
   364→        }
   365→        return "VARCHAR";
   366→    }

   368→    /** sql 模式列映射：管理员声明列 → QueryColumn（ref=key，外层子查询输出列名） */
   369→    private List<JoinSqlGenerator.QueryColumn> toQueryColumns(List<ColumnConfig> cols) {
   370→        List<JoinSqlGenerator.QueryColumn> out = new ArrayList<>();
   371→        for (ColumnConfig c : cols) {
   372→            out.add(new JoinSqlGenerator.QueryColumn(c.getKey(), c.getKey(),
   373→                    c.getColumnType() == null ? "" : c.getColumnType().toUpperCase(),
   374→                    Boolean.TRUE.equals(c.getSortable()), Boolean.TRUE.equals(c.getFilterable())));
   375→        }
   376→        return out;
   377→    }

   379→    /** 解析运行时参数（sql 模式透传）；null/空白 → 空 Map；非法 JSON → 400 */
   380→    @SuppressWarnings("unchecked")
   381→    private Map<String, Object> parseRuntimeParams(String paramsJson) {
   382→        if (paramsJson == null || paramsJson.isBlank()) {
   383→            return Map.of();
   384→        }
   385→        try {
   386→            Map<String, Object> map = objectMapper.readValue(paramsJson, Map.class);
   387→            return map == null ? Map.of() : map;
   388→        } catch (JsonProcessingException e) {
   389→            throw new BusinessException(400, "运行时参数 params 格式非法，应为 JSON 对象: " + e.getOriginalMessage());
   390→        }
   391→    }

   393→    /**
   394→     * 通用更新实现（乐观锁；不含装饰钩子链）。
   395→     */
   396→    @Transactional
   397→    public BizDataVO updateGeneric(String formKey, String id, Map<String, Object> data, Integer version) {
   398→        String tenantId = tenantProvider.getTenantId();
   399→        BizDataContext ctx = loadContext(formKey);
   400→
   401→        validateRequired(ctx.columns(), data);
   402→        int currentVersion = version == null ? 1 : version;
   403→
   404→        Map<String, Object> merged = serializeJsonColumns(data);
   405→        merged.putAll(resolvePickerValues(ctx, merged));
   406→
   407→        BizDataQueryBuilder.SqlAndParams update = BizDataQueryBuilder.buildUpdate(
   408→                ctx.tableName(), ctx.columnKeys(), merged, tenantId, id, currentVersion);
   409→        int affected = jdbcTemplate.update(update.sql(), update.params().toArray());
   410→
   411→        if (affected == 0) {
   412→            // 区分"记录不存在"（404）与"版本冲突"（409）
   413→            List<Map<String, Object>> exists = jdbcTemplate.queryForList(
   414→                    "SELECT id, version FROM " + ctx.tableName() + " WHERE id = ? AND tenant_id = ?", id, tenantId);
   415→            if (exists.isEmpty()) {
   416→                throw new BusinessException(404, "业务数据不存在: " + id);
   417→            }
   418→            throw new BusinessException(409, "数据已被他人修改，请刷新后重试");
   419→        }
   420→
   421→        // 子表行增量 diff（仅当请求携带子表字段时）
   422→        for (Map.Entry<String, BizDataContext.SubTableDef> e : ctx.subTables().entrySet()) {
   423→            Object raw = data.get(e.getKey());
   424→            if (raw instanceof List<?> rows) {
   425→                diffSubRows(e.getValue(), id, rows);
   426→            }
   427→        }
   428→
   429→        return findById(ctx.tableName(), tenantId, ctx, id);
   430→    }

   443→                    tenantId, id);

   447→        int affected = jdbcTemplate.update(delete.sql(), delete.params().toArray());
   448→        if (affected == 0) {
   449→            throw new BusinessException(404, "业务数据不存在: " + id);
   450→        }
   451→    }

   497→    /**
   498→     * 更新独立子表行（乐观锁：须携带当前 version）。
   499→     */
   500→    public Map<String, Object> updateSubRow(String formKey, String id, String field, String rowId,
   501→                                            Map<String, Object> data, Integer version) {
   502→        BizDataContext ctx = loadContext(formKey);
   503→        BizDataContext.SubTableDef def = requireSubTable(ctx, field);
   504→        requireMainRow(ctx, id);
   505→
   506→        // 仅允许更新子业务列（白名单 subKeys 过滤，防注入/防篡改内部列）
   507→        Map<String, Object> safe = new LinkedHashMap<>();
   508→        for (String k : def.subKeys()) {
   509→            if (data.containsKey(k)) {
   510→                safe.put(k, data.get(k));
   511→            }
   512→        }
   513→        if (safe.isEmpty()) {
   514→            throw new BusinessException(400, "更新内容不能为空: " + field);
   515→        }
   516→
   517→        StringBuilder set = new StringBuilder();
   518→        List<Object> params = new ArrayList<>();
   519→        for (Map.Entry<String, Object> e : safe.entrySet()) {
   520→            set.append(e.getKey()).append(" = ?, ");
   521→            params.add(e.getValue());
   522→        }
   523→        set.append("version = version + 1, updated_at = NOW()");
   524→        params.add(tenantProvider.getTenantId());
   525→        params.add(id);
   526→        params.add(rowId);
   527→        params.add(version == null ? 1 : version);
   528→
   529→        int affected = jdbcTemplate.update("UPDATE " + def.tableName() + " SET " + set
   530→                + " WHERE tenant_id = ? AND biz_id = ? AND id = ? AND version = ?", params.toArray());
   531→        if (affected == 0) {
   532→            throw new BusinessException(409, "子表行已被他人修改或不存在，请刷新后重试");
   533→        }
   534→        return readSubRows(def, id).stream()
   535→                .filter(r -> rowId.equals(String.valueOf(r.get("id"))))
   536→                .findFirst()
   537→                .orElseThrow(() -> new BusinessException(404, "子表行不存在: " + rowId));
   538→    }

   599→    @SuppressWarnings("unchecked")
   600→    private String resolvePickerText(BizDataContext ctx, ColumnConfig col, Object raw) {
   601→        try {
   602→            Map<String, Object> picker = objectMapper.readValue(col.getPickerConfig(), Map.class);
   603→            String sourceFormKey = picker.get("sourceFormKey") == null ? null : String.valueOf(picker.get("sourceFormKey"));
   604→            String displayField = picker.get("displayField") == null ? null : String.valueOf(picker.get("displayField"));
   605→            Object maxCountObj = picker.get("maxCount");
   606→
   607→            if (raw == null || String.valueOf(raw).isBlank()) {
   608→                return "";
   609→            }
   610→            // 值以 JSON 数组字符串存储（如 ["u1","u2"]；单选为 ["u1"]）
   611→            String rawStr = String.valueOf(raw);
   612→            List<String> ids;
   613→            try {
   614→                ids = objectMapper.readValue(rawStr,
   615→                        objectMapper.getTypeFactory().constructCollectionType(List.class, String.class));
   616→            } catch (JsonProcessingException e) {
   617→                throw new BusinessException(400, "data-picker 引用值格式非法（需 JSON 数组）: " + col.getKey());
   618→            }
   619→            ids.removeIf(String::isBlank);
   620→            if (ids.isEmpty()) {
   621→                return "";
   622→            }
   623→            // maxCount 校验：配置了（非 null 非空）且超限 → 400
   624→            if (maxCountObj != null) {
   625→                int maxCount;
   626→                try {
   627→                    maxCount = Integer.parseInt(String.valueOf(maxCountObj));
   628→                } catch (NumberFormatException e) {
   629→                    throw new BusinessException(400, "data-picker maxCount 配置非法: " + col.getKey());
   630→                }
   631→                if (maxCount > 0 && ids.size() > maxCount) {
   632→                    throw new BusinessException(400,
   633→                            "data-picker 引用数量超出限制（最多 " + maxCount + "）: " + col.getKey());
   634→                }
   635→            }
   636→
   637→            Map<String, String> texts = resolveDisplayTexts(sourceFormKey, ids, displayField);
   638→            List<String> ordered = new ArrayList<>();
   639→            for (String id : ids) {
   640→                String t = texts.get(id);
   641→                if (t == null) {
   642→                    throw new BusinessException(400, "引用的数据不存在: " + col.getKey() + "=" + id);
   643→                }
   644→                ordered.add(t);
   645→            }
   646→            // 冗余文本同样以 JSON 数组存储（与 id 数组顺序一致）
   647→            return objectMapper.writeValueAsString(ordered);
   648→        } catch (JsonProcessingException e) {
   649→            throw new BusinessException(400, "data-picker 配置或引用值非法: " + col.getKey());
   650→        }
   651→    }

   653→    /** 解析单个表单 column_config 中的 dataPicker 引用列，聚合到 result */
   654→    @SuppressWarnings("unchecked")
   655→    private void collectPickerRefs(FormDefinition def, Map<String, Map<String, Object>> result) {
   656→        String columnConfig = def.getColumnConfig();
   657→        if (columnConfig == null || columnConfig.isBlank()) {
   658→            return;
   659→        }
   660→        try {
   661→            List<Map<String, Object>> cols = objectMapper.readValue(columnConfig, List.class);
   662→            for (Map<String, Object> col : cols) {
   663→                Object pickerConfig = col.get("pickerConfig");
   664→                if (!(pickerConfig instanceof String s) || s.isBlank()) {
   665→                    continue;
   666→                }
   667→                Map<String, Object> picker = objectMapper.readValue(s, Map.class);
   668→                Object target = picker.get("sourceFormKey");
   669→                if (target == null || String.valueOf(target).isBlank()) {
   670→                    continue;
   671→                }
   672→                String targetKey = String.valueOf(target);
   673→                Map<String, Object> entry = result.computeIfAbsent(targetKey, k -> {
   674→                    Map<String, Object> m = new LinkedHashMap<>();
   675→                    m.put("count", 0);
   676→                    m.put("referencedBy", new ArrayList<String>());
   677→                    return m;
   678→                });
   679→                entry.put("count", ((Number) entry.get("count")).intValue() + 1);
   680→                ((List<String>) entry.get("referencedBy")).add(def.getKey());
   681→            }
   682→        } catch (JsonProcessingException e) {
   683→            // 非法 column_config 跳过（发布链路已校验，此处容错）
   684→        }
   685→    }

   687→    private BizDataVO toVO(BizDataContext ctx, Map<String, Object> row) {
   688→        Map<String, Object> data = new LinkedHashMap<>();
   689→        for (ColumnConfig c : ctx.columns()) {
   690→            Object v = row.get(c.getKey());
   691→            if (v != null) {
   692→                data.put(c.getKey(), "JSON".equals(c.getColumnType()) ? deserializeJsonValue(v) : v);
   693→            }
   694→        }
   695→        // embedded 模式：附加子表行（按 sort_no 升序），保留 id 供前端 diff
   696→        for (Map.Entry<String, BizDataContext.SubTableDef> e : ctx.subTables().entrySet()) {
   697→            BizDataContext.SubTableDef def = e.getValue();
   698→            if ("embedded".equals(def.subMode())) {
   699→                data.put(e.getKey(), readSubRows(def, String.valueOf(row.get("id"))));
   700→            }
   701→        }
   702→        Integer version = asInt(row.get("version"));
   703→        LocalDateTime createdAt = asDateTime(row.get("created_at"));
   704→        LocalDateTime updatedAt = asDateTime(row.get("updated_at"));
   705→        return new BizDataVO(String.valueOf(row.get("id")), data, version, createdAt, updatedAt);
   706→    }

   708→    /** config 模式行映射：主表列（toVO 逻辑）+ 虚拟列（virtualKey → joinField 值） */
   709→    private BizDataVO toJoinVO(BizDataContext ctx, List<JoinSqlGenerator.JoinConfig> joins, Map<String, Object> row) {
   710→        BizDataVO vo = toVO(ctx, row);
   711→        Map<String, Object> data = vo.getData();
   712→        for (JoinSqlGenerator.JoinConfig j : joins) {
   713→            Object v = row.get(j.virtualKey());
   714→            if (v != null) {
   715→                data.put(j.virtualKey(), v);
   716→            }
   717→        }
   718→        return vo;
   719→    }

   721→    /**
   722→     * sql 模式行映射：外层子查询输出列全量保留（可含聚合列），绕过 BizDataVO 系统列字段的仅主表列逻辑。
   723→     * <p>输出列名与声明列（columns）的 key 做大小写不敏感对齐：匹配某声明列 key（忽略大小写）时输出声明列精确 key，
   724→     * 保证返回 key 与字段元数据一致（字段 key 可为驼峰，如探测 JDBC 列标签 hireDate，前端按配置 key 取值）；
   725→     * 未匹配列统一转小写，消除 H2 等数据库对子查询输出列名规范化为大写的影响（与声明列小写 key 约定兼容）。
   726→     */
   727→    private BizDataVO toSqlVO(List<ColumnConfig> declared, Map<String, Object> row) {
   728→        Map<String, String> keyNormalizer = new HashMap<>();
   729→        if (declared != null) {
   730→            for (ColumnConfig c : declared) {
   731→                if (c.getKey() != null && !c.getKey().isBlank()) {
   732→                    // putIfAbsent：声明列存在仅大小写不同的重名时，保留先出现的精确 key
   733→                    keyNormalizer.putIfAbsent(c.getKey().toLowerCase(), c.getKey());
   734→                }
   735→            }
   736→        }
   737→        Map<String, Object> data = new LinkedHashMap<>();
   738→        for (Map.Entry<String, Object> e : row.entrySet()) {
   739→            String lower = e.getKey().toLowerCase();
   740→            data.put(keyNormalizer.getOrDefault(lower, lower), e.getValue());
   741→        }
   742→        data.remove("version");
   743→        data.remove("created_at");
   744→        data.remove("updated_at");
   745→        data.remove("tenant_id");
   746→        Integer version = asInt(row.get("version"));
   747→        LocalDateTime createdAt = asDateTime(row.get("created_at"));
   748→        LocalDateTime updatedAt = asDateTime(row.get("updated_at"));
   749→        // id 保留在 data 中（管理员 SQL 输出可能有非 id 主键/聚合值），BizDataVO.id 取行内 id 兜底
   750→        return new BizDataVO(String.valueOf(row.get("id")), data, version, createdAt, updatedAt);
   751→    }

   756→    private Map<String, Object> serializeJsonColumns(Map<String, Object> data) {
   757→        Map<String, Object> out = new LinkedHashMap<>(data);
   758→        for (Map.Entry<String, Object> e : data.entrySet()) {

   766→                throw new BusinessException(400, "字段 " + e.getKey() + " 无法序列化为 JSON: " + ex.getOriginalMessage());
   767→            }
   768→        }
   769→        return out;
   770→    }

   801→    /**
   802→     * 子表行增量 diff（update 场景）。
   803→     */
   804→    private void diffSubRows(BizDataContext.SubTableDef def, String bizId, List<?> rows) {
   805→        if (rows.size() > MAX_SUB_ROWS) {
   806→            throw new BusinessException(400, "子表行数超限（最多 " + MAX_SUB_ROWS + " 行）: " + def.tableName());
   807→        }
   808→        List<Map<String, Object>> existing = readSubRows(def, bizId);
   809→        Map<String, Map<String, Object>> existingById = new LinkedHashMap<>();
   810→        for (Map<String, Object> r : existing) {
   811→            existingById.put(String.valueOf(r.get("id")), r);
   812→        }
   813→
   814→        Set<String> keepIds = new HashSet<>();
   815→        int sortNo = 0;
   816→        for (Object row : rows) {
   817→            Map<String, Object> m = toRowMap(row, def.tableName());
   818→            Object rawId = m.get("id");
   819→            String rowId = rawId == null ? null : String.valueOf(rawId);
   820→            if (rowId != null && existingById.containsKey(rowId)) {
   821→                Map<String, Object> cur = existingById.get(rowId);
   822→                boolean changed = def.subKeys().stream()
   823→                        .anyMatch(k -> !Objects.equals(cur.get(k), m.get(k)));
   824→                if (changed || !Objects.equals(cur.get("sort_no"), sortNo)) {
   825→                    StringBuilder set = new StringBuilder();
   826→                    List<Object> params = new ArrayList<>();
   827→                    for (String k : def.subKeys()) {
   828→                        set.append(k).append(" = ?, ");
   829→                        params.add(m.get(k));
   830→                    }
   831→                    set.append("sort_no = ?");
   832→                    params.add(sortNo);
   833→                    params.add(tenantProvider.getTenantId());
   834→                    params.add(bizId);
   835→                    params.add(rowId);
   836→                    jdbcTemplate.update("UPDATE " + def.tableName() + " SET " + set
   837→                            + " WHERE tenant_id = ? AND biz_id = ? AND id = ?", params.toArray());
   838→                }
   839→                keepIds.add(rowId);
   840→            } else {
   841→                // 新行：剥离客户端传入的 id，走内部生成
   842→                Map<String, Object> newRow = new LinkedHashMap<>(m);
   843→                newRow.remove("id");
   844→                insertOneSubRow(def, bizId, newRow, sortNo);
   845→            }
   846→            sortNo++;
   847→        }
   848→
   849→        // 删除请求中不存在的现有行
   850→        if (existingById.size() > keepIds.size()) {
   851→            List<Object> params = new ArrayList<>();
   852→            params.add(tenantProvider.getTenantId());
   853→            params.add(bizId);
   854→            StringBuilder ph = new StringBuilder();
   855→            for (String id : existingById.keySet()) {
   856→                if (!keepIds.contains(id)) {
   857→                    if (ph.length() > 0) {
   858→                        ph.append(",");
   859→                    }
   860→                    ph.append("?");
   861→                    params.add(id);
   862→                }
   863→            }
   864→            jdbcTemplate.update("DELETE FROM " + def.tableName()
   865→                    + " WHERE tenant_id = ? AND biz_id = ? AND id IN (" + ph + ")", params.toArray());
   866→        }
   867→    }

   872→    private List<Map<String, Object>> readSubRows(BizDataContext.SubTableDef def, String bizId) {

   878→    /**
   879→     * 插入单行子表数据（id/biz_id/tenant_id/sort_no/version + 子业务列）。
   880→     */
   881→    private Map<String, Object> insertOneSubRow(BizDataContext.SubTableDef def, String bizId,
   882→                                                Map<String, Object> m, int sortNo) {
   883→        String rowId = UUID.randomUUID().toString().replace("-", "");
   884→        List<Object> params = new ArrayList<>();
   885→        params.add(rowId);
   886→        params.add(bizId);
   887→        params.add(tenantProvider.getTenantId());
   888→        params.add(sortNo);
   889→        params.add(1); // version
   890→
   891→        StringBuilder cols = new StringBuilder("id, biz_id, tenant_id, sort_no, version");
   892→        StringBuilder vals = new StringBuilder("?, ?, ?, ?, ?");
   893→        for (String k : def.subKeys()) {
   894→            cols.append(", ").append(k);
   895→            vals.append(", ?");
   896→            params.add(m.get(k));
   897→        }
   898→        jdbcTemplate.update("INSERT INTO " + def.tableName() + " (" + cols + ") VALUES (" + vals + ")",
   899→                params.toArray());
   900→
   901→        Map<String, Object> inserted = new LinkedHashMap<>(m);
   902→        inserted.put("id", rowId);
   903→        inserted.put("sort_no", sortNo);
   904→        return inserted;
   905→    }

   907→    /** 子表行必须是 JSON 对象；非法抛 400 */
   908→    @SuppressWarnings("unchecked")
   909→    private static Map<String, Object> toRowMap(Object row, String tableName) {
   910→        if (!(row instanceof Map<?, ?> m)) {
   911→            throw new BusinessException(400, "子表行数据格式非法（需对象）: " + tableName);
   912→        }
   913→        Map<String, Object> out = new LinkedHashMap<>();
   914→        for (Map.Entry<?, ?> e : m.entrySet()) {
   915→            out.put(String.valueOf(e.getKey()), e.getValue());
   916→        }
