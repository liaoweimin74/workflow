   137→/** 拉起结果 → 用户可读摘要 */
   138→function summarizeActions(
   139→  actions: Record<string, SpawnOutcome>,
   140→  services: ServiceStatus[],
   141→): { kind: ToastMsg["kind"]; text: string } {
   142→  const nameOf: Record<string, string> = { backend: "后端", frontend: "前端" };
   143→  const parts: string[] = [];
   144→  let kind: ToastMsg["kind"] = "success";
   145→  for (const [key, a] of Object.entries(actions)) {
   146→    const label = nameOf[key] ?? key;
   147→    switch (a.action) {
   148→      case "spawned":
   149→        parts.push(`${label}已拉起，启动中…`);
   150→        kind = "info";
   151→        break;
   152→      case "already-running":
   153→        parts.push(`${label}运行正常`);
   154→        break;
   155→      case "starting":
   156→        parts.push(`${label}启动中…`);
   157→        kind = "info";
   158→        break;
   159→      case "backoff":
   160→        parts.push(`${label}崩溃冷却中：${a.reason ?? ""}`);
   161→        kind = "error";
   162→        break;
   163→      case "blocked":
   164→        parts.push(`${label}未就绪：${a.reason ?? "前置条件缺失"}`);
   165→        kind = "error";
   166→        break;
   167→    }
   168→  }
   169→  if (parts.length === 0) return { kind: "info", text: "检查完成" };
   170→  return { kind, text: parts.join("；") };
   171→}
