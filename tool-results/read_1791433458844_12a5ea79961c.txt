    37→/**
    38→ * 独立逻辑流编排解释执行引擎（普通类，由 FlowableEngineConfig 装配为 Bean）。
    39→ *
    40→ * <p>执行语义：
    41→ * <ul>
    42→ *   <li>从 START 节点出发沿出边行走，到达 END 结束；多个 START 取第一个，无 START → FAILED；</li>
    43→ *   <li>CONDITION 节点按 branch=true/false 选出边（缺边 → FAILED）；非条件/结束节点缺出边 → FAILED「节点无出边」；</li>
    44→ *   <li>maxSteps=200：步数超出 → FAILED「超出最大执行步数(疑似死循环)」；</li>
    45→ *   <li>HTTP/BEAN/SCRIPT 三型复用既有执行器；结果经 resultVar 写回变量上下文；
    46→ *       BATCH 遍历集合并按循环体链（config.body[]，逐项顺序执行链上各节点）执行，
    47→ *       聚合结果列表写入 resultVar；legacy 单动作（actionType+actionConfig）仍兼容；
    48→ *       SUBFLOW 调用另一条已发布逻辑流（环检测 + 深度限制），其 outputVars 写 resultVar；</li>
    49→ *   <li>节点异常按 errorAction：FAIL_FLOW（默认）→ 整个流 FAILED 停止；
    50→ *       IGNORE_CONTINUE → 该节点 trace 记 FAILED 后继续走边；</li>
    51→ *   <li>输出 outputVars = 全部变量快照（容器结构深拷贝）。</li>
    52→ * </ul>
    53→ */
    54→public class LogicFlowEngine {

   130→    /** 子流程最大嵌套深度（防递归失控）。 */

   133→    /** 子流程调用链栈（同一线程内递归执行共享；run 入口初始化，finally 清理）。 */
   134→    private final ThreadLocal<Deque<String>> subflowStack = new ThreadLocal<>();
   135→
   136→    /**
   137→     * 执行编排（无调用链起点，子流程环检测从空栈开始）。节点级失败/结构性问题
   138→     * （缺 START、缺出边、未知类型、死循环等）均不抛出，统一收敛为
   139→     * {@code status=FAILED} 的 RunOutcome，保证运行历史完整留痕。
   140→     */
   141→    public RunOutcome run(LogicFlowDsl dsl, Map<String, Object> inputVars) {
   142→        return run(dsl, inputVars, null);
   143→    }
   144→
   145→    /**
   146→     * 执行编排（带所属流 id，用于子流程自引用/环检测：ownerFlowId 即在调用链上，
   147→     * 子流程若调回自身所属流将被拒绝）。
   148→     */
   149→    public RunOutcome run(LogicFlowDsl dsl, Map<String, Object> inputVars, String ownerFlowId) {
   150→        subflowStack.set(new ArrayDeque<>());
   151→        if (ownerFlowId != null && !ownerFlowId.isBlank()) {
   152→            subflowStack.get().push(ownerFlowId);
   153→        }
   154→        try {
   155→            return runInternal(dsl, inputVars);
   156→        } finally {
   157→            subflowStack.remove();
   158→        }
   159→    }
   160→
   161→    private RunOutcome runInternal(LogicFlowDsl dsl, Map<String, Object> inputVars) {
   162→        List<NodeTrace> traces = new ArrayList<>();
   163→        Map<String, Object> vars = new LinkedHashMap<>();
   164→        if (inputVars != null) {
   165→            vars.putAll(inputVars);
   166→        }
   167→        if (dsl == null || dsl.getNodes() == null || dsl.getNodes().isEmpty()) {
   168→            return fail(vars, traces, "DSL 为空：缺少节点定义");
   169→        }
   170→
   171→        Map<String, LogicFlowDsl.NodeDef> nodeById = new LinkedHashMap<>();
   172→        for (LogicFlowDsl.NodeDef node : dsl.getNodes()) {
   173→            nodeById.put(node.getId(), node);
   174→        }
   175→        Map<String, List<LogicFlowDsl.EdgeDef>> outEdges = new LinkedHashMap<>();
   176→        for (LogicFlowDsl.EdgeDef edge : dsl.getEdges() != null ? dsl.getEdges() : List.<LogicFlowDsl.EdgeDef>of()) {
   177→            outEdges.computeIfAbsent(edge.getSource(), k -> new ArrayList<>()).add(edge);
   178→        }
   179→
   180→        LogicFlowDsl.NodeDef start = dsl.getNodes().stream()
   181→                .filter(n -> n.getType() == NodeType.START)
   182→                .findFirst()
   183→                .orElse(null);
   184→        if (start == null) {
   185→            return fail(vars, traces, "缺少 START 节点");
   186→        }
   187→
   188→        LogicFlowDsl.NodeDef current = start;
   189→        int step = 0;
   190→        try {
   191→            return walk(current, step, nodeById, outEdges, vars, traces);
   192→        } catch (FlowAbortedException e) {
   193→            // 缺出边/边引用缺失节点/FAIL_FLOW 中断 → 统一收敛为 FAILED，引擎不向调用方抛异常
   194→            return fail(vars, traces, e.getMessage());
   195→        }
   196→    }
   197→
   198→    /**
   199→     * 主行走循环：从 current 出发沿边执行，直到 END（返回 SUCCESS outcome）
   200→     * 或因缺出边/FAIL_FLOW 等抛出 {@link FlowAbortedException} 中断。
   201→     */
   202→    private RunOutcome walk(LogicFlowDsl.NodeDef current, int step,
   203→                            Map<String, LogicFlowDsl.NodeDef> nodeById,
   204→                            Map<String, List<LogicFlowDsl.EdgeDef>> outEdges,
   205→                            Map<String, Object> vars,
   206→                            List<NodeTrace> traces) {
   207→        while (current != null) {
   208→            if (++step > MAX_STEPS) {
   209→                return fail(vars, traces, "超出最大执行步数(疑似死循环)");
   210→            }
   211→            NodeType type = current.getType();
   212→            if (type == null) {
   213→                return fail(vars, traces, "UNKNOWN_NODE_TYPE: null (node " + current.getId() + ")");
   214→            }
   215→            switch (type) {
   216→                case START -> {
   217→                    traces.add(new NodeTrace(current.getId(), current.getName(), type.name(),
   218→                            TRACE_SUCCESS, null, null, 0L));
   219→                    current = requireNext(nodeById, outEdges, current.getId());
   220→                }
   221→                case END -> {
   222→                    traces.add(new NodeTrace(current.getId(), current.getName(), type.name(),
   223→                            TRACE_SUCCESS, null, null, 0L));
   224→                    return new RunOutcome(STATUS_SUCCESS, snapshot(vars), List.copyOf(traces), null);
   225→                }
   226→                case CONDITION -> current = executeCondition(current, nodeById, outEdges, vars, traces);
   227→                default -> current = executeLogicNode(current, nodeById, outEdges, vars, traces);
   228→            }
   229→        }
   230→        // 不可达兜底（所有分支要么推进 current，要么已 return）
   231→        return fail(vars, traces, "执行中断于未知位置");
   232→    }

   238→    /**
   239→     * 执行 HTTP/BEAN/SCRIPT 三型逻辑节点，返回下一节点。
   240→     * 失败按 errorAction：FAIL_FLOW → 抛 {@link FlowAbortedException}（run 顶层收敛为 FAILED）；
   241→     * IGNORE_CONTINUE → trace 记 FAILED 后继续走边。
   242→     */
   243→    private LogicFlowDsl.NodeDef executeLogicNode(LogicFlowDsl.NodeDef node,
   244→                                                  Map<String, LogicFlowDsl.NodeDef> nodeById,
   245→                                                  Map<String, List<LogicFlowDsl.EdgeDef>> outEdges,
   246→                                                  Map<String, Object> vars,
   247→                                                  List<NodeTrace> traces) {
   248→        long begin = System.currentTimeMillis();
   249→        try {
   250→            Object result = dispatch(node, vars, traces);
   251→            long duration = System.currentTimeMillis() - begin;
   252→            String resultVar = node.getResultVar();
   253→            if (resultVar != null && !resultVar.isBlank()) {
   254→                vars.put(resultVar, result);
   255→            }
   256→            if (node.getType() == NodeType.SCRIPT) {
   257→                expandScriptOutputs(node, result, vars);
   258→            }
   259→            traces.add(new NodeTrace(node.getId(), node.getName(), node.getType().name(),
   260→                    TRACE_SUCCESS, result, null, duration));
   261→        } catch (Exception e) {
   262→            long duration = System.currentTimeMillis() - begin;
   263→            String error = e.getMessage() != null ? e.getMessage() : e.getClass().getSimpleName();
   264→            traces.add(new NodeTrace(node.getId(), node.getName(), node.getType().name(),
   265→                    TRACE_FAILED, null, error, duration));
   266→            if (!ACTION_IGNORE_CONTINUE.equalsIgnoreCase(node.getErrorAction())) {
   267→                throw new FlowAbortedException("节点执行失败: " + node.getId() + ": " + error);
   268→            }
   269→            log.debug("Logic flow node '{}' failed but continues (IGNORE_CONTINUE): {}", node.getId(), error);
   270→        }
   271→        return requireNext(nodeById, outEdges, node.getId());
   272→    }

   274→    /** 执行 CONDITION 节点：求值 → 按 branch=true/false 选边；返回下一节点。 */
   275→    private LogicFlowDsl.NodeDef executeCondition(LogicFlowDsl.NodeDef node,
   276→                                                  Map<String, LogicFlowDsl.NodeDef> nodeById,
   277→                                                  Map<String, List<LogicFlowDsl.EdgeDef>> outEdges,
   278→                                                  Map<String, Object> vars,
   279→                                                  List<NodeTrace> traces) {
   280→        long begin = System.currentTimeMillis();
   281→        boolean branch;
   282→        try {
   283→            JsonNode config = node.getConfig();
   284→            if (config == null || config.isNull()) {
   285→                throw new IllegalArgumentException("CONDITION 节点缺少 config: " + node.getId());
   286→            }
   287→            String variable = textOrNull(config, "variable");
   288→            String operator = textOrNull(config, "operator");
   289→            String value = textOrNull(config, "value");
   290→            branch = ConditionEvaluator.evaluate(variable, operator, value, vars, variableResolver);
   291→        } catch (Exception e) {
   292→            long duration = System.currentTimeMillis() - begin;
   293→            String error = e.getMessage() != null ? e.getMessage() : e.getClass().getSimpleName();
   294→            traces.add(new NodeTrace(node.getId(), node.getName(), NodeType.CONDITION.name(),
   295→                    TRACE_FAILED, null, error, duration));
   296→            if (!ACTION_IGNORE_CONTINUE.equalsIgnoreCase(node.getErrorAction())) {
   297→                throw new FlowAbortedException("条件评估失败: " + node.getId() + ": " + error);
   298→            }
   299→            // 容错继续：条件失败默认走 false 分支
   300→            branch = false;
   301→        }
   302→        long duration = System.currentTimeMillis() - begin;
   303→        if (node.getResultVar() != null && !node.getResultVar().isBlank()) {
   304→            vars.put(node.getResultVar(), branch);
   305→        }
   306→        traces.add(new NodeTrace(node.getId(), node.getName(), NodeType.CONDITION.name(),
   307→                TRACE_SUCCESS, String.valueOf(branch), null, duration));
   308→        return resolveBranchTarget(node, nodeById, outEdges.get(node.getId()), branch);
   309→    }

   311→    /** 四型逻辑 + 批处理/子流程分发。 */
   312→    private Object dispatch(LogicFlowDsl.NodeDef node, Map<String, Object> vars, List<NodeTrace> traces) {
   313→        return switch (node.getType()) {
   314→            case HTTP -> executeHttp(node, vars);
   315→            case BEAN -> executeBean(node, vars);
   316→            case SCRIPT -> executeScript(node, vars);
   317→            case BATCH -> executeBatch(node, vars, traces);
   318→            case SUBFLOW -> executeSubflow(node, vars);
   319→            case DATA_UPDATE -> executeDataUpdate(node, vars);
   320→            default -> throw new IllegalArgumentException("节点 " + node.getId() + " 不支持执行: " + node.getType());
   321→        };
   322→    }
   323→
   324→    private Object executeHttp(LogicFlowDsl.NodeDef node, Map<String, Object> vars) {
   325→        BackendLogicHttpConfig config = readConfig(node, BackendLogicHttpConfig.class);
   326→        if (config == null || config.getUrl() == null || config.getUrl().isBlank()) {
   327→            throw new IllegalArgumentException("HTTP 节点缺少 url 配置: " + node.getId());
   328→        }
   329→        String method = config.getMethod() == null || config.getMethod().isBlank()
   330→                ? "GET" : config.getMethod();
   331→        Map<String, String> headers = config.getHeaders() != null ? config.getHeaders() : Map.of();
   332→        List<ParamMapping> query = config.getQueryParams() != null ? config.getQueryParams() : List.of();
   333→        List<ParamMapping> body = config.getBodyParams() != null ? config.getBodyParams() : List.of();
   334→        // url/headers 的 {{var}} 展开由 HttpLogicExecutor 内部完成（占位符替换后、发请求前做白名单校验）
   335→        return httpExecutor.execute(config.getUrl(), method, headers, query, body, vars,
   336→                config.getConnTimeoutMs(), config.getReadTimeoutMs(), Math.max(0, config.getRetryCount()));
   337→    }
   338→
   339→    private Object executeBean(LogicFlowDsl.NodeDef node, Map<String, Object> vars) {
   340→        BackendLogicBeanConfig config = readConfig(node, BackendLogicBeanConfig.class);
   341→        if (config == null || isBlank(config.getBeanName()) || isBlank(config.getMethodName())) {
   342→            throw new IllegalArgumentException("BEAN 节点缺少 beanName/methodName 配置: " + node.getId());
   343→        }
   344→        List<ParamMapping> params = config.getParams() != null ? config.getParams() : List.of();
   345→        Object[] args = params.stream()
   346→                .map(pm -> vars.get(pm.source()))
   347→                .toArray();
   348→        // 未注册的 bean 方法由 BackendBeanRegistry 抛出 IllegalArgumentException → 走 errorAction
   349→        return backendBeanRegistry.invoke(config.getBeanName(), config.getMethodName(), args);
   350→    }
   351→
   352→    private Object executeScript(LogicFlowDsl.NodeDef node, Map<String, Object> vars) {
   353→        BackendLogicScriptConfig config = readConfig(node, BackendLogicScriptConfig.class);
   354→        if (config == null) {
   355→            throw new IllegalArgumentException("SCRIPT 节点缺少 config: " + node.getId());
   356→        }
   357→        if (!"groovy".equalsIgnoreCase(config.getLanguage())) {
   358→            throw new IllegalArgumentException("UNSUPPORTED_LANGUAGE: " + config.getLanguage());
   359→        }
   360→        return groovyScriptLogic.execute(config.getSource(), null, vars);
   361→    }
   362→
   363→    /**
   364→     * SCRIPT 节点多输出展开（outputs 声明式，与 resultVar 双轨并存）：
   365→     * <ul>
   366→     *   <li>脚本返回 Map：按声明逐 key 拆包写入扁平上下文；声明 key 缺失 → 跳过不写（防 null 覆盖上游变量）；</li>
   367→     *   <li>声明名与上游同名变量冲突 → log.warn 警告后放行（首版宽容策略）；</li>
   368→     *   <li>声明名与 resultVar 同名 → 跳过展开并警告（发布校验已拦截，运行期纵深防御）；</li>
   369→     *   <li>返回非 Map 而声明了 outputs → 抛 IllegalArgumentException（节点 FAILED，配置错误尽早暴露）。</li>
   370→     * </ul>
   371→     * 调用点须在 resultVar 写回之后、trace 记 SUCCESS 之前（失败归入节点异常路径）。
   372→     */
   373→    private void expandScriptOutputs(LogicFlowDsl.NodeDef node, Object result, Map<String, Object> vars) {
   374→        List<LogicFlowDsl.OutputVarDef> outputs = node.getOutputs();
   375→        if (outputs == null || outputs.isEmpty()) {
   376→            return;
   377→        }
   378→        if (!(result instanceof Map)) {
   379→            throw new IllegalArgumentException("SCRIPT 节点 " + node.getId()
   380→                    + " 配置了 outputs 但脚本未返回 Map（实际 " + (result == null ? "null" : result.getClass().getSimpleName())
   381→                    + "）：outputs 需脚本以 [key: value, ...] 形式返回");
   382→        }
   383→        Map<?, ?> map = (Map<?, ?>) result;
   384→        String resultVar = node.getResultVar();
   385→        for (LogicFlowDsl.OutputVarDef def : outputs) {
   386→            String name = def != null ? def.getName() : null;
   387→            if (name == null || name.isBlank()) {
   388→                continue;
   389→            }
   390→            if (name.equals(resultVar)) {
   391→                log.warn("SCRIPT 节点 '{}' 输出变量 '{}' 与 resultVar 同名，跳过展开（双轨不重叠）",
   392→                        node.getName() != null ? node.getName() : node.getId(), name);
   393→                continue;
   394→            }
   395→            if (!map.containsKey(name)) {
   396→                continue;
   397→            }
   398→            if (vars.containsKey(name)) {
   399→                log.warn("SCRIPT 节点 '{}' 输出变量 '{}' 覆盖上游同名变量（outputs 冲突警告放行）",
   400→                        node.getName() != null ? node.getName() : node.getId(), name);
   401→            }
   402→            vars.put(name, map.get(name));
   403→        }
   404→    }

   410→    /** 数据更新节点允许操作的动态表名前缀（业务表单动态表 / 审批表单数据表）。 */
   411→    private static final List<String> DATA_UPDATE_TABLE_PREFIXES = List.of("wf_biz_", "wf_form_data");
   412→    /** 数据更新节点 SET 禁改列（租户隔离列，防止跨租户污染）。 */
   413→    private static final String DATA_UPDATE_FORBIDDEN_COLUMN = "tenant_id";
   414→
   415→    /**
   416→     * 执行数据更新节点：纯配置 UPDATE 动态表。
   417→     * 表名/列名经元数据校验，值经参数绑定执行（防注入）；受影响行数返回写回 resultVar。
   418→     */
   419→    private Object executeDataUpdate(LogicFlowDsl.NodeDef node, Map<String, Object> vars) {
   420→        BackendDataUpdateConfig config = readConfig(node, BackendDataUpdateConfig.class);
   421→        if (config == null || isBlank(config.getTable())) {
   422→            throw new IllegalArgumentException("DATA_UPDATE 节点缺少 table 配置: " + node.getId());
   423→        }
   424→        if (config.getSetOps() == null || config.getSetOps().isEmpty()) {
   425→            throw new IllegalArgumentException("DATA_UPDATE 节点缺少 setOps 配置: " + node.getId());
   426→        }
   427→        if (jdbcTemplate == null || tableManager == null) {
   428→            throw new IllegalStateException("引擎未装配数据更新能力(JdbcTemplate/DynamicTableManager): " + node.getId());
   429→        }
   430→
   431→        String table = config.getTable().trim();
   432→        validateDataUpdateTable(table);
   433→        Map<String, ColumnInfo> columns = columnsOf(table);
   434→
   435→        StringBuilder sql = new StringBuilder("UPDATE ").append(table).append(" SET ");
   436→        List<Object> params = new ArrayList<>();
   437→        List<String> setParts = new ArrayList<>();
   438→        for (BackendDataUpdateConfig.SetOp setOp : config.getSetOps()) {
   439→            String column = requireColumn(columns, table, setOp.getColumn(), "SET");
   440→            if (DATA_UPDATE_FORBIDDEN_COLUMN.equalsIgnoreCase(column)) {
   441→                throw new IllegalArgumentException("DATA_UPDATE 节点禁止修改列 tenant_id: " + node.getId());
   442→            }
   443→            String mode = setOp.getMode() == null || setOp.getMode().isBlank()
   444→                    ? BackendDataUpdateConfig.MODE_SET : setOp.getMode().trim().toUpperCase(Locale.ROOT);
   445→            Object value = resolveDataUpdateValue(setOp.getValue(), vars);
   446→            switch (mode) {
   447→                case BackendDataUpdateConfig.MODE_SET -> setParts.add(column + " = ?");
   448→                case BackendDataUpdateConfig.MODE_ADD -> {
   449→                    setParts.add(column + " = " + column + " + ?");
   450→                    value = requireNumeric(value, node.getId(), column, "ADD");
   451→                }
   452→                case BackendDataUpdateConfig.MODE_SUB -> {
   453→                    setParts.add(column + " = " + column + " - ?");
   454→                    value = requireNumeric(value, node.getId(), column, "SUB");
   455→                }
   456→                default -> throw new IllegalArgumentException(
   457→                        "DATA_UPDATE 节点 setOp.mode 非法(须 SET/ADD/SUB): " + setOp.getMode());
   458→            }
   459→            params.add(value);
   460→        }
   461→        sql.append(String.join(", ", setParts));
   462→
   463→        List<BackendDataUpdateConfig.WhereCond> where = config.getWhere() != null
   464→                ? config.getWhere() : List.of();
   465→        if (!where.isEmpty()) {
   466→            sql.append(" WHERE ");
   467→            List<String> whereParts = new ArrayList<>();
   468→            for (BackendDataUpdateConfig.WhereCond cond : where) {
   469→                String column = requireColumn(columns, table, cond.getColumn(), "WHERE");
   470→                String op = cond.getOp() == null ? "" : cond.getOp().trim().toUpperCase(Locale.ROOT);
   471→                switch (op) {
   472→                    case "IS_NULL" -> whereParts.add(column + " IS NULL");
   473→                    case "NOT_NULL" -> whereParts.add(column + " IS NOT NULL");
   474→                    case "EQ", "NE", "GT", "GTE", "LT", "LTE" -> {
   475→                        String symbol = switch (op) {
   476→                            case "EQ" -> "=";
   477→                            case "NE" -> "<>";
   478→                            case "GT" -> ">";
   479→                            case "GTE" -> ">=";
   480→                            case "LT" -> "<";
   481→                            default -> "<=";
   482→                        };
   483→                        whereParts.add(column + " " + symbol + " ?");
   484→                        params.add(resolveDataUpdateValue(cond.getValue(), vars));
   485→                    }
   486→                    default -> throw new IllegalArgumentException(
   487→                            "DATA_UPDATE 节点 where.op 非法(须 EQ/NE/GT/GTE/LT/LTE/IS_NULL/NOT_NULL): " + cond.getOp());
   488→                }
   489→            }
   490→            if (whereParts.isEmpty()) {
   491→                throw new IllegalArgumentException("DATA_UPDATE 节点 WHERE 条件为空: " + node.getId());
   492→            }
   493→            sql.append(String.join(" AND ", whereParts));
   494→        }
   495→
   496→        int affected = jdbcTemplate.update(sql.toString(), params.toArray());
   497→        if (log.isDebugEnabled()) {
   498→            log.debug("DATA_UPDATE node '{}' -> {} rows: {}", node.getId(), affected, sql);
   499→        }
   500→        return affected;
   501→    }

   503→    /** 表名校验：合法标识符 + 存在 + 前缀白名单（仅平台动态数据表）。 */
   504→    private void validateDataUpdateTable(String table) {
   505→        if (!table.matches("[a-zA-Z_][a-zA-Z0-9_]*")) {
   506→            throw new IllegalArgumentException("DATA_UPDATE 表名非法: " + table);

   509→        if (!allowed) {
   510→            throw new IllegalArgumentException(
   511→                    "DATA_UPDATE 仅允许平台动态数据表(wf_biz_*/wf_form_data*): " + table);
   512→        }
   513→        if (!tableManager.tableExists(table)) {
   514→            throw new IllegalArgumentException("DATA_UPDATE 目标表不存在: " + table);
   515→        }
   516→    }

   526→    /** 列存在性校验（大小写不敏感），返回规范化列名。 */
   527→    private String requireColumn(Map<String, ColumnInfo> columns, String table, String column, String clause) {
   528→        if (column == null || column.isBlank()) {
   529→            throw new IllegalArgumentException("DATA_UPDATE " + clause + " 列名为空: " + table);
   530→        }
   531→        String name = column.trim();
   532→        if (!columns.containsKey(name.toLowerCase(Locale.ROOT))) {
   533→            throw new IllegalArgumentException("DATA_UPDATE 目标表缺少列: " + table + "." + name);
   534→        }
   535→        return name;

   538→    /**
   539→     * 取值解析：纯单占位符 {@code {{path}}} 返回变量原始对象（保留数值类型）；
   540→     * 含占位符的混合模板做字符串插值；其余按字面量返回。
   541→     * path 支持 {@code var} 与 {@code var.sub.sub}（逐层 Map 取值）。
   542→     */
   543→    private Object resolveDataUpdateValue(String expr, Map<String, Object> vars) {
   544→        if (expr == null) {
   545→            return null;
   546→        }
   547→        String trimmed = expr.trim();
   548→        java.util.regex.Matcher pure = java.util.regex.Pattern
   549→                .compile("\\{\\{\\s*([\\w]+(?:\\.[\\w]+)*)\\s*}}").matcher(trimmed);
   550→        if (pure.matches()) {
   551→            return resolvePath(pure.group(1), vars);
   552→        }
   553→        if (trimmed.contains("{{")) {
   554→            java.util.regex.Matcher matcher = java.util.regex.Pattern
   555→                    .compile("\\{\\{\\s*([\\w]+(?:\\.[\\w]+)*)\\s*}}").matcher(trimmed);
   556→            StringBuilder sb = new StringBuilder();
   557→            int last = 0;
   558→            while (matcher.find()) {
   559→                sb.append(trimmed, last, matcher.start());
   560→                Object value = resolvePath(matcher.group(1), vars);
   561→                sb.append(value != null ? value.toString() : "");
   562→                last = matcher.end();
   563→            }
   564→            sb.append(trimmed.substring(last));
   565→            return sb.toString();
   566→        }
   567→        return trimmed;
   568→    }
   569→
   570→    /** 点路径取值：首段为变量名，后续逐层 Map 取值；任一层缺失返回 null。 */
   571→    private Object resolvePath(String path, Map<String, Object> vars) {
   572→        String[] parts = path.split("\\.");
   573→        Object current = vars.get(parts[0]);
   574→        for (int i = 1; i < parts.length && current != null; i++) {
   575→            if (current instanceof Map<?, ?> map) {
   576→                current = map.get(parts[i]);
   577→            } else {
   578→                throw new IllegalArgumentException(
   579→                        "DATA_UPDATE 取值路径中间层非对象: " + path + " (于 " + parts[i - 1] + ")");
   580→            }
   581→        }
   582→        return current;
   583→    }

   585→    /** ADD/SUB 模式数值校验：Number 直通，字符串须可解析为 BigDecimal。 */
   586→    private Object requireNumeric(Object value, String nodeId, String column, String mode) {
   587→        if (value instanceof Number number) {
   588→            return new BigDecimal(number.toString());
   589→        }
   590→        if (value == null || value.toString().isBlank()) {
   591→            throw new IllegalArgumentException("DATA_UPDATE 节点 " + mode + " 模式列 " + column + " 值不能为空");
   592→        }
   593→        try {
   594→            return new BigDecimal(value.toString().trim());
   595→        } catch (NumberFormatException e) {
   596→            throw new IllegalArgumentException(
   597→                    "DATA_UPDATE 节点 " + mode + " 模式列 " + column + " 值须为数字: " + value);
   598→        }
   599→    }

   605→    /**
   606→     * 执行子流程节点：查目标已发布流 → 构建子入参 → 递归执行 → outputVars 作为节点结果。
   607→     *
   608→     * <p>config 形态：{@code {flowId, passAllVars=true, varsMapping:[{source,target}]}}。
   609→     *
   610→     * <p>语义：passAllVars=true（默认）时全量继承当前上下文，varsMapping 在此基础上
   611→     * 逐项覆盖/改名（后写胜出）；passAllVars=false 时仅传映射后的变量。
   612→     * 环检测：目标 flowId 已在调用链（含当前流自身）→ 拒绝；嵌套深度上限
   613→     * {@link #MAX_SUBFLOW_DEPTH}。子流程未发布/不存在 → 节点异常（走 errorAction）。
   614→     * 子流程自身轨迹不并入父轨迹；节点 result = 子流程 outputVars（写 resultVar 可选）。
   615→     */
   616→    private Object executeSubflow(LogicFlowDsl.NodeDef node, Map<String, Object> vars) {
   617→        JsonNode config = node.getConfig();
   618→        if (config == null || config.isNull()) {
   619→            throw new IllegalArgumentException("SUBFLOW 节点缺少 config: " + node.getId());
   620→        }
   621→        if (defRepository == null) {
   622→            throw new IllegalStateException("引擎未装配子流程仓库，无法执行 SUBFLOW: " + node.getId());
   623→        }
   624→        String targetId = textOrNull(config, "flowId");
   625→        if (targetId == null || targetId.isBlank()) {
   626→            throw new IllegalArgumentException("SUBFLOW 节点缺少 flowId 配置: " + node.getId());
   627→        }
   628→        Deque<String> stack = subflowStack.get();
   629→        if (stack == null) {
   630→            // 防御：run 内部必已初始化；直接调用 runInternal 时才可能到达
   631→            stack = new ArrayDeque<>();
   632→            subflowStack.set(stack);
   633→        }
   634→        if (stack.contains(targetId)) {
   635→            throw new IllegalArgumentException("检测到子流程循环调用: " + targetId);
   636→        }
   637→        if (stack.size() >= MAX_SUBFLOW_DEPTH) {
   638→            throw new IllegalArgumentException("超出子流程最大嵌套深度(" + MAX_SUBFLOW_DEPTH + "): " + targetId);
   639→        }
   640→        LogicFlowDef target = defRepository.findById(targetId)
   641→                .orElseThrow(() -> new IllegalArgumentException("目标子流程不存在: " + targetId));
   642→        if (!"PUBLISHED".equals(target.getStatus())) {
   643→            throw new IllegalArgumentException("目标子流程未发布，无法调用: " + target.getFlowKey());
   644→        }
   645→        LogicFlowDsl subDsl = LogicFlowDsl.parse(target.getDslJson(), objectMapper);
   646→
   647→        boolean passAllVars = !config.has("passAllVars") || config.get("passAllVars").asBoolean(true);
   648→        Map<String, Object> subVars = new LinkedHashMap<>();
   649→        if (passAllVars) {
   650→            subVars.putAll(vars);
   651→        }
   652→        JsonNode mapping = config.get("varsMapping");
   653→        if (mapping != null && mapping.isArray()) {
   654→            for (JsonNode pair : mapping) {
   655→                String source = textOrNull(pair, "source");
   656→                String targetName = textOrNull(pair, "target");
   657→                if (source == null || source.isBlank() || targetName == null || targetName.isBlank()) {
   658→                    continue; // 校验器已拦；运行期宽容跳过空映射
   659→                }
   660→                subVars.put(targetName, vars.get(source));
   661→            }
   662→        }
   663→
   664→        stack.push(targetId);
   665→        try {
   666→            RunOutcome sub = runInternal(subDsl, subVars);
   667→            if (STATUS_SUCCESS.equals(sub.status())) {
   668→                return sub.outputVars();
   669→            }
   670→            throw new IllegalStateException("子流程执行失败(" + target.getFlowKey() + "): "
   671→                    + (sub.errorMessage() != null ? sub.errorMessage() : "未知原因"));
   672→        } finally {
   673→            stack.pop();
   674→        }
   675→    }

   681→    /** 批处理单次最大迭代数（缺省值）。 */
   682→    public static final int BATCH_DEFAULT_MAX_ITEMS = 100;
   683→    /** 批处理迭代数硬上限（防失控）。 */
   684→    public static final int BATCH_HARD_MAX_ITEMS = 1000;
   685→    /** 批处理轨迹 result 中保留的单项结果/错误明细上限（防止轨迹膨胀）。 */
   686→    public static final int BATCH_TRACE_DETAIL_LIMIT = 20;
   687→    /** 循环体节点 trace 记录的迭代上限（只记首轮 + 失败项，防轨迹膨胀）。 */
   688→    private static final int BATCH_BODY_TRACE_ITERATIONS = 1;
   689→
   690→    /** 批处理循环体允许的节点类型（业务执行五型 + BATCH 嵌套）。 */
   691→    private static final Set<NodeType> BATCH_BODY_ALLOWED = EnumSet.of(
   692→            NodeType.HTTP, NodeType.BEAN, NodeType.SCRIPT, NodeType.DATA_UPDATE, NodeType.SUBFLOW,
   693→            NodeType.BATCH);
   694→
   695→    /** BATCH 嵌套深度运行时上限（发布校验限 3 层，此处纵深防御防绕过校验的自引用 DSL） */
   696→    private static final int MAX_BATCH_NESTING_DEPTH = 5;
   697→    private static final ThreadLocal<Integer> BATCH_DEPTH = ThreadLocal.withInitial(() -> 0);
   698→
   699→    /** 批处理汇总（写入 resultVar / 节点轨迹 result）。 */
   700→    record BatchSummary(int total, int succeeded, int failed, boolean truncated,
   701→                        List<Object> results, List<Map<String, Object>> errors) {
   702→    }
   703→
   704→    /**
   705→     * 执行批处理节点：解析集合 → 逐项执行循环体链（或 legacy 单动作）→ 聚合结果列表。
   706→     *
   707→     * <p>config 形态（body 模式，推荐）：{@code {collection, itemVar="item", indexVar="index",
   708→     * body:[{id?, type, name?, config, resultVar?, errorAction?}...], stopOnError=true, maxItems=100}}。
   709→     * legacy 模式（无 body）：{@code {actionType: HTTP|SCRIPT|BEAN, actionConfig:{...}}}。
   710→     *
   711→     * <p>语义：每项在「主变量上下文的副本」中执行（itemVar/indexVar 注入），
   712→     * 循环体链按序逐节点执行，节点 resultVar 写入副本，迭代后副本并回主上下文
   713→     * （后写胜出，脚本副作用跨项保留）；聚合结果按顺序写入 resultVar
   714→     * （汇总 Map：total/succeeded/failed/truncated/results/errors，results 取每项链末结果）；
   715→     * 链内节点 errorAction=IGNORE_CONTINUE → 跳过该步继续后续节点；
   716→     * 否则该项失败，stopOnError=true 时首个失败项向上抛出（由外层 errorAction 决定中断或跳过继续）。
   717→     */
   718→    private Object executeBatch(LogicFlowDsl.NodeDef node, Map<String, Object> vars, List<NodeTrace> traces) {
   719→        int depth = BATCH_DEPTH.get();
   720→        if (depth >= MAX_BATCH_NESTING_DEPTH) {
   721→            throw new IllegalArgumentException(
   722→                    "BATCH 嵌套过深（>" + MAX_BATCH_NESTING_DEPTH + " 层）：" + node.getId());
   723→        }
   724→        BATCH_DEPTH.set(depth + 1);
   725→        try {
   726→            return doExecuteBatch(node, vars, traces);
   727→        } finally {
   728→            BATCH_DEPTH.set(depth);
   729→        }
   730→    }
   731→
   732→    private Object doExecuteBatch(LogicFlowDsl.NodeDef node, Map<String, Object> vars, List<NodeTrace> traces) {
   733→        JsonNode config = node.getConfig();
   734→        if (config == null || config.isNull()) {
   735→            throw new IllegalArgumentException("BATCH 节点缺少 config: " + node.getId());
   736→        }
   737→        String collectionExpr = textOrNull(config, "collection");
   738→        if (collectionExpr == null || collectionExpr.isBlank()) {
   739→            throw new IllegalArgumentException("BATCH 节点缺少 collection 配置: " + node.getId());
   740→        }
   741→        JsonNode bodyNode = config.get("body");
   742→        List<LogicFlowDsl.NodeDef> body = bodyNode != null && bodyNode.isArray()
   743→                ? parseBatchBody(node, bodyNode)
   744→                : List.of();
   745→        String itemVar = orDefault(textOrNull(config, "itemVar"), "item");
   746→        String indexVar = orDefault(textOrNull(config, "indexVar"), "index");
   747→        boolean stopOnError = !config.has("stopOnError") || config.get("stopOnError").isNull()
   748→                || config.get("stopOnError").asBoolean(true);
   749→        int maxItems = normalizeMaxItems(config.get("maxItems"));
   750→        if (body.isEmpty() && (textOrNull(config, "actionType") == null || textOrNull(config, "actionType").isBlank())) {
   751→            throw new IllegalArgumentException(
   752→                    "BATCH 节点缺少循环体(body)或 legacy actionType 配置: " + node.getId());
   753→        }
   754→
   755→        List<Object> items = resolveCollection(collectionExpr, vars);
   756→        boolean truncated = items.size() > maxItems;
   757→        if (truncated) {
   758→            items = items.subList(0, maxItems);
   759→        }
   760→
   761→        List<Object> results = new ArrayList<>(items.size());
   762→        List<Map<String, Object>> errors = new ArrayList<>();
   763→        int succeeded = 0;
   764→        int failed = 0;
   765→        for (int i = 0; i < items.size(); i++) {
   766→            Object item = items.get(i);
   767→            Map<String, Object> childVars = new LinkedHashMap<>(vars);
   768→            childVars.put(itemVar, item);
   769→            childVars.put(indexVar, i);
   770→            try {
   771→                Object last = null;
   772→                if (!body.isEmpty()) {
   773→                    for (int k = 0; k < body.size(); k++) {
   774→                        last = executeBatchBodyStep(node, body.get(k), i, k, childVars, traces);
   775→                    }
   776→                } else {
   777→                    // legacy 单动作路径（兼容旧 DSL）
   778→                    String actionTypeName = textOrNull(config, "actionType");
   779→                    NodeType actionType;
   780→                    try {
   781→                        actionType = NodeType.fromJson(actionTypeName);
   782→                    } catch (IllegalArgumentException e) {
   783→                        throw new IllegalArgumentException("BATCH 节点 actionType 非法: " + actionTypeName);
   784→                    }
   785→                    if (actionType != NodeType.HTTP && actionType != NodeType.SCRIPT && actionType != NodeType.BEAN) {
   786→                        throw new IllegalArgumentException("BATCH 节点 actionType 仅支持 HTTP/SCRIPT/BEAN: " + actionTypeName);
   787→                    }
   788→                    LogicFlowDsl.NodeDef inner = new LogicFlowDsl.NodeDef();
   789→                    inner.setId(node.getId() + "#" + i);
   790→                    inner.setName(node.getName());
   791→                    inner.setType(actionType);
   792→                    inner.setConfig(config.get("actionConfig"));
   793→                    last = dispatch(inner, childVars, traces);
   794→                }
   795→                results.add(last);
   796→                succeeded++;
   797→                vars.putAll(childVars);
   798→            } catch (Exception e) {
   799→                failed++;
   800→                String error = e.getMessage() != null ? e.getMessage() : e.getClass().getSimpleName();
   801→                if (errors.size() < BATCH_TRACE_DETAIL_LIMIT) {
   802→                    Map<String, Object> err = new LinkedHashMap<>();
   803→                    err.put("index", i);
   804→                    err.put("error", error);
   805→                    errors.add(err);
   806→                }
   807→                if (stopOnError) {
   808→                    throw new IllegalArgumentException("BATCH 第 " + (i + 1) + "/" + items.size()
   809→                            + " 项执行失败: " + error);
   810→                }
   811→                log.debug("Batch node '{}' item {} failed but continues: {}", node.getId(), i, error);
   812→            }
   813→        }
   814→        List<Object> traceResults = results.size() > BATCH_TRACE_DETAIL_LIMIT
   815→                ? new ArrayList<>(results.subList(0, BATCH_TRACE_DETAIL_LIMIT)) : results;
   816→        return new BatchSummary(items.size(), succeeded, failed, truncated,
   817→                traceResults, errors);
   818→    }

   820→    /**
   821→     * 执行循环体链中的单步。链内节点失败语义：errorAction=IGNORE_CONTINUE → 记失败轨迹后
   822→     * 跳过该步继续链内后续节点；否则向上抛出（该项失败，由 stopOnError 决定整批走向）。
   823→     * trace 记录：首轮迭代（i==0）与失败步记录（画布循环体节点 id 不含 '#'，可点亮运行状态）。
   824→     */
   825→    private Object executeBatchBodyStep(LogicFlowDsl.NodeDef batchNode, LogicFlowDsl.NodeDef step,
   826→                                        int iteration, int stepIndex,
   827→                                        Map<String, Object> childVars, List<NodeTrace> traces) {
   828→        String stepId = step.getId() != null && !step.getId().isBlank()
   829→                ? step.getId() : batchNode.getId() + "#b" + stepIndex;
   830→        LogicFlowDsl.NodeDef inner = new LogicFlowDsl.NodeDef();
   831→        inner.setId(stepId);
   832→        inner.setName(step.getName());
   833→        inner.setType(step.getType());
   834→        inner.setConfig(step.getConfig());
   835→        inner.setResultVar(step.getResultVar());
   836→        inner.setOutputs(step.getOutputs());
   837→        inner.setErrorAction(step.getErrorAction());
   838→        boolean traceable = iteration < BATCH_BODY_TRACE_ITERATIONS;
   839→        long begin = System.currentTimeMillis();
   840→        try {
   841→            Object result = dispatch(inner, childVars, traces);
   842→            if (traceable && traces != null) {
   843→                traces.add(new NodeTrace(stepId, inner.getName(), inner.getType().name(),
   844→                        TRACE_SUCCESS, result, null, System.currentTimeMillis() - begin));
   845→            }
   846→            if (inner.getResultVar() != null && !inner.getResultVar().isBlank()) {
   847→                childVars.put(inner.getResultVar(), result);
   848→            }
   849→            if (inner.getType() == NodeType.SCRIPT) {
   850→                expandScriptOutputs(inner, result, childVars);
   851→            }
   852→            return result;
   853→        } catch (Exception e) {
   854→            String error = e.getMessage() != null ? e.getMessage() : e.getClass().getSimpleName();
   855→            if (ACTION_IGNORE_CONTINUE.equalsIgnoreCase(inner.getErrorAction())) {
   856→                if (traces != null) {
   857→                    traces.add(new NodeTrace(stepId, inner.getName(), inner.getType().name(),
   858→                            TRACE_FAILED, null, error, System.currentTimeMillis() - begin));
   859→                }
   860→                log.debug("Batch body step '{}' failed but continues: {}", stepId, error);
   861→                return null;
   862→            }
   863→            throw e;
   864→        }
   865→    }

   867→    /**
   868→     * 解析循环体配置（config.body 数组）为节点定义列表。
   869→     * 每项须为对象且 type ∈ HTTP/BEAN/SCRIPT/DATA_UPDATE/SUBFLOW，否则抛 IllegalArgumentException。
   870→     */
   871→    private List<LogicFlowDsl.NodeDef> parseBatchBody(LogicFlowDsl.NodeDef node, JsonNode bodyNode) {
   872→        List<LogicFlowDsl.NodeDef> steps = new ArrayList<>();
   873→        for (JsonNode entry : bodyNode) {
   874→            if (entry == null || entry.isNull() || !entry.isObject()) {
   875→                continue;
   876→            }
   877→            String typeName = textOrNull(entry, "type");
   878→            NodeType type;
   879→            try {
   880→                type = typeName == null ? null : NodeType.fromJson(typeName);
   881→            } catch (IllegalArgumentException e) {
   882→                type = null;
   883→            }
   884→            if (type == null) {
   885→                throw new IllegalArgumentException(
   886→                        "BATCH 节点 " + node.getId() + " 循环体步骤缺少/非法 type: " + typeName);
   887→            }
   888→            if (!BATCH_BODY_ALLOWED.contains(type)) {
   889→                throw new IllegalArgumentException(
   890→                        "BATCH 节点 " + node.getId() + " 循环体不支持节点类型: " + type);
   891→            }
   892→            LogicFlowDsl.NodeDef step = new LogicFlowDsl.NodeDef();
   893→            step.setId(textOrNull(entry, "id"));
   894→            step.setName(textOrNull(entry, "name"));
   895→            step.setType(type);
   896→            step.setConfig(entry.get("config"));
   897→            step.setResultVar(textOrNull(entry, "resultVar"));
   898→            step.setOutputs(parseOutputsList(entry.get("outputs")));
   899→            step.setErrorAction(textOrNull(entry, "errorAction"));
   900→            steps.add(step);
   901→        }
   902→        return steps;
   903→    }

   905→    /** outputs 数组 → OutputVarDef 列表（null/空安全；忽略无名行）。 */
   906→    private List<LogicFlowDsl.OutputVarDef> parseOutputsList(JsonNode outputsNode) {
   907→        if (outputsNode == null || !outputsNode.isArray() || outputsNode.isEmpty()) {
   908→            return null;
   909→        }
   910→        List<LogicFlowDsl.OutputVarDef> outputs = new ArrayList<>();
   911→        for (JsonNode item : outputsNode) {
   912→            LogicFlowDsl.OutputVarDef def = objectMapper.convertValue(item, LogicFlowDsl.OutputVarDef.class);
   913→            if (def != null && def.getName() != null && !def.getName().isBlank()) {
   914→                outputs.add(def);
   915→            }
   916→        }
   917→        return outputs.isEmpty() ? null : outputs;
   918→    }

   920→    /** maxItems 归一化：缺省 100，钳位 1~1000。 */
   921→    private int normalizeMaxItems(JsonNode raw) {
   922→        int value = raw == null || raw.isNull() ? BATCH_DEFAULT_MAX_ITEMS : raw.asInt(BATCH_DEFAULT_MAX_ITEMS);
   923→        return Math.max(1, Math.min(value, BATCH_HARD_MAX_ITEMS));
   924→    }

   926→    /**
   927→     * 解析批处理集合：{@code {{var}}} 精确形态直接取变量（List/数组/JSON 数组串均可）；
   928→     * 其余先占位符展开再按 JSON 数组字面量解析。解析失败/非集合抛 IllegalArgumentException。
   929→     */
   930→    private List<Object> resolveCollection(String expression, Map<String, Object> vars) {
   931→        String trimmed = expression.trim();
   932→        java.util.regex.Matcher direct = java.util.regex.Pattern
   933→                .compile("^\\{\\{\\s*(\\w+)\\s*}}$").matcher(trimmed);
   934→        if (direct.matches()) {
   935→            Object value = vars.get(direct.group(1));
   936→            if (value == null) {
   937→                throw new IllegalArgumentException("BATCH 集合变量不存在或为 null: " + direct.group(1));
   938→            }
   939→            return toItemList(value, direct.group(1));
   940→        }
   941→        String resolved = variableResolver.resolve(trimmed, vars);
   942→        try {
   943→            JsonNode array = objectMapper.readTree(resolved);
   944→            if (!array.isArray()) {
   945→                throw new IllegalArgumentException("BATCH 集合表达式解析结果不是 JSON 数组: " + trimmed);
   946→            }
   947→            List<Object> items = new ArrayList<>(array.size());
   948→            array.forEach(entry -> items.add(objectMapper.convertValue(entry, Object.class)));
   949→            return items;
   950→        } catch (IllegalArgumentException e) {
   951→            throw e;
   952→        } catch (Exception e) {
   953→            throw new IllegalArgumentException("BATCH 集合表达式不是合法 JSON 数组: " + trimmed);
   954→        }
   955→    }

   957→    /** 变量值 → 遍历列表（List/数组/Collection 原样转换；字符串按 JSON 数组解析）。 */
   958→    private List<Object> toItemList(Object value, String name) {
   959→        if (value instanceof List<?> list) {
   960→            return new ArrayList<>(list);
   961→        }
   962→        if (value instanceof Object[] array) {
   963→            return new ArrayList<>(List.of(array));
   964→        }
   965→        if (value instanceof java.util.Collection<?> collection) {
   966→            return new ArrayList<>(collection);
   967→        }
   968→        if (value instanceof String text) {
   969→            String trimmed = text.trim();
   970→            if (trimmed.startsWith("[")) {
   971→                try {
   972→                    JsonNode array = objectMapper.readTree(trimmed);
   973→                    if (array.isArray()) {
   974→                        List<Object> items = new ArrayList<>(array.size());
   975→                        array.forEach(entry -> items.add(objectMapper.convertValue(entry, Object.class)));
   976→                        return items;
   977→                    }
   978→                } catch (Exception ignored) {
   979→                    // 落入下方统一报错
   980→                }
   981→            }
   982→            throw new IllegalArgumentException("BATCH 集合变量不是数组: " + name);
   983→        }
   984→        throw new IllegalArgumentException("BATCH 集合变量类型不支持（需数组）: " + name);
   985→    }

   992→    private <T> T readConfig(LogicFlowDsl.NodeDef node, Class<T> type) {

  1009→    /** 非 CONDITION 节点的后继：取首条出边；缺出边/边引用缺失节点 → FAILED（经 FlowAbortedException 收敛）。 */
  1010→    private LogicFlowDsl.NodeDef requireNext(Map<String, LogicFlowDsl.NodeDef> nodeById,
  1011→                                             Map<String, List<LogicFlowDsl.EdgeDef>> outEdges,
  1012→                                             String nodeId) {
  1013→        List<LogicFlowDsl.EdgeDef> edges = outEdges.get(nodeId);
  1014→        if (edges == null || edges.isEmpty()) {
  1015→            throw new FlowAbortedException("节点无出边: " + nodeId);
  1016→        }
  1017→        String target = edges.get(0).getTarget();
  1018→        LogicFlowDsl.NodeDef next = target != null ? nodeById.get(target) : null;
  1019→        if (next == null) {
  1020→            throw new FlowAbortedException("边引用不存在的节点: " + target);
  1021→        }
  1022→        return next;
  1023→    }

  1025→    /** CONDITION 节点的后继：按 branch=true/false 选边；缺边 → FAILED。 */
  1026→    private LogicFlowDsl.NodeDef resolveBranchTarget(LogicFlowDsl.NodeDef node,
  1027→                                                     Map<String, LogicFlowDsl.NodeDef> nodeById,
  1028→                                                     List<LogicFlowDsl.EdgeDef> edges,
  1029→                                                     boolean branch) {
  1030→        String want = String.valueOf(branch);
  1031→        if (edges != null) {
  1032→            for (LogicFlowDsl.EdgeDef edge : edges) {
  1033→                if (want.equalsIgnoreCase(edge.getBranch() == null ? "" : edge.getBranch().trim())) {
  1034→                    LogicFlowDsl.NodeDef next = edge.getTarget() != null
  1035→                            ? nodeById.get(edge.getTarget()) : null;
  1036→                    if (next == null) {
  1037→                        throw new FlowAbortedException("边引用不存在的节点: " + edge.getTarget());
  1038→                    }
  1039→                    return next;
  1040→                }
  1041→            }
  1042→        }
  1043→        throw new FlowAbortedException("CONDITION 节点 " + node.getId() + " 缺少 branch=" + want + " 出边");
  1044→    }

  1055→    /** 变量快照：容器结构深拷贝（Map/List 递归复制），标量与不可变对象原样引用。 */
  1056→    private Map<String, Object> snapshot(Map<String, Object> vars) {
  1057→        Map<String, Object> copy = new LinkedHashMap<>();
  1058→        vars.forEach((key, value) -> copy.put(key, deepCopyValue(value)));
  1059→        return copy;
  1060→    }
  1061→
  1062→    private Object deepCopyValue(Object value) {
  1063→        if (value instanceof Map<?, ?> map) {
  1064→            Map<Object, Object> copy = new LinkedHashMap<>();
  1065→            map.forEach((k, v) -> copy.put(k, deepCopyValue(v)));
  1066→            return copy;
  1067→        }
  1068→        if (value instanceof List<?> list) {
  1069→            List<Object> copy = new ArrayList<>(list.size());
  1070→            for (Object item : list) {
  1071→                copy.add(deepCopyValue(item));
  1072→            }
  1073→            return copy;
  1074→        }
  1075→        return value;
  1076→    }
