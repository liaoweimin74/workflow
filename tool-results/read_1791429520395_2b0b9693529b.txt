    12→/** 构造画布节点的便捷工厂 */
    13→function makeNode(
    14→  id: string,
    15→  nodeType: FlowNode['data']['nodeType'],
    16→  x = 0,
    17→  y = 0,
    18→  extra: Partial<FlowNode['data']> = {}
    19→): FlowNode {
    20→  return {
    21→    id,
    22→    type: 'logic',
    23→    position: { x, y },
    24→    data: { nodeType, name: defaultNodeName(nodeType), ...extra },
    25→  }
    26→}
    27→
    28→function makeEdge(id: string, source: string, target: string, branch?: 'true' | 'false'): FlowEdge {
    29→  return { id, source, target, data: branch ? { branch } : undefined }
    30→}
