   297→/** 扁平选项 label 列表（select 类查询：label=value；树/级联/穿梭从 props.data/props.options 取） */
   298→function formOptionLabelItems(rule: any): { label: string; value: string }[] {
   299→  const list: any[] = rule?.options ?? rule?.props?.options ?? rule?.props?.data ?? []
   300→  const labels: string[] = []
   301→  const collect = (items: any[]) => {
   302→    for (const it of items) {
   303→      if (it && it.label !== undefined) labels.push(String(it.label))
   304→      if (Array.isArray(it.children)) collect(it.children)
   305→    }
   306→  }
   307→  collect(list)
   308→  return labels.map((l) => ({ label: l, value: l }))
   309→}

   387→/** JSON 数组 → 逗号拼接；非数组（旧逗号串/字符串）原样返回 */
   388→function formatArrayValue(v: unknown): unknown {
   389→  if (Array.isArray(v)) return v.join(', ')
   390→  if (typeof v === 'string') {
   391→    try {
   392→      const parsed = JSON.parse(v)
   393→      if (Array.isArray(parsed)) return parsed.join(', ')
   394→    } catch {
   395→      // 旧逗号串或普通字符串，原样
   396→    }
   397→    return v
   398→  }
   399→  return v
   400→}

   402→/** ColumnViewConfig → SearchTable TableColumn（formatter 字符串映射为函数） */

   415→      label: c.label || c.key,
   416→      minWidth: c.columnType === 'TEXT' || c.columnType === 'JSON' ? 200 : 120,

   422→              const text = row?.[c.key + '_text']
   423→              if (text !== undefined && text !== null && text !== '') return leafDisplayText(text)
   424→              return formatArrayValue(cellValue) as string
   425→            },
   426→          }
   427→        : {}),

   549→/** visible 表达式求值（$row.xxx → 行值，$param.xxx → 路由参数） */
   550→function evalVisible(expr: string, row: any): boolean {
   551→  try {
   552→    const resolved = expr
   553→      .replace(/\$row\.([\w]+)/g, (_: string, k: string) => {
   554→        const v = getNestedValue(row, k)
   555→        return v === null || v === undefined ? 'undefined' : JSON.stringify(v)
   556→      })
   557→      .replace(/\$param\.([\w]+)/g, (_: string, k: string) => {
   558→        const v = (window as any).__routeParams?.[k]
   559→        return v === null || v === undefined ? 'undefined' : JSON.stringify(v)
   560→      })
   561→    if (/[a-zA-Z_$]\s*\(/.test(resolved)) return false
   562→    return !!Function('"use strict"; return (' + resolved + ')')()
   563→  } catch {
   564→    return true
   565→  }
   566→}

   569→  return path.split('.').reduce((o, k) => o?.[k], obj)

   605→const fetchApi = async (params: { page: number; size: number; [key: string]: any }) => {
   606→  const dsId = resolvedRefId.value
   607→  if (!dsId) return { rows: [], total: 0 }
   608→
   609→  // 组装筛选条件：静态筛选（DsBindingConfigDialog 配置）+ 动作总线 set-filter + 搜索栏条件
   610→  const filterConditions: any[] = []
   611→  // 1. 静态筛选（组件级配置，始终生效；从 tableFilterStore 按页面内 dataSourceId 读取）
   612→  let staticFilter: any = props.dataSourceId ? tableFilterStore[props.dataSourceId] : undefined
   613→  if (staticFilter?.conditions) {
   614→    for (const c of staticFilter.conditions) {
   615→      if (c.column) {
   616→        // DsBindingConfigDialog 输出 fixedValue，兼容 value
   617→        const val = c.fixedValue ?? c.value ?? ''
   618→        filterConditions.push({ column: c.column, op: c.op || 'eq', value: val })
   619→      }
   620→    }
   621→  }
   622→  // 2. 动作总线 set-filter 注入的条件
   623→  if (currentFilter.value) {
   624→    for (const [column, value] of Object.entries(currentFilter.value)) {
   625→      if (value === '' || value === null || value === undefined) continue
   626→      filterConditions.push({ column, op: 'eq', value })
   627→    }
   628→  }
   629→  // 3. 搜索栏条件（数组值组件主列 → 映射 <key>_text 显示列查询 label；级联路径 label 数组 join('/') 匹配 _text 全路径）
   630→  for (const field of resolvedSearchFields.value) {
   631→    const raw = params[field.prop]
   632→    if (raw === '' || raw === null || raw === undefined) continue
   633→    const v = Array.isArray(raw) ? raw.join('/') : raw
   634→    filterConditions.push({ column: resolveSearchColumn(field.prop), op: 'like', value: v })
   635→  }
   636→  const query: Record<string, any> = props.designMode
   637→    // 设计态预览固定取首页且最多 10 条（含"不分页"场景，避免 size:-1 拉全量）
   638→    ? { page: 1, size: Math.min(params.size || 10, 10) }
   639→    : props.pagination === false
   640→      // 不分页：请求全部数据（后端 size<=0 跳过 LIMIT），保留排序
   641→      ? { size: -1 }
   642→          : { page: Math.max(1, params.page), size: params.size }
   643→  // 排序状态透传（SearchTable 内部维护，服务器端排序）
   644→  if (params.sort) query.sort = params.sort
   645→  if (params.order) query.order = params.order
   646→  if (filterConditions.length > 0) {
   647→    query.filter = JSON.stringify({ logic: 'AND', conditions: filterConditions })
   648→  }
   649→  const res: any = await dataSourceApi.queryData(dsId, query)
   650→  const rows = (res?.data?.records || []).map((r: any) => ({ ...(r.data || {}), id: r.id, version: r.version }))
   651→  records.value = rows
   652→  emit('loaded', rows)
   653→  return { rows, total: res?.data?.total || 0 }
   654→}

   675→function handleActionClick(btn: any, row: any) {
   676→  // 有绑定事件 → 优先执行事件链（内建/自定义都一样，对齐 PageRenderer）
   677→  if (btn.events?.length) {
   678→    for (const ev of btn.events) {
   679→      for (const action of ev.actions || []) {
   680→        void dispatchButtonAction(action, row)
   681→      }
   682→    }
   683→    return
   684→  }
   685→  // 表格-容器联动触发器：派发到页面动作总线（动作链消费时跳过默认行为，避免双弹窗）
   686→  const linkageTrigger = LINKAGE_TRIGGERS[btn.key]
   687→  if (linkageTrigger && actionBus) {
   688→    const consumed = actionBus.dispatch(
   689→      linkageTrigger,
   690→      row ? { node: row, row, source: props.dataSourceId } : { source: props.dataSourceId },
   691→    )
   692→    if (consumed) return
   693→  }
   694→  // 无事件 → 默认行为
   695→  // 存在同数据源的数据容器 → 打开容器表单（自动关联，无需动作链）；否则本地默认表单
   696→  const dsId = props.dataSourceId
   697→  const linked = actionBus?.hasLinkedContainer?.(dsId)
   698→  if (btn.key === 'create') {
   699→    if (linked && dsId) actionBus?.openLinkedContainer?.(dsId, 'create', null)
   700→    else openLocalForm('create')
   701→  } else if (btn.key === 'edit') {
   702→    if (linked && dsId) actionBus?.openLinkedContainer?.(dsId, 'edit', row)
   703→    else openLocalForm('edit', row)
   704→  } else if (btn.key === 'delete') {
   705→    handleDelete(row)
   706→  } else if (btn.key === 'view') {
   707→    if (linked && dsId) actionBus?.openLinkedContainer?.(dsId, 'view', row)
   708→    else openDetail(row)
   709→  }
   710→}

   713→/** 模板变量替换：$row.字段（当前行）/ $param.参数（路由参数） */
   714→function resolveTemplate(tpl: string, row: any): string {
   715→  return tpl
   716→    .replace(/\$row\.([\w]+)/g, (_: string, k: string) => {
   717→      const v = row?.[k]
   718→      return v === null || v === undefined ? '' : String(v)
   719→    })
   720→    .replace(/\$param\.([\w]+)/g, (_: string, k: string) => (route.query?.[k] == null ? '' : String(route.query[k])))
   721→}

   723→/** 解析动作参数 [{key,value}] → {key: 模板替换后的值} */
   724→function resolveParams(params: { key: string; value: string }[], row: any): Record<string, any> {
   725→  const out: Record<string, any> = {}
   726→  for (const p of params || []) {
   727→    out[p.key] = resolveTemplate(p.value, row)
   728→  }
   729→  return out
   730→}

   744→/** 执行单个事件链动作（ViewDesigner 风格：type/params） */
   745→async function dispatchButtonAction(action: { type: string; params?: { key: string; value: string }[] }, row: any) {
   746→  const resolved = resolveParams(action.params || [], row)
   747→  switch (action.type) {
   748→    case 'open-detail':
   749→      openDetail(row)
   750→      break
   751→    case 'open-link':
   752→      if (resolved.url) router.push(resolved.url)
   753→      break
   754→    case 'open-create':
   755→      openLocalForm('create')
   756→      break
   757→    case 'edit':
   758→      if (row) openLocalForm('edit', row)
   759→      break
   760→    case 'delete':
   761→      if (row) await handleDelete(row)
   762→      break
   763→    case 'refresh':
   764→      tableRef.value?.fetchList()
   765→      break
   766→    case 'export':
   767→      exportData()
   768→      break
   769→    case 'message':
   770→      ElMessage({
   771→        message: resolved.text || resolved.message || '提示',
   772→        type: (resolved.type as any) || 'info',
   773→      })
   774→      break
   775→    case 'set-filter':
   776→      setFilter(resolved)
   777→      break
   778→    case 'set-page': {
   779→      const page = parseInt(resolved.page, 10)
   780→      if (!isNaN(page) && page > 0) {
   781→        tableRef.value?.setQuery({ page }, false)
   782→      }
   783→      break
   784→    }
   785→    case 'set-sort':
   786→      tableRef.value?.sort(resolved.field || resolved.prop || '', resolved.order || 'ascending')
   787→      break
   788→    case 'clear-selection':
   789→      tableRef.value?.clearSelection()
   790→      break
   791→    case 'script': {
   792→      const source: string = resolved.source || ''
   793→      if (!source) {
   794→        console.warn('[page-table] 脚本动作缺少 source 参数')
   795→        break
   796→      }
   797→      if (!isScriptEventEnabled()) {
   798→        console.warn('[page-table] 脚本事件未启用（设置 VITE_PAGE_SCRIPT_ENABLED=true 开启）')
   799→        break
   800→      }
   801→      await executeScript(source, {
   802→        row,
   803→        params: route.query || {},
   804→        selectedRows: row ? [row] : [],
   805→        ds: {
   806→          query: (filter?: Record<string, any>) => {
   807→            if (filter) setFilter(filter)
   808→            else refresh()
   809→          },
   810→          detail: (id: string) => (resolvedRefId.value ? dataSourceApi.getData(resolvedRefId.value, id) : Promise.reject(new Error('数据源未绑定'))),
   811→          create: (data: Record<string, unknown>) =>
   812→            resolvedRefId.value ? dataSourceApi.createData(resolvedRefId.value, data) : Promise.reject(new Error('数据源未绑定')),
   813→          update: (id: string, data: Record<string, unknown>) =>
   814→            resolvedRefId.value ? dataSourceApi.updateData(resolvedRefId.value, id, data, row?.version) : Promise.reject(new Error('数据源未绑定')),
   815→          remove: (id: string) =>
   816→            resolvedRefId.value ? dataSourceApi.deleteData(resolvedRefId.value, id) : Promise.reject(new Error('数据源未绑定')),
   817→        },
   818→        api: { pageKey: props.pageKey || '', dataSourceId: resolvedRefId.value || '' },
   819→        actions: {
   820→          refresh: () => refresh(),
   821→          openDetail: () => openDetail(row),
   822→          openCreate: () => openLocalForm('create'),
   823→          openEdit: () => (row ? openLocalForm('edit', row) : undefined),
   824→          remove: (id: string) =>
   825→            resolvedRefId.value ? dataSourceApi.deleteData(resolvedRefId.value, id) : Promise.reject(new Error('数据源未绑定')),
   826→        },
   827→        $: { message: (msg: string, type = 'info') => ElMessage({ message: msg, type: type as any }) },
   828→      })
   829→      break
   830→    }
   831→    default:
   832→      console.warn('[page-table] 未知动作类型:', action.type)
   833→  }
   834→}

   935→// 数据源 refId 解析/变更：重新加载元数据；
   936→// 仅「已解析 → 另一 refId」的真实换绑才置 useMetadataColumns（旧列配置失效），
   937→// 首次异步解析（'' → id）不置位，保证用户已保存的列顺序/宽度等配置生效
   938→// （设计态额外补发取数：配置数据源确定/切换后立即渲染表格；运行态由 _pendingFirstFetch 保证首次请求单次触发，不重复取数）
   939→watch(() => resolvedRefId.value, (val, oldVal) => {
   940→  if (oldVal && val && oldVal !== val) {
   941→    useMetadataColumns.value = true
   942→  }
   943→  loadMetadata()
   944→  if (props.designMode) nextTick(() => { tableRef.value?.fetchList() })
   945→})

   949→watch(() => props.columns, (cols) => {
   950→  if (!cols || cols.length === 0) return
   951→  const metaKeys = new Set(metaColumns.value.map((c) => c.key))
   952→  if (metaColumns.value.length > 0 && !cols.some((c: any) => metaKeys.has(c.key ?? c.prop))) return
   953→  useMetadataColumns.value = false
   954→})
