   173→function runBackendLogic(inst: InstRow, ir: DiagramIR, deploy: DeployRow, nodeId: string, trigger: "ENTER" | "COMPLETE", ctx: Record<string, UelValue>): void {
   174→  const cfg = configMapOf(deploy)[nodeId];
   175→  const items = cfg?.backendLogic ?? [];
   176→  const procCfg = ir.processConfig;
   177→  // __PROCESS__ 级 backendLogic 目前设计器不存在，预留
   178→  for (const item of items) {
   179→    if (!item.enabled || item.trigger !== trigger) continue;
   180→    try {
   181→      if (item.type === "http" && item.http) {
   182→        // {{var}} 占位替换（对齐 VariableResolver），执行后 resultVar 回写
   183→        const url = new URL(subst(item.http.url, ctx));
   184→        for (const q of item.http.queryParams ?? []) url.searchParams.set(q.target, String(ctx[q.source] ?? ""));
   185→        const method = (item.http.method ?? "GET").toUpperCase();
   186→        let body: string | undefined;
   187→        if (["POST", "PUT", "DELETE"].includes(method)) {
   188→          const bodyObj: Record<string, unknown> = {};
   189→          for (const b of item.http.bodyParams ?? []) bodyObj[b.target] = ctx[b.source] ?? null;
   190→          body = JSON.stringify(bodyObj);
   191→        }
   192→        // 同步 fetch 在 bun 下可 await——引擎为同步流，采用 fire-and-forget + TODO：重试/超时对齐后续补
   193→        void fetch(url, { method, headers: item.http.headers, body }).then(async (r) => {
   194→          const text = await r.text();
   195→          if (item.resultVar) setVar(inst.ID, item.resultVar, text);
   196→        }).catch(() => {
   197→          if (item.errorAction === "FAIL_FLOW") console.error(`[engine] backendLogic FAIL_FLOW http ${url}`);
   198→        });
   199→      } else if (item.type === "bean") {
   200→        // TODO: 白名单反射调用对齐 @BackendLogicBean；当前记录日志
   201→        console.log(`[engine] backendLogic bean ${item.bean?.beanName}.${item.bean?.methodName} @ ${nodeId}`);
   202→        if (item.resultVar) setVar(inst.ID, item.resultVar, null);
   203→      } else if (item.type === "script") {
   204→        // TODO: groovy→js 转译执行；当前记录日志
   205→        console.log(`[engine] backendLogic script(${item.script?.language ?? "groovy"}) @ ${nodeId}`);
   206→        if (item.resultVar) setVar(inst.ID, item.resultVar, null);
   207→      }
   208→    } catch (e) {
   209→      if (item.errorAction === "FAIL_FLOW") throw new EngineException(`后置逻辑执行失败: ${(e as Error).message}`);
   210→      console.error("[engine] backendLogic error(IGNORE_CONTINUE):", (e as Error).message);
   211→    }
   212→  }
   213→  void procCfg;
   214→}

   216→function subst(tpl: string, ctx: Record<string, UelValue>): string {
   217→  return tpl.replace(/\{\{\s*([\w$]+)\s*\}\}/g, (_m, name: string) => {
   218→    const v = ctx[name];
   219→    return v === null || v === undefined ? "" : String(v);
   220→  });
   221→}

   229→interface Assignment { assignee: string | null; candidates: string[]; mi: { userId: string; index: number }[]; }

   231→/**
   232→ * dept_head 主管解析（Task 16，Node 版增强——Java 从未实现）：
   233→ * 发起人所在组织主管 → 无主管沿父组织逐级向上找（最多 10 层）。
   234→ * 返回用户 ID 字符串（SYS_* 主键 Long→String 语义）。
   235→ */
   236→export function resolveDeptHead(userId: string): string | null {
   237→  const u = queryOne<{ ORG_ID: string | null }>("SELECT ORG_ID FROM SYS_USER WHERE ID = ? AND IS_DELETED = 0", [String(userId)]);
   238→  let orgId = u?.ORG_ID != null ? String(u.ORG_ID) : null;
   239→  for (let hops = 0; orgId && hops < 10; hops++) {
   240→    const org = queryOne<{ LEADER_ID: string | null; PARENT_ID: string | null }>(
   241→      "SELECT LEADER_ID, PARENT_ID FROM SYS_ORGANIZATION WHERE ID = ? AND IS_DELETED = 0", [orgId]);
   242→    if (!org) return null;
   243→    if (org.LEADER_ID) return String(org.LEADER_ID);
   244→    orgId = org.PARENT_ID != null ? String(org.PARENT_ID) : null;
   245→  }
   246→  return null;
   247→}

   249→function resolveAssignment(inst: InstRow, deploy: DeployRow, node: NodeIR): Assignment {
   250→  const vars = getVariables(inst);
   251→  const cfg = configMapOf(deploy)[node.id];
   252→  const approval = cfg?.approval;
   253→  const ctx = evalCtx(inst);
   254→
   255→  // 1) 配置 multiMode（会签/或签/依次）→ MI：approverList = approval.userIds
   256→  if (approval?.multiMode && String(approval.multiMode) !== "" && approval.userIds?.length) {
   257→    return { assignee: null, candidates: [], mi: approval.userIds.map((u, i) => ({ userId: String(u), index: i + 1 })) };
   258→  }
   259→  // 1.5) 配置 type=dept_head：发起人所在部门主管（逐级向上；Node 版增强，见 resolveDeptHead）
   260→  if (approval?.type === "dept_head") {
   261→    const starterId = String((vars["initiator"] as string | null) ?? inst.START_USER ?? "");
   262→    const head = starterId ? resolveDeptHead(starterId) : null;
   263→    if (head) return { assignee: head, candidates: [], mi: [] };
   264→    // 找不到主管：宽松落空任务，由管理员在管理端认领（与 expression 失败同策略）
   265→  }
   266→  // 2) 配置 userIds：单人 assignee / 多人候选（对齐 Java：设计器老配置无 type 字段，仅凭 userIds 分配）
   267→  if (approval?.userIds?.length) {
   268→    const ids = approval.userIds.map(String);
   269→    if (ids.length === 1) return { assignee: ids[0], candidates: [], mi: [] };
   270→    return { assignee: null, candidates: ids, mi: [] };
   271→  }
   272→  // 2.5) 配置 expression：${var} 变量表达式求值出审批人（设计器"表达式"分配；Java 未实现，此为 Node 版增强）
   273→  const expr = (approval?.expression ?? "") as string;
   274→  if (expr && expr.trim()) {
   275→    const resolved = resolveVarRef(expr.trim(), ctx);
   276→    if (resolved) return { assignee: resolved, candidates: [], mi: [] };
   277→    // 表达式求值失败：宽松落空（与 XML assignee 变量缺失一致），由管理员在管理端认领
   278→  }
   279→  // 3) XML 属性：assignee（变量或固定 ID）
   280→  if (node.assignee) {
   281→    const resolved = resolveVarRef(node.assignee, ctx);
   282→    if (resolved !== null) return { assignee: resolved, candidates: [], mi: [] };
   283→    // ${manager} 变量缺失：Flowable 会以异常失败；这里宽松落到候选空任务（TODO 对齐 FAIL）
   284→    return { assignee: null, candidates: [], mi: [] };
   285→  }
   286→  // 4) XML candidateUsers
   287→  if (node.candidateUsers?.length) {
   288→    const ids = node.candidateUsers.map((u) => resolveVarRef(u, ctx) ?? u);
   289→    if (ids.length === 1) return { assignee: ids[0], candidates: [], mi: [] };
   290→    return { assignee: null, candidates: ids, mi: [] };
   291→  }
   292→  void vars;
   293→  return { assignee: null, candidates: [], mi: [] };
   294→}

   301→function createTask(inst: InstRow, node: NodeIR, a: { assignee?: string | null; candidates?: string[]; miGroupId?: string | null; miIndex?: number | null; miTotal?: number | null }): TaskRow {
   302→  const deploy = inst.DEPLOY_ID ? loadDeployById(inst.DEPLOY_ID)! : loadLatestDeployByKey(inst.PROC_KEY)!;
   303→  const ir = getIR(deploy);
   304→  const id = uuid32();
   305→  const now = nowStr();
   306→  const candidates = a.candidates ?? [];
   307→  exec(
   308→    `INSERT INTO WF_TASK_INST (ID, PROC_INST_ID, PROC_KEY, NODE_KEY, NODE_NAME, ASSIGNEE, OWNER, CANDIDATES_JSON, STATUS, PRIORITY, START_TIME, FORM_DEF_ID, MI_GROUP_ID, MI_INDEX, MI_TOTAL, TENANT_ID)
   309→     VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)`,
   310→    [id, inst.ID, inst.PROC_KEY, node.id, node.name ?? node.id, a.assignee ?? null, null,
   311→      candidates.length ? JSON.stringify(candidates) : null, "PENDING", null, now,
   312→      taskFormDefId(deploy, ir, node.id), a.miGroupId ?? null, a.miIndex ?? null, a.miTotal ?? null, inst.TENANT_ID ?? "default"],
   313→  );
   314→  for (const uid of candidates) {
   315→    exec("INSERT INTO WF_TASK_CANDIDATE (ID, TASK_ID, PROC_INST_ID, USER_ID, CREATED_AT) VALUES (?,?,?,?,?)",
   316→      [uuid32(), id, inst.ID, uid, now]);
   317→  }
   318→  const task = queryOne<TaskRow>("SELECT * FROM WF_TASK_INST WHERE ID = ?", [id])!;
   319→  recordActivity(inst, ir, node.id, "RUNNING", { assignee: a.assignee ?? null, miGroupId: a.miGroupId ?? null });
   320→  return task;
   321→}

   335→function finishInstance(inst: InstRow, ir: DiagramIR, deploy: DeployRow): void {
   336→  const remaining = queryOne<{ C: number }>("SELECT COUNT(*) C FROM WF_EXEC_TOKEN WHERE PROC_INST_ID = ? AND STATUS = 'ACTIVE'", [inst.ID]);
   337→  if ((remaining?.C ?? 0) > 0) return;
   338→  const now = nowStr();
   339→  exec("UPDATE WF_PROC_INST SET STATUS = 'COMPLETED', END_TIME = ?, CURRENT_NODE_KEYS = NULL WHERE ID = ?", [now, inst.ID]);
   340→  // endEvent 节点 COMPLETE 后置逻辑（对齐 ACTIVITY_COMPLETED(endEvent)）
   341→  const endNodes = ir.process.nodes.filter((n) => n.type === "endEvent");
   342→  for (const n of endNodes) runBackendLogic(inst, ir, deploy, n.id, "COMPLETE", evalCtx(inst));
   343→  // callActivity 语义（Task 16）：子实例完结 → outParams 回写父实例 + 父令牌离开调用活动继续推进
   344→  if (inst.PARENT_ID) resumeParentAfterChild(getInst(inst.ID));
   345→}

   362→ * - 变量：initiator 继承父 START_USER，再按 inParams（source=父变量 → target=子变量）映射；
   363→ * - START_USER/BUSINESS_KEY/TENANT 继承父实例，PARENT_ID/PARENT_NODE_KEY 回链；
   364→ * - 子实例在父事务内同步推进（泵为重入安全：外层循环每令牌重读最新状态）。
   365→ */
   366→function createChildInstance(parent: InstRow, parentDeploy: DeployRow, node: NodeIR, calledElement: string): InstRow {
   367→  const childDeploy = loadLatestDeployByKey(calledElement);
   368→  if (!childDeploy) throw new EngineException(`子流程定义未部署: ${calledElement}`);
   369→  const childIr = getIR(childDeploy);
   370→  const startNode = childIr.process.nodes.find((n) => n.type === "startEvent");
   371→  if (!startNode) throw new EngineException(`子流程 ${calledElement} 缺少开始事件`);
   372→
   373→  const inParams = configMapOf(parentDeploy)[node.id]?.callActivity?.inParams ?? [];
   374→  const parentVars = getVariables(parent);
   375→  const childVars: Record<string, UelValue> = { initiator: parent.START_USER };
   376→  for (const m of inParams) {
   377→    if (m.source && m.target && m.source in parentVars) childVars[m.target] = parentVars[m.source];
   378→  }
   379→
   380→  const childId = uuid32();
   381→  const now = nowStr();
   382→  exec(
   383→    `INSERT INTO WF_PROC_INST (ID, PROC_KEY, PROC_NAME, PROC_VERSION, DRAFT_ID, DEPLOY_ID, BUSINESS_KEY, TITLE, START_USER, START_TIME, STATUS, TENANT_ID, VARIABLES_JSON, PARENT_ID, PARENT_NODE_KEY)
   384→     VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)`,
   385→    [childId, childDeploy.PROC_KEY, childDeploy.PROC_NAME, childDeploy.VERSION, childDeploy.DRAFT_ID, childDeploy.ID,
   386→      parent.BUSINESS_KEY, `${parent.TITLE ?? parent.PROC_NAME ?? parent.PROC_KEY} / 子流程:${childDeploy.PROC_NAME ?? calledElement}`,
   387→      parent.START_USER, now, "RUNNING", parent.TENANT_ID ?? "default", JSON.stringify(childVars), parent.ID, node.id],
   388→  );
   389→  exec("INSERT INTO WF_EXEC_TOKEN (ID, PROC_INST_ID, NODE_KEY, STATUS, CREATED_AT) VALUES (?,?,?,?,?)",
   390→    [uuid32(), childId, startNode.id, "ACTIVE", now]);
   391→  // 父图：调用活动进入 RUNNING（完结由 resumeParentAfterChild 收尾）
   392→  recordActivity(parent, getIR(parentDeploy), node.id, "RUNNING", {});
   393→
   394→  const childInst = getInst(childId);
   395→  recordActivity(childInst, childIr, startNode.id, "COMPLETED", { endTime: now });
   396→  runBackendLogic(childInst, childIr, childDeploy, startNode.id, "ENTER", evalCtx(childInst));
   397→  pump(childInst, childDeploy);
   398→  return getInst(childId);
   399→}

   401→/**
   402→ * 子实例完结 → 恢复父实例：outParams 回写 → 调用活动轨迹完结 → 令牌沿默认出线离开并泵父实例。
   403→ * 幂等：令牌已被推进（同步完成场景）则无操作。
   404→ */
   405→function resumeParentAfterChild(child: InstRow): void {
   406→  if (!child.PARENT_ID || !child.PARENT_NODE_KEY) return;
   407→  const parent = queryOne<InstRow>("SELECT * FROM WF_PROC_INST WHERE ID = ?", [child.PARENT_ID]);
   408→  if (!parent || parent.STATUS !== "RUNNING") return;
   409→  const parentDeploy = parent.DEPLOY_ID ? loadDeployById(parent.DEPLOY_ID) : loadLatestDeployByKey(parent.PROC_KEY);
   410→  if (!parentDeploy) return;
   411→  const parentIr = getIR(parentDeploy);
   412→  const parentNodeKey = child.PARENT_NODE_KEY;
   413→
   414→  // 1) outParams：子变量 source → 父变量 target（后续网关条件可依赖）
   415→  const outParams = configMapOf(parentDeploy)[parentNodeKey]?.callActivity?.outParams ?? [];
   416→  if (outParams.length) {
   417→    const childVars = getVariables(child);
   418→    const parentVars = getVariables(parent);
   419→    for (const m of outParams) {
   420→      if (m.source && m.target && m.source in childVars) parentVars[m.target] = childVars[m.source];
   421→    }
   422→    saveVariables(parent.ID, parentVars);
   423→  }
   424→  // 2) 调用活动轨迹完结
   425→  completeRunningActivities(parent.ID, parentNodeKey);
   426→  // 3) 令牌离开调用活动
   427→  const token = queryOne<TokenRow>(
   428→    "SELECT * FROM WF_EXEC_TOKEN WHERE PROC_INST_ID = ? AND NODE_KEY = ? AND STATUS = 'ACTIVE' ORDER BY CREATED_AT LIMIT 1",
   429→    [parent.ID, parentNodeKey]);
   430→  if (token) {
   431→    const f = defaultOut(parentIr, parentNodeKey);
   432→    if (f) {
   433→      // outParams 已回写，重载父实例行再推进（避免后续网关/分配用旧变量快照）
   434→      const freshParent = getInst(parent.ID);
   435→      moveToken(freshParent, parentIr, token, f);
   436→      pump(freshParent, parentDeploy);
   437→    }
   438→  }
   439→}

   442→function runningChildIds(instId: string): string[] {
   443→  return query<InstRow>("SELECT ID FROM WF_PROC_INST WHERE PARENT_ID = ? AND STATUS = 'RUNNING'", [instId]).map((r) => r.ID);
   444→}

   446→/** 深度收集全部运行中后代实例（级联终止/挂起用，防环） */
   447→export function runningDescendantIds(instId: string): string[] {
   448→  const out: string[] = [];
   449→  const seen = new Set<string>([instId]);
   450→  const queue = [instId];
   451→  while (queue.length) {
   452→    const cur = queue.shift()!;
   453→    for (const c of runningChildIds(cur)) {
   454→      if (seen.has(c)) continue;
   455→      seen.add(c);
   456→      out.push(c);
   457→      queue.push(c);
   458→    }
   459→  }
   460→  return out;
   461→}

   463→/**
   464→ * 令牌泵：循环处理所有可推进的 ACTIVE 令牌，直到全部等待（userTask）或实例结束。
   465→ * 每步处理一个令牌的一个原子动作，避免重入。
   466→ */
   467→function pump(inst: InstRow, deploy: DeployRow): void {
   468→  const ir = getIR(deploy);
   469→  let guard = 0;
   470→  for (;;) {
   471→    if (++guard > 10000) throw new EngineException("推进步数超限（疑似死循环图）");
   472→    const tokens = activeTokens(inst.ID);
   473→    if (!tokens.length) { finishInstance(inst, ir, deploy); return; }
   474→    let progressed = false;
   475→    let finished = false;
   476→    for (const snap of tokens) {
   477→      // 重入安全：callActivity 同步完成子实例时会重入泵推进本令牌，
   478→      // 快照可能已过期——重读最新状态，非 ACTIVE / 节点已变则跳过，避免双重推进。
   479→      const fresh = queryOne<TokenRow>("SELECT * FROM WF_EXEC_TOKEN WHERE ID = ?", [snap.ID]);
   480→      if (!fresh || fresh.STATUS !== "ACTIVE" || fresh.NODE_KEY !== snap.NODE_KEY) continue;
   481→      const token = fresh;
   482→      const node = ir.nodeById[token.NODE_KEY];
   483→      if (!node) { setTokenStatus(token.ID, "DONE"); progressed = true; continue; }
   484→      switch (node.type) {
   485→        case "startEvent": {
   486→          completeRunningActivities(inst.ID, node.id);
   487→          const f = defaultOut(ir, node.id);
   488→          if (!f) throw new EngineException(`开始事件 ${node.id} 无出线`);
   489→          moveToken(inst, ir, token, f);
   490→          progressed = true;
   491→          break;
   492→        }
   493→        case "endEvent": {
   494→          setTokenStatus(token.ID, "DONE");
   495→          completeRunningActivities(inst.ID, node.id);
   496→          progressed = true;
   497→          finished = true;
   498→          break;
   499→        }
   500→        case "userTask": {
   501→          if (enterUserTask(inst, deploy, ir, token, node)) progressed = true;
   502→          break;
   503→        }
   504→        case "serviceTask": {
   505→          completeRunningActivities(inst.ID, node.id);
   506→          runBackendLogic(inst, ir, deploy, node.id, "ENTER", evalCtx(inst));
   507→          runBackendLogic(inst, ir, deploy, node.id, "COMPLETE", evalCtx(inst));
   508→          const f = defaultOut(ir, node.id);
   509→          if (!f) throw new EngineException(`服务任务 ${node.id} 无出线`);
   510→          moveToken(inst, ir, token, f);
   511→          progressed = true;
   512→          break;
   513→        }
   514→        case "exclusiveGateway": {
   515→          const ctx = evalCtx(inst);
   516→          const outs = ir.outgoing[node.id] ?? [];
   517→          let chosen: FlowIR | null = null;
   518→          for (const f of outs) {
   519→            if (!f.condition) { if (!chosen) chosen = f; continue; }
   520→            if (evalExpression(f.condition, ctx) === true) { chosen = f; break; }
   521→          }
   522→          if (!chosen) throw new EngineException(`排他网关 ${node.id} 无满足条件的出线`);
   523→          completeRunningActivities(inst.ID, node.id);
   524→          moveToken(inst, ir, token, chosen);
   525→          progressed = true;
   526→          break;
   527→        }
   528→        case "parallelGateway":
   529→        case "inclusiveGateway": {
   530→          const outs = ir.outgoing[node.id] ?? [];
   531→          const ins = ir.incomingCount[node.id] ?? 1;
   532→          const atNode = activeTokens(inst.ID).filter((t) => t.NODE_KEY === node.id);
   533→          if (ins > 1 && atNode.length < ins) break; // 汇合等待
   534→          // 汇合完成：消耗全部到达令牌
   535→          for (const t of atNode) setTokenStatus(t.ID, "DONE");
   536→          completeRunningActivities(inst.ID, node.id);
   537→          let targets: FlowIR[];
   538→          if (node.type === "parallelGateway") targets = outs;
   539→          else {
   540→            const ctx = evalCtx(inst);
   541→            targets = outs.filter((f) => !f.condition || evalExpression(f.condition, ctx) === true);
   542→            if (!targets.length) targets = outs.filter((f) => !f.condition);
   543→            if (!targets.length) throw new EngineException(`包含网关 ${node.id} 无可用出线`);
   544→          }
   545→          if (!targets.length) throw new EngineException(`并行网关 ${node.id} 无出线`);
   546→          for (const f of targets) {
   547→            exec("INSERT INTO WF_EXEC_TOKEN (ID, PROC_INST_ID, NODE_KEY, STATUS, CREATED_AT) VALUES (?,?,?,?,?)",
   548→              [uuid32(), inst.ID, f.targetRef, "ACTIVE", nowStr()]);
   549→          }
   550→          progressed = true;
   551→          break;
   552→        }
   553→        case "callActivity": {
   554→          const called = calledElementOf(deploy, ir, node.id);
   555→          if (!called) throw new EngineException(`调用活动 ${node.id} 未配置子流程（calledElement）`);
   556→          const child = queryOne<InstRow>(
   557→            "SELECT * FROM WF_PROC_INST WHERE PARENT_ID = ? AND PARENT_NODE_KEY = ? ORDER BY START_TIME DESC LIMIT 1",
   558→            [inst.ID, node.id]);
   559→          if (child && child.STATUS === "RUNNING") {
   560→            // 等待子流程：令牌停在调用活动，本轮不推进
   561→            break;
   562→          }
   563→          if (!child) {
   564→            // 首次到达：建子实例并在父事务内同步推进（子流程若一路直达终点，
   565→            // finishInstance → resumeParentAfterChild 会重入泵把本令牌推离，外层快照检查会跳过）
   566→            createChildInstance(inst, deploy, node, called);
   567→            progressed = true;
   568→          } else {
   569→            // 兜底：子实例已完结但令牌仍在此（finishInstance 恢复链路中断的补挂）
   570→            resumeParentAfterChild(getInst(child.ID));
   571→            progressed = true;
   572→          }
   573→          break;
   574→        }
   575→        case "subProcess": {
   576→          // TODO: 嵌套图展开；当前按透传记录（遗留缺口）
   577→          completeRunningActivities(inst.ID, node.id);
   578→          const f = defaultOut(ir, node.id);
   579→          if (f) { moveToken(inst, ir, token, f); progressed = true; }
   580→          break;
   581→        }
   582→      }
   583→      if (finished) break;
   584→    }
   585→    if (finished) { finishInstance(inst, ir, deploy); return; }
   586→    if (!progressed) return;
   587→  }
   588→}

   590→/** userTask 到达：建任务/展开 MI。返回是否产生推进（供泵判定）。 */
   591→function enterUserTask(inst: InstRow, deploy: DeployRow, ir: DiagramIR, token: TokenRow, node: NodeIR): boolean {
   592→  const existing = queryOne<{ C: number }>(
   593→    "SELECT COUNT(*) C FROM WF_TASK_INST WHERE PROC_INST_ID = ? AND NODE_KEY = ? AND STATUS = 'PENDING' AND MI_GROUP_ID IS ?",
   594→    [inst.ID, node.id, token.MI_GROUP_ID ?? null]);
   595→  if ((existing?.C ?? 0) > 0) return false; // 已在等待
   596→
   597→  if (token.MI_GROUP_ID) return false; // MI 子令牌等待各自任务
   598→
   599→  const assignment = resolveAssignment(inst, deploy, node);
   600→  if (assignment.mi.length > 1 || (assignment.mi.length === 1 && (configMapOf(deploy)[node.id]?.approval?.multiMode ?? "") !== "")) {
   601→    // MI：每审批人一个子令牌 + 一个任务
   602→    const groupId = uuid32();
   603→    const total = assignment.mi.length;
   604→    for (const sub of assignment.mi) {
   605→      const subToken = uuid32();
   606→      exec("INSERT INTO WF_EXEC_TOKEN (ID, PROC_INST_ID, NODE_KEY, STATUS, MI_GROUP_ID, MI_INDEX, CREATED_AT) VALUES (?,?,?,?,?,?,?)",
   607→        [subToken, inst.ID, node.id, "ACTIVE", groupId, sub.index, nowStr()]);
   608→      createTask(inst, node, { assignee: sub.userId, miGroupId: groupId, miIndex: sub.index, miTotal: total });
   609→    }
   610→    setTokenStatus(token.ID, "DONE"); // 组根令牌已展开
   611→    // MI 审批人注入语义（MultiInstanceApproverListener）：approverList + rejected=false
   612→    setVar(inst.ID, "approverList", assignment.mi.map((m) => m.userId));
   613→    if (getVariables(inst)["rejected"] === undefined) setVar(inst.ID, "rejected", false);
   614→    return true;
   615→  }
   616→  createTask(inst, node, { assignee: assignment.assignee, candidates: assignment.candidates });
   617→  return true;
   618→}

   629→/** C.1 启动：按 key 最新版本 → 建实例 → 首节点 → 发起人自动 complete → 推进 */
   630→export function startProcess(input: {
   631→  procKey: string; starter: string; variables?: Record<string, UelValue>; businessKey?: string | null;
   632→  formDefId?: string | null; tenantId?: string | null;
   633→}): StartResult {
   634→  const deploy = loadLatestDeployByKey(input.procKey);
   635→  if (!deploy) throw new BusinessException(`流程定义不存在: ${input.procKey}`, 404);
   636→  const ir = getIR(deploy);
   637→  const startNode = ir.process.nodes.find((n) => n.type === "startEvent");
   638→  if (!startNode) throw new EngineException("缺少开始事件");
   639→  const now = nowStr();
   640→  const vars: Record<string, UelValue> = { initiator: input.starter, ...(input.variables ?? {}) };
   641→  const instId = uuid32();
   642→  const tenant = input.tenantId ?? "default";
   643→
   644→  tx(() => {
   645→    exec(
   646→      `INSERT INTO WF_PROC_INST (ID, PROC_KEY, PROC_NAME, PROC_VERSION, DRAFT_ID, DEPLOY_ID, BUSINESS_KEY, TITLE, START_USER, START_TIME, STATUS, TENANT_ID, VARIABLES_JSON)
   647→       VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?)`,
   648→      [instId, deploy.PROC_KEY, deploy.PROC_NAME, deploy.VERSION, deploy.DRAFT_ID, deploy.ID,
   649→        input.businessKey ?? null, input.businessKey ?? deploy.PROC_NAME, input.starter, now, "RUNNING", tenant, JSON.stringify(vars)],
   650→    );
   651→    exec("INSERT INTO WF_EXEC_TOKEN (ID, PROC_INST_ID, NODE_KEY, STATUS, CREATED_AT) VALUES (?,?,?,?,?)",
   652→      [uuid32(), instId, startNode.id, "ACTIVE", now]);
   653→    const inst = getInst(instId);
   654→    recordActivity(inst, ir, startNode.id, "COMPLETED", { endTime: now });
   655→    // PROCESS_STARTED → start 节点 ENTER 后置逻辑
   656→    runBackendLogic(inst, ir, deploy, startNode.id, "ENTER", evalCtx(inst));
   657→  });
   658→
   659→  const inst = getInst(instId);
   660→  pump(inst, deploy);
   661→
   662→  // 发起人节点自动 complete：定位 initiator 节点（wf:nodeRole=initiator 兜底第一个 userTask）上 PENDING 任务
   663→  const irAfter = getIR(deploy);
   664→  const initiatorNode =
   665→    irAfter.process.nodes.find((n) => n.type === "userTask" && isInitiatorNodeWith(deploy, n)) ??
   666→    irAfter.process.nodes.find((n) => n.type === "userTask");
   667→  if (initiatorNode) {
   668→    const t = queryOne<TaskRow>(
   669→      "SELECT * FROM WF_TASK_INST WHERE PROC_INST_ID = ? AND NODE_KEY = ? AND STATUS = 'PENDING' ORDER BY START_TIME LIMIT 1",
   670→      [instId, initiatorNode.id]);
   671→    if (t) {
   672→      completeTask({ taskId: t.ID, userId: input.starter, variables: {}, comment: null, action: "submit" });
   673→    }
   674→  }
   675→
   676→  // 表单数据落库（form_data，非快照）——对齐 ProcessInstanceController.start
   677→  if (input.formDefId) {
   678→    try {
   679→      exec(
   680→        `INSERT INTO WF_FORM_DATA (ID, CREATED_AT, CREATED_BY, DATA_JSON, FORM_DEF_ID, FORM_VERSION, IS_SNAPSHOT, PROCESS_INSTANCE_ID, TENANT_ID)
   681→         SELECT ?, ?, ?, ?, ?, COALESCE(PUBLISHED_VERSION, VERSION), 0, ?, ? FROM WF_FORM_DEF WHERE ID = ?`,
   682→        [uuid32(), nowStr(), input.starter, JSON.stringify(vars), input.formDefId, instId, tenant, input.formDefId],
   683→      );
   684→    } catch { /* 表单保存失败不影响流程启动 */ }
   685→  }
   686→  return { id: instId, processDefinitionId: deploy.ID, processDefinitionKey: deploy.PROC_KEY, businessKey: input.businessKey ?? null, tenantId: tenant };
   687→}

   689→export interface CompleteResult { processFinished: boolean; nextTask: { id: string; name: string; assignee: string | null } | null; }

   691→/** C.3 完成任务：变量合并 → MI 计数（all/any + rejected 短路）→ advance */
   692→export function completeTask(input: {
   693→  taskId: string; userId: string; variables?: Record<string, UelValue>; comment?: string | null; action?: string;
   694→}): CompleteResult {
   695→  return tx(() => {
   696→    const task = getTask(input.taskId);
   697→    if (task.STATUS !== "PENDING") throw new BusinessException("任务已完成或已取消", 400);
   698→    const inst = getInst(task.PROC_INST_ID);
   699→    if (inst.STATUS !== "RUNNING") throw new BusinessException("流程实例不在运行中", 400);
   700→    const deploy = inst.DEPLOY_ID ? loadDeployById(inst.DEPLOY_ID)! : loadLatestDeployByKey(inst.PROC_KEY)!;
   701→    const ir = getIR(deploy);
   702→    const node = ir.nodeById[task.NODE_KEY];
   703→    const now = nowStr();
   704→    const action = input.action ?? "approve";
   705→
   706→    // 委派任务：先 resolveTask 语义（owner 收回任务，不完成）
   707→    if (task.DELEGATION === "PENDING") {
   708→      exec("UPDATE WF_TASK_INST SET ASSIGNEE = COALESCE(OWNER, ASSIGNEE), DELEGATION = NULL WHERE ID = ?", [task.ID]);
   709→      addComment(inst, task.ID, input.userId, "resolve", input.comment ?? null, task.OWNER);
   710→      return { processFinished: false, nextTask: nextTaskOf(inst.ID) };
   711→    }
   712→
   713→    // 变量合并
   714→    if (input.variables && Object.keys(input.variables).length) {
   715→      const vars = getVariables(inst);
   716→      Object.assign(vars, input.variables);
   717→      saveVariables(inst.ID, vars);
   718→    }
   719→    exec("UPDATE WF_TASK_INST SET STATUS = 'COMPLETED', END_TIME = ?, COMPLETED_BY = ? WHERE ID = ?",
   720→      [now, input.userId, task.ID]);
   721→    addComment(inst, task.ID, input.userId, action, input.comment ?? null);
   722→    completeRunningActivities(inst.ID, task.NODE_KEY, "COMPLETED", task.MI_GROUP_ID ?? undefined);
   723→    // TASK_COMPLETED → 节点 COMPLETE 后置逻辑
   724→    if (node) runBackendLogic(inst, ir, deploy, node.id, "COMPLETE", evalCtx(getInst(inst.ID)));
   725→
   726→    const token = queryOne<TokenRow>(
   727→      "SELECT * FROM WF_EXEC_TOKEN WHERE PROC_INST_ID = ? AND NODE_KEY = ? AND STATUS = 'ACTIVE' ORDER BY CREATED_AT LIMIT 1",
   728→      [inst.ID, task.NODE_KEY]);
   729→
   730→    if (task.MI_GROUP_ID) {
   731→      // MI 计数：all=全部完成 / any=≥1 完成；rejected=true 任一触发整体终止（C.4）
   732→      const groupTasks = query<TaskRow>("SELECT * FROM WF_TASK_INST WHERE MI_GROUP_ID = ?", [task.MI_GROUP_ID]);
   733→      const pending = groupTasks.filter((t) => t.STATUS === "PENDING");
   734→      const completed = groupTasks.filter((t) => t.STATUS === "COMPLETED");
   735→      const rejected = getVariables(getInst(inst.ID))["rejected"] === true;
   736→      const mode = configMapOf(deploy)[task.NODE_KEY]?.approval?.multiMode ?? "countersign";
   737→      const done = rejected || (mode === "or_sign" ? completed.length >= 1 : pending.length === 0);
   738→      if (!done) {
   739→        // 取消被拒绝短路时其余任务
   740→        if (rejected) {
   741→          for (const p of pending) {
   742→            exec("UPDATE WF_TASK_INST SET STATUS = 'CANCELED', END_TIME = ?, DELETE_REASON = 'rejected' WHERE ID = ?", [now, p.ID]);
   743→            completeRunningActivities(inst.ID, p.NODE_KEY, "CANCELED", p.MI_GROUP_ID ?? undefined);
   744→          }
   745→        }
   746→        return { processFinished: false, nextTask: nextTaskOf(inst.ID) };
   747→      }
   748→      // 完成条件满足：清空组令牌，创建组延续令牌并立即沿默认出线离开（不得重入 userTask）
   749→      exec("UPDATE WF_EXEC_TOKEN SET STATUS = 'DONE' WHERE PROC_INST_ID = ? AND MI_GROUP_ID = ?", [inst.ID, task.MI_GROUP_ID]);
   750→      for (const p of pending) {
   751→        exec("UPDATE WF_TASK_INST SET STATUS = 'CANCELED', END_TIME = ?, DELETE_REASON = 'mi-completed' WHERE ID = ?", [now, p.ID]);
   752→        completeRunningActivities(inst.ID, p.NODE_KEY, "CANCELED", p.MI_GROUP_ID ?? undefined);
   753→      }
   754→      const contId = uuid32();
   755→      exec("INSERT INTO WF_EXEC_TOKEN (ID, PROC_INST_ID, NODE_KEY, STATUS, CREATED_AT) VALUES (?,?,?,?,?)",
   756→        [contId, inst.ID, task.NODE_KEY, "ACTIVE", now]);
   757→      const contToken = queryOne<TokenRow>("SELECT * FROM WF_EXEC_TOKEN WHERE ID = ?", [contId])!;
   758→      const contFlow = defaultOut(ir, task.NODE_KEY);
   759→      if (contFlow) moveToken(getInst(inst.ID), ir, contToken, contFlow);
   760→      pump(getInst(inst.ID), deploy);
   761→      return { processFinished: isFinished(inst.ID), nextTask: nextTaskOf(inst.ID) };
   762→    }
   763→
   764→    // 单实例任务：令牌沿出线推进
   765→    if (token) {
   766→      const f = defaultOut(ir, task.NODE_KEY);
   767→      if (!f) throw new EngineException(`节点 ${task.NODE_KEY} 无出线`);
   768→      moveToken(getInst(inst.ID), ir, token, f);
   769→    }
   770→    pump(getInst(inst.ID), deploy);
   771→    return { processFinished: isFinished(inst.ID), nextTask: nextTaskOf(inst.ID) };
   772→  });
   773→}

   776→  const inst = queryOne<{ STATUS: string }>("SELECT STATUS FROM WF_PROC_INST WHERE ID = ?", [instId]);
   777→  return inst?.STATUS === "COMPLETED" || inst?.STATUS === "CANCELED" || inst?.STATUS === "REJECTED";
   778→}

   793→/** 驳回 = 退回发起人节点（changeActivityState 语义） */
   794→export function rejectTask(taskId: string, userId: string, comment: string | null): CompleteResult {
   795→  return tx(() => {
   796→    const task = getTask(taskId);
   797→    if (task.STATUS !== "PENDING") throw new BusinessException("任务已完成或已取消", 400);
   798→    const inst = getInst(task.PROC_INST_ID);
   799→    if (inst.STATUS !== "RUNNING") throw new BusinessException("流程实例不在运行中", 400);
   800→    const deploy = inst.DEPLOY_ID ? loadDeployById(inst.DEPLOY_ID)! : loadLatestDeployByKey(inst.PROC_KEY)!;
   801→    const ir = getIR(deploy);
   802→    const initiatorNode = resolveInitiatorNode(ir);
   803→    if (!initiatorNode) throw new EngineException("未找到发起人节点，无法驳回");
   804→    if (initiatorNode.id === task.NODE_KEY) throw new BusinessException("当前已在发起节点，不能驳回", 400);
   805→
   806→    const now = nowStr();
   807→    // 设变量 rejected=true（终止 MI）；取消当前令牌
   808→    setVar(inst.ID, "rejected", true);
   809→    exec("UPDATE WF_EXEC_TOKEN SET STATUS = 'CANCELED' WHERE PROC_INST_ID = ?", [inst.ID]);
   810→    exec("UPDATE WF_TASK_INST SET STATUS = 'CANCELED', END_TIME = ?, DELETE_REASON = 'reject' WHERE STATUS = 'PENDING' AND PROC_INST_ID = ?",
   811→      [now, inst.ID]);
   812→    exec("UPDATE WF_ACTIVITY_INST SET END_TIME = ?, STATUS = 'CANCELED' WHERE PROC_INST_ID = ? AND STATUS = 'RUNNING'", [now, inst.ID]);
   813→    // 级联取消运行中子实例（callActivity 链不能孤儿运行）
   814→    for (const cid of runningDescendantIds(inst.ID)) {
   815→      terminateInstanceInner(cid, "reject", "CANCELED");
   816→    }
   817→    addComment(inst, task.ID, userId, "reject", comment ?? null);
   818→
   819→    // changeActivityState：发起人节点新令牌 + 新任务
   820→    const assignee = getVariables(inst)["initiator"] as string | null ?? inst.START_USER;
   821→    exec("INSERT INTO WF_EXEC_TOKEN (ID, PROC_INST_ID, NODE_KEY, STATUS, CREATED_AT) VALUES (?,?,?,?,?)",
   822→      [uuid32(), inst.ID, initiatorNode.id, "ACTIVE", now]);
   823→    const newInst = getInst(inst.ID);
   824→    createTask(newInst, initiatorNode, { assignee });
   825→    exec("UPDATE WF_PROC_INST SET CURRENT_NODE_KEYS = ? WHERE ID = ?", [JSON.stringify([initiatorNode.id]), inst.ID]);
   826→    // variableMappings 写回（发起人重新填报后变量刷新）——TODO dataMappings 聚合
   827→    return { processFinished: false, nextTask: nextTaskOf(inst.ID) };
   828→  });
   829→}

   831→/**
   832→ * 发起人撤回（recall）：把运行中实例拉回发起人节点重新提交。
   833→ * 守卫：仅发起人本人 / 实例运行中 / 发起节点之后的审批节点尚无完成记录（否则审批已实质开始）。
   834→ */
   835→export function recallInstance(procInstId: string, userId: string, comment: string | null): CompleteResult {
   836→  return tx(() => {
   837→    const inst = getInst(procInstId);
   838→    if (inst.STATUS !== "RUNNING") throw new BusinessException("流程实例不在运行中，无法撤回", 400);
   839→    // 发起人取值链：START_USER → 变量 initiator（Flowable 历史数据可能未记录 START_USER）
   840→    const recallVars = getVariables(inst);
   841→    const initiatorId = String(inst.START_USER ?? (recallVars["initiator"] as string | null) ?? "");
   842→    if (!initiatorId) throw new BusinessException("无法确定流程发起人，无法撤回", 400);
   843→    if (initiatorId !== String(userId)) throw new BusinessException("仅流程发起人可以撤回", 403);
   844→    const deploy = inst.DEPLOY_ID ? loadDeployById(inst.DEPLOY_ID)! : loadLatestDeployByKey(inst.PROC_KEY)!;
   845→    const ir = getIR(deploy);
   846→    const initiatorNode = resolveInitiatorNode(ir);
   847→    if (!initiatorNode) throw new EngineException("未找到发起人节点，无法撤回");
   848→    if (initiatorNode.id === (currentNodeKeys(inst.ID)[0] ?? "")) {
   849→      throw new BusinessException("流程仍在发起节点，无需撤回", 400);
   850→    }
   851→    // 守卫：后续审批节点已有完成记录 → 审批已实质开始
   852→    const doneElsewhere = queryOne<{ C: number }>(
   853→      `SELECT COUNT(*) AS C FROM WF_ACTIVITY_INST
   854→       WHERE PROC_INST_ID = ? AND END_TIME IS NOT NULL AND NODE_TYPE = 'userTask' AND NODE_KEY != ?`,
   855→      [inst.ID, initiatorNode.id],
   856→    );
   857→    if ((doneElsewhere?.C ?? 0) > 0) throw new BusinessException("审批已进入后续节点完成，不可撤回", 400);
   858→
   859→    const now = nowStr();
   860→    const pending = query<TaskRow>(`SELECT * FROM WF_TASK_INST WHERE PROC_INST_ID = ? AND STATUS = 'PENDING'`, [inst.ID]);
   861→    setVar(inst.ID, "recalled", true);
   862→    exec("UPDATE WF_EXEC_TOKEN SET STATUS = 'CANCELED' WHERE PROC_INST_ID = ?", [inst.ID]);
   863→    exec("UPDATE WF_TASK_INST SET STATUS = 'CANCELED', END_TIME = ?, DELETE_REASON = 'recall' WHERE STATUS = 'PENDING' AND PROC_INST_ID = ?",
   864→      [now, inst.ID]);
   865→    exec("UPDATE WF_ACTIVITY_INST SET END_TIME = ?, STATUS = 'CANCELED' WHERE PROC_INST_ID = ? AND STATUS = 'RUNNING'", [now, inst.ID]);
   866→    // 级联取消运行中子实例（callActivity 链不能孤儿运行）
   867→    for (const cid of runningDescendantIds(inst.ID)) {
   868→      terminateInstanceInner(cid, "recall", "CANCELED");
   869→    }
   870→    addComment(inst, pending[0]?.ID ?? null, userId, "recall", comment ?? null);
   871→
   872→    // 回到发起人节点重新提交
   873→    const assignee = getVariables(inst)["initiator"] as string | null ?? inst.START_USER;
   874→    exec("INSERT INTO WF_EXEC_TOKEN (ID, PROC_INST_ID, NODE_KEY, STATUS, CREATED_AT) VALUES (?,?,?,?,?)",
   875→      [uuid32(), inst.ID, initiatorNode.id, "ACTIVE", now]);
   876→    const fresh = getInst(inst.ID);
   877→    createTask(fresh, initiatorNode, { assignee });
   878→    exec("UPDATE WF_PROC_INST SET CURRENT_NODE_KEYS = ? WHERE ID = ?", [JSON.stringify([initiatorNode.id]), inst.ID]);
   879→    return { processFinished: false, nextTask: nextTaskOf(inst.ID) };
   880→  });
   881→}

   899→/** 级联终止：实例 + 全部运行中后代子实例（callActivity 链，防环） */
   900→function terminateCascade(instId: string, reason: string, status: string, seen: Set<string>): void {
   901→  if (seen.has(instId)) return;
   902→  seen.add(instId);
   903→  terminateInstanceInner(instId, reason, status);
   904→  for (const c of runningChildIds(instId)) terminateCascade(c, reason, status, seen);
   905→}
   906→
   907→function terminateInstanceInner(instId: string, reason: string, status: string): void {
   908→  const now = nowStr();
   909→  exec("UPDATE WF_TASK_INST SET STATUS = 'CANCELED', END_TIME = ?, DELETE_REASON = ? WHERE PROC_INST_ID = ? AND STATUS = 'PENDING'", [now, reason, instId]);
   910→  exec("UPDATE WF_EXEC_TOKEN SET STATUS = 'CANCELED' WHERE PROC_INST_ID = ? AND STATUS = 'ACTIVE'", [instId]);
   911→  exec("UPDATE WF_ACTIVITY_INST SET END_TIME = ?, STATUS = 'CANCELED' WHERE PROC_INST_ID = ? AND STATUS = 'RUNNING'", [now, instId]);
   912→  exec("UPDATE WF_PROC_INST SET STATUS = ?, END_TIME = ?, DELETE_REASON = ?, CURRENT_NODE_KEYS = NULL WHERE ID = ?",
   913→    [status, now, reason, instId]);
   914→}

   916→export function suspendInstance(instId: string): void {
   917→  // 级联挂起：子实例一并挂起（对齐 Flowable 挂起级联语义）
   918→  for (const id of [instId, ...runningDescendantIds(instId)]) {
   919→    exec("UPDATE WF_PROC_INST SET STATUS = 'SUSPENDED' WHERE ID = ? AND STATUS = 'RUNNING'", [id]);
   920→  }
   921→}

   923→export function resumeInstance(instId: string): void {
   924→  for (const id of [instId, ...runningDescendantIdsSuspended(instId)]) {
   925→    exec("UPDATE WF_PROC_INST SET STATUS = 'RUNNING' WHERE ID = ? AND STATUS = 'SUSPENDED'", [id]);
   926→  }
   927→}

   929→/** 深度收集挂起中的后代实例（恢复用） */
   930→function runningDescendantIdsSuspended(instId: string): string[] {
   931→  const out: string[] = [];
   932→  const seen = new Set<string>([instId]);
   933→  const queue = [instId];
   934→  while (queue.length) {
   935→    const cur = queue.shift()!;
   936→    const kids = query<InstRow>("SELECT ID FROM WF_PROC_INST WHERE PARENT_ID = ? AND STATUS = 'SUSPENDED'", [cur]).map((r) => r.ID);
   937→    for (const c of kids) {
   938→      if (seen.has(c)) continue;
   939→      seen.add(c);
   940→      out.push(c);
   941→      queue.push(c);
   942→    }
   943→  }
   944→  return out;
   945→}

   949→/** 加签：MI=addMultiInstanceExecution（每人一个新子实例，受完成条件约束）；非 MI=addCandidateUser */
   950→export function addSign(taskId: string, userId: string, targetUserIds: string[], comment: string | null): void {
   951→  tx(() => {
   952→    const task = getTask(taskId);
   953→    if (task.STATUS !== "PENDING") throw new BusinessException("任务已完成或已取消", 400);
   954→    const inst = getInst(task.PROC_INST_ID);
   955→    const deploy = inst.DEPLOY_ID ? loadDeployById(inst.DEPLOY_ID)! : loadLatestDeployByKey(inst.PROC_KEY)!;
   956→    const ir = getIR(deploy);
   957→    const node = ir.nodeById[task.NODE_KEY];
   958→    const now = nowStr();
   959→    if (task.MI_GROUP_ID) {
   960→      const total = queryOne<{ C: number }>("SELECT COUNT(*) C FROM WF_TASK_INST WHERE MI_GROUP_ID = ?", [task.MI_GROUP_ID])?.C ?? 0;
   961→      targetUserIds.forEach((uid, i) => {
   962→        const idx = total + i + 1;
   963→        exec("INSERT INTO WF_EXEC_TOKEN (ID, PROC_INST_ID, NODE_KEY, STATUS, MI_GROUP_ID, MI_INDEX, CREATED_AT) VALUES (?,?,?,?,?,?,?)",
   964→          [uuid32(), inst.ID, task.NODE_KEY, "ACTIVE", task.MI_GROUP_ID, idx, now]);
   965→        createTask(inst, node, { assignee: uid, miGroupId: task.MI_GROUP_ID!, miIndex: idx, miTotal: null });
   966→      });
   967→      // 更新组内任务 MI_TOTAL
   968→      const newTotal = total + targetUserIds.length;
   969→      exec("UPDATE WF_TASK_INST SET MI_TOTAL = ? WHERE MI_GROUP_ID = ?", [newTotal, task.MI_GROUP_ID]);
   970→    } else {
   971→      for (const uid of targetUserIds) {
   972→        exec("INSERT INTO WF_TASK_CANDIDATE (ID, TASK_ID, PROC_INST_ID, USER_ID, CREATED_AT) VALUES (?,?,?,?,?)",
   973→          [uuid32(), task.ID, inst.ID, uid, now]);
   974→      }
   975→      const existing: string[] = task.CANDIDATES_JSON ? JSON.parse(task.CANDIDATES_JSON) : [];
   976→      const merged = Array.from(new Set([...existing, ...targetUserIds]));
   977→      exec("UPDATE WF_TASK_INST SET CANDIDATES_JSON = ? WHERE ID = ?", [JSON.stringify(merged), task.ID]);
   978→    }
   979→    addComment(inst, task.ID, userId, "add_sign", comment ?? null, targetUserIds.join(","));
   980→  });
   981→}

   983→/** 转签 forwardSign：MI 实例级换人 = 删旧实例（不计完成）+ 加新实例 */
   984→export function forwardSign(taskId: string, userId: string, fromUserId: string, toUserId: string, comment: string | null): void {
   985→  tx(() => {
   986→    const task = getTask(taskId);
   987→    if (task.STATUS !== "PENDING") throw new BusinessException("任务已完成或已取消", 400);
   988→    const inst = getInst(task.PROC_INST_ID);
   989→    const deploy = inst.DEPLOY_ID ? loadDeployById(inst.DEPLOY_ID)! : loadLatestDeployByKey(inst.PROC_KEY)!;
   990→    const ir = getIR(deploy);
   991→    const node = ir.nodeById[task.NODE_KEY];
   992→    const now = nowStr();
   993→    if (!task.MI_GROUP_ID) throw new BusinessException("转签仅支持多实例（会签/或签）任务", 400);
   994→    // 删旧实例（不计完成数）：任务 CANCELED + 子令牌 DONE
   995→    const oldTasks = query<TaskRow>("SELECT * FROM WF_TASK_INST WHERE MI_GROUP_ID = ? AND ASSIGNEE = ? AND STATUS = 'PENDING'", [task.MI_GROUP_ID, fromUserId]);
   996→    for (const t of oldTasks) {
   997→      exec("UPDATE WF_TASK_INST SET STATUS = 'CANCELED', END_TIME = ?, DELETE_REASON = 'forward_sign' WHERE ID = ?", [now, t.ID]);
   998→      exec("UPDATE WF_EXEC_TOKEN SET STATUS = 'DONE' WHERE PROC_INST_ID = ? AND NODE_KEY = ? AND MI_GROUP_ID = ? AND STATUS = 'ACTIVE' AND MI_INDEX = ?",
   999→        [inst.ID, t.NODE_KEY, t.MI_GROUP_ID, t.MI_INDEX]);
  1000→      completeRunningActivities(inst.ID, t.NODE_KEY, "CANCELED", t.MI_GROUP_ID ?? undefined);
  1001→    }
  1002→    // 加新实例
  1003→    const total = queryOne<{ C: number }>("SELECT COUNT(*) C FROM WF_TASK_INST WHERE MI_GROUP_ID = ?", [task.MI_GROUP_ID])?.C ?? 0;
  1004→    exec("INSERT INTO WF_EXEC_TOKEN (ID, PROC_INST_ID, NODE_KEY, STATUS, MI_GROUP_ID, MI_INDEX, CREATED_AT) VALUES (?,?,?,?,?,?,?)",
  1005→      [uuid32(), inst.ID, task.NODE_KEY, "ACTIVE", task.MI_GROUP_ID, total + 1, now]);
  1006→    createTask(inst, node, { assignee: toUserId, miGroupId: task.MI_GROUP_ID, miIndex: total + 1, miTotal: null });
  1007→    addComment(inst, task.ID, userId, "forward_sign", comment ?? null, toUserId);
  1008→  });
  1009→}

  1011→/** 转办 transfer：setAssignee（唯一强制权限校验 allowTransfer） */
  1012→export function transferTask(taskId: string, userId: string, toUser: string, reason: string | null): void {
  1013→  tx(() => {
  1014→    const task = getTask(taskId);
  1015→    if (task.STATUS !== "PENDING") throw new BusinessException("任务已完成或已取消", 400);
  1016→    if (String(toUser) === String(task.ASSIGNEE)) throw new BusinessException("转办目标与当前办理人相同", 400);
  1017→    const inst = getInst(task.PROC_INST_ID);
  1018→    const deploy = inst.DEPLOY_ID ? loadDeployById(inst.DEPLOY_ID)! : loadLatestDeployByKey(inst.PROC_KEY)!;
  1019→    const ir = getIR(deploy);
  1020→    const node = ir.nodeById[task.NODE_KEY];
  1021→    const cfg = node ? configMapOf(deploy)[node.id] : null;
  1022→    // extractOperations：流程级 AND 节点级（仅 transfer 强制校验，对齐 Java 现状）
  1023→    const procAllow = ir.processConfig?.approvalPolicy?.operations?.allowTransfer ?? true;
  1024→    const nodeAllow = cfg?.operations?.allowTransfer ?? true;
  1025→    if (!(procAllow && nodeAllow)) throw new BusinessException("当前节点不允许转办", 403);
  1026→    exec("UPDATE WF_TASK_INST SET ASSIGNEE = ? WHERE ID = ?", [toUser, task.ID]);
  1027→    exec("UPDATE WF_ACTIVITY_INST SET ASSIGNEE = ? WHERE PROC_INST_ID = ? AND NODE_KEY = ? AND STATUS = 'RUNNING'", [toUser, inst.ID, task.NODE_KEY]);
  1028→    exec("INSERT INTO WF_TASK_TRANSFER (ID, CREATED_AT, FROM_USER, PROCESS_INSTANCE_ID, REASON, TASK_ID, TENANT_ID, TO_USER) VALUES (?,?,?,?,?,?,?,?)",
  1029→      [uuid32(), nowStr(), userId, inst.ID, reason, task.ID, inst.TENANT_ID ?? "default", toUser]);
  1030→    addComment(inst, task.ID, userId, "transfer", reason, toUser);
  1031→  });
  1032→}

  1059→}

  1064→export interface PredictionResult { nodes: PredictionNode[]; branchStops: PredictionNode[]; finished: boolean; }
  1065→
  1066→/** 静态遍历 + 表达式求值：无条件出线继续深入；有条件出线按变量求值，不可判定即标记分支停止；到 endEvent 停止；visited 防环 */
  1067→export function predict(procDefId: string | null, procKey: string | null, startUser: string, variables: Record<string, UelValue>): PredictionResult {
  1068→  const deploy = procDefId ? loadDeployById(procDefId) : procKey ? loadLatestDeployByKey(procKey) : null;
  1069→  if (!deploy) throw new BusinessException("流程定义不存在", 404);
  1070→  const ir = getIR(deploy);
  1071→  const start = ir.process.nodes.find((n) => n.type === "startEvent");
  1072→  if (!start) throw new EngineException("缺少开始事件");
  1073→  const nodes: PredictionNode[] = [];
  1074→  const branchStops: PredictionNode[] = [];
  1075→  const visited = new Set<string>();
  1076→  const ctx: Record<string, UelValue> = { initiator: startUser, ...variables };
  1077→  let finished = false;
  1078→
  1079→  const assigneesOf = (dep: DeployRow, node: NodeIR): string[] => {
  1080→    const cfg = configMapOf(dep)[node.id];
  1081→    if (cfg?.approval?.type === "dept_head") {
  1082→      const head = resolveDeptHead(startUser);
  1083→      if (head) return [head];
  1084→    }
  1085→    if (cfg?.approval?.userIds?.length) return cfg.approval.userIds.map(String);
  1086→    if (node.assignee) {
  1087→      const r = resolveVarRef(node.assignee, ctx);
  1088→      if (r) return [r];
  1089→    }
  1090→    return node.candidateUsers ?? [];
  1091→  };
  1092→
  1093→  // walk(dep, g, ...): 支持跨图遍历——callActivity 展开子流程图（深度限 3，防循环调用）
  1094→  const walk = (dep: DeployRow, g: DiagramIR, nodeId: string, visited: Set<string>, depth: number): void => {
  1095→    if (visited.has(nodeId)) return;
  1096→    visited.add(nodeId);
  1097→    const node = g.nodeById[nodeId];
  1098→    if (!node) return;
  1099→    if (node.type === "endEvent") { finished = true; return; }
  1100→    if (node.type === "userTask") nodes.push({ nodeId: node.id, nodeName: node.name ?? node.id, nodeType: node.type, assignees: assigneesOf(dep, node) });
  1101→    if (node.type === "serviceTask") nodes.push({ nodeId: node.id, nodeName: node.name ?? node.id, nodeType: node.type, assignees: [] });
  1102→    if (node.type === "callActivity") {
  1103→      nodes.push({ nodeId: node.id, nodeName: node.name ?? node.id, nodeType: node.type, assignees: [] });
  1104→      const called = configMapOf(dep)[node.id]?.callActivity?.calledElement ?? node.calledElement;
  1105→      if (called && depth < 3) {
  1106→        const childDeploy = loadLatestDeployByKey(called);
  1107→        const childStart = childDeploy ? getIR(childDeploy).process.nodes.find((n) => n.type === "startEvent") : undefined;
  1108→        if (childDeploy && childStart) walk(childDeploy, getIR(childDeploy), childStart.id, new Set(), depth + 1);
  1109→      }
  1110→      const f0 = defaultOut(g, node.id);
  1111→      if (f0) walk(dep, g, f0.targetRef, visited, depth);
  1112→      return;
  1113→    }
  1114→    if (node.type === "exclusiveGateway" || node.type === "inclusiveGateway") {
  1115→      const outs = g.outgoing[node.id] ?? [];
  1116→      const condOuts = outs.filter((f) => f.condition);
  1117→      const freeOuts = outs.filter((f) => !f.condition);
  1118→      if (!condOuts.length) { for (const f of freeOuts) walk(dep, g, f.targetRef, visited, depth); return; }
  1119→      const ctxFull = { ...ctx };
  1120→      const matched = condOuts.filter((f) => {
  1121→        try { return evalExpression(f.condition!, ctxFull) === true; } catch { return false; }
  1122→      });
  1123→      if (node.type === "exclusiveGateway") {
  1124→        if (matched.length) walk(dep, g, matched[0].targetRef, visited, depth);
  1125→        else if (freeOuts.length) walk(dep, g, freeOuts[0].targetRef, visited, depth);
  1126→        else branchStops.push({ nodeId: node.id, nodeName: node.name ?? node.id, nodeType: node.type, assignees: [] });
  1127→      } else {
  1128→        if (matched.length) for (const f of matched) walk(dep, g, f.targetRef, visited, depth);
  1129→        else if (freeOuts.length) for (const f of freeOuts) walk(dep, g, f.targetRef, visited, depth);
  1130→        else branchStops.push({ nodeId: node.id, nodeName: node.name ?? node.id, nodeType: node.type, assignees: [] });
  1131→      }
  1132→      return;
  1133→    }
  1134→    if (node.type === "parallelGateway") {
  1135→      for (const f of g.outgoing[node.id] ?? []) walk(dep, g, f.targetRef, visited, depth);
  1136→      return;
  1137→    }
  1138→    const f = defaultOut(g, node.id);
  1139→    if (f) walk(dep, g, f.targetRef, visited, depth);
  1140→  };
  1141→  walk(deploy, ir, start.id, visited, 0);
  1142→  return { nodes, branchStops, finished };
  1143→}

  1147→export function pendingTasksOfUser(userId: string): TaskRow[] {
  1148→  const byAssignee = query<TaskRow>("SELECT * FROM WF_TASK_INST WHERE ASSIGNEE = ? AND STATUS = 'PENDING' ORDER BY START_TIME DESC", [String(userId)]);
  1149→  const cand = query<{ TASK_ID: string }>("SELECT TASK_ID FROM WF_TASK_CANDIDATE WHERE USER_ID = ?", [String(userId)]);
  1150→  let byCandidate: TaskRow[] = [];
  1151→  if (cand.length) {
  1152→    const ph = cand.map(() => "?").join(",");
  1153→    byCandidate = query<TaskRow>(
  1154→      `SELECT * FROM WF_TASK_INST WHERE ID IN (${ph}) AND STATUS = 'PENDING' AND ASSIGNEE IS NULL ORDER BY START_TIME DESC`,
  1155→      cand.map((c) => c.TASK_ID));
  1156→  }
  1157→  return [...byAssignee, ...byCandidate];
  1158→}
