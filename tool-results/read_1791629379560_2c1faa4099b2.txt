   422→async function load() {
   423→  try {
   424→    // preview=true（视图设计器预览）：取最新 DRAFT 定义而非已发布版本
   425→    const preview = route.query.preview === 'true'
   426→    const res = await pageApi.getPageByKey(pageKey.value, preview)
   427→    page.value = res.data
   428→    if (res.data.type === 'PAGE') {
   429→      // 自定义页面（阶段二）：交给 PageRendererPage 渲染（数据源注册 + 动作总线）
   430→      return
   431→    }
   432→    if (res.data.type !== 'VIEW') {
   433→      error.value = '未知页面类型'
   434→      return
   435→    }
   436→    if (!parseSchema(res.data.schema)) {
   437→      error.value = '页面配置异常，请联系管理员'
   438→      return
   439→    }
   440→    // 数据源协议：绑定 dataSourceId 时拉取 metadata（列 + writable），定义（FORM 反查 formKey）延迟到首次表单打开时加载
   441→    if (res.data.dataSourceId) {
   442→      try {
   443→        dataSourceMeta.value = (await dataSourceApi.getMetadata(res.data.dataSourceId)).data
   444→      } catch {
   445→        dataSourceMeta.value = null
   446→      }
   447→    }

   455→/** 解析编译产物 {rule, option}，按 rule.type 分段提取 */
   456→function parseSchema(schema: string): boolean {
   457→  try {
   458→    const parsed = JSON.parse(schema || '{}')
   459→    const rule: any[] = Array.isArray(parsed) ? parsed : (parsed.rule || [])
   460→    // 显示方式（编译产物顶层 display；缺省表格）
   461→    displayMode.value = parsed.display === 'card' ? 'card' : 'table'
   462→    detailOption.value = Array.isArray(parsed) ? {} : (parsed.option || {})
   463→    searchRules.value = rule.filter((r) => r.type === 'input' || r.type === 'datePicker')
   464→    // 视图级可排序字段（编译产物顶层 sortableFields；未声明=跟随数据源全部可排字段）
   465→    sortableFieldKeys.value = Array.isArray(parsed.sortableFields) ? parsed.sortableFields : []
   466→    // 分页配置（编译产物顶层 pagination；未声明使用默认）
   467→    const pag = parsed.pagination || {}
   468→    paginationConfig.value = {
   469→      show: pag.show !== false,
   470→      pageSize: Number(pag.pageSize) > 0 ? Number(pag.pageSize) : 20,
   471→      pageSizes: Array.isArray(pag.pageSizes) && pag.pageSizes.length ? [...pag.pageSizes] : [10, 20, 50],
   472→    }
   473→    const tableRule = rule.find((r) => r.type === 'table')
   474→    tableColumns.value = (tableRule?.props?.columns || []) as CompiledColumn[]
   475→    // 表格增强（Task ⑤）：从顶层原始 schema 读取（ViewCompiler 编译产物不含这些字段，
   476→    // mergeCompiled 保留原始字段，预览/发布后均可读；未配置时全部关闭 = 现状行为）
   477→    const rawColumns: any[] = Array.isArray(parsed.columns) ? parsed.columns : []
   478→    summarySpecs.value = rawColumns
   479→      .filter((c: any) => c && !c.hidden && c.aggregate)
   480→      .map((c: any) => ({ key: c.key, aggregate: c.aggregate }))
   481→    headerFilterOn.value = featureEnabled(parsed.headerFilter)
   482→    headerFilters.value = {}
   483→    batchEnabled.value = featureEnabled(parsed.batch)
   484→    batchDeleteFlag.value = typeof parsed.batch === 'boolean' ? true : parsed.batch?.delete !== false
   485→    tableStyle.value = tableRule?.props?.style as CardStyle | undefined
   486→    cardStyle.value = tableRule?.props?.cardStyle as CardStyle | undefined
   487→    const actionsRule = rule.find((r) => r.type === '__page_actions')
   488→    actionConfig.value = actionsRule?.props || {}
   489→    const detailRule = rule.find((r) => r.type === '__page_detail')
   490→    if (detailRule?.props?.enabled) {
   491→      detailConfig.value = detailRule.props
   492→    }
   493→    const eventsRule = rule.find((r) => r.type === '__page_events')
   494→    eventsList.value = eventsRule?.events || []
   495→    // 查询默认值（编译产物 rule.value）
   496→    queryDefaults.value = Object.fromEntries(searchRules.value.map((r) => [r.field, r.value ?? '']))
   497→    Object.keys(queryDefaults.value).forEach((k) => { query[k] = queryDefaults.value[k] })
   498→    return true
   499→  } catch {
   500→    return false
   501→  }
   502→}

   504→/** 构建结构化 filter {logic, conditions:[{column,op,value}]}；空条件返回 undefined */
   505→function buildFilter(params: Record<string, any>): string | undefined {
   506→  const conditions: { column: string; op: string; value: any }[] = []
   507→  for (const r of searchRules.value) {
   508→    const v = params[r.field]
   509→    if (v === undefined || v === null || v === '') continue
   510→    if (r.matchType === 'like') conditions.push({ column: r.field, op: 'like', value: v })
   511→    else if (r.matchType === 'range') conditions.push({ column: r.field, op: 'range', value: v })
   512→    else conditions.push({ column: r.field, op: 'eq', value: v })
   513→  }
   514→  // 表头筛选条件（Task ⑤：漏斗按列筛选；多选→in / 区间→range；与查询条件 AND 合并）
   515→  for (const [hfKey, hfValue] of Object.entries(headerFilters.value)) {
   516→    const cond = buildHeaderFilterCondition(hfKey, hfValue)
   517→    if (cond) conditions.push(cond as { column: string; op: string; value: any })
   518→  }
   519→  if (!conditions.length) return undefined
   520→  return JSON.stringify({ logic: 'AND', conditions })
   521→}

   523→/** SearchTable 数据获取：page 为 1-based，转后端 0-based；搜索条件从 params（含外部注入 query）提取 */
   524→const searchTableFetchApi = async (params: QueryParams): Promise<{ rows: any[]; total: number }> => {
   525→  // 不分页（pagination.show=false）：请求全部数据（后端 size<=0 跳过 LIMIT），保留排序
   526→  const p: Record<string, any> = paginationConfig.value.show === false
   527→    ? { size: -1 }
   528→    : { page: params.page || 1, size: params.size || 20 }
   529→  // 排序状态透传（SearchTable 内部维护，服务器端排序）
   530→  if (params.sort) p.sort = params.sort
   531→  if (params.order) p.order = params.order
   532→  const filter = buildFilter(params)
   533→  if (filter) p.filter = filter
   534→  const res = await pageApi.queryPageData(pageKey.value, p)
   535→  const data = res.data as any
   536→  // 最近一次取数的行：汇总行统计 + 表头筛选文本候选读取
   537→  lastRows.value = data.records || []
   538→  triggerEvents('refresh', 'table', { row: null, params: route.query || {} })
   539→  return { rows: data.records || [], total: data.total || 0 }
   540→}

   589→/** 行级按钮可见性：根据 visible 表达式判断（$row 变量替换求值） */
   590→function isButtonVisibleForRow(b: { key: string; visible?: string; events?: any[] }, row: any): boolean {
   591→  // 先通过全局可见性检查
   592→  if (!isActionVisible(b)) return false
   593→  // 无 visible 表达式 → 始终显示
   594→  if (!b.visible) return true
   595→  // 替换 $row.xxx 变量后求值
   596→  try {
   597→    const expr = b.visible.replace(/\$row\.([\w]+)/g, (_: string, k: string) => {
   598→      const v = row?.data != null && typeof row.data === 'object' ? row.data[k] : row?.[k]
   599→      return v === null || v === undefined ? 'undefined' : JSON.stringify(v)
   600→    }).replace(/\$param\.([\w]+)/g, (_: string, k: string) => {
   601→      const v = route.query?.[k]
   602→      return v === null || v === undefined ? 'undefined' : JSON.stringify(v)
   603→    })
   604→    // 安全求值：仅允许比较运算，禁止函数调用
   605→    if (/[a-zA-Z_$]\s*\(/.test(expr)) return false
   606→    return !!Function('"use strict"; return (' + expr + ')')()
   607→  } catch {
   608→    return true // 求值失败时默认显示
   609→  }
   610→}

   826→function toButtonConfig(b: { key: string; label: string; style: 'button' | 'icon' | 'text'; icon?: string; events?: any[] }): ActionButtonConfig {
   827→  const handlers: Record<string, (row?: any) => void> = {
   828→    create: () => openCreate(),
   829→    edit: (row?: any) => openEdit(row),
   830→    delete: (row?: any) => handleDelete(row),
   831→    view: (row?: any) => openDetail(row),
   832→  }
   833→  const type = b.key === 'create' ? 'primary' : b.key === 'delete' ? 'danger' : ''
   834→  const isBuiltin = BUILTIN_ACTION_KEYS.includes(b.key)
   835→  const hasEvents = !!(b.events && b.events.length > 0)
   836→  // 有绑定事件 → 优先执行事件链（内建/自定义都一样）
   837→  const onClick = hasEvents
   838→    ? (row?: any) => handleCustomButton(b, row)
   839→    : isBuiltin
   840→      ? handlers[b.key]
   841→      : () => {}
   842→  return {
   843→    label: b.label,
   844→    type: type as '' | 'primary' | 'danger',
   845→    icon: (b.icon && iconMap[b.icon]) || defaultIcons[b.key],
   846→    style: b.style,
   847→    link: b.style === 'text',
   848→    circle: b.style === 'icon',
   849→    onClick,
   850→  }

  1123→/** 模板变量替换：$row.字段（当前行，优先 row.data 内层）/ $param.参数（路由参数） */
  1124→function resolveTemplate(tpl: string, ctx: { row: any; params: Record<string, any> }): string {
  1125→  return tpl
  1126→    .replace(/\$row\.([\w]+)/g, (_, k) => {
  1127→      const row = ctx.row || {}
  1128→      const v = row.data != null && typeof row.data === 'object' ? row.data[k] : row[k]
  1129→      return v === null || v === undefined ? '' : String(v)
  1130→    })
  1131→    .replace(/\$param\.([\w]+)/g, (_, k) => (ctx.params?.[k] == null ? '' : String(ctx.params[k])))
  1132→}
  1133→
  1134→/** 解析动作参数 [{key,value}] → {key: 模板替换后的值} */
  1135→function resolveParams(
  1136→  params: { key: string; value: string }[],
  1137→  ctx: { row: any; params: Record<string, any> },
  1138→): Record<string, any> {
  1139→  const out: Record<string, any> = {}
  1140→  for (const p of params || []) {
  1141→    out[p.key] = resolveTemplate(p.value, ctx)
  1142→  }
  1143→  return out
  1144→}

  1146→/** 动作类型分发 */
  1147→async function dispatchAction(
  1148→  action: { type: string; params?: { key: string; value: string }[] },
  1149→  ctx: { row: any; params: Record<string, any> },
  1150→) {
  1151→  const resolved = resolveParams(action.params || [], ctx)
  1152→  switch (action.type) {
  1153→    case 'open-detail':
  1154→      await openDetail(ctx.row)
  1155→      if (resolved.title) detailTitle.value = resolved.title
  1156→      if (resolved.width) detailWidth.value = resolved.width
  1157→      break
  1158→    case 'open-link':
  1159→      if (resolved.url) router.push(resolved.url)
  1160→      break
  1161→    case 'open-create':
  1162→      openCreate()
  1163→      break
  1164→    case 'edit':
  1165→      currentRow.value = ctx.row
  1166→      openEdit()
  1167→      break
  1168→    case 'delete': {
  1169→      currentRow.value = ctx.row
  1170→      await handleDelete()
  1171→      break
  1172→    }
  1173→    case 'refresh':
  1174→      searchTableRef.value?.fetchList()
  1175→      break
  1176→    case 'export':
  1177→      exportData()
  1178→      break
  1179→    case 'message':
  1180→      ElMessage({
  1181→        message: resolved.text || resolved.message || '提示',
  1182→        type: (resolved.type as any) || 'info',
  1183→      })
  1184→      break
  1185→    case 'set-filter':
  1186→      for (const [k, v] of Object.entries(resolved)) {
  1187→        if (k in query) query[k] = v
  1188→      }
  1189→      searchTableRef.value?.setQuery({ ...query })
  1190→      break
  1191→    case 'set-sort': {
  1192→      // 设置表格排序：params 包含 field 和 order
  1193→      const field = resolved.field || resolved.prop || ''
  1194→      const order = resolved.order || 'ascending'
  1195→      searchTableRef.value?.sort(field, order)
  1196→      break
  1197→    }
  1198→    case 'set-page': {
  1199→      // 设置分页：params 包含 page（1-based），不重置页
  1200→      const page = parseInt(resolved.page, 10)
  1201→      if (!isNaN(page) && page > 0) {
  1202→        searchTableRef.value?.setQuery({ page }, false)
  1203→      }
  1204→      break
  1205→    }
  1206→    case 'clear-selection': {
  1207→      // 清空行选择
  1208→      searchTableRef.value?.clearSelection()
  1209→      selectedRows.value = []
  1210→      break
  1211→    }
  1212→    case 'script': {
  1213→      const source: string = resolved.source || ''
  1214→      if (!source) {
  1215→        console.warn('[page] 脚本动作缺少 source 参数')
  1216→        break
  1217→      }
  1218→      if (!isScriptEventEnabled()) {
  1219→        console.warn('[page] 脚本事件未启用（设置 VITE_PAGE_SCRIPT_ENABLED=true 开启）')
  1220→        break
  1221→      }
  1222→      await executeScript(source, {
  1223→        row: ctx.row,
  1224→        params: ctx.params || {},
  1225→        selectedRows: currentRow.value ? [currentRow.value] : [],
  1226→        ds: {
  1227→          query: (filter?: Record<string, any>) => {
  1228→            if (filter) for (const [k, v] of Object.entries(filter)) query[k] = v
  1229→            return searchTableRef.value?.fetchList() ?? Promise.resolve()
  1230→          },
  1231→          detail: (id: string) => (boundFormKey.value ? bizDataApi.detail(boundFormKey.value, id) : null),
  1232→          create: (data: Record<string, unknown>) =>
  1233→            boundFormKey.value ? bizDataApi.create(boundFormKey.value, data) : Promise.reject(new Error('数据源只读')),
  1234→          update: (id: string, data: Record<string, unknown>) =>
  1235→            boundFormKey.value
  1236→              ? bizDataApi.update(boundFormKey.value, id, data, currentRow.value?.version || 1)
  1237→              : Promise.reject(new Error('数据源只读')),
  1238→          remove: (id: string) =>
  1239→            boundFormKey.value ? bizDataApi.remove(boundFormKey.value, id) : Promise.reject(new Error('数据源只读')),
  1240→        },
  1241→        api: { formKey: boundFormKey.value || '', pageKey: pageKey.value },
  1242→        actions: {
  1243→          refresh: () => searchTableRef.value?.fetchList() ?? Promise.resolve(),
  1244→          openDetail: () => openDetail(currentRow.value),
  1245→          openCreate: () => openCreate(),
  1246→          openEdit: () => openEdit(),
  1247→          remove: (id: string) => (boundFormKey.value ? bizDataApi.remove(boundFormKey.value, id) : Promise.reject(new Error('数据源只读'))),
  1248→        },
  1249→        $: { message: (msg: string, type = 'info') => ElMessage({ message: msg, type: type as any }) },
  1250→      })
  1251→      break
  1252→    }
  1253→    default:
  1254→      console.warn('[page] 未知动作类型:', action.type)
  1255→  }
  1256→}

  1268→/** 按触发器 + 挂接组件执行事件链；target 为空 = 通配（匹配所有组件） */
  1269→function triggerEvents(trigger: string, target: string, ctx: EventContext) {
  1270→  for (const ev of eventsList.value) {

  1272→    if (ev.target && ev.target !== target) continue
  1273→    for (const action of ev.actions || []) {
  1274→      dispatchAction(action, { row: ctx.row, params: ctx.params || {} })
  1275→    }
  1276→  }
