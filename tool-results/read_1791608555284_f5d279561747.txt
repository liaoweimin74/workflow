    75→    /**
    76→     * 加载表单运行时上下文（表名 + 列 + 子表元数据）。
    77→     */
    78→    public BizDataContext loadContext(String formKey) {
    79→        if (formKey == null || !FORM_KEY_PATTERN.matcher(formKey).matches()) {
    80→            throw new BusinessException(400, "非法表单 key: " + formKey);
    81→        }
    82→        String tableName = "wf_biz_" + formKey;
    83→        if (!tableManager.tableExists(tableName)) {
    84→            throw new BusinessException(404, "业务表单数据表不存在: " + formKey);
    85→        }
    86→        List<ColumnConfig> columns = formDefService.getBusinessColumnsByKey(formKey);
    87→        // 主表列：排除子表字段（子表字段映射独立物理表，非主表列）
    88→        List<String> keys = columns.stream()
    89→                .filter(c -> c.getSubColumns() == null || c.getSubColumns().isEmpty())
    90→                .map(ColumnConfig::getKey)
    91→                .toList();
    92→        Map<String, BizDataContext.SubTableDef> subTables = new LinkedHashMap<>();
    93→        for (ColumnConfig c : columns) {
    94→            if (c.getSubColumns() != null && !c.getSubColumns().isEmpty()) {
    95→                String mode = c.getSubMode() == null || c.getSubMode().isBlank() ? "embedded" : c.getSubMode();
    96→                List<String> subKeys = c.getSubColumns().stream().map(ColumnConfig::getKey).toList();
    97→                subTables.put(c.getKey(), new BizDataContext.SubTableDef(
    98→                        "wf_biz_" + formKey + "_" + c.getKey(), mode, c.getSubColumns(), subKeys));
    99→            }
   100→        }
   101→        return new BizDataContext(tableName, formKey, columns, keys, subTables);
   102→    }

   119→    public void validateRequired(List<ColumnConfig> columns, Map<String, Object> data) {
   120→        Map<String, Object> safe = data == null ? Map.of() : data;
   121→        for (ColumnConfig c : columns) {
   122→            if (c.isRequired()) {
   123→                Object v = safe.get(c.getKey());
   124→                if (v == null || (v instanceof String s && s.isBlank())) {
   125→                    throw new BusinessException(400, "必填字段不能为空: " + c.getLabel());
   126→                }
   127→            }
   128→        }
   129→    }

   131→    /**
   132→     * 遍历 data-picker 引用列：校验 id 存在并生成 {@code <key>_text} 展示缓存文本。
   133→     * 不修改原 data，返回附加字段（{@code <key>_text} → 文本）；引用值为空时返回空文本。
   134→     */
   135→    @SuppressWarnings("unchecked")
   136→    public Map<String, Object> resolvePickerValues(BizDataContext ctx, Map<String, Object> data) {
   137→        Map<String, Object> extra = new LinkedHashMap<>();
   138→        for (ColumnConfig col : ctx.columns()) {
   139→            String pickerConfig = col.getPickerConfig();
   140→            if (pickerConfig == null || pickerConfig.isBlank()) {
   141→                continue;
   142→            }
   143→            // 仅 data-picker 引用列生成冗余文本；LookupPicker 列跳过
   144→            if (!isDataPickerColumn(ctx, col)) {
   145→                continue;
   146→            }
   147→            String key = col.getKey();
   148→            Object raw = data.get(key);
   149→            String text = resolvePickerText(ctx, col, raw);
   150→            extra.put(key + "_text", text);
   151→        }
   152→        return extra;
   153→    }

   158→    public Map<String, String> resolveByFormKey(String formKey, List<String> ids, String displayField) {
   159→        BizDataContext ctx = loadContext(formKey);
   160→        String field = displayField;
   161→        if (field == null || field.isBlank()) {
   162→            field = ctx.columns().stream()
   163→                    .filter(c -> !c.isHidden() && c.getPickerConfig() == null)
   164→                    .map(ColumnConfig::getKey)
   165→                    .findFirst()
   166→                    .orElseThrow(() -> new BusinessException(400, "目标表单无可解析的显示字段"));
   167→        }
   168→        return resolveDisplayTexts(formKey, ids, field);
   169→    }

   171→    /**
   172→     * 批量解析被引用记录的显示文本（id → displayField 值）。
   173→     */
   174→    public Map<String, String> resolveDisplayTexts(String sourceFormKey, List<String> ids, String displayField) {
   175→        if (sourceFormKey == null || !FORM_KEY_PATTERN.matcher(sourceFormKey).matches()) {
   176→            throw new BusinessException(400, "非法目标表单 key: " + sourceFormKey);
   177→        }
   178→        if (ids == null || ids.isEmpty()) {
   179→            return Map.of();
   180→        }
   181→        if (displayField == null || !displayField.matches("^[a-zA-Z][a-zA-Z0-9_]{0,63}$")) {
   182→            throw new BusinessException(400, "非法显示字段: " + displayField);
   183→        }
   184→        String table = "wf_biz_" + sourceFormKey;
   185→        String placeholders = String.join(",", java.util.Collections.nCopies(ids.size(), "?"));
   186→        String sql = "SELECT id, " + displayField + " FROM " + table
   187→                + " WHERE tenant_id = ? AND id IN (" + placeholders + ")";
   188→        List<Object> params = new ArrayList<>();
   189→        params.add(tenantProvider.getTenantId());
   190→        params.addAll(ids);
   191→
   192→        Map<String, String> result = new LinkedHashMap<>();
   193→        for (Map<String, Object> row : jdbcTemplate.queryForList(sql, params.toArray())) {
   194→            Object v = row.get(displayField);
   195→            result.put(String.valueOf(row.get("id")), v == null ? "" : String.valueOf(v));
   196→        }
   197→        return result;
   198→    }

   223→    /**
   224→     * 通用新增实现（不含装饰钩子链；覆盖 handler 委托此方法执行通用插入）。
   225→     */
   226→    @Transactional
   227→    public BizDataVO createGeneric(String formKey, Map<String, Object> data) {
   228→        String tenantId = tenantProvider.getTenantId();
   229→        BizDataContext ctx = loadContext(formKey);
   230→
   231→        validateRequired(ctx.columns(), data);
   232→
   233→        // data-picker 引用校验与冗余文本生成（不改原 data，返回附加字段）
   234→        Map<String, Object> merged = serializeJsonColumns(data, ctx.columns());
   235→        normalizeDateTimeColumns(merged, ctx.columns());
   236→        merged.putAll(resolvePickerValues(ctx, merged));
   237→
   238→        BizDataQueryBuilder.SqlAndParams insert = BizDataQueryBuilder.buildInsert(
   239→                ctx.tableName(), ctx.columnKeys(), merged, tenantId);
   240→        jdbcTemplate.update(insert.sql(), insert.params().toArray());
   241→
   242→        // 子表行写入（随主表创建批量插入）
   243→        String bizId = insertedId(insert);
   244→        for (Map.Entry<String, BizDataContext.SubTableDef> e : ctx.subTables().entrySet()) {
   245→            Object raw = data.get(e.getKey());
   246→            if (raw instanceof List<?> rows) {
   247→                writeSubRows(e.getValue(), bizId, rows);
   248→            }
   249→        }
   250→
   251→        // 新行 id 由 buildInsert 内部生成，查询返回
   252→        return findById(ctx.tableName(), tenantId, ctx, bizId);
   253→    }

   255→    /**
   256→     * 通用分页查询实现。
   257→     */
   258→    public BizDataPageVO queryGeneric(String formKey, BizDataQueryRequest req) {
   259→        String tenantId = tenantProvider.getTenantId();
   260→        BizDataContext ctx = loadContext(formKey);
   261→
   262→        Map<String, Object> filters = parseFilter(req.getFilter());
   263→        // 列类型映射（key → columnType）：JSON 数组列筛选走 JSON 函数（JSON_CONTAINS/JSON_OVERLAPS）
   264→        Map<String, String> columnTypeOf = ctx.columns().stream()
   265→                .collect(Collectors.toMap(ColumnConfig::getKey,
   266→                        c -> c.getColumnType() == null ? "" : c.getColumnType().toUpperCase(), (a, b) -> a));
   267→        int page = Math.max(req.getPage(), 1);
   268→        // size <= 0 表示不分页取全部（buildSelect 跳过 LIMIT/OFFSET）；正数沿用原钳制上限
   269→        int size = req.getSize() <= 0 ? req.getSize() : Math.min(Math.max(req.getSize(), 1), 100);
   270→
   271→        try {
   272→            BizDataQueryBuilder.SqlAndParams count = BizDataQueryBuilder.buildCount(
   273→                    ctx.tableName(), ctx.columnKeys(), columnTypeOf, tenantId, filters, req.getKeyword(), req.getKeywordColumn());
   274→
   275→            BizDataQueryBuilder.SqlAndParams select = BizDataQueryBuilder.buildSelect(
   276→                    ctx.tableName(), ctx.columnKeys(), columnTypeOf, tenantId, filters,
   277→                    req.getKeyword(), req.getKeywordColumn(), req.getSort(), req.getOrder(), page - 1, size);
   278→
   279→            return sqlQueryEngine.execPage(page, size, count, select, row -> toVO(ctx, row));
   280→        } catch (IllegalArgumentException e) {
   281→            throw new BusinessException(400, e.getMessage());
   282→        }
   283→    }

   286→     * config 模式分页查询实现（声明式 JOIN，含虚拟列）。
   287→     */
   288→    public BizDataPageVO queryJoinConfig(String formKey, BizDataQueryRequest req,
   289→                                         List<JoinSqlGenerator.JoinConfig> joins) {
   290→        String tenantId = tenantProvider.getTenantId();
   291→        BizDataContext ctx = loadContext(formKey);
   292→
   293→        List<JoinSqlGenerator.QueryColumn> columns = buildJoinColumns(ctx, joins);
   294→        Map<String, Object> filters = parseFilter(req.getFilter());
   295→        int page = Math.max(req.getPage(), 1);
   296→        // size <= 0 表示不分页取全部（buildSelect 跳过 LIMIT/OFFSET）；正数沿用原钳制上限
   297→        int size = req.getSize() <= 0 ? req.getSize() : Math.min(Math.max(req.getSize(), 1), 100);
   298→
   299→        try {
   300→            // 运行时兜底校验（对齐 Node 新版）：必填/标识符格式/唯一/主表冲突 —— 存量脏配置快速 400 而非畸形 SQL
   301→            List<String> mainColumnsWithId = new ArrayList<>(ctx.columnKeys());
   302→            mainColumnsWithId.add("id"); // SELECT m.* 已带主键，虚拟列同名会重复列错误
   303→            JoinSqlGenerator.validate(joins, mainColumnsWithId);
   304→
   305→            BizDataQueryBuilder.SqlAndParams count = JoinSqlGenerator.buildCount(
   306→                    ctx.tableName(), tenantId, joins, columns, filters, req.getKeyword(), req.getKeywordColumn());
   307→
   308→            BizDataQueryBuilder.SqlAndParams select = JoinSqlGenerator.buildSelect(
   309→                    ctx.tableName(), tenantId, joins, columns, filters,
   310→                    req.getKeyword(), req.getKeywordColumn(), req.getSort(), req.getOrder(), page - 1, size);
   311→
   312→            return sqlQueryEngine.execPage(page, size, count, select, row -> toJoinVO(ctx, joins, row));
   313→        } catch (IllegalArgumentException e) {
   314→            throw new BusinessException(400, e.getMessage());
   315→        }
   316→    }

   328→        List<JoinSqlGenerator.QueryColumn> columns = toQueryColumns(cfg.columns());
   329→        Map<String, Object> filters = parseFilter(req.getFilter());
   330→        Map<String, Object> runtimeParams = parseRuntimeParams(req.getParams());
   331→        int page = Math.max(req.getPage(), 1);

   333→        int size = req.getSize() <= 0 ? req.getSize() : Math.min(Math.max(req.getSize(), 1), 100);

   336→            SqlQueryEngine.WrappedQuery wq = SqlTemplateEngine.wrap(
   337→                    cfg.query(), tenantId, columns, filters, req.getKeyword(), req.getKeywordColumn(),
   338→                    req.getSort(), req.getOrder(), page, size, cfg.declaredParams(), runtimeParams);

   340→                    row -> toSqlVO(cfg.columns(), row));

   407→    /** 聚合行集 → VO（对齐 Node {@code String(row.__k ?? '')} + {@code Number(row.__v ?? 0)}）。 */
   408→    private static AggregateResultVO toAggregateResult(List<Map<String, Object>> rows) {
   409→        List<AggregateRowVO> out = new ArrayList<>(rows.size());
   410→        for (Map<String, Object> row : rows) {
   411→            Object k = row.get("__k");
   412→            out.add(AggregateRowVO.of(k == null ? "" : String.valueOf(k), row.get("__v")));
   413→        }
   414→        return AggregateResultVO.of(out);
   415→    }

   417→    /** 构建查询列映射：主表列（ref="m."+key，默认全可排可筛）+ 虚拟列（ref=alias+"."+joinField，能力取 join 声明） */
   418→    private List<JoinSqlGenerator.QueryColumn> buildJoinColumns(BizDataContext ctx,
   419→                                                                List<JoinSqlGenerator.JoinConfig> joins) {
   420→        Map<String, String> typeOf = new HashMap<>();
   421→        for (ColumnConfig c : ctx.columns()) {
   422→            typeOf.put(c.getKey(), c.getColumnType() == null ? "" : c.getColumnType().toUpperCase());
   423→        }
   424→        List<JoinSqlGenerator.QueryColumn> columns = new ArrayList<>();
   425→        for (String key : ctx.columnKeys()) {
   426→            columns.add(new JoinSqlGenerator.QueryColumn(key, "m." + key, typeOf.getOrDefault(key, ""), true, true));
   427→        }
   428→        for (JoinSqlGenerator.JoinGroup g : JoinSqlGenerator.group(joins)) {
   429→            for (JoinSqlGenerator.JoinConfig m : g.members()) {
   430→                List<ColumnConfig> targetCols = resolveJoinTargets(m);
   431→                columns.add(new JoinSqlGenerator.QueryColumn(m.virtualKey(), g.alias() + "." + m.joinField(),
   432→                        joinColumnType(targetCols, m.joinField()), m.sortable(), m.filterable()));
   433→                // 目标表单含 <joinField>_text 冗余文本列（dataPicker 引用列）→ 带出 <virtualKey>_text，供前端引用渲染显示文本
   434→                if (findJoinTarget(targetCols, m.joinField() + "_text") != null) {
   435→                    columns.add(new JoinSqlGenerator.QueryColumn(m.virtualKey() + "_text",
   436→                            g.alias() + "." + m.joinField() + "_text",
   437→                            joinColumnType(targetCols, m.joinField() + "_text"), false, false));
   438→                }
   439→            }
   440→        }
   441→        return columns;
   442→    }

   445→     * config 模式 SQL 预览：生成主表 + JOIN 虚拟列完整 SELECT（无筛选/无关键词/默认排序/不分页）。
   446→     * 校验 formKey 合法且主表存在（loadContext）；目标表存在性由保存校验负责，预览不重复校验。
   447→     * <p>

   449→     * ⚠️ 目标 key 安全校验：非内建白名单 key 必须匹配 FORM key 模式，否则
   450→     * {@code wf_biz_<key>} 拼接会产生畸形/可注入表名 —— 生成器不做这层，这里拦。
   451→     * alias 兜底：前端不录入 alias（由保存侧/运行时分配），预览按序临时分配（j1/j2/...），
   452→     * 避免 null 引用拼进 SQL（对齐 NodeJS previewJoinSql）。

   454→    public JoinPreviewVO previewJoinSql(String formKey, List<JoinSqlGenerator.JoinConfig> joins) {
   455→        BizDataContext ctx = loadContext(formKey);
   456→        if (joins != null) {
   457→            for (JoinSqlGenerator.JoinConfig join : joins) {
   458→                String targetKey = join.targetFormKey() == null ? "" : join.targetFormKey();
   459→                if (!JoinTargetCatalog.isJoinTargetSystemKey(targetKey)
   460→                        && !FORM_KEY_PATTERN.matcher(targetKey).matches()) {
   461→                    throw new BusinessException(400, "非法关联目标: " + targetKey);
   462→                }
   463→            }
   464→        }
   465→        List<JoinSqlGenerator.JoinConfig> aliasedJoins = withAutoAliases(joins);
   466→        String tenantId = tenantProvider.getTenantId();
   467→        List<JoinSqlGenerator.QueryColumn> columns = buildJoinColumns(ctx, aliasedJoins);
   468→        BizDataQueryBuilder.SqlAndParams select = JoinSqlGenerator.buildSelect(
   469→                ctx.tableName(), tenantId, aliasedJoins, columns, Map.of(),
   470→                null, null, null, null, 0, 0);
   471→        return new JoinPreviewVO(select.sql(), select.params());
   472→    }

   474→    /** alias 兜底分配：空/非法/重复 alias 按 j1/j2/... 找空位（复用 {@link FormQueryConfig#ensureAlias}）。 */
   475→    private static List<JoinSqlGenerator.JoinConfig> withAutoAliases(List<JoinSqlGenerator.JoinConfig> joins) {
   476→        if (joins == null || joins.isEmpty()) {
   477→            return List.of();
   478→        }
   479→        List<JoinSqlGenerator.JoinConfig> out = new ArrayList<>(joins.size());
   480→        Set<String> used = new HashSet<>();
   481→        int idx = 0;
   482→        for (JoinSqlGenerator.JoinConfig j : joins) {
   483→            idx++;
   484→            out.add(new JoinSqlGenerator.JoinConfig(
   485→                    FormQueryConfig.ensureAlias(j.alias(), used, idx),
   486→                    j.targetFormKey(), j.localField(), j.foreignField(),
   487→                    j.joinField(), j.virtualKey(), j.label(),
   488→                    j.sortable(), j.filterable()));
   489→        }
   490→        return out;
   491→    }

   567→     * 通用更新实现（乐观锁；不含装饰钩子链）。
   568→     */
   569→    @Transactional
   570→    public BizDataVO updateGeneric(String formKey, String id, Map<String, Object> data, Integer version) {
   571→        String tenantId = tenantProvider.getTenantId();
   572→        BizDataContext ctx = loadContext(formKey);
   573→
   574→        validateRequired(ctx.columns(), data);
   575→        int currentVersion = version == null ? 1 : version;
   576→
   577→        Map<String, Object> merged = serializeJsonColumns(data, ctx.columns());
   578→        normalizeDateTimeColumns(merged, ctx.columns());
   579→        merged.putAll(resolvePickerValues(ctx, merged));
   580→
   581→        BizDataQueryBuilder.SqlAndParams update = BizDataQueryBuilder.buildUpdate(
   582→                ctx.tableName(), ctx.columnKeys(), merged, tenantId, id, currentVersion);
   583→        int affected = jdbcTemplate.update(update.sql(), update.params().toArray());
   584→
   585→        if (affected == 0) {
   586→            // 区分"记录不存在"（404）与"版本冲突"（409）
   587→            List<Map<String, Object>> exists = jdbcTemplate.queryForList(
   588→                    "SELECT id, version FROM " + ctx.tableName() + " WHERE id = ? AND tenant_id = ?", id, tenantId);
   589→            if (exists.isEmpty()) {
   590→                throw new BusinessException(404, "业务数据不存在: " + id);
   591→            }
   592→            throw new BusinessException(409, "数据已被他人修改，请刷新后重试");
   593→        }
   594→
   595→        // 子表行增量 diff（仅当请求携带子表字段时）
   596→        for (Map.Entry<String, BizDataContext.SubTableDef> e : ctx.subTables().entrySet()) {
   597→            Object raw = data.get(e.getKey());
   598→            if (raw instanceof List<?> rows) {
   599→                diffSubRows(e.getValue(), id, rows);
   600→            }
   601→        }
   602→
   603→        return findById(ctx.tableName(), tenantId, ctx, id);
   604→    }

   671→    /**
   672→     * 更新独立子表行（乐观锁：须携带当前 version）。
   673→     */
   674→    public Map<String, Object> updateSubRow(String formKey, String id, String field, String rowId,
   675→                                            Map<String, Object> data, Integer version) {
   676→        BizDataContext ctx = loadContext(formKey);
   677→        BizDataContext.SubTableDef def = requireSubTable(ctx, field);
   678→        requireMainRow(ctx, id);
   679→
   680→        // 仅允许更新子业务列（白名单 subKeys 过滤，防注入/防篡改内部列）
   681→        Map<String, Object> safe = new LinkedHashMap<>();
   682→        for (String k : def.subKeys()) {
   683→            if (data.containsKey(k)) {
   684→                safe.put(k, data.get(k));
   685→            }
   686→        }
   687→        if (safe.isEmpty()) {
   688→            throw new BusinessException(400, "更新内容不能为空: " + field);
   689→        }
   690→
   691→        StringBuilder set = new StringBuilder();
   692→        List<Object> params = new ArrayList<>();
   693→        for (Map.Entry<String, Object> e : safe.entrySet()) {
   694→            set.append(e.getKey()).append(" = ?, ");
   695→            params.add(e.getValue());
   696→        }
   697→        set.append("version = version + 1, updated_at = NOW()");
   698→        params.add(tenantProvider.getTenantId());
   699→        params.add(id);
   700→        params.add(rowId);
   701→        params.add(version == null ? 1 : version);
   702→
   703→        int affected = jdbcTemplate.update("UPDATE " + def.tableName() + " SET " + set
   704→                + " WHERE tenant_id = ? AND biz_id = ? AND id = ? AND version = ?", params.toArray());
   705→        if (affected == 0) {
   706→            throw new BusinessException(409, "子表行已被他人修改或不存在，请刷新后重试");
   707→        }
   708→        return readSubRows(def, id).stream()
   709→                .filter(r -> rowId.equals(String.valueOf(r.get("id"))))
   710→                .findFirst()
   711→                .orElseThrow(() -> new BusinessException(404, "子表行不存在: " + rowId));
   712→    }

   773→    @SuppressWarnings("unchecked")
   774→    private String resolvePickerText(BizDataContext ctx, ColumnConfig col, Object raw) {
   775→        try {
   776→            Map<String, Object> picker = objectMapper.readValue(col.getPickerConfig(), Map.class);
   777→            String sourceFormKey = picker.get("sourceFormKey") == null ? null : String.valueOf(picker.get("sourceFormKey"));
   778→            String displayField = picker.get("displayField") == null ? null : String.valueOf(picker.get("displayField"));
   779→            Object maxCountObj = picker.get("maxCount");
   780→
   781→            if (raw == null || String.valueOf(raw).isBlank()) {
   782→                return "";
   783→            }
   784→            // 值以 JSON 数组字符串存储（如 ["u1","u2"]；单选为 ["u1"]）
   785→            String rawStr = String.valueOf(raw);
   786→            List<String> ids;
   787→            try {
   788→                ids = objectMapper.readValue(rawStr,
   789→                        objectMapper.getTypeFactory().constructCollectionType(List.class, String.class));
   790→            } catch (JsonProcessingException e) {
   791→                throw new BusinessException(400, "data-picker 引用值格式非法（需 JSON 数组）: " + col.getKey());
   792→            }
   793→            ids.removeIf(String::isBlank);
   794→            if (ids.isEmpty()) {
   795→                return "";
   796→            }
   797→            // maxCount 校验：配置了（非 null 非空）且超限 → 400
   798→            if (maxCountObj != null) {
   799→                int maxCount;
   800→                try {
   801→                    maxCount = Integer.parseInt(String.valueOf(maxCountObj));
   802→                } catch (NumberFormatException e) {
   803→                    throw new BusinessException(400, "data-picker maxCount 配置非法: " + col.getKey());
   804→                }
   805→                if (maxCount > 0 && ids.size() > maxCount) {
   806→                    throw new BusinessException(400,
   807→                            "data-picker 引用数量超出限制（最多 " + maxCount + "）: " + col.getKey());
   808→                }
   809→            }
   810→
   811→            Map<String, String> texts = resolveDisplayTexts(sourceFormKey, ids, displayField);
   812→            List<String> ordered = new ArrayList<>();
   813→            for (String id : ids) {
   814→                String t = texts.get(id);
   815→                if (t == null) {
   816→                    throw new BusinessException(400, "引用的数据不存在: " + col.getKey() + "=" + id);
   817→                }
   818→                ordered.add(t);
   819→            }
   820→            // 冗余文本同样以 JSON 数组存储（与 id 数组顺序一致）
   821→            return objectMapper.writeValueAsString(ordered);
   822→        } catch (JsonProcessingException e) {
   823→            throw new BusinessException(400, "data-picker 配置或引用值非法: " + col.getKey());
   824→        }
   825→    }

   827→    /** 解析单个表单 column_config 中的 dataPicker 引用列，聚合到 result */
   828→    @SuppressWarnings("unchecked")
   829→    private void collectPickerRefs(FormDefinition def, Map<String, Map<String, Object>> result) {
   830→        String columnConfig = def.getColumnConfig();
   831→        if (columnConfig == null || columnConfig.isBlank()) {
   832→            return;
   833→        }
   834→        try {
   835→            List<Map<String, Object>> cols = objectMapper.readValue(columnConfig, List.class);
   836→            for (Map<String, Object> col : cols) {
   837→                Object pickerConfig = col.get("pickerConfig");
   838→                if (!(pickerConfig instanceof String s) || s.isBlank()) {
   839→                    continue;
   840→                }
   841→                Map<String, Object> picker = objectMapper.readValue(s, Map.class);
   842→                Object target = picker.get("sourceFormKey");
   843→                if (target == null || String.valueOf(target).isBlank()) {
   844→                    continue;
   845→                }
   846→                String targetKey = String.valueOf(target);
   847→                Map<String, Object> entry = result.computeIfAbsent(targetKey, k -> {
   848→                    Map<String, Object> m = new LinkedHashMap<>();
   849→                    m.put("count", 0);
   850→                    m.put("referencedBy", new ArrayList<String>());
   851→                    return m;
   852→                });
   853→                entry.put("count", ((Number) entry.get("count")).intValue() + 1);
   854→                ((List<String>) entry.get("referencedBy")).add(def.getKey());
   855→            }
   856→        } catch (JsonProcessingException e) {
   857→            // 非法 column_config 跳过（发布链路已校验，此处容错）
   858→        }
   859→    }

   861→    private BizDataVO toVO(BizDataContext ctx, Map<String, Object> row) {
   862→        Map<String, Object> data = new LinkedHashMap<>();
   863→        for (ColumnConfig c : ctx.columns()) {
   864→            Object v = row.get(c.getKey());
   865→            if (v != null) {
   866→                data.put(c.getKey(), "JSON".equals(c.getColumnType()) ? deserializeJsonValue(v) : v);
   867→            }
   868→        }
   869→        // embedded 模式：附加子表行（按 sort_no 升序），保留 id 供前端 diff
   870→        for (Map.Entry<String, BizDataContext.SubTableDef> e : ctx.subTables().entrySet()) {
   871→            BizDataContext.SubTableDef def = e.getValue();
   872→            if ("embedded".equals(def.subMode())) {
   873→                data.put(e.getKey(), readSubRows(def, String.valueOf(row.get("id"))));
   874→            }
   875→        }
   876→        Integer version = asInt(row.get("version"));
   877→        LocalDateTime createdAt = asDateTime(row.get("created_at"));
   878→        LocalDateTime updatedAt = asDateTime(row.get("updated_at"));
   879→        return new BizDataVO(String.valueOf(row.get("id")), data, version, createdAt, updatedAt);
   880→    }

   882→    /** config 模式行映射：主表列（toVO 逻辑）+ 虚拟列（virtualKey → joinField 值，含 dataPicker 冗余 _text） */
   883→    private BizDataVO toJoinVO(BizDataContext ctx, List<JoinSqlGenerator.JoinConfig> joins, Map<String, Object> row) {
   884→        BizDataVO vo = toVO(ctx, row);
   885→        Map<String, Object> data = vo.getData();
   886→        for (JoinSqlGenerator.JoinConfig j : joins) {
   887→            Object v = row.get(j.virtualKey());

   889→                data.put(j.virtualKey(), v);

   891→            Object vt = row.get(j.virtualKey() + "_text");
   892→            if (vt != null) {
   893→                data.put(j.virtualKey() + "_text", vt);
   894→            }
   895→        }

   899→    /**
   900→     * sql 模式行映射：外层子查询输出列全量保留（可含聚合列），绕过 BizDataVO 系统列字段的仅主表列逻辑。
   901→     * <p>输出列名与声明列（columns）的 key 做大小写不敏感对齐：匹配某声明列 key（忽略大小写）时输出声明列精确 key，
   902→     * 保证返回 key 与字段元数据一致（字段 key 可为驼峰，如探测 JDBC 列标签 hireDate，前端按配置 key 取值）；
   903→     * 未匹配列统一转小写，消除 H2 等数据库对子查询输出列名规范化为大写的影响（与声明列小写 key 约定兼容）。
   904→     */
   905→    private BizDataVO toSqlVO(List<ColumnConfig> declared, Map<String, Object> row) {
   906→        Map<String, String> keyNormalizer = new HashMap<>();
   907→        if (declared != null) {
   908→            for (ColumnConfig c : declared) {
   909→                if (c.getKey() != null && !c.getKey().isBlank()) {
   910→                    // putIfAbsent：声明列存在仅大小写不同的重名时，保留先出现的精确 key
   911→                    keyNormalizer.putIfAbsent(c.getKey().toLowerCase(), c.getKey());
   912→                }
   913→            }
   914→        }
   915→        Map<String, Object> data = new LinkedHashMap<>();
   916→        for (Map.Entry<String, Object> e : row.entrySet()) {
   917→            String lower = e.getKey().toLowerCase();
   918→            data.put(keyNormalizer.getOrDefault(lower, lower), e.getValue());
   919→        }
   920→        data.remove("version");
   921→        data.remove("created_at");
   922→        data.remove("updated_at");
   923→        data.remove("tenant_id");
   924→        Integer version = asInt(row.get("version"));
   925→        LocalDateTime createdAt = asDateTime(row.get("created_at"));
   926→        LocalDateTime updatedAt = asDateTime(row.get("updated_at"));
   927→        // id 保留在 data 中（管理员 SQL 输出可能有非 id 主键/聚合值），BizDataVO.id 取行内 id 兜底
   928→        return new BizDataVO(String.valueOf(row.get("id")), data, version, createdAt, updatedAt);
   929→    }

   931→    /**
   932→     * 对非字符串值（数组/List/Map）序列化为 JSON 字符串（供参数绑定存储）。
   933→     *
   934→     * ⚠️ JSON 列（`columnType == 'JSON'`，物理表 `longtext CHECK (json_valid(...))`）的字符串值需归一：
   935→     * 单选 select / 树选等组件的 value 是裸字符串（如 'annual'），直接入库即撞 CHECK
   936→     * （报 `CONSTRAINT <表>.<列> failed`）。规则与 Node 侧逐条对齐：
   937→     *   - 空白字符串 → null（required 列已被 validateRequired 拦截，能到这里必是可空列）；
   938→     *   - 非法 JSON 的非空字符串 → Jackson writeValueAsString 包成 JSON 字符串文档；
   939→     *   - 合法 JSON（含数字/布尔/null 字面量文本）→ 原样（readValue 成功即 json_valid 必过）。
   940→     * 读取侧 deserializeJsonValue 对 "\"annual\"" parse 回 annual，回显不变。
   941→     */
   942→    private Map<String, Object> serializeJsonColumns(Map<String, Object> data, List<ColumnConfig> columns) {
   943→        Set<String> jsonKeys = new HashSet<>();
   944→        for (ColumnConfig c : columns) {
   945→            if (c.getColumnType() != null && "JSON".equalsIgnoreCase(c.getColumnType())) {
   946→                jsonKeys.add(c.getKey());
   947→            }
   948→        }
   949→        Map<String, Object> out = new LinkedHashMap<>(data);
   950→        for (Map.Entry<String, Object> e : data.entrySet()) {
   951→            Object v = e.getValue();
   952→            if (v == null) {
   953→                continue;
   954→            }
   955→            if (v instanceof String s) {
   956→                if (!jsonKeys.contains(e.getKey())) {
   957→                    continue; // 非 JSON 列：字符串为旧格式容错
   958→                }
   959→                if (s.isBlank()) {
   960→                    out.put(e.getKey(), null);
   961→                    continue;
   962→                }
   963→                if (!isValidJsonText(s)) {
   964→                    out.put(e.getKey(), writeJsonOr400(e.getKey(), s));
   965→                }
   966→                continue;
   967→            }
   968→            out.put(e.getKey(), writeJsonOr400(e.getKey(), v));
   969→        }
   970→        return out;
   971→    }

   973→    /**
   974→     * DATE / DATETIME 列值归一（就地修改 `data`；与 Node 侧 `normalizeDateTimeColumns` 逐条对齐）。
   975→     *
   976→     * ⚠️ 为什么需要：前端日期选择器（form-create `datePicker`，未配置 value-format）
   977→     *    的 modelValue 是 Date 对象，JSON 序列化后是**带时区的 ISO 字符串**
   978→     *    （如 `2026-09-24T16:00:00.000Z`，即东八区 2026-09-25 00:00）。直接入库：
   979→     *    - DATE 列（物理 `date`）：MariaDB 严格模式拒绝 datetime 字符串 ——
   980→     *      `Incorrect date value: '2026-09-24T16:00:00.000Z'`；
   981→     *    - DATETIME 列：即便库接受了，存的也是 UTC 壁钟时间，回显差 8 小时。
   982→     *    因此写入前按**业务时区 Asia/Shanghai** 归一：DATE → `yyyy-MM-dd`，
   983→     *    DATETIME → `yyyy-MM-dd HH:mm:ss`。纯日期/本地时间字符串原样保留；
   984→     *    不可解析字符串原样透传（交给 DB 报 400 兜底）。
   985→     */
   986→    private void normalizeDateTimeColumns(Map<String, Object> data, List<ColumnConfig> columns) {
   987→        Set<String> dateKeys = new HashSet<>();
   988→        Set<String> dateTimeKeys = new HashSet<>();
   989→        for (ColumnConfig c : columns) {
   990→            String t = c.getColumnType() == null ? "" : c.getColumnType().toUpperCase();
   991→            if ("DATE".equals(t)) {
   992→                dateKeys.add(c.getKey());
   993→            } else if ("DATETIME".equals(t) || "TIMESTAMP".equals(t)) {
   994→                dateTimeKeys.add(c.getKey());
   995→            }
   996→        }
   997→        if (dateKeys.isEmpty() && dateTimeKeys.isEmpty()) {
   998→            return;
   999→        }
  1000→        for (Map.Entry<String, Object> e : data.entrySet()) {
  1001→            if (!(e.getValue() instanceof String s)) {
  1002→                continue;
  1003→            }
  1004→            String trimmed = s.trim();
  1005→            if (trimmed.isEmpty()) {
  1006→                continue;
  1007→            }
  1008→            if (dateKeys.contains(e.getKey())) {
  1009→                e.setValue(normalizeDateColumnValue(trimmed, true));
  1010→            } else if (dateTimeKeys.contains(e.getKey())) {
  1011→                e.setValue(normalizeDateColumnValue(trimmed, false));
  1012→            }
  1013→        }
  1014→    }

  1016→    /** 纯日期文本（本地语义，不做时区换算）。 */
  1017→    private static final java.util.regex.Pattern PLAIN_DATE_RE =
  1018→            java.util.regex.Pattern.compile("^\\d{4}-\\d{2}-\\d{2}$");
  1019→    /** 本地时间文本（`yyyy-MM-dd HH:mm[:ss]`，无时区语义）。 */
  1020→    private static final java.util.regex.Pattern PLAIN_DATETIME_RE =
  1021→            java.util.regex.Pattern.compile("^\\d{4}-\\d{2}-\\d{2} \\d{2}:\\d{2}(:\\d{2})?$");
  1022→
  1023→    /**
  1024→     * 单值归一（对齐 Node `normalizeDateColumnValue`）：
  1025→     * 纯日期原样（DATE）/补零点（DATETIME）；本地时间补秒；
  1026→     * 可解析时刻按 Asia/Shanghai 换算；不可解析原样。
  1027→     */
  1028→    private String normalizeDateColumnValue(String trimmed, boolean dateOnly) {
  1029→        if (PLAIN_DATE_RE.matcher(trimmed).matches()) {
  1030→            return dateOnly ? trimmed : trimmed + " 00:00:00";
  1031→        }
  1032→        if (!dateOnly && PLAIN_DATETIME_RE.matcher(trimmed).matches()) {
  1033→            return trimmed.length() == 16 ? trimmed + ":00" : trimmed;
  1034→        }
  1035→        java.time.Instant instant = tryParseInstant(trimmed);
  1036→        if (instant == null) {
  1037→            return trimmed;
  1038→        }
  1039→        java.time.ZonedDateTime zdt = instant.atZone(BIZ_ZONE);
  1040→        java.time.LocalDate date = zdt.toLocalDate();
  1041→        if (dateOnly) {
  1042→            return date.format(java.time.format.DateTimeFormatter.ISO_LOCAL_DATE);
  1043→        }
  1044→        return date.format(java.time.format.DateTimeFormatter.ISO_LOCAL_DATE)
  1045→                + " " + zdt.toLocalTime().format(java.time.format.DateTimeFormatter.ofPattern("HH:mm:ss"));
  1046→    }

  1048→    /** 宽松时刻解析：ISO 日期时间（含 Z/偏移/纳秒）→ Instant；失败 null。 */
  1049→    private java.time.Instant tryParseInstant(String s) {
  1050→        try {
  1051→            return java.time.OffsetDateTime.parse(s).toInstant();
  1052→        } catch (java.time.format.DateTimeParseException ignored) {
  1053→            // 落到下一格式
  1054→        }
  1055→        try {
  1056→            return java.time.LocalDateTime.parse(s).atZone(BIZ_ZONE).toInstant();
  1057→        } catch (java.time.format.DateTimeParseException ignored) {
  1058→            return null;
  1059→        }
  1060→    }

  1110→    /**
  1111→     * 子表行增量 diff（update 场景）。
  1112→     */
  1113→    private void diffSubRows(BizDataContext.SubTableDef def, String bizId, List<?> rows) {
  1114→        if (rows.size() > MAX_SUB_ROWS) {
  1115→            throw new BusinessException(400, "子表行数超限（最多 " + MAX_SUB_ROWS + " 行）: " + def.tableName());
  1116→        }
  1117→        List<Map<String, Object>> existing = readSubRows(def, bizId);
  1118→        Map<String, Map<String, Object>> existingById = new LinkedHashMap<>();
  1119→        for (Map<String, Object> r : existing) {
  1120→            existingById.put(String.valueOf(r.get("id")), r);
  1121→        }
  1122→
  1123→        Set<String> keepIds = new HashSet<>();
  1124→        int sortNo = 0;
  1125→        for (Object row : rows) {
  1126→            Map<String, Object> m = toRowMap(row, def.tableName());
  1127→            Object rawId = m.get("id");
  1128→            String rowId = rawId == null ? null : String.valueOf(rawId);
  1129→            if (rowId != null && existingById.containsKey(rowId)) {
  1130→                Map<String, Object> cur = existingById.get(rowId);
  1131→                boolean changed = def.subKeys().stream()
  1132→                        .anyMatch(k -> !Objects.equals(cur.get(k), m.get(k)));
  1133→                if (changed || !Objects.equals(cur.get("sort_no"), sortNo)) {
  1134→                    StringBuilder set = new StringBuilder();
  1135→                    List<Object> params = new ArrayList<>();
  1136→                    for (String k : def.subKeys()) {
  1137→                        set.append(k).append(" = ?, ");
  1138→                        params.add(m.get(k));
  1139→                    }
  1140→                    set.append("sort_no = ?");
  1141→                    params.add(sortNo);
  1142→                    params.add(tenantProvider.getTenantId());
  1143→                    params.add(bizId);
  1144→                    params.add(rowId);
  1145→                    jdbcTemplate.update("UPDATE " + def.tableName() + " SET " + set
  1146→                            + " WHERE tenant_id = ? AND biz_id = ? AND id = ?", params.toArray());
  1147→                }
  1148→                keepIds.add(rowId);
  1149→            } else {
  1150→                // 新行：剥离客户端传入的 id，走内部生成
  1151→                Map<String, Object> newRow = new LinkedHashMap<>(m);
  1152→                newRow.remove("id");
  1153→                insertOneSubRow(def, bizId, newRow, sortNo);
  1154→            }
  1155→            sortNo++;
  1156→        }
  1157→
  1158→        // 删除请求中不存在的现有行
  1159→        if (existingById.size() > keepIds.size()) {
  1160→            List<Object> params = new ArrayList<>();
  1161→            params.add(tenantProvider.getTenantId());
  1162→            params.add(bizId);
  1163→            StringBuilder ph = new StringBuilder();
  1164→            for (String id : existingById.keySet()) {
  1165→                if (!keepIds.contains(id)) {
  1166→                    if (ph.length() > 0) {
  1167→                        ph.append(",");
  1168→                    }
  1169→                    ph.append("?");
  1170→                    params.add(id);
  1171→                }
  1172→            }
  1173→            jdbcTemplate.update("DELETE FROM " + def.tableName()
  1174→                    + " WHERE tenant_id = ? AND biz_id = ? AND id IN (" + ph + ")", params.toArray());
  1175→        }
  1176→    }

  1188→     * 插入单行子表数据（id/biz_id/tenant_id/sort_no/version + 子业务列）。

  1190→    private Map<String, Object> insertOneSubRow(BizDataContext.SubTableDef def, String bizId,
  1191→                                                Map<String, Object> m, int sortNo) {
  1192→        String rowId = UUID.randomUUID().toString().replace("-", "");
  1193→        List<Object> params = new ArrayList<>();
  1194→        params.add(rowId);
  1195→        params.add(bizId);
  1196→        params.add(tenantProvider.getTenantId());
  1197→        params.add(sortNo);
  1198→        params.add(1); // version
  1199→
  1200→        StringBuilder cols = new StringBuilder("id, biz_id, tenant_id, sort_no, version");
  1201→        StringBuilder vals = new StringBuilder("?, ?, ?, ?, ?");
  1202→        for (String k : def.subKeys()) {
  1203→            cols.append(", ").append(k);
  1204→            vals.append(", ?");
  1205→            params.add(m.get(k));
  1206→        }
  1207→        jdbcTemplate.update("INSERT INTO " + def.tableName() + " (" + cols + ") VALUES (" + vals + ")",
  1208→                params.toArray());
  1209→
  1210→        Map<String, Object> inserted = new LinkedHashMap<>(m);
  1211→        inserted.put("id", rowId);
  1212→        inserted.put("sort_no", sortNo);
  1213→        return inserted;
  1214→    }
