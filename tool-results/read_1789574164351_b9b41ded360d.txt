    93→    @Override
    94→    public DataSourceMetadata metadata(DataSourceDefinition ds) {
    95→        return switch (ds.getType()) {
    96→            case "FORM" -> {
    97→                List<ColumnConfig> cols = new ArrayList<>(formDefService.getBusinessColumnsByKey(ds.getFormKey()));
    98→                FormQueryConfig cfg = FormQueryConfig.parse(ds.getParams(), objectMapper);
    99→                if (cfg.isConfigMode()) {
   100→                    appendJoinColumns(cols, cfg.joins());
   101→                } else if (cfg.isSqlMode()) {
   102→                    appendDeclaredColumns(cols, cfg.columns());
   103→                }
   104→                SortableResolver.resolve(cols);
   105→                DataSourceMetadata m = new DataSourceMetadata(cols, true);
   106→                m.setFormKey(ds.getFormKey());
   107→                yield m;
   108→            }
   109→            case "WORKFLOW" -> {
   110→                List<ColumnConfig> cols = workflowQueryService.columnsFor(ds.getFormKey());
   111→                SortableResolver.resolve(cols);
   112→                DataSourceMetadata m = new DataSourceMetadata(cols, false);
   113→                m.setFormKey(ds.getFormKey());
   114→                yield m;
   115→            }
   116→            case "SYSTEM" -> {
   117→                List<ColumnConfig> cols = "user-tree".equals(ds.getSourceKey())
   118→                        ? copyWithSortableFalse(USER_COLUMNS)
   119→                        : copyWithSortableFalse(DEPT_COLUMNS);
   120→                yield new DataSourceMetadata(cols, false);
   121→            }
   122→            case "API" -> {
   123→                DataSourceMetadata m = apiMetadata(ds);
   124→                DataSourceMetadata copy = new DataSourceMetadata(copyWithSortableFalse(m.getColumns()), m.isWritable());
   125→                copy.setFormKey(m.getFormKey());
   126→                yield copy;
   127→            }
   128→            case "SQL" -> {
   129→                // 字段元数据 = params.columns（单一来源；探测/主表单覆盖/行内编辑由前端写入，无表单列默认合并）
   130→                FormQueryConfig cfg = FormQueryConfig.parse(ds.getParams(), objectMapper);
   131→                List<ColumnConfig> cols = cfg.isConfigMode() ? List.of() : cfg.columns();
   132→                boolean writable = ds.getFormKey() != null && !ds.getFormKey().isBlank();
   133→                DataSourceMetadata m = new DataSourceMetadata(cols, writable);
   134→                if (ds.getFormKey() != null && !ds.getFormKey().isBlank()) {
   135→                    m.setFormKey(ds.getFormKey());
   136→                }
   137→                yield m;
   138→            }
   139→            default -> throw unsupported(ds, "metadata");
   140→        };
   141→    }

   143→    /** 复制列并强制标记 sortable=false（SYSTEM/API 数据源不可排序；避免就地污染共享常量列）。 */
   144→    private List<ColumnConfig> copyWithSortableFalse(List<ColumnConfig> src) {
   145→        List<ColumnConfig> out = new ArrayList<>();
   146→        if (src != null) {
   147→            for (ColumnConfig c : src) {
   148→                ColumnConfig copy = new ColumnConfig();
   149→                copy.setKey(c.getKey());
   150→                copy.setLabel(c.getLabel());
   151→                copy.setColumnType(c.getColumnType());
   152→                copy.setSortable(false);
   153→                out.add(copy);
   154→            }
   155→        }
   156→        return out;
   157→    }

   159→    @Override
   160→    public BizDataPageVO query(DataSourceDefinition ds, BizDataQueryRequest req) {
   161→        return switch (ds.getType()) {
   162→            case "FORM" -> {
   163→                router.resolve(ds, "list");
   164→                FormQueryConfig cfg = FormQueryConfig.parse(ds.getParams(), objectMapper);
   165→                if (cfg.isConfigMode()) {
   166→                    yield bizDataService.queryJoin(ds.getFormKey(), req, cfg.joins());
   167→                }
   168→                if (cfg.isSqlMode()) {
   169→                    yield bizDataService.querySql(ds.getFormKey(), req, cfg);
   170→                }
   171→                yield bizDataService.query(ds.getFormKey(), req);
   172→            }
   173→            case "WORKFLOW" -> workflowQueryService.query(ds.getFormKey(), req);
   174→            case "SYSTEM" -> {
   175→                router.resolve(ds, "list");
   176→                yield systemQuery(ds, req);
   177→            }
   178→            case "API" -> apiQuery(ds, req);
   179→            case "SQL" -> {
   180→                router.resolve(ds, "list");
   181→                FormQueryConfig cfg = FormQueryConfig.parse(ds.getParams(), objectMapper);
   182→                if (cfg.isVisualMode() || cfg.isSqlMode()) {
   183→                    // 绕过表单 covering handler：SQL 数据源执行管理员显式 SQL，不被绑定表单业务定制劫持
   184→                    yield bizDataService.querySqlRaw(ds.getFormKey(), req, cfg);
   185→                }
   186→                throw new BusinessException(400, "SQL 数据源缺少查询配置");
   187→            }
   188→            default -> throw unsupported(ds, "query");
   189→        };
   190→    }

   276→    /** 追加 config 模式 JOIN 虚拟列到 metadata；与主表列 key 重复的 virtualKey 跳过（防御性去重）。 */
   277→    private void appendJoinColumns(List<ColumnConfig> cols, List<JoinConfig> joins) {
   278→        Set<String> existing = cols.stream().map(ColumnConfig::getKey).collect(Collectors.toSet());
   279→        for (JoinConfig j : joins) {
   280→            if (j.virtualKey() == null || j.virtualKey().isBlank() || existing.contains(j.virtualKey())) {
   281→                continue;
   282→            }
   283→            ColumnConfig c = new ColumnConfig();
   284→            c.setKey(j.virtualKey());
   285→            c.setLabel(j.label() == null || j.label().isBlank() ? j.virtualKey() : j.label());
   286→            c.setColumnType(resolveJoinColumnType(j));
   287→            c.setSortable(j.sortable());
   288→            c.setFilterable(j.filterable());
   289→            cols.add(c);
   290→            existing.add(c.getKey());
   291→        }
   292→    }
   293→
   294→    /** 追加 sql 模式管理员声明列到 metadata；与主表列 key 重复的声明列跳过。 */
   295→    private void appendDeclaredColumns(List<ColumnConfig> cols, List<ColumnConfig> declared) {
   296→        Set<String> existing = cols.stream().map(ColumnConfig::getKey).collect(Collectors.toSet());
   297→        for (ColumnConfig d : declared) {
   298→            if (d.getKey() == null || d.getKey().isBlank() || existing.contains(d.getKey())) {
   299→                continue;
   300→            }
   301→            ColumnConfig c = new ColumnConfig();
   302→            c.setKey(d.getKey());
   303→            c.setLabel(d.getLabel() == null || d.getLabel().isBlank() ? d.getKey() : d.getLabel());
   304→            c.setColumnType(d.getColumnType());
   305→            c.setSortable(d.getSortable());
   306→            c.setFilterable(d.getFilterable());
   307→            cols.add(c);
   308→            existing.add(c.getKey());
   309→        }
   310→    }

   312→    /** 虚拟列 columnType：目标表单 joinField 的列类型；查不到 fallback "VARCHAR"。 */
   313→    private String resolveJoinColumnType(JoinConfig j) {
   314→        try {
   315→            List<ColumnConfig> target = formDefService.getBusinessColumnsByKey(j.targetFormKey());
   316→            if (target != null) {
   317→                for (ColumnConfig c : target) {
   318→                    if (j.joinField().equals(c.getKey()) && c.getColumnType() != null && !c.getColumnType().isBlank()) {
   319→                        return c.getColumnType().toUpperCase();
   320→                    }
   321→                }
   322→            }
   323→        } catch (BusinessException ignored) {
   324→            // 目标表单不可解析时回退
   325→        }
   326→        return "VARCHAR";
   327→    }

   346→    private BizDataPageVO queryDeptTree() {
   347→        List<BizDataVO> rows = new ArrayList<>();
   348→        for (TreeNode node : organizationService.tree()) {
   349→            collectNode(node, rows);
   350→        }
   351→        return new BizDataPageVO(rows, rows.size(), 0, rows.size());
   352→    }

   354→    private void collectNode(TreeNode node, List<BizDataVO> out) {
   355→        Map<String, Object> data = new LinkedHashMap<>();
   356→        data.put("id", String.valueOf(node.id()));
   357→        data.put("parentId", node.parentId() == null ? "" : String.valueOf(node.parentId()));
   358→        data.put("label", node.label() == null ? "" : node.label());
   359→        data.put("code", node.code() == null ? "" : node.code());
   360→        out.add(new BizDataVO(String.valueOf(node.id()), data, null, null, null));
   361→        if (node.children() != null) {
   362→            for (TreeNode child : node.children()) {
   363→                collectNode(child, out);
   364→            }
   365→        }
   366→    }

   368→    private BizDataPageVO queryUsers(BizDataQueryRequest req) {
   369→        int userServicePage = Math.max(req.getPage(), 1);
   370→        UserQueryRequest query = new UserQueryRequest(
   371→                req.getKeyword(), null, null, null, null, null, userServicePage, req.getSize());

   373→        List<BizDataVO> rows = new ArrayList<>();
   374→        for (UserVO u : page.getRows() == null ? List.<UserVO>of() : page.getRows()) {
   375→            Map<String, Object> data = new LinkedHashMap<>();
   376→            data.put("id", String.valueOf(u.id()));
   377→            data.put("username", u.username());
   378→            data.put("nickname", u.nickname() == null ? "" : u.nickname());
   379→            data.put("orgId", u.orgId() == null ? "" : String.valueOf(u.orgId()));
   380→            data.put("orgName", u.orgName() == null ? "" : u.orgName());

   389→    private DataSourceMetadata apiMetadata(DataSourceDefinition ds) {
   390→        Map<String, Object> params = parseParams(ds);
   391→        List<ColumnConfig> columns = new ArrayList<>();
   392→        JsonNode colsNode = jsonNode(params, "columns");
   393→        if (colsNode != null && colsNode.isArray()) {
   394→            for (JsonNode node : colsNode) {
   395→                columns.add(objectMapper.convertValue(node, ColumnConfig.class));
   396→            }
   397→        }
   398→        DataSourceMetadata m = new DataSourceMetadata(columns, writable(params));
   399→        m.setFormKey(ds.getFormKey());
   400→        return m;
   401→    }

   403→    private BizDataPageVO apiQuery(DataSourceDefinition ds, BizDataQueryRequest req) {
   404→        Map<String, Object> params = parseParams(ds);
   405→        Map<String, Object> op = operation(params, "list");
   406→        if (op == null) {
   407→            throw new BusinessException(400, "API 数据源未配置 list 操作: " + ds.getName());
   408→        }
   409→        String action = requireAction(op, "list");
   410→        String method = opMethod(op);
   411→        Map<String, Object> vars = new HashMap<>();
   412→        if (params.get("data") instanceof Map<?, ?> data) {
   413→            for (Map.Entry<?, ?> e : data.entrySet()) {
   414→                vars.put(String.valueOf(e.getKey()), e.getValue());
   415→            }
   416→        }
   417→        vars.put("page", req.getPage());
   418→        vars.put("size", req.getSize());
   419→        String kw = req.getKeyword();
   420→        vars.put("keyword", kw == null ? "" : kw);
   421→        String searchParam = str(params.get("searchParam"));
   422→        if (kw != null && !kw.isBlank() && searchParam != null && !searchParam.isBlank()) {
   423→            vars.put(searchParam, kw);
   424→        }
   425→        Object raw = httpExecutor.execute(action, method, headers(params), List.of(), List.of(),
   426→                vars, DEFAULT_TIMEOUT_MS, DEFAULT_TIMEOUT_MS, DEFAULT_RETRY);
   427→        return toPageVO(raw, op, req);
   428→    }

   482→    @SuppressWarnings("unchecked")
   483→    private Map<String, Object> parseParams(DataSourceDefinition ds) {
   484→        String json = ds.getParams();
   485→        if (json == null || json.isBlank()) {
   486→            throw new BusinessException(400, "API 数据源缺少 params: " + ds.getName());
   487→        }
   488→        try {
   489→            JsonNode node = objectMapper.readTree(json);
   490→            if (!node.isObject()) {
   491→                throw new BusinessException(400, "API 数据源 params 必须是 JSON 对象: " + ds.getName());
   492→            }
   493→            return objectMapper.convertValue(node, Map.class);
   494→        } catch (JsonProcessingException e) {
   495→            throw new BusinessException(400, "API 数据源 params 不是合法 JSON: " + ds.getName());
   496→        }
   497→    }

   499→    @SuppressWarnings("unchecked")
   500→    private Map<String, Object> operation(Map<String, Object> params, String name) {
   501→        Object op = params.get(name);
   502→        if (op instanceof Map<?, ?> m) return (Map<String, Object>) m;
   503→        if ("list".equals(name) && params.get("action") != null) {
   504→            Map<String, Object> legacy = new LinkedHashMap<>();
   505→            legacy.put("action", params.get("action"));
   506→            legacy.put("method", params.get("method"));
   507→            legacy.put("parse", params.get("parse"));
   508→            legacy.put("totalParse", params.get("totalParse"));
   509→            return legacy;
   510→        }
   511→        return null;
   512→    }

   545→    private JsonNode jsonNode(Map<String, Object> params, String key) {
   546→        Object v = params.get(key);
   547→        if (v == null) return null;
   548→        if (v instanceof JsonNode jn) return jn;
   549→        try { return objectMapper.valueToTree(v); } catch (IllegalArgumentException e) { return null; }
   550→    }

   552→    private JsonNode toJsonNode(Object raw) {
   553→        if (raw instanceof String s) {
   554→            try { return objectMapper.readTree(s); }
   555→            catch (JsonProcessingException e) { throw new BusinessException(400, "外部 API 响应不是合法 JSON"); }
   556→        }
   557→        return objectMapper.valueToTree(raw);
   558→    }

   560→    private JsonNode walkPath(JsonNode root, String path) {
   561→        JsonNode cur = root;
   562→        for (String part : path.split("\\.")) {
   563→            if (cur == null || cur.isMissingNode() || !cur.isObject()) return null;
   564→            cur = cur.get(part);
   565→        }
   566→        return cur;
   567→    }

   569→    private BizDataPageVO toPageVO(Object raw, Map<String, Object> op, BizDataQueryRequest req) {
   570→        int normalizedPage = Math.max(req.getPage(), 1);
   571→        JsonNode root = toJsonNode(raw);
   572→        JsonNode recordsNode = root;
   573→        String parse = str(op.get("parse"));
   574→        if (parse != null && !parse.isBlank()) {
   575→            recordsNode = walkPath(root, parse);
   576→        }
   577→        if (recordsNode == null || !recordsNode.isArray()) {
   578→            return new BizDataPageVO(List.of(), 0, normalizedPage, req.getSize());
   579→        }
   580→        List<BizDataVO> records = new ArrayList<>();
   581→        for (JsonNode item : recordsNode) {
   582→            Map<String, Object> data = objectMapper.convertValue(item, Map.class);
   583→            Object id = item.has("id") ? item.get("id").asText() : null;
   584→            records.add(new BizDataVO(id == null ? "" : String.valueOf(id), data, null, null, null));
   585→        }
   586→        long total = records.size();
   587→        String totalParse = str(op.get("totalParse"));
   588→        if (totalParse != null && !totalParse.isBlank()) {
   589→            JsonNode totalNode = walkPath(root, totalParse);
   590→            if (totalNode != null && totalNode.isNumber()) total = totalNode.asLong();
   591→        }
   592→        return new BizDataPageVO(records, total, normalizedPage, req.getSize());
   593→    }
