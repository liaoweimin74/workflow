   143→    /**
   144→     * 查询待办任务列表，返回 TaskTodoVO 分页（含关联字段）。
   145→     *
   146→     * <p>批量查询 ProcessInstance + UserService 避免 N+1。
   147→     *
   148→     * @param assignee 办理人
   149→     * @param pageable 分页
   150→     * @param filter   过滤参数（processName, initiator, createTimeStart/End）
   151→     * @return TaskTodoVO 分页
   152→     */
   153→    public Page<TaskTodoVO> listTodoTasksVO(String assignee, Pageable pageable, TaskTodoFilter filter) {
   154→        String tenantId = tenantProvider.getTenantId();
   155→        var query = flowableTaskService.createTaskQuery()
   156→                .taskTenantId(tenantId)
   157→                .taskAssignee(assignee)
   158→                .orderByTaskCreateTime()
   159→                .desc();
   160→
   161→        // createTime 范围过滤（Flowable 原生支持 taskCreatedAfter/Before）
   162→        if (filter != null) {
   163→            if (filter.createTimeStart() != null) {
   164→                query.taskCreatedAfter(parseDate(filter.createTimeStart()));
   165→            }
   166→            if (filter.createTimeEnd() != null) {
   167→                query.taskCreatedBefore(parseDate(filter.createTimeEnd()));
   168→            }
   169→        }
   170→
   171→        long total = query.count();
   172→        List<Task> tasks = query.listPage((int) pageable.getOffset(), pageable.getPageSize());
   173→
   174→        List<TaskTodoVO> vos = assembleTodoVOs(tasks);
   175→
   176→        // 内存过滤 processName / initiator（Flowable TaskQuery 不直接支持）
   177→        if (filter != null) {
   178→            vos = vos.stream()
   179→                    .filter(vo -> matchesProcessName(vo, filter.processName()))
   180→                    .filter(vo -> matchesInitiator(vo, filter.initiator()))
   181→                    .toList();
   182→        }
   183→
   184→        return new PageImpl<>(vos, pageable, total);
   185→    }

   187→    /**
   188→     * 查询已办任务列表，返回 TaskDoneVO 分页（含关联字段）。
   189→     *
   190→     * @param userId   办理人
   191→     * @param pageable 分页
   192→     * @param filter   过滤参数（processName, initiator, endTimeStart/End, approveResult）
   193→     * @return TaskDoneVO 分页
   194→     */
   195→    public Page<TaskDoneVO> listHistoricTasksVO(String userId, Pageable pageable, TaskDoneFilter filter) {
   196→        String tenantId = tenantProvider.getTenantId();
   197→        var query = historyService.createHistoricTaskInstanceQuery()
   198→                .taskTenantId(tenantId)
   199→                .taskAssignee(userId)
   200→                .finished()
   201→                .orderByHistoricTaskInstanceEndTime()
   202→                .desc();
   203→
   204→        // endTime 范围过滤
   205→        if (filter != null) {
   206→            if (filter.endTimeStart() != null) {
   207→                query.taskCompletedAfter(parseDate(filter.endTimeStart()));
   208→            }
   209→            if (filter.endTimeEnd() != null) {
   210→                query.taskCompletedBefore(parseDate(filter.endTimeEnd()));
   211→            }
   212→        }
   213→
   214→        long total = query.count();
   215→        List<HistoricTaskInstance> tasks = new ArrayList<>(query.listPage((int) pageable.getOffset(), pageable.getPageSize()));
   216→
   217→        // 补充：用户操作过但未完成的任务（转办/委派/加签/转签后任务已换人），
   218→        // 从 wf_task_comment 按 user_id 反查 taskId，也应出现在已办中
   219→        Set<String> commentTaskIds = commentRepository.findByUserId(userId).stream()
   220→                .map(WfTaskComment::getTaskId)
   221→                .filter(Objects::nonNull)
   222→                .collect(Collectors.toSet());
   223→
   224→        Set<String> existingTaskIds = tasks.stream()
   225→                .map(HistoricTaskInstance::getId)
   226→                .collect(Collectors.toSet());
   227→
   228→        // 只查不在已办列表中的任务（去重）
   229→        Set<String> missingTaskIds = commentTaskIds.stream()
   230→                .filter(id -> !existingTaskIds.contains(id))
   231→                .collect(Collectors.toSet());
   232→
   233→        if (!missingTaskIds.isEmpty()) {
   234→            List<HistoricTaskInstance> commentTasks = historyService
   235→                    .createHistoricTaskInstanceQuery()
   236→                    .taskTenantId(tenantId)
   237→                    .taskIds(missingTaskIds)
   238→                    .orderByHistoricTaskInstanceEndTime()
   239→                    .desc()
   240→                    .list();
   241→            tasks.addAll(commentTasks);
   242→        }
   243→
   244→        List<TaskDoneVO> vos = assembleDoneVOs(tasks);
   245→
   246→        // 内存过滤 processName / initiator / approveResult
   247→        if (filter != null) {
   248→            vos = vos.stream()
   249→                    .filter(vo -> matchesProcessName(vo, filter.processName()))
   250→                    .filter(vo -> matchesInitiator(vo, filter.initiator()))
   251→                    .filter(vo -> filter.approveResult() == null || filter.approveResult().equals(vo.getApproveResult()))
   252→                    .toList();
   253→        }
   254→
   255→        // 排序：按 endTime 或 createTime 倒序
   256→        vos = vos.stream()
   257→                .sorted(Comparator.comparing(TaskDoneVO::getEndTime,
   258→                        Comparator.nullsFirst(Comparator.naturalOrder())).reversed())
   259→                .toList();
   260→
   261→        return new PageImpl<>(vos, pageable, Math.max(total, vos.size()));
   262→    }

   264→    // ==================== VO 组装（批量查询，避免 N+1）====================
   265→
   266→    private List<TaskTodoVO> assembleTodoVOs(List<Task> tasks) {
   267→        if (tasks.isEmpty()) {
   268→            return List.of();
   269→        }
   270→
   271→        // 1. 批量收集 processInstanceIds
   272→        Set<String> processInstanceIds = tasks.stream()
   273→                .map(Task::getProcessInstanceId)
   274→                .filter(Objects::nonNull)
   275→                .collect(Collectors.toSet());
   276→
   277→        // 2. 批量查询 ProcessInstance
   278→        Map<String, ProcessInstance> piMap = batchQueryProcessInstances(processInstanceIds);
   279→
   280→        // 3. 批量收集 processDefinitionIds → 查 ProcessDefinition 名称
   281→        Set<String> processDefinitionIds = tasks.stream()
   282→                .map(Task::getProcessDefinitionId)
   283→                .filter(Objects::nonNull)
   284→                .collect(Collectors.toSet());
   285→        Map<String, ProcessDefinition> pdMap = batchQueryProcessDefinitions(processDefinitionIds);
   286→
   287→        // 4. 批量查询 initiator 变量 + 解析用户名
   288→        Map<String, String> initiatorMap = batchQueryInitiators(processInstanceIds, piMap);
   289→        Map<String, String> initiatorNameMap = batchQueryInitiatorNames(initiatorMap.values());
   290→
   291→        // 5. 批量查询催办标记（哪些 task 已有催办记录）
   292→        Set<String> taskIds = tasks.stream().map(Task::getId).filter(Objects::nonNull).collect(Collectors.toSet());
   293→        Set<String> remindedTaskIds = batchQueryRemindedTaskIds(taskIds);
   294→
   295→        // 6. 组装 VO
   296→        return tasks.stream().map(task -> {
   297→            TaskTodoVO vo = new TaskTodoVO();
   298→            vo.setTaskId(task.getId());
   299→            vo.setProcessInstanceId(task.getProcessInstanceId());
   300→            vo.setProcessDefinitionId(task.getProcessDefinitionId());
   301→            vo.setCurrentNodeName(task.getName());
   302→            vo.setAssignee(task.getAssignee());
   303→            if (task.getCreateTime() != null) {
   304→                vo.setCreateTime(formatDate(task.getCreateTime()));
   305→            }
   306→
   307→            ProcessInstance pi = piMap.get(task.getProcessInstanceId());
   308→            if (pi != null) {
   309→                vo.setBusinessKey(pi.getBusinessKey());
   310→            }
   311→
   312→            ProcessDefinition pd = pdMap.get(task.getProcessDefinitionId());
   313→            if (pd != null) {
   314→                vo.setProcessName(pd.getName() != null ? pd.getName() : pd.getKey());
   315→            }
   316→
   317→            String initiator = initiatorMap.get(task.getProcessInstanceId());
   318→            vo.setInitiator(initiator);
   319→            vo.setInitiatorName(initiatorNameMap.get(initiator));
   320→
   321→            vo.setReminded(remindedTaskIds.contains(task.getId()));
   322→
   323→            return vo;
   324→        }).toList();
   325→    }

   327→    private List<TaskDoneVO> assembleDoneVOs(List<HistoricTaskInstance> tasks) {
   328→        if (tasks.isEmpty()) {
   329→            return List.of();
   330→        }
   331→
   332→        // 1. 批量收集 processInstanceIds
   333→        Set<String> processInstanceIds = tasks.stream()
   334→                .map(HistoricTaskInstance::getProcessInstanceId)
   335→                .filter(Objects::nonNull)
   336→                .collect(Collectors.toSet());
   337→
   338→        // 2. 批量查询 HistoricProcessInstance
   339→        Map<String, HistoricProcessInstance> piMap = batchQueryHistoricProcessInstances(processInstanceIds);
   340→
   341→        // 3. 批量收集 processDefinitionIds → 查 ProcessDefinition 名称
   342→        Set<String> processDefinitionIds = tasks.stream()
   343→                .map(HistoricTaskInstance::getProcessDefinitionId)
   344→                .filter(Objects::nonNull)
   345→                .collect(Collectors.toSet());
   346→        Map<String, ProcessDefinition> pdMap = batchQueryProcessDefinitions(processDefinitionIds);
   347→
   348→        // 4. 批量查询 initiator 变量 + 解析用户名
   349→        Map<String, String> initiatorMap = batchQueryHistoricInitiators(processInstanceIds);
   350→        Map<String, String> initiatorNameMap = batchQueryInitiatorNames(initiatorMap.values());
   351→
   352→        // 4b. 批量查询审批意见（approveResult）
   353→        Map<String, String> approveResultMap = new HashMap<>();
   354→        for (HistoricTaskInstance task : tasks) {
   355→            if (task.getId() == null) continue;
   356→            List<WfTaskComment> comments = commentRepository.findByTaskId(task.getId());
   357→            if (!comments.isEmpty()) {
   358→                // 取最后一条操作
   359→                approveResultMap.put(task.getId(), comments.get(comments.size() - 1).getAction());
   360→            }
   361→        }
   362→
   363→        // 4c. 批量查询流程当前待办节点（currentNode，区别于办理节点 currentNodeName）
   364→        Map<String, String> currentNodeMap = new HashMap<>();
   365→        for (String pid : processInstanceIds) {
   366→            List<Task> activeTasks = flowableTaskService.createTaskQuery()
   367→                    .processInstanceId(pid)
   368→                    .active()
   369→                    .list();
   370→            if (activeTasks != null && !activeTasks.isEmpty()) {
   371→                String names = activeTasks.stream()
   372→                        .map(Task::getName)
   373→                        .filter(n -> n != null && !n.isBlank())
   374→                        .distinct()
   375→                        .collect(Collectors.joining("、"));
   376→                if (!names.isBlank()) {
   377→                    currentNodeMap.put(pid, names);
   378→                }
   379→            }
   380→        }
   381→
   382→        // 5. 组装 VO
   383→        return tasks.stream().map(task -> {
   384→            TaskDoneVO vo = new TaskDoneVO();
   385→            vo.setTaskId(task.getId());
   386→            vo.setProcessInstanceId(task.getProcessInstanceId());
   387→            vo.setProcessDefinitionId(task.getProcessDefinitionId());
   388→            vo.setCurrentNodeName(task.getName());
   389→            vo.setAssignee(task.getAssignee());
   390→            if (task.getStartTime() != null) {
   391→                vo.setCreateTime(formatDate(task.getStartTime()));
   392→            }
   393→            if (task.getEndTime() != null) {
   394→                vo.setEndTime(formatDate(task.getEndTime()));
   395→            }
   396→
   397→            HistoricProcessInstance pi = piMap.get(task.getProcessInstanceId());
   398→            if (pi != null) {
   399→                vo.setBusinessKey(pi.getBusinessKey());
   400→            }
   401→
   402→            ProcessDefinition pd = pdMap.get(task.getProcessDefinitionId());
   403→            if (pd != null) {
   404→                vo.setProcessName(pd.getName() != null ? pd.getName() : pd.getKey());
   405→            }
   406→
   407→            String initiator = initiatorMap.get(task.getProcessInstanceId());
   408→            vo.setInitiator(initiator);
   409→            vo.setInitiatorName(initiatorNameMap.get(initiator));
   410→
   411→            // approveResult: 从 wf_task_comment 取该任务的最新操作
   412→            vo.setApproveResult(approveResultMap.get(task.getId()));
   413→
   414→            // currentNode: 流程当前待办节点（非办理节点）
   415→            vo.setCurrentNode(currentNodeMap.get(task.getProcessInstanceId()));
   416→
   417→            return vo;
   418→        }).toList();
   419→    }

   456→    /**
   457→     * 批量查询运行中流程实例的 initiator 变量。
   458→     */
   459→    private Map<String, String> batchQueryInitiators(Set<String> processInstanceIds,
   460→                                                     Map<String, ProcessInstance> piMap) {
   461→        if (processInstanceIds.isEmpty()) {
   462→            return Map.of();
   463→        }
   464→        Map<String, String> result = new HashMap<>();
   465→        for (String piId : processInstanceIds) {
   466→            try {
   467→                Object initiator = runtimeService.getVariable(piId, "initiator");
   468→                if (initiator != null) {
   469→                    result.put(piId, String.valueOf(initiator));
   470→                }
   471→            } catch (Exception e) {
   472→                // 流程实例可能已结束，变量不可查
   473→            }
   474→        }
   475→        return result;
   476→    }

   478→    /**
   479→     * 批量查询历史流程实例的 initiator 变量。
   480→     */
   481→    private Map<String, String> batchQueryHistoricInitiators(Set<String> processInstanceIds) {
   482→        if (processInstanceIds.isEmpty()) {
   483→            return Map.of();
   484→        }
   485→        Map<String, String> result = new HashMap<>();
   486→        for (String piId : processInstanceIds) {
   487→            try {
   488→                HistoricVariableInstance hv = historyService.createHistoricVariableInstanceQuery()
   489→                        .processInstanceId(piId)
   490→                        .variableName("initiator")
   491→                        .singleResult();
   492→                if (hv != null && hv.getValue() != null) {
   493→                    result.put(piId, String.valueOf(hv.getValue()));
   494→                }
   495→            } catch (Exception e) {
   496→                // ignore
   497→            }
   498→        }
   499→        return result;
   500→    }

   502→    /**
   503→     * 批量查询用户姓名。
   504→     */
   505→    private Map<String, String> batchQueryInitiatorNames(Collection<String> initiatorIds) {
   506→        if (initiatorIds == null || initiatorIds.isEmpty()) {
   507→            return Map.of();
   508→        }
   509→        List<Long> userIds = initiatorIds.stream()
   510→                .filter(Objects::nonNull)
   511→                .map(id -> {
   512→                    try {
   513→                        return Long.parseLong(id);
   514→                    } catch (NumberFormatException e) {
   515→                        return null;
   516→                    }
   517→                })
   518→                .filter(Objects::nonNull)
   519→                .toList();
   520→
   521→        if (userIds.isEmpty()) {
   522→            return Map.of();
   523→        }
   524→
   525→        List<UserVO> users = userService.findByIds(userIds);
   526→        return users.stream()
   527→                .collect(Collectors.toMap(
   528→                        u -> String.valueOf(u.id()),
   529→                        u -> u.nickname() != null ? u.nickname() : u.username(),
   530→                        (a, b) -> a));
   531→    }

   607→    /**
   608→     * 从运行时 Task 构建 TaskDetailVO。
   609→     */
   610→    private TaskDetailVO buildTaskDetailFromRuntime(Task task) {
   611→        TaskDetailVO vo = new TaskDetailVO();
   612→        vo.setTaskId(task.getId());
   613→        vo.setName(task.getName());
   614→        vo.setDescription(task.getDescription());
   615→        vo.setAssignee(task.getAssignee());
   616→        // assignee userId → nickname
   617→        if (task.getAssignee() != null) {
   618→            Map<String, String> assigneeNameMap = batchQueryInitiatorNames(List.of(task.getAssignee()));
   619→            vo.setAssigneeName(assigneeNameMap.get(task.getAssignee()));
   620→        }
   621→        vo.setProcessInstanceId(task.getProcessInstanceId());
   622→        vo.setProcessDefinitionId(task.getProcessDefinitionId());
   623→        if (task.getCreateTime() != null) {
   624→            vo.setCreateTime(formatDate(task.getCreateTime()));
   625→        }
   626→
   627→        // ProcessInstance → businessKey + initiator（复用 batch 查询模式）
   628→        String processInstanceId = task.getProcessInstanceId();
   629→        if (processInstanceId != null) {
   630→            Set<String> piIdSet = Set.of(processInstanceId);
   631→
   632→            // 1. 复用 batchQueryProcessInstances 获取运行中 ProcessInstance
   633→            Map<String, ProcessInstance> piMap = batchQueryProcessInstances(piIdSet);
   634→            ProcessInstance pi = piMap.get(processInstanceId);
   635→
   636→            if (pi != null) {
   637→                // 流程运行中：直接取 businessKey + initiator 变量
   638→                vo.setBusinessKey(pi.getBusinessKey());
   639→
   640→                Map<String, String> initiatorMap = batchQueryInitiators(piIdSet, piMap);
   641→                String initiator = initiatorMap.get(processInstanceId);
   642→                if (initiator != null) {
   643→                    vo.setInitiator(initiator);
   644→                    Map<String, String> nameMap = batchQueryInitiatorNames(List.of(initiator));
   645→                    vo.setInitiatorName(nameMap.get(initiator));
   646→                }
   647→            } else {
   648→                // 2. 流程已结束：fallback 查 HistoricProcessInstance 获取 businessKey
   649→                try {
   650→                    HistoricProcessInstance hpi = historyService.createHistoricProcessInstanceQuery()
   651→                            .processInstanceId(processInstanceId)
   652→                            .singleResult();
   653→                    if (hpi != null) {
   654→                        vo.setBusinessKey(hpi.getBusinessKey());
   655→                    }
   656→                } catch (Exception e) {
   657→                    // 历史查询失败，忽略
   658→                }
   659→
   660→                // initiator 变量也从历史变量中获取
   661→                Map<String, String> initiatorMap = batchQueryHistoricInitiators(piIdSet);
   662→                String initiator = initiatorMap.get(processInstanceId);
   663→                if (initiator != null) {
   664→                    vo.setInitiator(initiator);
   665→                    Map<String, String> nameMap = batchQueryInitiatorNames(List.of(initiator));
   666→                    vo.setInitiatorName(nameMap.get(initiator));
   667→                }
   668→            }
   669→        }
   670→
   671→        // ProcessDefinition → processName
   672→        String processDefinitionId = task.getProcessDefinitionId();
   673→        if (processDefinitionId != null) {
   674→            Map<String, ProcessDefinition> pdMap = batchQueryProcessDefinitions(Set.of(processDefinitionId));
   675→            ProcessDefinition pd = pdMap.get(processDefinitionId);
   676→            if (pd != null) {
   677→                vo.setProcessName(pd.getName() != null ? pd.getName() : pd.getKey());
   678→                vo.setProcessVersion(pd.getVersion());
   679→            }
   680→
   681→            // formKey 从 BpmnModel 中当前任务的 UserTask 节点提取
   682→            String formKey = extractFormKey(processDefinitionId, task.getTaskDefinitionKey());
   683→            vo.setFormKey(formKey);
   684→
   685→// 判断是否为发起节点
   686→            try {
   687→                String initiatorNodeId = initiatorNodeResolver.resolve(processDefinitionId);
   688→                vo.setIsInitiatorTask(initiatorNodeId != null
   689→                        && initiatorNodeId.equals(task.getTaskDefinitionKey()));
   690→            } catch (Exception e) {
   691→                vo.setIsInitiatorTask(false);
   692→            }
   693→            // 表单字段权限 + 操作权限配置
   694→            FormConfigResult formConfig = extractFormConfig(processDefinitionId, task.getTaskDefinitionKey());
   695→            vo.setFieldPermissions(formConfig != null ? formConfig.getFieldPermissions() : null);
   696→            vo.setOperations(extractOperations(processDefinitionId, task.getTaskDefinitionKey()));
   697→        }
   698→
   699→        // variables
   700→        try {
   701→            Map<String, Object> variables = flowableTaskService.getVariables(task.getId());
   702→            vo.setVariables(variables);
   703→        } catch (Exception e) {
   704→            vo.setVariables(Map.of());
   705→        }
   706→
   707→        // mappedData：跨表单映射聚合（未配置映射时 merge 返回空 → 置 null）
   708→        if (task.getProcessDefinitionId() != null && task.getProcessInstanceId() != null) {
   709→            try {
   710→                Map<String, Object> mappedData = formDataMerger.merge(
   711→                        task.getProcessDefinitionId(), task.getTaskDefinitionKey(), task.getProcessInstanceId());
   712→                vo.setMappedData(mappedData.isEmpty() ? null : mappedData);
   713→            } catch (Exception e) {
   714→                log.warn("Failed to merge mappedData for task [{}]: {}", task.getId(), e.getMessage());
   715→                vo.setMappedData(null);
   716→            }
   717→        }
   718→
   719→        return vo;
   720→    }

   868→    /**
   869→     * 解析任务节点的表单配置（formDefId + fieldPermissions）。
   870→     *
   871→     * <p>解析逻辑：
   872→     * <ol>
   873→     *   <li>优先从节点配置（NodeConfig, nodeId=taskDefKey）读取 form.formDefId 和 form.fieldPermissions</li>
   874→     *   <li>节点未配置 formDefId 时，从流程级配置（NodeConfig, nodeId=__PROCESS__）读取 form.formDefId 和 form.fieldPermissions</li>
   875→     *   <li>均未配置时返回 null</li>
   876→     * </ol>
   877→     * 表单和字段权限作为整体从同一配置层取，不跨层合并。
   878→     *
   879→     * @param processDefinitionId 流程定义 ID
   880→     * @param taskDefinitionKey   任务定义键（BPMN 节点 ID）
   881→     * @return 表单配置，未找到时返回 null
   882→     */
   883→    public FormConfigResult extractFormConfig(String processDefinitionId, String taskDefinitionKey) {
   884→        if (processDefinitionId == null || taskDefinitionKey == null) {
   885→            return null;
   886→        }
   887→        try {
   888→            // 精确匹配该部署版本的 NodeConfig 快照（部署时由当前配置复制生成）
   889→            List<NodeConfig> configs = nodeConfigRepository.findByProcessDefinitionId(processDefinitionId);
   890→            FormConfigResult taskCfg = null;
   891→            FormConfigResult processCfg = null;
   892→
   893→            for (NodeConfig nc : configs) {
   894→                FormConfigResult cfg = parseFormConfigFromJson(nc.getConfigJson());
   895→                if (cfg == null || cfg.getFormDefId() == null) continue;
   896→
   897→                if (taskDefinitionKey.equals(nc.getNodeId())) {
   898→                    taskCfg = cfg;
   899→                } else if ("__PROCESS__".equals(nc.getNodeId())) {
   900→                    processCfg = cfg;
   901→                }
   902→            }
   903→
   904→            // 节点表单优先，没有则用流程默认表单
   905→            FormConfigResult selected = taskCfg != null ? taskCfg : processCfg;
   906→            if (selected != null && selected.getFieldPermissions() == null) {
   907→                selected.setFieldPermissions(new HashMap<>());
   908→            }
   909→            return selected;
   910→        } catch (Exception e) {
   911→            log.warn("从 NodeConfig 解析表单配置失败", e);
   912→            return null;
   913→        }
   914→    }

   916→    /**
   917→     * 解析任务节点的操作权限配置。
   918→     *
   919→     * <p>叠加流程级与节点级配置（AND 规则）：从该部署版本的 {@code __PROCESS__} 节点配置读取
   920→     * 流程级总控（JSON 路径 {@code approvalPolicy.operations}，未配置视为全开），从节点配置
   921→     * （NodeConfig, nodeId=taskDefKey）读取节点级 operations，每个开关取两者的 AND。
   922→     * 节点未配置 operations 时按节点级默认值处理。默认值：
   923→     * <ul>
   924→     *   <li>节点级：allowReject: true、allowTransfer: true、allowAddSign: false、allowDelegate: false</li>
   925→     *   <li>流程级：allowReject/allowAddSign/allowTransfer/allowDelegate 均为 true</li>
   926→     * </ul>
   927→     *
   928→     * @param processDefinitionId 流程定义 ID
   929→     * @param taskDefinitionKey   任务定义键（BPMN 节点 ID）
   930→     * @return 操作权限配置（永不为 null）
   931→     */
   932→    public OperationsConfig extractOperations(String processDefinitionId, String taskDefinitionKey) {
   933→        if (processDefinitionId == null || taskDefinitionKey == null) {
   934→            return new OperationsConfig();
   935→        }
   936→        try {
   937→            // 精确匹配该部署版本的 NodeConfig 快照（与 extractFormConfig 保持一致）
   938→            List<NodeConfig> configs = nodeConfigRepository.findByProcessDefinitionId(processDefinitionId);
   939→            // 流程级总控（__PROCESS__），未配置视为全开
   940→            OperationsConfig processLevel = new OperationsConfig();
   941→            boolean hasProcessLevel = false;
   942→            for (NodeConfig nc : configs) {
   943→                if ("__PROCESS__".equals(nc.getNodeId())) {
   944→                    processLevel = parseProcessOperations(nc.getConfigJson());
   945→                    hasProcessLevel = true;
   946→                    break;
   947→                }
   948→            }
   949→            // 节点级
   950→            OperationsConfig nodeLevel = new OperationsConfig();
   951→            for (NodeConfig nc : configs) {
   952→                if (taskDefinitionKey.equals(nc.getNodeId())) {
   953→                    nodeLevel = parseOperationsFromConfig(nc.getConfigJson());
   954→                    break;
   955→                }
   956→            }
   957→            if (!hasProcessLevel) {
   958→                // 流程级全开时等价于节点级
   959→                return nodeLevel;
   960→            }
   961→            // 流程级 && 节点级（AND 合并）
   962→            OperationsConfig result = new OperationsConfig();
   963→            result.setAllowReject(processLevel.isAllowReject() && nodeLevel.isAllowReject());
   964→            result.setAllowAddSign(processLevel.isAllowAddSign() && nodeLevel.isAllowAddSign());
   965→            result.setAllowTransfer(processLevel.isAllowTransfer() && nodeLevel.isAllowTransfer());
   966→            result.setAllowDelegate(processLevel.isAllowDelegate() && nodeLevel.isAllowDelegate());
   967→            return result;
   968→        } catch (Exception e) {
   969→            log.warn("从 NodeConfig 解析操作配置失败", e);
   970→            return new OperationsConfig();
   971→        }
   972→    }

   974→    /**
   975→     * 从 NodeConfig JSON 中解析 operations，缺失字段用默认值补全。
   976→     */
   977→    private OperationsConfig parseOperationsFromConfig(String configJson) {
   978→        OperationsConfig result = new OperationsConfig();
   979→        try {
   980→            JsonNode json = objectMapper.readTree(configJson);
   981→            JsonNode ops = json.get("operations");
   982→            if (ops == null || !ops.isObject()) {
   983→                return result;
   984→            }
   985→            if (ops.has("allowReject")) result.setAllowReject(ops.get("allowReject").asBoolean());
   986→            if (ops.has("allowAddSign")) result.setAllowAddSign(ops.get("allowAddSign").asBoolean());
   987→            if (ops.has("allowTransfer")) result.setAllowTransfer(ops.get("allowTransfer").asBoolean());
   988→            if (ops.has("allowDelegate")) result.setAllowDelegate(ops.get("allowDelegate").asBoolean());
   989→        } catch (Exception e) {
   990→            log.warn("从 NodeConfig 解析 operations JSON 失败: {}", e.getMessage());
   991→        }
   992→        return result;
   993→    }

   995→    /**
   996→     * 从 {@code __PROCESS__} 配置解析流程级操作权限（JSON 路径 {@code approvalPolicy.operations}）。
   997→     * 未配置 operations 时返回全开默认值。
   998→     */
   999→    private OperationsConfig parseProcessOperations(String configJson) {
  1000→        OperationsConfig result = new OperationsConfig();
  1001→        result.setAllowReject(true);
  1002→        result.setAllowAddSign(true);
  1003→        result.setAllowTransfer(true);
  1004→        result.setAllowDelegate(true);
  1005→        try {
  1006→            JsonNode json = objectMapper.readTree(configJson);
  1007→            JsonNode ops = json.path("approvalPolicy").path("operations");
  1008→            if (ops.isObject()) {
  1009→                if (ops.has("allowReject")) result.setAllowReject(ops.get("allowReject").asBoolean());
  1010→                if (ops.has("allowAddSign")) result.setAllowAddSign(ops.get("allowAddSign").asBoolean());
  1011→                if (ops.has("allowTransfer")) result.setAllowTransfer(ops.get("allowTransfer").asBoolean());
  1012→                if (ops.has("allowDelegate")) result.setAllowDelegate(ops.get("allowDelegate").asBoolean());
  1013→            }

  1020→    /**
  1021→     * 从 NodeConfig JSON 中解析完整表单配置（formDefId + fieldPermissions）。
  1022→     *
  1023→     * @param configJson NodeConfig 的 config_json
  1024→     * @return 表单配置；JSON 无 form 节点或无 formDefId 时返回 null
  1025→     */
  1026→    private FormConfigResult parseFormConfigFromJson(String configJson) {
  1027→        try {
  1028→            JsonNode json = objectMapper.readTree(configJson);
  1029→            JsonNode form = json.get("form");
  1030→            if (form == null || !form.has("formDefId")) {
  1031→                return null;
  1032→            }
  1033→            String val = form.get("formDefId").asText();
  1034→            if (val == null || val.isEmpty()) {
  1035→                return null;
  1036→            }
  1037→            FormConfigResult result = new FormConfigResult();
  1038→            result.setFormDefId(val);
  1039→            Map<String, String> permissions = new HashMap<>();
  1040→            JsonNode permNode = form.get("fieldPermissions");
  1041→            if (permNode != null && permNode.isObject()) {
  1042→                permNode.fields().forEachRemaining(e -> permissions.put(e.getKey(), e.getValue().asText()));
  1043→            }
  1044→            result.setFieldPermissions(permissions);
  1045→            return result;
  1046→        } catch (Exception e) {
  1047→            log.warn("从 NodeConfig 解析表单配置 JSON 失败: {}", e.getMessage());
  1048→            return null;
  1049→        }
  1050→    }

  1078→    /**
  1079→     * 完成任务并返回下一个任务信息，同时写入审批意见。
  1080→     *
  1081→     * @param taskId    任务 ID
  1082→     * @param variables 流程变量
  1083→     * @param userId    操作人 ID（用于审批意见记录）
  1084→     * @param comment   审批意见（可为 null）
  1085→     * @return 包含下一个任务信息和流程结束标志的响应
  1086→     */
  1087→    @Transactional
  1088→    public CompleteTaskResponse completeTaskWithResponse(String taskId, Map<String, Object> variables,
  1089→                                                          String userId, String comment) {
  1090→        // 1. 查当前任务获取 processInstanceId
  1091→        Task currentTask = flowableTaskService.createTaskQuery()
  1092→                .taskId(taskId)
  1093→                .singleResult();
  1094→
  1095→        if (currentTask == null) {
  1096→            throw new IllegalStateException("Task not found: " + taskId);
  1097→        }
  1098→
  1099→        String processInstanceId = currentTask.getProcessInstanceId();
  1100→
  1101→        // 2. 委派状态的任务需要先 resolve
  1102→        org.flowable.task.api.DelegationState state = currentTask.getDelegationState();
  1103→        if (state != null) {
  1104→            flowableTaskService.resolveTask(taskId);
  1105→        }
  1106→
  1107→        // 3. 完成任务
  1108→        flowableTaskService.complete(taskId, variables);
  1109→
  1110→        // 3b. 流程变量映射写入（form:* 源需在任务数据流转后读取）
  1111→        try {
  1112→            if (currentTask.getProcessDefinitionId() != null) {
  1113→                variableMappingWriter.write(currentTask.getProcessDefinitionId(), processInstanceId);
  1114→            }
  1115→        } catch (Exception e) {
  1116→            log.warn("Failed to write variable mappings after complete task [{}]: {}", taskId, e.getMessage());
  1117→        }
  1118→
  1119→        // 4. 写入审批意见
  1120→        if (userId != null) {
  1121→            saveTaskComment(taskId, processInstanceId, userId, "approve", comment);
  1122→        }
  1123→
  1124→        // 4. 查流程是否仍在运行
  1125→        ProcessInstance runningInstance = runtimeService.createProcessInstanceQuery()
  1126→                .processInstanceId(processInstanceId)
  1127→                .singleResult();
  1128→
  1129→        boolean processFinished = (runningInstance == null);
  1130→
  1131→        // 5. 如果流程未结束，查下一个任务
  1132→        String nextTaskId = null;
  1133→        String nextTaskName = null;
  1134→        String nextTaskAssignee = null;
  1135→        String nextTaskDefinitionKey = null;
  1136→
  1137→        if (!processFinished) {
  1138→            Task nextTask = flowableTaskService.createTaskQuery()
  1139→                    .processInstanceId(processInstanceId)
  1140→                    .singleResult();
  1141→
  1142→            if (nextTask != null) {
  1143→                nextTaskId = nextTask.getId();
  1144→                nextTaskName = nextTask.getName();
  1145→                nextTaskAssignee = nextTask.getAssignee();
  1146→                nextTaskDefinitionKey = nextTask.getTaskDefinitionKey();
  1147→            }
  1148→        }
  1149→
  1150→        return CompleteTaskResponse.builder()
  1151→                .processInstanceId(processInstanceId)
  1152→                .processFinished(processFinished)
  1153→                .nextTaskId(nextTaskId)
  1154→                .nextTaskName(nextTaskName)
  1155→                .nextTaskAssignee(nextTaskAssignee)
  1156→                .nextTaskDefinitionKey(nextTaskDefinitionKey)
  1157→                .build();
  1158→    }
