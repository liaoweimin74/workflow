   188→   188→/** procDefId（Node 契约 `{key}:{version}:{rowId}`）解析 */
   189→   189→function parseProcDefId(id: string): { key: string; version: number; rowId: number } | null {
   190→   190→  const i1 = id.lastIndexOf(':');
   191→   191→  if (i1 <= 0) return null;
   192→   192→  const rowId = Number(id.slice(i1 + 1));
   193→   193→  if (!Number.isInteger(rowId) || rowId <= 0) return null;
   194→   194→  const i2 = id.lastIndexOf(':', i1 - 1);
   195→   195→  if (i2 < 0) return null;
   196→   196→  const version = Number(id.slice(i2 + 1, i1));
   197→   197→  if (!Number.isInteger(version) || version <= 0) return null;
   198→   198→  return { key: id.slice(0, i2), version, rowId };
   199→   199→}

   259→   259→/** ProcessDefinitionController#extractFormConfig 移植 */
   260→   260→function extractFormConfig(configJson: string): FormConfig | null {
   261→   261→  let json: Record<string, unknown>;
   262→   262→  try {
   263→   263→    json = JSON.parse(configJson) as Record<string, unknown>;
   264→   264→  } catch {
   265→   265→    return null;
   266→   266→  }
   267→   267→  if (json == null || typeof json !== 'object') return null;
   268→   268→  const form = json['form'];
   269→   269→  if (form == null || typeof form !== 'object' || Array.isArray(form)) return null;
   270→   270→  const f = form as Record<string, unknown>;
   271→   271→  if (f['formDefId'] == null) return null;
   272→   272→  const val = typeof f['formDefId'] === 'string' ? f['formDefId'] : String(f['formDefId']);
   273→   273→  if (val === '') return null;
   274→   274→  const permNode = f['fieldPermissions'];
   275→   275→  let permissions: Record<string, string> | null = null;
   276→   276→  if (permNode != null && typeof permNode === 'object' && !Array.isArray(permNode)) {
   277→   277→    permissions = {};
   278→   278→    for (const [k, v] of Object.entries(permNode as Record<string, unknown>)) {
   279→   279→      permissions[k] = typeof v === 'string' ? v : String(v);
   280→   280→    }
   281→   281→    permissions = javaHashMapOrdered(permissions) as Record<string, string>;
   282→   282→  }
   283→   283→  return { formDefId: val, fieldPermissions: permissions };
   284→   284→}

   286→   286→/**
   287→   287→ * ProcessDefinitionController#resolveFormDefIds 移植：
   288→   288→ * 优先级 发起人节点表单 > 流程默认表单（表单与字段权限同层取，不跨层合并）。
   289→   289→ * 返回插入序：formDefId, fieldPermissions, initiatorFormDefId, processFormDefId。
   290→   290→ */
   291→   291→function resolveFormDefIds(procDefId: string, deployRow: Row | null): Record<string, unknown> {
   292→   292→  const out: Record<string, unknown> = {
   293→   293→    formDefId: null,
   294→   294→    fieldPermissions: null,
   295→   295→    initiatorFormDefId: null,
   296→   296→    processFormDefId: null,
   297→   297→  };
   298→   298→  const configs = snapshotConfigRows(procDefId);
   299→   299→  let processCfg: FormConfig | null = null;
   300→   300→  let initiatorCfg: FormConfig | null = null;
   301→   301→
   302→   302→  for (const nc of configs) {
   303→   303→    try {
   304→   304→      const cfg = extractFormConfig(String(nc['config_json'] ?? ''));
   305→   305→      if (cfg == null) continue;
   306→   306→      if (String(nc['node_id']) === '__PROCESS__') {
   307→   307→        processCfg = cfg;
   308→   308→      }
   309→   309→    } catch {
   310→   310→      // 忽略解析错误
   311→   311→    }
   312→   312→  }
   313→   313→
   314→   314→  const xml = deployedXmlOf(procDefId, deployRow);
   315→   315→  const initiatorNodeId = xml == null ? null : extractInitiatorNodeId(parseBpmnToIr(xml));
   316→   316→  if (initiatorNodeId != null) {
   317→   317→    for (const nc of configs) {
   318→   318→      if (String(nc['node_id']) === initiatorNodeId) {
   319→   319→        try {
   320→   320→          initiatorCfg = extractFormConfig(String(nc['config_json'] ?? ''));
   321→   321→        } catch {
   322→   322→          // 忽略解析错误
   323→   323→        }
   324→   324→        break;
   325→   325→      }
   326→   326→    }
   327→   327→  }
   328→   328→
   329→   329→  const effective = initiatorCfg != null ? initiatorCfg : processCfg;
   330→   330→  out['formDefId'] = effective != null ? effective.formDefId : null;
   331→   331→  out['fieldPermissions'] = effective != null ? effective.fieldPermissions : null;
   332→   332→  out['initiatorFormDefId'] = initiatorCfg != null ? initiatorCfg.formDefId : null;
   333→   333→  out['processFormDefId'] = processCfg != null ? processCfg.formDefId : null;
   334→   334→  return out;
   335→   335→}

   339→   339→function isSameAsLastDeployment(draft: Row, effectiveBpmnXml: string, nodeConfigMap: Record<string, string>): boolean {
   340→   340→  if (trimToNull(draft['deployed_xml'] == null ? null : String(draft['deployed_xml'])) !== trimToNull(effectiveBpmnXml)) {
   341→   341→    return false; // XML 变化
   342→   342→  }
   343→   343→  const procDefId = draft['process_definition_id'] == null ? null : String(draft['process_definition_id']);
   344→   344→  if (procDefId == null) {
   345→   345→    return true; // 无历史部署记录且 XML 相同，视为未变化
   346→   346→  }
   347→   347→  const snapshotMap = configMap(all(`SELECT "node_id", "config_json" FROM wf_node_config WHERE "process_def_id" = ? AND "process_definition_id" = ?`, [
   348→   348→    String(draft['id']),
   349→   349→    procDefId,
   350→   350→  ]));
   351→   351→  return mapEquals(snapshotMap, nodeConfigMap);
   352→   352→}

   354→   354→function doDeploy(tenant: string, draftId: string): Record<string, unknown> {
   355→   355→  const draft = draftById(draftId, tenant);
   356→   356→
   357→   357→  // 当前编辑中配置 + MultiInstanceBpmnRewriter 改写（会签/或签 → MI；单实例 → assignee/candidateUsers）
   358→   358→  const nodeConfigMap = configMap(currentConfigRows(draftId));
   359→   359→  let effectiveBpmnXml = rewriteMultiInstance(draft['bpmn_xml'] == null ? null : String(draft['bpmn_xml']), nodeConfigMap);
   360→   360→  // 注入 StartEvent/EndEvent 默认名称
   361→   361→  effectiveBpmnXml = injectEventNames(effectiveBpmnXml);
   362→   362→
   363→   363→  // 变化检测：hash 优先，历史数据（deployed_config_hash 为空）降级比较
   364→   364→  const currentHash = deployHash(effectiveBpmnXml, nodeConfigMap);
   365→   365→  const storedHash = trimToNull(draft['deployed_config_hash'] == null ? null : String(draft['deployed_config_hash']));
   366→   366→  const unchanged = storedHash != null ? storedHash === currentHash : isSameAsLastDeployment(draft, effectiveBpmnXml, nodeConfigMap);
   367→   367→  if (unchanged) {
   368→   368→    throw new BusinessException('流程数据未变化，无需部署', 400);
   369→   369→  }
   370→   370→
   371→   371→  // Flowable 引擎校验/解析对位：XML 解析失败 → 部署失败（消息含 BPMN → 流程定义不完整）
   372→   372→  const key = String(draft['process_key']);
   373→   373→  let ir;
   374→   374→  try {
   375→   375→    ir = parseBpmnToIr(effectiveBpmnXml, { nodeConfigs: nodeConfigMap });
   376→   376→  } catch (e) {
   377→   377→    const msg = e instanceof Error ? e.message : String(e);
   378→   378→    if (msg.includes('BPMN')) {
   379→   379→      throw new BusinessException(`流程定义不完整：${msg}`, 400);
   380→   380→    }
   381→   381→    throw new BusinessException(`部署失败：${msg || '未知错误'}`, 400);
   382→   382→  }
   383→   383→
   384→   384→  const now = nowText();
   385→   385→  const resourceName = `${key}.bpmn20.xml`;
   386→   386→  const diagramResourceName = hasDiagramInfo(effectiveBpmnXml) ? `${key}.${key}.png` : null;
   387→   387→  // Flowable ProcessDefinition.category = BPMN targetNamespace（非 deployment category）
   388→   388→  const category = extractTargetNamespace(effectiveBpmnXml);
   389→   389→
   390→   390→  run(
   391→   391→    `INSERT INTO WF_PROC_DEPLOY (NAME, KEY_, CATEGORY, VERSION, DEPLOY_TIME, RESOURCE_NAME, DIAGRAM_RESOURCE_NAME, TENANT_ID, SUSPENSION_STATE, BPMN_XML, CONFIG_JSON)
   392→   392→     VALUES (?, ?, ?, ?, ?, ?, ?, ?, 1, ?, ?)`,
   393→   393→    [String(draft['name']), key, category, 0, now, resourceName, diagramResourceName, tenant, effectiveBpmnXml, JSON.stringify(ir)],
   394→   394→  );
   395→   395→  const rowId = Number(one(`SELECT last_insert_rowid() AS ID`)!['ID']);
   396→   396→
   397→   397→  let procDefId: string | null = null;
   398→   398→  let version = Number(draft['version'] ?? 0);
   399→   399→
   400→   400→  if (ir.procKey != null) {
   401→   401→    // version = 同 key+租户 已部署最大版本 + 1（Flowable 语义，首版 = 1）
   402→   402→    version = Number(one(`SELECT MAX(VERSION) AS V FROM WF_PROC_DEPLOY WHERE KEY_ = ? AND TENANT_ID = ? AND ID != ?`, [key, tenant, rowId])!['V'] ?? 0) + 1;
   403→   403→    procDefId = `${key}:${version}:${rowId}`;
   404→   404→    run(`UPDATE WF_PROC_DEPLOY SET VERSION = ? WHERE ID = ?`, [version, rowId]);
   405→   405→  }
   406→   406→
   407→   407→  run(
   408→   408→    `UPDATE wf_process_draft SET "status" = 'DEPLOYED', "deploy_id" = ?, "last_deployed_at" = ?, "deployed_xml" = ?, "deployed_config_hash" = ?, "process_definition_id" = ?, "version" = ?, "updated_at" = ? WHERE "id" = ?`,
   409→   409→    [String(rowId), now, effectiveBpmnXml, currentHash, procDefId, version, now, draftId],
   410→   410→  );
   411→   411→
   412→   412→  if (procDefId != null) {
   413→   413→    // 版本快照：复制当前配置绑定新部署版本（幂等：先删同版本旧快照）
   414→   414→    const currentRows = currentConfigRows(draftId);
   415→   415→    if (currentRows.length > 0) {
   416→   416→      run(`DELETE FROM wf_node_config WHERE "process_def_id" = ? AND "process_definition_id" = ?`, [draftId, procDefId]);
   417→   417→      const nodeTypes = parseNodeTypes(effectiveBpmnXml);
   418→   418→      for (const src of currentRows) {
   419→   419→        const nodeId = String(src['node_id']);
   420→   420→        const nodeType = nodeId === '__PROCESS__' ? 'process' : (nodeTypes[nodeId] ?? 'unknown');
   421→   421→        insertNodeConfig(randomUUID().replace(/-/g, ''), tenant, draftId, procDefId, nodeId, nodeType, String(src['config_json'] ?? ''));
   422→   422→      }
   423→   423→    }
   424→   424→    // 表单映射校验（字段存在性/变量名唯一/循环引用），失败抛 IllegalArgumentError 阻止部署
   425→   425→    validateFormMappings(procDefId, effectiveBpmnXml);
   426→   426→  }
   427→   427→
   428→   428→  return draftJson(one(`SELECT * FROM wf_process_draft WHERE "id" = ?`, [draftId])!);
   429→   429→}
