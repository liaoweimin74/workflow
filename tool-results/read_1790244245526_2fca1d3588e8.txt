    69→    /**
    70→     * 加载表单运行时上下文（表名 + 列 + 子表元数据）。
    71→     */
    72→    public BizDataContext loadContext(String formKey) {
    73→        if (formKey == null || !FORM_KEY_PATTERN.matcher(formKey).matches()) {
    74→            throw new BusinessException(400, "非法表单 key: " + formKey);
    75→        }
    76→        String tableName = "wf_biz_" + formKey;
    77→        if (!tableManager.tableExists(tableName)) {
    78→            throw new BusinessException(404, "业务表单数据表不存在: " + formKey);
    79→        }
    80→        List<ColumnConfig> columns = formDefService.getBusinessColumnsByKey(formKey);
    81→        // 主表列：排除子表字段（子表字段映射独立物理表，非主表列）
    82→        List<String> keys = columns.stream()
    83→                .filter(c -> c.getSubColumns() == null || c.getSubColumns().isEmpty())
    84→                .map(ColumnConfig::getKey)
    85→                .toList();
    86→        Map<String, BizDataContext.SubTableDef> subTables = new LinkedHashMap<>();
    87→        for (ColumnConfig c : columns) {
    88→            if (c.getSubColumns() != null && !c.getSubColumns().isEmpty()) {
    89→                String mode = c.getSubMode() == null || c.getSubMode().isBlank() ? "embedded" : c.getSubMode();
    90→                List<String> subKeys = c.getSubColumns().stream().map(ColumnConfig::getKey).toList();
    91→                subTables.put(c.getKey(), new BizDataContext.SubTableDef(
    92→                        "wf_biz_" + formKey + "_" + c.getKey(), mode, c.getSubColumns(), subKeys));
    93→            }
    94→        }
    95→        return new BizDataContext(tableName, formKey, columns, keys, subTables);
    96→    }

   113→    public void validateRequired(List<ColumnConfig> columns, Map<String, Object> data) {
   114→        Map<String, Object> safe = data == null ? Map.of() : data;
   115→        for (ColumnConfig c : columns) {
   116→            if (c.isRequired()) {
   117→                Object v = safe.get(c.getKey());
   118→                if (v == null || (v instanceof String s && s.isBlank())) {
   119→                    throw new BusinessException(400, "必填字段不能为空: " + c.getLabel());
   120→                }
   121→            }
   122→        }

   125→    /**
   126→     * 遍历 data-picker 引用列：校验 id 存在并生成 {@code <key>_text} 展示缓存文本。
   127→     * 不修改原 data，返回附加字段（{@code <key>_text} → 文本）；引用值为空时返回空文本。
   128→     */
   129→    @SuppressWarnings("unchecked")
   130→    public Map<String, Object> resolvePickerValues(BizDataContext ctx, Map<String, Object> data) {
   131→        Map<String, Object> extra = new LinkedHashMap<>();
   132→        for (ColumnConfig col : ctx.columns()) {
   133→            String pickerConfig = col.getPickerConfig();
   134→            if (pickerConfig == null || pickerConfig.isBlank()) {
   135→                continue;
   136→            }
   137→            // 仅 data-picker 引用列生成冗余文本；LookupPicker 列跳过
   138→            if (!isDataPickerColumn(ctx, col)) {
   139→                continue;
   140→            }
   141→            String key = col.getKey();
   142→            Object raw = data.get(key);
   143→            String text = resolvePickerText(ctx, col, raw);
   144→            extra.put(key + "_text", text);
   145→        }
   146→        return extra;
   147→    }

   153→        BizDataContext ctx = loadContext(formKey);

   157→                    .filter(c -> !c.isHidden() && c.getPickerConfig() == null)

   161→        }

   165→    /**
   166→     * 批量解析被引用记录的显示文本（id → displayField 值）。
   167→     */
   168→    public Map<String, String> resolveDisplayTexts(String sourceFormKey, List<String> ids, String displayField) {
   169→        if (sourceFormKey == null || !FORM_KEY_PATTERN.matcher(sourceFormKey).matches()) {
   170→            throw new BusinessException(400, "非法目标表单 key: " + sourceFormKey);
   171→        }
   172→        if (ids == null || ids.isEmpty()) {
   173→            return Map.of();
   174→        }
   175→        if (displayField == null || !displayField.matches("^[a-zA-Z][a-zA-Z0-9_]{0,63}$")) {
   176→            throw new BusinessException(400, "非法显示字段: " + displayField);
   177→        }
   178→        String table = "wf_biz_" + sourceFormKey;
   179→        String placeholders = String.join(",", java.util.Collections.nCopies(ids.size(), "?"));
   180→        String sql = "SELECT id, " + displayField + " FROM " + table
   181→                + " WHERE tenant_id = ? AND id IN (" + placeholders + ")";
   182→        List<Object> params = new ArrayList<>();
   183→        params.add(tenantProvider.getTenantId());
   184→        params.addAll(ids);
   185→
   186→        Map<String, String> result = new LinkedHashMap<>();
   187→        for (Map<String, Object> row : jdbcTemplate.queryForList(sql, params.toArray())) {
   188→            Object v = row.get(displayField);
   189→            result.put(String.valueOf(row.get("id")), v == null ? "" : String.valueOf(v));
   190→        }
   191→        return result;
   192→    }

   217→    /**
   218→     * 通用新增实现（不含装饰钩子链；覆盖 handler 委托此方法执行通用插入）。
   219→     */
   220→    @Transactional
   221→    public BizDataVO createGeneric(String formKey, Map<String, Object> data) {
   222→        String tenantId = tenantProvider.getTenantId();
   223→        BizDataContext ctx = loadContext(formKey);
   224→
   225→        validateRequired(ctx.columns(), data);
   226→
   227→        // data-picker 引用校验与冗余文本生成（不改原 data，返回附加字段）
   228→        Map<String, Object> merged = serializeJsonColumns(data);
   229→        merged.putAll(resolvePickerValues(ctx, merged));
   230→
   231→        BizDataQueryBuilder.SqlAndParams insert = BizDataQueryBuilder.buildInsert(
   232→                ctx.tableName(), ctx.columnKeys(), merged, tenantId);
   233→        jdbcTemplate.update(insert.sql(), insert.params().toArray());
   234→
   235→        // 子表行写入（随主表创建批量插入）
   236→        String bizId = insertedId(insert);
   237→        for (Map.Entry<String, BizDataContext.SubTableDef> e : ctx.subTables().entrySet()) {
   238→            Object raw = data.get(e.getKey());
   239→            if (raw instanceof List<?> rows) {
   240→                writeSubRows(e.getValue(), bizId, rows);
   241→            }
   242→        }
   243→
   244→        // 新行 id 由 buildInsert 内部生成，查询返回
   245→        return findById(ctx.tableName(), tenantId, ctx, bizId);
   246→    }

   248→    /**
   249→     * 通用分页查询实现。
   250→     */
   251→    public BizDataPageVO queryGeneric(String formKey, BizDataQueryRequest req) {
   252→        String tenantId = tenantProvider.getTenantId();
   253→        BizDataContext ctx = loadContext(formKey);
   254→
   255→        Map<String, Object> filters = parseFilter(req.getFilter());
   256→        // 列类型映射（key → columnType）：JSON 数组列筛选走 JSON 函数（JSON_CONTAINS/JSON_OVERLAPS）
   257→        Map<String, String> columnTypeOf = ctx.columns().stream()
   258→                .collect(Collectors.toMap(ColumnConfig::getKey,
   259→                        c -> c.getColumnType() == null ? "" : c.getColumnType().toUpperCase(), (a, b) -> a));
   260→        int page = Math.max(req.getPage(), 1);
   261→        // size <= 0 表示不分页取全部（buildSelect 跳过 LIMIT/OFFSET）；正数沿用原钳制上限
   262→        int size = req.getSize() <= 0 ? req.getSize() : Math.min(Math.max(req.getSize(), 1), 100);
   263→
   264→        try {
   265→            BizDataQueryBuilder.SqlAndParams count = BizDataQueryBuilder.buildCount(
   266→                    ctx.tableName(), ctx.columnKeys(), columnTypeOf, tenantId, filters, req.getKeyword(), req.getKeywordColumn());
   267→
   268→            BizDataQueryBuilder.SqlAndParams select = BizDataQueryBuilder.buildSelect(
   269→                    ctx.tableName(), ctx.columnKeys(), columnTypeOf, tenantId, filters,
   270→                    req.getKeyword(), req.getKeywordColumn(), req.getSort(), req.getOrder(), page - 1, size);
   271→
   272→            return sqlQueryEngine.execPage(page, size, count, select, row -> toVO(ctx, row));
   273→        } catch (IllegalArgumentException e) {
   274→            throw new BusinessException(400, e.getMessage());
   275→        }
   276→    }

   306→    /**
   307→     * sql 模式分页查询实现（管理员 SQL 模板包裹，运行时参数白名单透传）。
   308→     * <p>仅校验 formKey 合法性；不校验主表单物理表（管理员 SQL 独立定义，可跨表/聚合）。
   309→     */
   310→    public BizDataPageVO querySqlTemplate(String formKey, BizDataQueryRequest req, FormQueryConfig cfg) {
   311→        // SQL 类型无 formKey 时跳过校验（管理员 SQL 独立定义，不依赖表单）
   312→        if (formKey != null && !FORM_KEY_PATTERN.matcher(formKey).matches()) {
   313→            throw new BusinessException(400, "非法表单 key: " + formKey);
   314→        }
   315→        String tenantId = tenantProvider.getTenantId();
   316→        List<JoinSqlGenerator.QueryColumn> columns = toQueryColumns(cfg.columns());
   317→        Map<String, Object> filters = parseFilter(req.getFilter());
   318→        Map<String, Object> runtimeParams = parseRuntimeParams(req.getParams());
   319→        int page = Math.max(req.getPage(), 1);
   320→        // size <= 0 表示不分页取全部；正数沿用原钳制上限
   321→        int size = req.getSize() <= 0 ? req.getSize() : Math.min(Math.max(req.getSize(), 1), 100);
   322→
   323→        try {
   324→            SqlQueryEngine.WrappedQuery wq = SqlTemplateEngine.wrap(
   325→                    cfg.query(), tenantId, columns, filters, req.getKeyword(), req.getKeywordColumn(),
   326→                    req.getSort(), req.getOrder(), page, size, cfg.declaredParams(), runtimeParams);
   327→            return sqlQueryEngine.execPage(page, size, wq.count(), wq.select(),
   328→                    row -> toSqlVO(cfg.columns(), row));
   329→        } catch (IllegalArgumentException e) {
   330→            throw new BusinessException(400, e.getMessage());
   331→        }
   332→    }

   334→    /** 构建查询列映射：主表列（ref="m."+key，默认全可排可筛）+ 虚拟列（ref=alias+"."+joinField，能力取 join 声明） */
   335→    private List<JoinSqlGenerator.QueryColumn> buildJoinColumns(BizDataContext ctx,
   336→                                                                List<JoinSqlGenerator.JoinConfig> joins) {
   337→        Map<String, String> typeOf = new HashMap<>();
   338→        for (ColumnConfig c : ctx.columns()) {
   339→            typeOf.put(c.getKey(), c.getColumnType() == null ? "" : c.getColumnType().toUpperCase());
   340→        }
   341→        List<JoinSqlGenerator.QueryColumn> columns = new ArrayList<>();
   342→        for (String key : ctx.columnKeys()) {
   343→            columns.add(new JoinSqlGenerator.QueryColumn(key, "m." + key, typeOf.getOrDefault(key, ""), true, true));
   344→        }
   345→        for (JoinSqlGenerator.JoinGroup g : JoinSqlGenerator.group(joins)) {
   346→            for (JoinSqlGenerator.JoinConfig m : g.members()) {
   347→                List<ColumnConfig> targetCols = resolveJoinTargets(m);
   348→                columns.add(new JoinSqlGenerator.QueryColumn(m.virtualKey(), g.alias() + "." + m.joinField(),
   349→                        joinColumnType(targetCols, m.joinField()), m.sortable(), m.filterable()));
   350→                // 目标表单含 <joinField>_text 冗余文本列（dataPicker 引用列）→ 带出 <virtualKey>_text，供前端引用渲染显示文本
   351→                if (findJoinTarget(targetCols, m.joinField() + "_text") != null) {
   352→                    columns.add(new JoinSqlGenerator.QueryColumn(m.virtualKey() + "_text",
   353→                            g.alias() + "." + m.joinField() + "_text",
   354→                            joinColumnType(targetCols, m.joinField() + "_text"), false, false));
   355→                }
   356→            }
   357→        }
   358→        return columns;
   359→    }

   363→     * 校验 formKey 合法且主表存在（loadContext）；目标表单存在性由保存校验负责，预览不重复校验。

   375→    /** 目标表单列列表；解析失败/不存在 → 空列表（调用方统一按"查不到"处理）。 */
   376→    private List<ColumnConfig> resolveJoinTargets(JoinSqlGenerator.JoinConfig j) {
   377→        try {
   378→            List<ColumnConfig> target = formDefService.getBusinessColumnsByKey(j.targetFormKey());
   379→            return target == null ? List.of() : target;
   380→        } catch (BusinessException ignored) {
   381→            // 目标表单不可解析时回退
   382→            return List.of();
   383→        }
   384→    }

   386→    /** 目标列查找：按 key 匹配；查不到返回 null。 */
   387→    private static ColumnConfig findJoinTarget(List<ColumnConfig> cols, String key) {
   388→        if (cols == null) {
   389→            return null;
   390→        }
   391→        for (ColumnConfig c : cols) {
   392→            if (key.equals(c.getKey())) {
   393→                return c;
   394→            }
   395→        }
   396→        return null;
   397→    }

   399→    /** 目标列 columnType：查不到 fallback "VARCHAR"（查询与 metadata 两处一致）。 */
   400→    private static String joinColumnType(List<ColumnConfig> cols, String key) {
   401→        ColumnConfig c = findJoinTarget(cols, key);
   402→        return c != null && c.getColumnType() != null && !c.getColumnType().isBlank()
   403→                ? c.getColumnType().toUpperCase()
   404→                : "VARCHAR";
   405→    }

   407→    /** sql 模式列映射：管理员声明列 → QueryColumn（ref=key，外层子查询输出列名） */
   408→    private List<JoinSqlGenerator.QueryColumn> toQueryColumns(List<ColumnConfig> cols) {
   409→        List<JoinSqlGenerator.QueryColumn> out = new ArrayList<>();
   410→        for (ColumnConfig c : cols) {
   411→            out.add(new JoinSqlGenerator.QueryColumn(c.getKey(), c.getKey(),
   412→                    c.getColumnType() == null ? "" : c.getColumnType().toUpperCase(),
   413→                    Boolean.TRUE.equals(c.getSortable()), Boolean.TRUE.equals(c.getFilterable())));
   414→        }
   415→        return out;
   416→    }

   418→    /** 解析运行时参数（sql 模式透传）；null/空白 → 空 Map；非法 JSON → 400 */
   419→    @SuppressWarnings("unchecked")
   420→    private Map<String, Object> parseRuntimeParams(String paramsJson) {
   421→        if (paramsJson == null || paramsJson.isBlank()) {
   422→            return Map.of();
   423→        }
   424→        try {
   425→            Map<String, Object> map = objectMapper.readValue(paramsJson, Map.class);
   426→            return map == null ? Map.of() : map;
   427→        } catch (JsonProcessingException e) {
   428→            throw new BusinessException(400, "运行时参数 params 格式非法，应为 JSON 对象: " + e.getOriginalMessage());
   429→        }
   430→    }

   432→    /**
   433→     * 通用更新实现（乐观锁；不含装饰钩子链）。
   434→     */
   435→    @Transactional
   436→    public BizDataVO updateGeneric(String formKey, String id, Map<String, Object> data, Integer version) {
   437→        String tenantId = tenantProvider.getTenantId();
   438→        BizDataContext ctx = loadContext(formKey);
   439→
   440→        validateRequired(ctx.columns(), data);
   441→        int currentVersion = version == null ? 1 : version;
   442→
   443→        Map<String, Object> merged = serializeJsonColumns(data);
   444→        merged.putAll(resolvePickerValues(ctx, merged));
   445→
   446→        BizDataQueryBuilder.SqlAndParams update = BizDataQueryBuilder.buildUpdate(
   447→                ctx.tableName(), ctx.columnKeys(), merged, tenantId, id, currentVersion);
   448→        int affected = jdbcTemplate.update(update.sql(), update.params().toArray());
   449→
   450→        if (affected == 0) {
   451→            // 区分"记录不存在"（404）与"版本冲突"（409）
   452→            List<Map<String, Object>> exists = jdbcTemplate.queryForList(
   453→                    "SELECT id, version FROM " + ctx.tableName() + " WHERE id = ? AND tenant_id = ?", id, tenantId);
   454→            if (exists.isEmpty()) {
   455→                throw new BusinessException(404, "业务数据不存在: " + id);
   456→            }
   457→            throw new BusinessException(409, "数据已被他人修改，请刷新后重试");
   458→        }
   459→
   460→        // 子表行增量 diff（仅当请求携带子表字段时）
   461→        for (Map.Entry<String, BizDataContext.SubTableDef> e : ctx.subTables().entrySet()) {
   462→            Object raw = data.get(e.getKey());
   463→            if (raw instanceof List<?> rows) {
   464→                diffSubRows(e.getValue(), id, rows);
   465→            }
   466→        }
   467→
   468→        return findById(ctx.tableName(), tenantId, ctx, id);
   469→    }

   537→     * 更新独立子表行（乐观锁：须携带当前 version）。
   538→     */
   539→    public Map<String, Object> updateSubRow(String formKey, String id, String field, String rowId,
   540→                                            Map<String, Object> data, Integer version) {
   541→        BizDataContext ctx = loadContext(formKey);
   542→        BizDataContext.SubTableDef def = requireSubTable(ctx, field);
   543→        requireMainRow(ctx, id);
   544→
   545→        // 仅允许更新子业务列（白名单 subKeys 过滤，防注入/防篡改内部列）
   546→        Map<String, Object> safe = new LinkedHashMap<>();
   547→        for (String k : def.subKeys()) {
   548→            if (data.containsKey(k)) {
   549→                safe.put(k, data.get(k));
   550→            }
   551→        }
   552→        if (safe.isEmpty()) {
   553→            throw new BusinessException(400, "更新内容不能为空: " + field);
   554→        }
   555→
   556→        StringBuilder set = new StringBuilder();
   557→        List<Object> params = new ArrayList<>();
   558→        for (Map.Entry<String, Object> e : safe.entrySet()) {
   559→            set.append(e.getKey()).append(" = ?, ");
   560→            params.add(e.getValue());
   561→        }
   562→        set.append("version = version + 1, updated_at = NOW()");
   563→        params.add(tenantProvider.getTenantId());
   564→        params.add(id);
   565→        params.add(rowId);
   566→        params.add(version == null ? 1 : version);
   567→
   568→        int affected = jdbcTemplate.update("UPDATE " + def.tableName() + " SET " + set
   569→                + " WHERE tenant_id = ? AND biz_id = ? AND id = ? AND version = ?", params.toArray());
   570→        if (affected == 0) {
   571→            throw new BusinessException(409, "子表行已被他人修改或不存在，请刷新后重试");
   572→        }
   573→        return readSubRows(def, id).stream()
   574→                .filter(r -> rowId.equals(String.valueOf(r.get("id"))))
   575→                .findFirst()
   576→                .orElseThrow(() -> new BusinessException(404, "子表行不存在: " + rowId));
   577→    }

   638→    @SuppressWarnings("unchecked")
   639→    private String resolvePickerText(BizDataContext ctx, ColumnConfig col, Object raw) {
   640→        try {
   641→            Map<String, Object> picker = objectMapper.readValue(col.getPickerConfig(), Map.class);
   642→            String sourceFormKey = picker.get("sourceFormKey") == null ? null : String.valueOf(picker.get("sourceFormKey"));
   643→            String displayField = picker.get("displayField") == null ? null : String.valueOf(picker.get("displayField"));
   644→            Object maxCountObj = picker.get("maxCount");
   645→
   646→            if (raw == null || String.valueOf(raw).isBlank()) {
   647→                return "";
   648→            }
   649→            // 值以 JSON 数组字符串存储（如 ["u1","u2"]；单选为 ["u1"]）
   650→            String rawStr = String.valueOf(raw);
   651→            List<String> ids;
   652→            try {
   653→                ids = objectMapper.readValue(rawStr,
   654→                        objectMapper.getTypeFactory().constructCollectionType(List.class, String.class));
   655→            } catch (JsonProcessingException e) {
   656→                throw new BusinessException(400, "data-picker 引用值格式非法（需 JSON 数组）: " + col.getKey());
   657→            }
   658→            ids.removeIf(String::isBlank);
   659→            if (ids.isEmpty()) {
   660→                return "";
   661→            }
   662→            // maxCount 校验：配置了（非 null 非空）且超限 → 400
   663→            if (maxCountObj != null) {
   664→                int maxCount;
   665→                try {
   666→                    maxCount = Integer.parseInt(String.valueOf(maxCountObj));
   667→                } catch (NumberFormatException e) {
   668→                    throw new BusinessException(400, "data-picker maxCount 配置非法: " + col.getKey());
   669→                }
   670→                if (maxCount > 0 && ids.size() > maxCount) {
   671→                    throw new BusinessException(400,
   672→                            "data-picker 引用数量超出限制（最多 " + maxCount + "）: " + col.getKey());
   673→                }
   674→            }
   675→
   676→            Map<String, String> texts = resolveDisplayTexts(sourceFormKey, ids, displayField);
   677→            List<String> ordered = new ArrayList<>();
   678→            for (String id : ids) {
   679→                String t = texts.get(id);
   680→                if (t == null) {
   681→                    throw new BusinessException(400, "引用的数据不存在: " + col.getKey() + "=" + id);
   682→                }
   683→                ordered.add(t);
   684→            }
   685→            // 冗余文本同样以 JSON 数组存储（与 id 数组顺序一致）
   686→            return objectMapper.writeValueAsString(ordered);
   687→        } catch (JsonProcessingException e) {
   688→            throw new BusinessException(400, "data-picker 配置或引用值非法: " + col.getKey());
   689→        }
   690→    }

   692→    /** 解析单个表单 column_config 中的 dataPicker 引用列，聚合到 result */
   693→    @SuppressWarnings("unchecked")
   694→    private void collectPickerRefs(FormDefinition def, Map<String, Map<String, Object>> result) {
   695→        String columnConfig = def.getColumnConfig();
   696→        if (columnConfig == null || columnConfig.isBlank()) {
   697→            return;
   698→        }
   699→        try {
   700→            List<Map<String, Object>> cols = objectMapper.readValue(columnConfig, List.class);
   701→            for (Map<String, Object> col : cols) {
   702→                Object pickerConfig = col.get("pickerConfig");
   703→                if (!(pickerConfig instanceof String s) || s.isBlank()) {
   704→                    continue;
   705→                }
   706→                Map<String, Object> picker = objectMapper.readValue(s, Map.class);
   707→                Object target = picker.get("sourceFormKey");
   708→                if (target == null || String.valueOf(target).isBlank()) {
   709→                    continue;
   710→                }
   711→                String targetKey = String.valueOf(target);
   712→                Map<String, Object> entry = result.computeIfAbsent(targetKey, k -> {
   713→                    Map<String, Object> m = new LinkedHashMap<>();
   714→                    m.put("count", 0);
   715→                    m.put("referencedBy", new ArrayList<String>());
   716→                    return m;
   717→                });
   718→                entry.put("count", ((Number) entry.get("count")).intValue() + 1);
   719→                ((List<String>) entry.get("referencedBy")).add(def.getKey());
   720→            }
   721→        } catch (JsonProcessingException e) {
   722→            // 非法 column_config 跳过（发布链路已校验，此处容错）
   723→        }
   724→    }

   726→    private BizDataVO toVO(BizDataContext ctx, Map<String, Object> row) {
   727→        Map<String, Object> data = new LinkedHashMap<>();
   728→        for (ColumnConfig c : ctx.columns()) {
   729→            Object v = row.get(c.getKey());
   730→            if (v != null) {
   731→                data.put(c.getKey(), "JSON".equals(c.getColumnType()) ? deserializeJsonValue(v) : v);
   732→            }
   733→        }
   734→        // embedded 模式：附加子表行（按 sort_no 升序），保留 id 供前端 diff
   735→        for (Map.Entry<String, BizDataContext.SubTableDef> e : ctx.subTables().entrySet()) {
   736→            BizDataContext.SubTableDef def = e.getValue();
   737→            if ("embedded".equals(def.subMode())) {
   738→                data.put(e.getKey(), readSubRows(def, String.valueOf(row.get("id"))));
   739→            }
   740→        }
   741→        Integer version = asInt(row.get("version"));
   742→        LocalDateTime createdAt = asDateTime(row.get("created_at"));
   743→        LocalDateTime updatedAt = asDateTime(row.get("updated_at"));
   744→        return new BizDataVO(String.valueOf(row.get("id")), data, version, createdAt, updatedAt);
   745→    }

   747→    /** config 模式行映射：主表列（toVO 逻辑）+ 虚拟列（virtualKey → joinField 值，含 dataPicker 冗余 _text） */
   748→    private BizDataVO toJoinVO(BizDataContext ctx, List<JoinSqlGenerator.JoinConfig> joins, Map<String, Object> row) {
   749→        BizDataVO vo = toVO(ctx, row);
   750→        Map<String, Object> data = vo.getData();

   752→            Object v = row.get(j.virtualKey());
   753→            if (v != null) {
   754→                data.put(j.virtualKey(), v);
   755→            }
   756→            Object vt = row.get(j.virtualKey() + "_text");
   757→            if (vt != null) {
   758→                data.put(j.virtualKey() + "_text", vt);
   759→            }
   760→        }
   761→        return vo;
   762→    }

   764→    /**
   765→     * sql 模式行映射：外层子查询输出列全量保留（可含聚合列），绕过 BizDataVO 系统列字段的仅主表列逻辑。
   766→     * <p>输出列名与声明列（columns）的 key 做大小写不敏感对齐：匹配某声明列 key（忽略大小写）时输出声明列精确 key，
   767→     * 保证返回 key 与字段元数据一致（字段 key 可为驼峰，如探测 JDBC 列标签 hireDate，前端按配置 key 取值）；
   768→     * 未匹配列统一转小写，消除 H2 等数据库对子查询输出列名规范化为大写的影响（与声明列小写 key 约定兼容）。
   769→     */
   770→    private BizDataVO toSqlVO(List<ColumnConfig> declared, Map<String, Object> row) {
   771→        Map<String, String> keyNormalizer = new HashMap<>();
   772→        if (declared != null) {
   773→            for (ColumnConfig c : declared) {
   774→                if (c.getKey() != null && !c.getKey().isBlank()) {
   775→                    // putIfAbsent：声明列存在仅大小写不同的重名时，保留先出现的精确 key
   776→                    keyNormalizer.putIfAbsent(c.getKey().toLowerCase(), c.getKey());
   777→                }
   778→            }
   779→        }
   780→        Map<String, Object> data = new LinkedHashMap<>();
   781→        for (Map.Entry<String, Object> e : row.entrySet()) {
   782→            String lower = e.getKey().toLowerCase();
   783→            data.put(keyNormalizer.getOrDefault(lower, lower), e.getValue());
   784→        }
   785→        data.remove("version");
   786→        data.remove("created_at");
   787→        data.remove("updated_at");
   788→        data.remove("tenant_id");
   789→        Integer version = asInt(row.get("version"));
   790→        LocalDateTime createdAt = asDateTime(row.get("created_at"));
   791→        LocalDateTime updatedAt = asDateTime(row.get("updated_at"));
   792→        // id 保留在 data 中（管理员 SQL 输出可能有非 id 主键/聚合值），BizDataVO.id 取行内 id 兜底
   793→        return new BizDataVO(String.valueOf(row.get("id")), data, version, createdAt, updatedAt);
   794→    }

   844→    /**
   845→     * 子表行增量 diff（update 场景）。
   846→     */
   847→    private void diffSubRows(BizDataContext.SubTableDef def, String bizId, List<?> rows) {
   848→        if (rows.size() > MAX_SUB_ROWS) {
   849→            throw new BusinessException(400, "子表行数超限（最多 " + MAX_SUB_ROWS + " 行）: " + def.tableName());
   850→        }
   851→        List<Map<String, Object>> existing = readSubRows(def, bizId);
   852→        Map<String, Map<String, Object>> existingById = new LinkedHashMap<>();
   853→        for (Map<String, Object> r : existing) {
   854→            existingById.put(String.valueOf(r.get("id")), r);
   855→        }
   856→
   857→        Set<String> keepIds = new HashSet<>();
   858→        int sortNo = 0;
   859→        for (Object row : rows) {
   860→            Map<String, Object> m = toRowMap(row, def.tableName());
   861→            Object rawId = m.get("id");
   862→            String rowId = rawId == null ? null : String.valueOf(rawId);
   863→            if (rowId != null && existingById.containsKey(rowId)) {
   864→                Map<String, Object> cur = existingById.get(rowId);
   865→                boolean changed = def.subKeys().stream()
   866→                        .anyMatch(k -> !Objects.equals(cur.get(k), m.get(k)));
   867→                if (changed || !Objects.equals(cur.get("sort_no"), sortNo)) {
   868→                    StringBuilder set = new StringBuilder();
   869→                    List<Object> params = new ArrayList<>();
   870→                    for (String k : def.subKeys()) {
   871→                        set.append(k).append(" = ?, ");
   872→                        params.add(m.get(k));
   873→                    }
   874→                    set.append("sort_no = ?");
   875→                    params.add(sortNo);
   876→                    params.add(tenantProvider.getTenantId());
   877→                    params.add(bizId);
   878→                    params.add(rowId);
   879→                    jdbcTemplate.update("UPDATE " + def.tableName() + " SET " + set
   880→                            + " WHERE tenant_id = ? AND biz_id = ? AND id = ?", params.toArray());
   881→                }
   882→                keepIds.add(rowId);
   883→            } else {
   884→                // 新行：剥离客户端传入的 id，走内部生成
   885→                Map<String, Object> newRow = new LinkedHashMap<>(m);
   886→                newRow.remove("id");
   887→                insertOneSubRow(def, bizId, newRow, sortNo);
   888→            }
   889→            sortNo++;
   890→        }
   891→
   892→        // 删除请求中不存在的现有行
   893→        if (existingById.size() > keepIds.size()) {
   894→            List<Object> params = new ArrayList<>();
   895→            params.add(tenantProvider.getTenantId());
   896→            params.add(bizId);
   897→            StringBuilder ph = new StringBuilder();
   898→            for (String id : existingById.keySet()) {
   899→                if (!keepIds.contains(id)) {
   900→                    if (ph.length() > 0) {
   901→                        ph.append(",");
   902→                    }
   903→                    ph.append("?");
   904→                    params.add(id);
   905→                }
   906→            }
   907→            jdbcTemplate.update("DELETE FROM " + def.tableName()
   908→                    + " WHERE tenant_id = ? AND biz_id = ? AND id IN (" + ph + ")", params.toArray());
   909→        }

   921→    /**
   922→     * 插入单行子表数据（id/biz_id/tenant_id/sort_no/version + 子业务列）。
   923→     */
   924→    private Map<String, Object> insertOneSubRow(BizDataContext.SubTableDef def, String bizId,
   925→                                                Map<String, Object> m, int sortNo) {
   926→        String rowId = UUID.randomUUID().toString().replace("-", "");
   927→        List<Object> params = new ArrayList<>();
   928→        params.add(rowId);
   929→        params.add(bizId);
   930→        params.add(tenantProvider.getTenantId());
   931→        params.add(sortNo);
   932→        params.add(1); // version
   933→
   934→        StringBuilder cols = new StringBuilder("id, biz_id, tenant_id, sort_no, version");
   935→        StringBuilder vals = new StringBuilder("?, ?, ?, ?, ?");
   936→        for (String k : def.subKeys()) {
   937→            cols.append(", ").append(k);
   938→            vals.append(", ?");
   939→            params.add(m.get(k));
   940→        }
   941→        jdbcTemplate.update("INSERT INTO " + def.tableName() + " (" + cols + ") VALUES (" + vals + ")",
   942→                params.toArray());
   943→
   944→        Map<String, Object> inserted = new LinkedHashMap<>(m);
   945→        inserted.put("id", rowId);
   946→        inserted.put("sort_no", sortNo);
   947→        return inserted;
   948→    }

   951→    @SuppressWarnings("unchecked")
   952→    private static Map<String, Object> toRowMap(Object row, String tableName) {
   953→        if (!(row instanceof Map<?, ?> m)) {
   954→            throw new BusinessException(400, "子表行数据格式非法（需对象）: " + tableName);
   955→        }
   956→        Map<String, Object> out = new LinkedHashMap<>();
   957→        for (Map.Entry<?, ?> e : m.entrySet()) {
   958→            out.put(String.valueOf(e.getKey()), e.getValue());
   959→        }
   960→        return out;
   961→    }
