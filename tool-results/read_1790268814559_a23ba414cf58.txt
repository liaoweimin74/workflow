    56→ *                只接受 config/sql，所以 `visual` 分支在系统里不可达）
    57→ */

    66→  /**
    67→   * 创建数据源。
    68→   *
    69→   * ⚠️ 状态规则：**API/SQL 创建即 ENABLED**（手动配置的数据源视为已发布），
    70→   *    其余类型一律 **DRAFT**（FORM/WORKFLOW 由表单事件建的才直接 ENABLED）。
    71→   * ⚠️ FORM/WORKFLOW 的 `sourceKey` **恒等于 formKey**（formKey 权威，忽略入参 sourceKey）。
    72→   */
    73→  async create(request: {
    74→    name: string | null
    75→    type: string | null
    76→    formKey: string | null
    77→    sourceKey: string | null
    78→    params: string | null
    79→  }): Promise<Record<string, unknown>> {
    80→    const tenantId = getTenantId()
    81→    const { name, type, formKey, sourceKey, params } = request
    82→
    83→    if (type === null || type.trim() === '') {
    84→      throw new BusinessException(400, '数据源类型 type 必填')
    85→    }
    86→    if (!SUPPORTED_TYPES.has(type)) {
    87→      throw new BusinessException(400, `不支持的数据源类型: ${type}`)
    88→    }
    89→    if (name === null || name.trim() === '') {
    90→      throw new BusinessException(400, '数据源名称不能为空')
    91→    }
    92→    if (await this.repository.existsByTenantIdAndName(tenantId, name)) {
    93→      throw new BusinessException(400, `数据源名称已存在: ${name}`)
    94→    }
    95→
    96→    // FORM/WORKFLOW：sourceKey 恒等于 formKey（formKey 权威）；其余类型以入参 sourceKey 为准
    97→    const formBound = type === TYPE_FORM || type === TYPE_WORKFLOW
    98→    const effSourceKey = formBound ? formKey : sourceKey
    99→    this.validateRequiredFields(type, formKey, effSourceKey, params)
   100→    if (formBound && !(await this.formDefRepository.existsByKey(String(formKey), tenantId))) {
   101→      throw new BusinessException(400, `绑定的表单不存在: ${String(formKey)}`)
   102→    }
   103→    if (effSourceKey === null || effSourceKey.trim() === '') {
   104→      throw new BusinessException(400, '数据源必须填写 sourceKey')
   105→    }
   106→    if (await this.repository.existsByTenantIdAndSourceKey(tenantId, effSourceKey)) {
   107→      throw new BusinessException(400, `数据源标识 sourceKey 已存在: ${effSourceKey}`)
   108→    }
   109→
   110→    let storedParams: string | null
   111→    if (type === TYPE_FORM || type === TYPE_SYSTEM) {
   112→      if (type === TYPE_FORM && hasQueryModeSegment(params)) {
   113→        await this.validateFormQueryConfig(params, formKey)
   114→        storedParams = mergeQueryConfig(generateParams(type, String(formKey), sourceKey), String(params))
   115→      } else {
   116→        storedParams = generateParams(type, String(formKey), sourceKey)
   117→      }
   118→    } else {
   119→      storedParams = params
   120→    }
   121→
   122→    const now = new Date()
   123→    const row: DataSourceRow = {
   124→      id: randomBytes(16).toString('hex'),
   125→      tenant_id: tenantId,
   126→      name,
   127→      type,
   128→      form_key: formKey,
   129→      source_key: effSourceKey,
   130→      // create 不设 formId（只有表单事件同步那条路径会设）
   131→      form_id: null,
   132→      params: storedParams,
   133→      // API/SQL 为手动配置：创建即发布（ENABLED）；其余类型仍 DRAFT
   134→      status: type === TYPE_API || type === TYPE_SQL ? STATUS_ENABLED : STATUS_DRAFT,
   135→      created_by: null,
   136→      created_at: now,
   137→      updated_at: now,
   138→    }
   139→    await this.repository.insertDefinition(row)
   140→    return toDto(row)
   141→  }
   142→
   143→  /**
   144→   * 原地更新数据源（`null` 表示不更新）。
   145→   *
   146→   * ⚠️ 已启用且「类型或绑定表单」变化时要重新校验发布状态 —— 这是 Java 里
   147→   *    `bindChanged` 那一段，容易漏。
   148→   * ⚠️ 系统内建数据源（`tenant_id = BUILT_IN_TENANT`）受保护：改名/换绑定都会
   149→   *    破坏预置语义与设计器引用，直接 400。
   150→   */
   151→  async update(
   152→    id: string,
   153→    request: {
   154→      name: string | null
   155→      type: string | null
   156→      formKey: string | null
   157→      sourceKey: string | null
   158→      params: string | null
   159→    },
   160→  ): Promise<Record<string, unknown>> {
   161→    const tenantId = getTenantId()
   162→    const current = await this.requireById(id)
   163→    this.requireNotBuiltIn(current, '修改')
   164→    const { name, type, formKey, sourceKey, params } = request
   165→
   166→    const newType = type === null || type.trim() === '' ? current.type : type
   167→    if (type !== null && type.trim() !== '' && !SUPPORTED_TYPES.has(newType)) {
   168→      throw new BusinessException(400, `不支持的数据源类型: ${newType}`)
   169→    }
   170→    if (
   171→      name !== null &&
   172→      name.trim() !== '' &&
   173→      name !== current.name &&
   174→      (await this.repository.existsByTenantIdAndName(tenantId, name))
   175→    ) {
   176→      throw new BusinessException(400, `数据源名称已存在: ${name}`)
   177→    }
   178→
   179→    const newFormKey = formKey === null ? current.form_key : formKey
   180→    const newSourceKey = sourceKey === null ? current.source_key : sourceKey
   181→    const formBound = newType === TYPE_FORM || newType === TYPE_WORKFLOW
   182→    // FORM/WORKFLOW：sourceKey 恒等于 formKey（formKey 权威），忽略入参 sourceKey 差异
   183→    const effNewSourceKey = formBound ? newFormKey : newSourceKey
   184→    // FORM：入参 params 非空时与 generateParams 的端点段合并（对齐 create；
   185→    // queryMode/joins/query/columns/params 由前端供给，端点段系统权威重建）
   186→    let newParams = params === null ? current.params : params
   187→    if (newType === TYPE_FORM && params !== null && params.trim() !== '') {
   188→      newParams = mergeQueryConfig(generateParams(newType, String(newFormKey), effNewSourceKey), String(newParams))
   189→    }
   190→    this.validateRequiredFields(newType, newFormKey, effNewSourceKey, newParams)
   191→    if (formBound && !(await this.formDefRepository.existsByKey(String(newFormKey), tenantId))) {
   192→      throw new BusinessException(400, `绑定的表单不存在: ${String(newFormKey)}`)
   193→    }
   194→    if (newType === TYPE_FORM) {
   195→      await this.validateFormQueryConfig(newParams, newFormKey)
   196→    }
   197→    // sourceKey 变更（不等于当前值）时校验租户内唯一；保持不变则跳过（自身不算冲突）
   198→    if (
   199→      effNewSourceKey !== current.source_key &&
   200→      (await this.repository.existsByTenantIdAndSourceKey(tenantId, String(effNewSourceKey)))
   201→    ) {
   202→      throw new BusinessException(400, `数据源标识 sourceKey 已存在: ${String(effNewSourceKey)}`)
   203→    }
   204→
   205→    // 已启用数据源若变更类型/绑定对象，须重新校验发布状态
   206→    const bindChanged =
   207→      current.type !== TYPE_FORM || (formKey !== null && formKey !== current.form_key)
   208→    if (current.status === STATUS_ENABLED && bindChanged) {
   209→      if (newType === TYPE_FORM) {
   210→        await this.requirePublishedForm(tenantId, String(newFormKey))
   211→      } else if (newType === TYPE_WORKFLOW) {
   212→        await this.requireWorkflowForm(tenantId, String(newFormKey))
   213→      }
   214→    }
   215→
   216→    const patch: Partial<DataSourceRow> = {
   217→      name: name === null ? current.name : name,
   218→      type: newType,
   219→      form_key: newFormKey,
   220→      source_key: effNewSourceKey,
   221→      params: newParams,
   222→    }
   223→    await this.repository.replaceDefinition(id, patch, new Date())
   224→    return toDto({ ...current, ...patch })
   225→  }
   226→
   227→  /**
   228→   * 启用数据源。
   229→   *
   230→   * ⚠️ FORM 须绑定**已发布**表单；WORKFLOW 须绑定**已发布且非 BUSINESS** 的表单
   231→   *    （业务表单没有流程实例，无法跨实例聚合）。
   232→   * ⚠️ FORM/SYSTEM 若 `params` 为空，启用时会补上自动生成的端点配置。
   233→   */
   234→  async enable(id: string): Promise<Record<string, unknown>> {
   235→    const tenantId = getTenantId()
   236→    const ds = await this.requireById(id)
   237→    this.validateRequiredFields(ds.type, ds.form_key, ds.source_key, ds.params)
   238→    if (ds.type === TYPE_FORM) {
   239→      await this.requirePublishedForm(tenantId, String(ds.form_key))
   240→      await this.validateFormQueryConfig(ds.params, ds.form_key)
   241→    } else if (ds.type === TYPE_WORKFLOW) {
   242→      await this.requireWorkflowForm(tenantId, String(ds.form_key))
   243→    }
   244→    const patch: Partial<DataSourceRow> = { status: STATUS_ENABLED }
   245→    if ((ds.type === TYPE_FORM || ds.type === TYPE_SYSTEM) && ds.params === null) {
   246→      patch.params = generateParams(ds.type, String(ds.form_key), ds.source_key)
   247→    }
   248→    await this.repository.replaceDefinition(id, patch, new Date())
   249→    return toDto({ ...ds, ...patch })
   250→  }
   251→
   252→  /**
   253→   * 禁用数据源（**不做引用校验**，也不影响已发布页面运行）。
   254→   * 系统内建数据源受保护：禁用会让设计器/页面取数失败，直接 400。
   255→   */
   256→  async disable(id: string): Promise<Record<string, unknown>> {
   257→    const ds = await this.requireById(id)
   258→    this.requireNotBuiltIn(ds, '禁用')
   259→    await this.repository.replaceDefinition(id, { status: STATUS_DISABLED }, new Date())
   260→    return toDto({ ...ds, status: STATUS_DISABLED })
   261→  }
   262→
   263→  /**
   264→   * 删除数据源（任意状态可删，**被页面引用时拒绝**）。
   265→   *
   266→   * 引用统计覆盖两种绑定方式，取**并集**（任一命中即拒绝）：
   267→   *   1. `PageDefinition.dataSourceId` 列
   268→   *   2. PAGE 类型页面 `schema.dataSources[].refId`（软删除的 ARCHIVED 页面不算）
   269→   * ⚠️ Java 的短路顺序：列引用 > 0 时直接返回该数，不再扫 schema ——
   270→   *    所以报出来的「被 N 个页面引用」里的 N 语义**不是总数**。照抄。
   271→   * ⚠️ 系统内建数据源**不允许删除**（预置语义；重启会被 seeder 补回，删除是假象）。
   272→   */
   273→  async remove(id: string): Promise<void> {
   274→    const ds = await this.requireById(id)
   275→    this.requireNotBuiltIn(ds, '删除')
   276→    const refCount = await this.countRefs(ds.tenant_id, id)
   277→    if (refCount > 0) {
   278→      throw new BusinessException(400, `数据源已被 ${refCount} 个页面引用，无法删除`)
   279→    }
   280→    await this.repository.deleteDataSource(id)
   281→  }

   283→  /** 取数据源（含 SYSTEM 跨租户可见）；不存在 → 业务 404。 */
   284→  private async requireById(id: string): Promise<DataSourceRow> {
   285→    const row = await this.repository.findByIdAccessible(id, getTenantId())
   286→    if (row === null) throw new BusinessException(404, `数据源不存在: ${id}`)
   287→    return row
   288→  }
   289→
   290→  /**
   291→   * 内建保护：`tenant_id = BUILT_IN_TENANT` 的预置行不允许修改/删除/禁用。
   292→   * enable 不拦（对已 ENABLED 的内建行是幂等空操作，且拦了反而让前端开关卡死）。
   293→   */
   294→  private requireNotBuiltIn(row: DataSourceRow, action: string): void {
   295→    if (row.tenant_id === BUILT_IN_TENANT) {
   296→      throw new BusinessException(400, `系统内建数据源不允许${action}: ${row.name}`)
   297→    }
   298→  }

   300→  /** 统计引用指定数据源的页面数（列 ∪ schema）。 */
   301→  private async countRefs(tenantId: string, dataSourceId: string): Promise<number> {
   302→    const { rows: columnPages } = await this.pageDefRepository.findPage(
   303→      tenantId,
   304→      { status: null, name: null, type: null, dataSourceId },
   305→      0,
   306→      Number.MAX_SAFE_INTEGER,
   307→    )
   308→    if (columnPages.length > 0) return columnPages.length
   309→
   310→    // PAGE 类型页面的引用声明在 schema.dataSources[].refId（dataSourceId 列为空），需扫描；
   311→    // 页面软删除（ARCHIVED）后不再使用，其 schema 引用不阻塞删除
   312→    const { rows: pageTypePages } = await this.pageDefRepository.findPage(
   313→      tenantId,
   314→      { status: null, name: null, type: 'PAGE' },
   315→      0,
   316→      Number.MAX_SAFE_INTEGER,
   317→    )
   318→    let schemaRefs = 0
   319→    for (const page of pageTypePages) {
   320→      if (page.status === 'ARCHIVED') continue
   321→      if (schemaRefsDataSource(page.schema, dataSourceId)) schemaRefs++
   322→    }
   323→    return schemaRefs
   324→  }

   326→  private async requirePublishedForm(tenantId: string, formKey: string): Promise<void> {
   327→    const published = await this.formDefRepository.findLatestPublishedByKey(formKey, tenantId)
   328→    if (published === null) {
   329→      throw new BusinessException(400, `绑定的表单未发布，无法启用: ${formKey}`)
   330→    }
   331→  }

   333→  private async requireWorkflowForm(tenantId: string, formKey: string): Promise<void> {
   334→    if (!(await this.formDefRepository.existsByKey(formKey, tenantId))) {
   335→      throw new BusinessException(400, `表单不存在: ${formKey}`)
   336→    }
   337→    const published = await this.formDefRepository.findLatestPublishedByKey(formKey, tenantId)
   338→    if (published === null) {
   339→      throw new BusinessException(400, `工作流表单必须先发布: ${formKey}`)
   340→    }
   341→    if (published.type === 'BUSINESS') {
   342→      throw new BusinessException(400, `业务表单不可配置为工作流表单数据源: ${formKey}`)
   343→    }
   344→  }

   346→  /** 按类型校验必填项（对齐 Java `validateRequiredFields`，错误消息逐字一致）。 */
   347→  private validateRequiredFields(
   348→    type: string,
   349→    formKey: string | null,
   350→    sourceKey: string | null,
   351→    params: string | null,
   352→  ): void {
   353→    if (type === TYPE_FORM || type === TYPE_WORKFLOW) {
   354→      if (formKey === null || formKey.trim() === '') {
   355→        throw new BusinessException(400, `${type} 类型数据源必须绑定表单 formKey`)
   356→      }
   357→      return
   358→    }
   359→    if (type === TYPE_SYSTEM || type === TYPE_API || type === TYPE_SQL) {
   360→      if (sourceKey === null || sourceKey.trim() === '') {
   361→        throw new BusinessException(400, `${type} 类型数据源必须填写 sourceKey`)
   362→      }
   363→      if (type === TYPE_SYSTEM && !SYSTEM_SOURCE_KEYS.has(sourceKey)) {
   364→        throw new BusinessException(400, `未注册的系统数据源: ${sourceKey}`)
   365→      }
   366→      if (type === TYPE_API) {
   367→        // LookupFetchConfig 契约：params 须为 JSON 对象且 action 必填
   368→        if (params === null || params.trim() === '') {
   369→          throw new BusinessException(400, 'API 数据源参数 params 必须包含 action（API 路径）')
   370→        }
   371→        let parsed: unknown
   372→        try {
   373→          parsed = JSON.parse(params)
   374→        } catch (error) {
   375→          throw new BusinessException(
   376→            400,
   377→            `API 数据源参数 params 必须是合法 JSON: ${error instanceof Error ? error.message : String(error)}`,
   378→          )
   379→        }
   380→        if (parsed === null || typeof parsed !== 'object' || Array.isArray(parsed)) {
   381→          throw new BusinessException(400, 'API 数据源参数 params 必须是 JSON 对象')
   382→        }
   383→        const action = (parsed as Record<string, unknown>).action
   384→        if (action === null || action === undefined || String(action).trim() === '') {
   385→          throw new BusinessException(400, 'API 数据源参数 params 必须包含 action（API 路径）')
   386→        }
   387→      }
   388→      // SQL：仅要求 sourceKey（query 配置在 params 中，DRAFT 阶段可为空）
   389→    }
   390→  }
   391→
   392→  /**
   393→   * FORM 数据源 `params` 的 `queryMode` 配置段校验。
   394→   *
   395→   * 无 queryMode 段直接返回（单表查询，向后兼容）；
   396→   * `config` 校验 joins；`sql` 复用 `SqlTemplateEngine.validate`（SELECT / `:tenantId` /
   397→   * 列匹配 / 参数白名单）；未知模式 → 400。
   398→   *
   399→   * ⚠️ `validate` 抛的是 `IllegalArgumentException` 形态（HTTP 400），这里统一转成
   400→   *    `BusinessException(400, 同文案)` —— 两条路径的错误码一致，且消息逐字保留。
   401→   */
   402→  private async validateFormQueryConfig(params: string | null, formKey: string | null): Promise<void> {
   403→    if (params === null || params.trim() === '') return
   404→    let root: unknown
   405→    try {
   406→      root = JSON.parse(params)
   407→    } catch (error) {
   408→      throw new BusinessException(
   409→        400,
   410→        `数据源参数 params 必须是合法 JSON: ${error instanceof Error ? error.message : String(error)}`,
   411→      )
   412→    }
   413→    if (root === null || typeof root !== 'object' || Array.isArray(root)) {
   414→      throw new BusinessException(400, '数据源参数 params 必须是 JSON 对象')
   415→    }
   416→    const record = root as Record<string, unknown>
   417→    const modeNode = record.queryMode
   418→    if (modeNode === null || modeNode === undefined || String(modeNode).trim() === '') return
   419→    const mode = String(modeNode)
   420→    if (mode === 'config') {
   421→      await this.validateConfigJoins(record.joins, formKey)
   422→      return
   423→    }
   424→    if (mode === 'sql') {
   425→      this.validateSqlConfig(record)
   426→      return
   427→    }
   428→    throw new BusinessException(400, `未知查询模式 queryMode: ${mode}（支持 config / sql）`)
   429→  }

   431→  /** sql 模式：`query`/`columns`/参数白名单复用 `SqlTemplateEngine.validate`。 */
   432→  private validateSqlConfig(root: Record<string, unknown>): void {
   433→    const query = textOf(root, 'query')
   434→    const columns = parseSqlColumns(root.columns)
   435→    const declaredParams = parseStringList(root.params)
   436→    try {
   437→      validate(query, columns, declaredParams)
   438→    } catch (error) {
   439→      throw new BusinessException(400, error instanceof Error ? error.message : String(error))
   440→    }

   443→  /** config 模式：joins[] 结构校验（别名/目标表/字段/虚拟列唯一性）。
   444→   *
   445→   * alias 可缺省：前端不录入，由运行时 parseJoins 自动分配（j1/j2/...）；
   446→   * 传入则校验格式与唯一性。目标表支持内建数据源（join-target-catalog 白名单）：
   447→   * SYSTEM 目标不查 form_def，且 foreignField/joinField 必须是目录内物理列。
   448→   *
   449→   * 新增（对齐 Java 新版）：主表关联字段格式 + **存在性**校验（避免手输错列名保存成功、
   450→   * 运行时才爆 SQL 错）；FORM 目标的 foreignField/joinField 存在性校验（与 SYSTEM 目标
   451→   * 白名单对称）；virtualKey 与主表列冲突校验（SELECT m.* 同名重复列）。
   452→   * 主表/目标表单未发布时存在性校验**优雅降级跳过**（启用时 requirePublishedForm 兜底），
   453→   * 格式校验始终生效。 */
   454→  private async validateConfigJoins(joins: unknown, mainFormKey: string | null): Promise<void> {
   455→    if (!Array.isArray(joins) || joins.length === 0) {
   456→      throw new BusinessException(400, 'queryMode=config 时必须配置至少一个关联 joins')
   457→    }
   458→    const tenantId = getTenantId()
   459→    // 主表物理列候选：业务列 + 系统列（id/created_at/updated_at；SELECT m.* 全量带出，
   460→    // virtualKey 同名会重复列）。主表单未发布 → null（存在性校验降级跳过）
   461→    const mainColumns = await this.publishedColumnKeys(mainFormKey, tenantId, true)
   462→    const virtualKeys = new Set<string>()
   463→    const aliases = new Set<string>()
   464→    let idx = 0
   465→    for (const item of joins) {
   466→      idx++
   467→      if (item === null || typeof item !== 'object' || Array.isArray(item)) {
   468→        throw new BusinessException(400, `joins 第 ${idx} 项必须是对象`)
   469→      }
   470→      const join = item as Record<string, unknown>
   471→      const alias = textOf(join, 'alias')
   472→      if (alias !== null) {
   473→        if (!/^[a-zA-Z_][a-zA-Z0-9_]*$/.test(alias)) {
   474→          throw new BusinessException(400, `joins 第 ${idx} 项 alias 非法: ${alias}`)
   475→        }
   476→        if (aliases.has(alias)) {
   477→          throw new BusinessException(400, `joins 第 ${idx} 项 alias 重复: ${alias}`)
   478→        }
   479→        aliases.add(alias)
   480→      }
   481→      const targetFormKey = textOf(join, 'targetFormKey')
   482→      if (targetFormKey === null || targetFormKey.trim() === '') {
   483→        throw new BusinessException(400, `joins 第 ${idx} 项必须指定目标表 targetFormKey`)
   484→      }
   485→      let foreignCandidates: ReadonlySet<string> | null = null
   486→      if (isJoinTargetSystemKey(targetFormKey)) {
   487→        // 内建目标：物理列白名单来自 join-target-catalog（不走 form_def）
   488→        const system = joinTargetSystemByKey(targetFormKey)
   489→        foreignCandidates = system !== null ? new Set(system.columns.map((c) => c.key)) : null
   490→      } else {
   491→        if (!/^[a-zA-Z][a-zA-Z0-9_]{0,63}$/.test(targetFormKey)) {
   492→          throw new BusinessException(400, `非法关联目标: ${targetFormKey}`)
   493→        }
   494→        if (!(await this.formDefRepository.existsByKey(targetFormKey, tenantId))) {
   495→          throw new BusinessException(400, `目标表单不存在: ${targetFormKey}`)
   496→        }
   497→        // FORM 目标：物理列候选 = 目标表单业务列 + id（对齐前端 targetFormColumns 候选）
   498→        const targetColumns = await this.publishedColumnKeys(targetFormKey, tenantId, false)
   499→        if (targetColumns !== null) foreignCandidates = new Set(targetColumns)
   500→      }
   501→      const foreignField = textOf(join, 'foreignField')
   502→      const localField = textOf(join, 'localField')
   503→      requireJoinField(join, 'foreignField', '目标表关联字段', idx)
   504→      requireJoinField(join, 'localField', '主表关联字段', idx)
   505→      requireJoinField(join, 'joinField', '显示字段', idx)
   506→      // 主表关联字段：格式 + 存在性（引用列必须在主表物理列中，杜绝运行时畸形 SQL）
   507→      if (localField !== null && !JOIN_FIELD_PATTERN.test(localField)) {
   508→        throw new BusinessException(400, `joins 第 ${idx} 项主表关联字段非法: ${localField}`)
   509→      }
   510→      if (mainColumns !== null && localField !== null && !mainColumns.has(localField)) {
   511→        throw new BusinessException(400, `joins 第 ${idx} 项主表关联字段不在绑定表单列中: ${localField}`)
   512→      }
   513→      if (foreignCandidates !== null) {
   514→        if (foreignField === null || !foreignCandidates.has(foreignField)) {
   515→          const label = isJoinTargetSystemKey(targetFormKey)
   516→            ? `joins 第 ${idx} 项目标表关联字段不在内建数据源物理列中: ${String(foreignField)}`
   517→            : `joins 第 ${idx} 项目标表关联字段不在目标表单列中: ${String(foreignField)}`
   518→          throw new BusinessException(400, label)
   519→        }
   520→        const joinFieldValue = textOf(join, 'joinField')
   521→        if (joinFieldValue === null || !foreignCandidates.has(joinFieldValue)) {
   522→          const label = isJoinTargetSystemKey(targetFormKey)
   523→            ? `joins 第 ${idx} 项显示字段不在内建数据源物理列中: ${String(joinFieldValue)}`
   524→            : `joins 第 ${idx} 项显示字段不在目标表单列中: ${String(joinFieldValue)}`
   525→          throw new BusinessException(400, label)
   526→        }
   527→      }
   528→      requireJoinField(join, 'label', '显示名称', idx)
   529→      const virtualKey = textOf(join, 'virtualKey')
   530→      if (virtualKey === null || virtualKey.trim() === '') {
   531→        throw new BusinessException(400, `joins 第 ${idx} 项必须指定虚拟列标识 virtualKey`)
   532→      }
   533→      if (virtualKeys.has(virtualKey)) {
   534→        throw new BusinessException(400, `虚拟列 virtualKey 重复: ${virtualKey}`)
   535→      }
   536→      if (mainColumns !== null && mainColumns.has(virtualKey)) {
   537→        throw new BusinessException(400, `joins 第 ${idx} 项虚拟列 virtualKey 与主表列冲突: ${virtualKey}`)
   538→      }
   539→      virtualKeys.add(virtualKey)
   540→    }
   541→  }

   543→  /**
   544→   * 已发布表单的物理列候选集合：业务列（column_config）+ 系统列。
   545→   *
   546→   * @param formKey    表单 key（null/空白 → null）
   547→   * @param tenantId   租户
   548→   * @param withTimeColumns 是否附加 created_at/updated_at（主表候选要；目标表 JOIN 候选不要）
   549→   * @returns 未发布 → null（调用方降级跳过存在性校验）；id 始终在候选中
   550→   */
   551→  private async publishedColumnKeys(
   552→    formKey: string | null,
   553→    tenantId: string,
   554→    withTimeColumns: boolean,
   555→  ): Promise<Set<string> | null> {
   556→    if (formKey === null || String(formKey).trim() === '') return null
   557→    const published = await this.formDefRepository.findLatestPublishedByKey(String(formKey), tenantId)
   558→    if (published === null) return null
   559→    const keys = new Set<string>(['id'])
   560→    if (withTimeColumns) {
   561→      keys.add('created_at')
   562→      keys.add('updated_at')
   563→    }
   564→    for (const column of parseBusinessColumnConfig(published.column_config)) {
   565→      if (column.key !== null && column.key.trim() !== '') keys.add(column.key)
   566→    }
   567→    return keys
   568→  }
   569→}

   571→/** 单个 join 的必填字段校验（对齐 Java `requireJoinField`）。 */
   572→function requireJoinField(
   573→  join: Record<string, unknown>,
   574→  field: string,
   575→  label: string,
   576→  idx: number,
   577→): void {
   578→  const value = textOf(join, field)
   579→  if (value === null || value.trim() === '') {
   580→    throw new BusinessException(400, `joins 第 ${idx} 项必须指定${label} ${field}`)
   581→  }
   582→}

   584→function textOf(node: Record<string, unknown>, field: string): string | null {
   585→  const value = node[field]

   589→/** `params` 是否含 `queryMode` 配置段（非合法 JSON 视为无，交给后续校验报错）。 */

   605→/**
   606→ * 为 FORM/SYSTEM 数据源生成 `params` JSON（只读配置，UI 不可编辑）。
   607→ *
   608→ * ⚠️ 这是一个**字符串**字段，且 Java 用 `ObjectNode.toString()` 产出**紧凑 JSON**、
   609→ *    键顺序是插入顺序。所以下面的书写顺序就是契约的一部分 —— 不能改写键的顺序，
   610→ *    也不能用 `JSON.stringify` 之外的格式化方式。
   611→ */
   612→export function generateParams(
   613→  type: string,
   614→  formKey: string,
   615→  sourceKey: string | null,
   616→): string {
   617→  if (type === TYPE_FORM) {
   618→    const base = `/api/v1/biz-data/${formKey}`
   619→    return JSON.stringify({
   620→      list: { action: base, method: 'GET', parse: 'records', totalParse: 'total' },
   621→      create: { action: base, method: 'POST' },
   622→      get: { action: `${base}/{id}`, method: 'GET' },
   623→      update: { action: `${base}/{id}`, method: 'PUT' },
   624→      delete: { action: `${base}/{id}`, method: 'DELETE' },
   625→    })
   626→  }
   627→  if (type === TYPE_SYSTEM) {
   628→    return JSON.stringify({
   629→      list: { action: `/api/v1/internal/system/${mapSystemInternalKey(sourceKey)}`, method: 'GET' },
   630→    })
   631→  }
   632→  return JSON.stringify({})
   633→}

   635→/**
   636→ * sql 模式声明列解析（对齐 Java `DataSourceDefinitionService.parseSqlColumns`）。
   637→ *
   638→ * ⚠️ `ref = key`（不是 `m.key`）—— 这是 `SqlTemplateEngine` 的语义：
   639→ *    引用的是外层子查询的输出列名。布尔字段用 `boolVal`（仅 `true` 算 true）。
   640→ */
   641→function parseSqlColumns(node: unknown): QueryColumn[] {
   642→  const out: QueryColumn[] = []
   643→  if (!Array.isArray(node)) return out
   644→  for (const item of node) {
   645→    if (item === null || typeof item !== 'object' || Array.isArray(item)) continue
   646→    const record = item as Record<string, unknown>
   647→    const key = textOf(record, 'key')
   648→    out.push({
   649→      key: String(key),
   650→      ref: String(key),
   651→      columnType: textOf(record, 'columnType') ?? '',
   652→      sortable: record.sortable === true,
   653→      filterable: record.filterable === true,
   654→    })
   655→  }
   656→  return out
   657→}

   659→/** 字符串数组解析（对齐 Java `parseStringList`：只收非空白文本项）。 */
   660→function parseStringList(node: unknown): string[] {
   661→  const out: string[] = []
   662→  if (!Array.isArray(node)) return out
   663→  for (const item of node) {
   664→    if (typeof item === 'string' && item.trim() !== '') out.push(item)
   665→  }
   666→  return out
   667→}

   669→/**
   670→ * sourceKey → internal API 路径段（dept-tree→dept-tree，user-tree→users；
   671→ * 新 6 个见 `mapSystemInternalPath`）。未知 key 报错文案逐字保留。
   672→ */
   673→function mapSystemInternalKey(sourceKey: string | null): string {
   674→  const mapped = mapSystemInternalPath(sourceKey)
   675→  if (mapped === '') {
   676→    throw new BusinessException(400, `未注册的系统数据源: ${String(sourceKey)}`)
   677→  }
   678→  return mapped

   681→/** 合并：生成端点 params 之上叠加 queryMode 配置段。 */
   682→export function mergeQueryConfig(generated: string, params: string): string {
   683→  let out: Record<string, unknown>
   684→  let input: Record<string, unknown>
   685→  try {
   686→    out = JSON.parse(generated) as Record<string, unknown>
   687→    input = JSON.parse(params) as Record<string, unknown>
   688→  } catch (error) {
   689→    throw new BusinessException(
   690→      400,
   691→      `数据源参数 params 必须是合法 JSON: ${error instanceof Error ? error.message : String(error)}`,
   692→    )
   693→  }
   694→  for (const field of ['queryMode', 'joins', 'query', 'columns', 'params']) {
   695→    if (field in input) out[field] = input[field]
   696→  }
   697→  return JSON.stringify(out)
   698→}

   700→/** schema 的 `dataSources[].refId` 是否指向指定数据源。 */
   701→export function schemaRefsDataSource(schema: string | null, dataSourceId: string): boolean {
   702→  if (schema === null || schema.trim() === '') return false
   703→  try {
   704→    const root: unknown = JSON.parse(schema)
   705→    if (root === null || typeof root !== 'object') return false
   706→    const dataSources = (root as Record<string, unknown>).dataSources
   707→    if (!Array.isArray(dataSources)) return false
   708→    return dataSources.some((ds) => {
   709→      if (ds === null || typeof ds !== 'object') return false
   710→      const refId = (ds as Record<string, unknown>).refId
   711→      return refId !== null && refId !== undefined && String(refId) === dataSourceId
   712→    })
   713→  } catch {
   714→    return false
   715→  }
   716→}
