    79→    /**
    80→     * 创建数据源（默认 DRAFT）。
    81→     * 校验：type 必填且合法；同租户 name 唯一；按类型必填项（FORM→formKey + 表单存在；
    82→     * SYSTEM/API→sourceKey；API→params 须为合法 JSON）。

    84→     * 注意：此方法仅供系统内部调用，用户不能直接创建数据源。
    85→     */
    86→    @Transactional
    87→    public DataSourceDefinition create(String name, String type, String formKey, String sourceKey, String params) {
    88→        String tenantId = tenantProvider.getTenantId();
    89→
    90→        if (type == null || type.isBlank()) {
    91→            throw new BusinessException(400, "数据源类型 type 必填");
    92→        }
    93→        if (!SUPPORTED_TYPES.contains(type)) {
    94→            throw new BusinessException(400, "不支持的数据源类型: " + type);
    95→        }
    96→        if (name == null || name.isBlank()) {
    97→            throw new BusinessException(400, "数据源名称不能为空");
    98→        }
    99→        if (dsRepository.existsByTenantIdAndName(tenantId, name)) {
   100→            throw new BusinessException(400, "数据源名称已存在: " + name);
   101→        }
   102→        // FORM/WORKFLOW：sourceKey 恒等于 formKey（formKey 权威）；其余类型以入参 sourceKey 为准
   103→        boolean formBound = TYPE_FORM.equals(type) || TYPE_WORKFLOW.equals(type);
   104→        String effSourceKey = formBound ? formKey : sourceKey;
   105→        validateRequiredFields(type, formKey, effSourceKey, params);
   106→        if (formBound && !formDefRepository.existsByTenantIdAndKey(tenantId, formKey)) {
   107→            throw new BusinessException(400, "绑定的表单不存在: " + formKey);
   108→        }
   109→        if (effSourceKey == null || effSourceKey.isBlank()) {
   110→            throw new BusinessException(400, "数据源必须填写 sourceKey");
   111→        }
   112→        if (dsRepository.existsByTenantIdAndSourceKey(tenantId, effSourceKey)) {
   113→            throw new BusinessException(400, "数据源标识 sourceKey 已存在: " + effSourceKey);
   114→        }
   115→
   116→        DataSourceDefinition ds = new DataSourceDefinition();
   117→        ds.setId(UUID.randomUUID().toString().replace("-", ""));
   118→        ds.setTenantId(tenantId);
   119→        ds.setName(name);
   120→        ds.setType(type);
   121→        ds.setFormKey(formKey);
   122→        ds.setSourceKey(effSourceKey);
   123→        if (TYPE_FORM.equals(type) || TYPE_SYSTEM.equals(type)) {
   124→            if (TYPE_FORM.equals(type) && hasQueryModeSegment(params)) {
   125→                // FORM 带 queryMode 配置段：校验后与自动生成端点合并保存
   126→                validateFormQueryConfig(params);
   127→                ds.setParams(mergeQueryConfig(generateParams(type, formKey, sourceKey), params));
   128→            } else {
   129→                ds.setParams(generateParams(type, formKey, sourceKey));
   130→            }
   131→        } else {
   132→            ds.setParams(params);
   133→        }
   134→        // API/SQL 为手动配置的数据源：创建即发布（ENABLED）；其余类型仍 DRAFT（防御：手动创建仅 API/SQL）
   135→        boolean manualPublish = TYPE_API.equals(type) || TYPE_SQL.equals(type);
   136→        ds.setStatus(manualPublish ? STATUS_ENABLED : STATUS_DRAFT);
   137→        return dsRepository.save(ds);
   138→    }

   140→    /**
   141→     * 原地更新数据源（name/type/formKey/sourceKey/params；null 表示不更新）。
   142→     * 若当前 ENABLED 且 type/formKey 变更，重新校验（FORM 须仍指向已发布表单）。
   143→     * 
   144→     * 注意：此方法仅供系统内部调用，用户不能直接编辑数据源。
   145→     */
   146→    @Transactional
   147→    public DataSourceDefinition update(String id, String name, String type, String formKey,
   148→                                       String sourceKey, String params) {
   149→        String tenantId = tenantProvider.getTenantId();
   150→        DataSourceDefinition ds = getById(id);
   151→
   152→        String newType = type == null || type.isBlank() ? ds.getType() : type;
   153→        if (type != null && !type.isBlank() && !SUPPORTED_TYPES.contains(newType)) {
   154→            throw new BusinessException(400, "不支持的数据源类型: " + newType);
   155→        }
   156→        if (name != null && !name.isBlank() && !name.equals(ds.getName())
   157→                && dsRepository.existsByTenantIdAndName(tenantId, name)) {
   158→            throw new BusinessException(400, "数据源名称已存在: " + name);
   159→        }
   160→
   161→        String newFormKey = formKey == null ? ds.getFormKey() : formKey;
   162→        String newSourceKey = sourceKey == null ? ds.getSourceKey() : sourceKey;
   163→        String newParams = params == null ? ds.getParams() : params;
   164→        // FORM/WORKFLOW：sourceKey 恒等于 formKey（formKey 权威），忽略入参 sourceKey 差异
   165→        boolean formBound = TYPE_FORM.equals(newType) || TYPE_WORKFLOW.equals(newType);
   166→        String effNewSourceKey = formBound ? newFormKey : newSourceKey;
   167→        validateRequiredFields(newType, newFormKey, effNewSourceKey, newParams);
   168→        if (formBound && !formDefRepository.existsByTenantIdAndKey(tenantId, newFormKey)) {
   169→            throw new BusinessException(400, "绑定的表单不存在: " + newFormKey);
   170→        }
   171→        if (TYPE_FORM.equals(newType)) {
   172→            // FORM 查询配置段（queryMode/joins/query/columns/params）保存校验
   173→            validateFormQueryConfig(newParams);
   174→        }
   175→        // sourceKey 变更（不等于当前值）时校验租户内唯一；保持不变则跳过（自身不算冲突）
   176→        if (!java.util.Objects.equals(effNewSourceKey, ds.getSourceKey())
   177→                && dsRepository.existsByTenantIdAndSourceKey(tenantId, effNewSourceKey)) {
   178→            throw new BusinessException(400, "数据源标识 sourceKey 已存在: " + effNewSourceKey);
   179→        }
   180→
   181→        // 已启用数据源若变更类型/绑定对象，须重新校验发布状态
   182→        boolean bindChanged = !TYPE_FORM.equals(ds.getType()) || (formKey != null && !formKey.equals(ds.getFormKey()));
   183→        if (STATUS_ENABLED.equals(ds.getStatus()) && bindChanged) {
   184→            if (TYPE_FORM.equals(newType)) {
   185→                requirePublishedForm(tenantId, newFormKey);
   186→            } else if (TYPE_WORKFLOW.equals(newType)) {
   187→                requireWorkflowForm(tenantId, newFormKey);
   188→            }
   189→        }
   190→
   191→        ds.setName(name == null ? ds.getName() : name);
   192→        ds.setType(newType);
   193→        ds.setFormKey(newFormKey);
   194→        ds.setSourceKey(effNewSourceKey);
   195→        ds.setParams(newParams);
   196→        return dsRepository.save(ds);
   197→    }

   200→     * 启用数据源：校验按类型必填项齐全；FORM 类型须绑定已发布表单。成功置 ENABLED。

   202→     * 注意：此方法仅供系统内部调用，用户不能直接启用数据源。

   206→        String tenantId = tenantProvider.getTenantId();
   207→        DataSourceDefinition ds = getById(id);
   208→        validateRequiredFields(ds.getType(), ds.getFormKey(), ds.getSourceKey(), ds.getParams());
   209→        if (TYPE_FORM.equals(ds.getType())) {
   210→            requirePublishedForm(tenantId, ds.getFormKey());
   211→            validateFormQueryConfig(ds.getParams());
   212→        } else if (TYPE_WORKFLOW.equals(ds.getType())) {
   213→            requireWorkflowForm(tenantId, ds.getFormKey());
   214→        }
   215→        ds.setStatus(STATUS_ENABLED);
   216→        if ((TYPE_FORM.equals(ds.getType()) || TYPE_SYSTEM.equals(ds.getType())) && ds.getParams() == null) {
   217→            ds.setParams(generateParams(ds.getType(), ds.getFormKey(), ds.getSourceKey()));
   218→        }
   219→        return dsRepository.save(ds);
   220→    }

   234→    /**
   235→     * 删除数据源：任意状态可删除（ENABLED/DISABLED 均可），但被页面引用时拒绝（400）。
   236→     * 引用统计覆盖两种绑定方式：
   237→     * 1) PageDefinition.dataSourceId 列（VIEW 新协议，迁移器回填）；
   238→     * 2) PAGE 类型页面 schema.dataSources[].refId（设计器 dataSources 声明）。
   239→     * 两者取并集，任一命中即拒绝删除。
   240→     * 
   241→     * 注意：此方法仅供系统内部调用，用户不能直接删除数据源。
   242→     */
   243→    @Transactional
   244→    public void delete(String id) {
   245→        DataSourceDefinition ds = getById(id);
   246→        String tenantId = ds.getTenantId();
   247→        long refCount = countRefs(tenantId, id);
   248→        if (refCount > 0) {
   249→            throw new BusinessException(400, "数据源已被 " + refCount + " 个页面引用，无法删除");
   250→        }
   251→        dsRepository.delete(ds);
   252→    }

   254→    /**
   255→     * 统计当前租户内引用指定数据源的页面数（dataSourceId 列 + PAGE schema dataSources[].refId 并集）。
   256→     */
   257→    private long countRefs(String tenantId, String dataSourceId) {
   258→        long columnRefs = pageRepository.countByTenantIdAndDataSourceId(tenantId, dataSourceId);
   259→        if (columnRefs > 0) {
   260→            return columnRefs;
   261→        }
   262→        // PAGE 类型页面引用声明在 schema.dataSources[].refId（dataSourceId 列为空），需扫描；
   263→        // 页面软删除（ARCHIVED）后不再使用，其 schema 引用不阻塞删除
   264→        long schemaRefs = 0;
   265→        Page<PageDefinition> pages = pageRepository
   266→                .findByTenantIdAndTypeOrderByUpdatedAtDesc(tenantId, "PAGE", PageRequest.of(0, Integer.MAX_VALUE));
   267→        for (PageDefinition page : pages.getContent()) {
   268→            if ("ARCHIVED".equals(page.getStatus())) {
   269→                continue;
   270→            }
   271→            if (schemaRefsDataSource(page.getSchema(), dataSourceId)) {
   272→                schemaRefs++;
   273→            }
   274→        }
   275→        return schemaRefs;
   276→    }

   278→    /**
   279→     * 判断页面 schema 的 dataSources[].refId 是否指向指定数据源。
   280→     */
   281→    private boolean schemaRefsDataSource(String schema, String dataSourceId) {
   282→        if (schema == null || schema.isBlank()) {
   283→            return false;
   284→        }
   285→        try {
   286→            JsonNode root = objectMapper.readTree(schema);
   287→            JsonNode dataSources = root.path("dataSources");
   288→            if (!dataSources.isArray()) {
   289→                return false;
   290→            }
   291→            for (JsonNode ds : dataSources) {
   292→                String refId = ds.path("refId").asText(null);
   293→                if (dataSourceId.equals(refId)) {
   294→                    return true;
   295→                }
   296→            }
   297→        } catch (JsonProcessingException e) {
   298→            return false;
   299→        }
   300→        return false;
   301→    }

   387→    /** 按数据源类型找适配器（数据源须 ENABLED） */
   388→    private DataSourceAdapter adapterOf(String id) {
   389→        DataSourceDefinition ds = getById(id);
   390→        if (!STATUS_ENABLED.equals(ds.getStatus())) {
   391→            throw new BusinessException(400, "数据源未启用，无法访问: " + ds.getName());
   392→        }
   393→        for (DataSourceAdapter adapter : adapters) {
   394→            if (adapter.supports(ds.getType())) {
   395→                return adapter;
   396→            }
   397→        }
   398→        throw new BusinessException(400, "数据源类型未启用: " + ds.getType());
   399→    }

   401→    // ==================== 参数自动生成 ====================
   402→
   403→    /**
   404→     * 为 FORM/SYSTEM 数据源自动生成 params JSON（只读配置，UI 不可编辑）。
   405→     * - FORM：list/get/create/update/delete → /api/v1/biz-data/{formKey}[/{id}]
   406→     * - SYSTEM：list → /api/v1/internal/system/{internalKey}，internalKey 由 sourceKey 映射
   407→     */
   408→    private String generateParams(String type, String formKey, String sourceKey) {
   409→        ObjectNode params = objectMapper.getNodeFactory().objectNode();
   410→        if (TYPE_FORM.equals(type)) {
   411→            String base = "/api/v1/biz-data/" + formKey;
   412→            JsonNode list = params.putObject("list")
   413→                    .put("action", base).put("method", "GET")
   414→                    .put("parse", "records").put("totalParse", "total");
   415→            params.putObject("create").put("action", base).put("method", "POST");
   416→            params.putObject("get").put("action", base + "/{id}").put("method", "GET");
   417→            params.putObject("update").put("action", base + "/{id}").put("method", "PUT");
   418→            params.putObject("delete").put("action", base + "/{id}").put("method", "DELETE");
   419→        } else if (TYPE_SYSTEM.equals(type)) {
   420→            params.putObject("list")
   421→                    .put("action", "/api/v1/internal/system/" + mapSystemInternalKey(sourceKey))
   422→                    .put("method", "GET");
   423→        }
   424→        return params.toString();
   425→    }
   426→
   427→    /** sourceKey → internal API 路径 key 映射（dept-tree→dept-tree，user-tree→users） */
   428→    private String mapSystemInternalKey(String sourceKey) {
   429→        return switch (sourceKey) {
   430→            case "dept-tree" -> "dept-tree";
   431→            case "user-tree" -> "users";
   432→            default -> throw new BusinessException(400, "未注册的系统数据源: " + sourceKey);
   433→        };
   434→    }

   438→    /** 判断 params 是否含 queryMode 配置段（非合法 JSON 视为无，由后续校验处理） */
   439→    private boolean hasQueryModeSegment(String params) {
   440→        if (params == null || params.isBlank()) {
   441→            return false;
   442→        }
   443→        try {
   444→            JsonNode root = objectMapper.readTree(params);
   445→            return root != null && root.isObject() && root.has("queryMode");
   446→        } catch (JsonProcessingException e) {
   447→            return false;
   448→        }
   449→    }

   451→    /**
   452→     * FORM 数据源查询配置段保存校验（Task 5）。
   453→     * <ul>
   454→     *   <li>无 queryMode 段 → 向后兼容，跳过校验</li>
   455→     *   <li>config：joins 非空；每项 alias 合法、targetFormKey 表单存在、必填字段齐全、virtualKey 唯一</li>
   456→     *   <li>sql：query/columns 合法性复用 SqlTemplateEngine.validate（SELECT / :tenantId / 列匹配 / 参数白名单）</li>
   457→     *   <li>未知 queryMode → 400</li>
   458→     * </ul>
   459→     */
   460→    private void validateFormQueryConfig(String params) {
   461→        if (params == null || params.isBlank()) {
   462→            return;
   463→        }
   464→        JsonNode root;
   465→        try {
   466→            root = objectMapper.readTree(params);
   467→        } catch (JsonProcessingException e) {
   468→            throw new BusinessException(400, "数据源参数 params 必须是合法 JSON: " + e.getOriginalMessage());
   469→        }
   470→        if (root == null || !root.isObject()) {
   471→            throw new BusinessException(400, "数据源参数 params 必须是 JSON 对象");
   472→        }
   473→        JsonNode modeNode = root.get("queryMode");
   474→        if (modeNode == null || modeNode.isNull() || modeNode.asText().isBlank()) {
   475→            return; // 无 queryMode 段 → 单表查询，向后兼容
   476→        }
   477→        String mode = modeNode.asText();
   478→        if ("config".equals(mode)) {
   479→            validateConfigJoins(root.get("joins"));
   480→        } else if ("sql".equals(mode)) {
   481→            validateSqlConfig(root);
   482→        } else {
   483→            throw new BusinessException(400, "未知查询模式 queryMode: " + mode + "（支持 config / sql）");
   484→        }
   485→    }
   486→
   487→    /** config 模式：joins[] 结构校验（别名/目标表单/字段/虚拟列唯一性） */
   488→    private void validateConfigJoins(JsonNode joins) {
   489→        if (joins == null || !joins.isArray() || joins.isEmpty()) {
   490→            throw new BusinessException(400, "queryMode=config 时必须配置至少一个关联 joins");
   491→        }
   492→        String tenantId = tenantProvider.getTenantId();
   493→        Set<String> virtualKeys = new HashSet<>();
   494→        int idx = 0;
   495→        for (JsonNode j : joins) {
   496→            idx++;
   497→            if (j == null || !j.isObject()) {
   498→                throw new BusinessException(400, "joins 第 " + idx + " 项必须是对象");
   499→            }
   500→            String targetFormKey = text(j, "targetFormKey");
   501→            if (targetFormKey == null || targetFormKey.isBlank()) {
   502→                throw new BusinessException(400, "joins 第 " + idx + " 项必须指定目标表单 targetFormKey");
   503→            }
   504→            if (!formDefRepository.existsByTenantIdAndKey(tenantId, targetFormKey)) {
   505→                throw new BusinessException(400, "目标表单不存在: " + targetFormKey);
   506→            }
   507→            requireJoinField(j, "localField", "主表关联字段", idx);
   508→            requireJoinField(j, "foreignField", "目标表关联字段", idx);
   509→            requireJoinField(j, "joinField", "显示字段", idx);
   510→            requireJoinField(j, "label", "显示名称", idx);
   511→            String virtualKey = text(j, "virtualKey");
   512→            if (virtualKey == null || virtualKey.isBlank()) {
   513→                throw new BusinessException(400, "joins 第 " + idx + " 项必须指定虚拟列标识 virtualKey");
   514→            }
   515→            if (!virtualKeys.add(virtualKey)) {
   516→                throw new BusinessException(400, "虚拟列 virtualKey 重复: " + virtualKey);
   517→            }
   518→        }
   519→    }
   520→
   521→    private void requireJoinField(JsonNode j, String field, String label, int idx) {
   522→        String v = text(j, field);
   523→        if (v == null || v.isBlank()) {
   524→            throw new BusinessException(400, "joins 第 " + idx + " 项必须指定" + label + " " + field);
   525→        }
   526→    }

   528→    /** sql 模式：query/columns/参数白名单复用 SqlTemplateEngine.validate（IllegalArgumentException → 400） */
   529→    private void validateSqlConfig(JsonNode root) {
   530→        String query = text(root, "query");
   531→        List<JoinSqlGenerator.QueryColumn> columns = parseSqlColumns(root.get("columns"));
   532→        List<String> declaredParams = parseStringList(root.get("params"));
   533→        try {
   534→            SqlTemplateEngine.validate(query, columns, declaredParams);
   535→        } catch (IllegalArgumentException e) {
   536→            throw new BusinessException(400, e.getMessage());
   537→        }
   538→    }
   539→
   540→    private List<JoinSqlGenerator.QueryColumn> parseSqlColumns(JsonNode node) {
   541→        List<JoinSqlGenerator.QueryColumn> out = new ArrayList<>();
   542→        if (node == null || !node.isArray()) {
   543→            return out;
   544→        }
   545→        for (JsonNode n : node) {
   546→            if (n == null || !n.isObject()) {
   547→                continue;
   548→            }
   549→            out.add(new JoinSqlGenerator.QueryColumn(
   550→                    text(n, "key"), text(n, "key"), text(n, "columnType"),
   551→                    boolVal(n, "sortable"), boolVal(n, "filterable")));
   552→        }
   553→        return out;
   554→    }
   555→
   556→    private List<String> parseStringList(JsonNode node) {
   557→        List<String> out = new ArrayList<>();
   558→        if (node == null || !node.isArray()) {
   559→            return out;
   560→        }
   561→        for (JsonNode n : node) {
   562→            if (n != null && n.isTextual() && !n.asText().isBlank()) {
   563→                out.add(n.asText());
   564→            }
   565→        }
   566→        return out;
   567→    }

   569→    /** 合并：生成端点 params 之上叠加 queryMode 配置段（config 的 joins / sql 的 query+columns+params 白名单） */
   570→    private String mergeQueryConfig(String generated, String params) {
   571→        try {
   572→            ObjectNode out = (ObjectNode) objectMapper.readTree(generated);
   573→            JsonNode input = objectMapper.readTree(params);
   574→            for (String field : List.of("queryMode", "joins", "query", "columns", "params")) {
   575→                if (input.has(field)) {
   576→                    out.set(field, input.get(field));
   577→                }
   578→            }
   579→            return out.toString();
   580→        } catch (JsonProcessingException e) {
   581→            throw new BusinessException(400, "数据源参数 params 必须是合法 JSON: " + e.getOriginalMessage());
   582→        }
   583→    }
   584→
   585→    private static String text(JsonNode node, String field) {
   586→        JsonNode v = node.get(field);
   587→        return v == null || v.isNull() ? null : v.asText();
   588→    }
   589→
   590→    private static boolean boolVal(JsonNode node, String field) {
   591→        JsonNode v = node.get(field);
   592→        return v != null && v.isBoolean() && v.asBoolean();
   593→    }

   597→    /** 按类型校验必填项：FORM/WORKFLOW→formKey（sourceKey 由 formKey 派生）；SYSTEM/API/SQL→sourceKey；API→params 合法 JSON */
   598→    private void validateRequiredFields(String type, String formKey, String sourceKey, String params) {
   599→        if (TYPE_FORM.equals(type) || TYPE_WORKFLOW.equals(type)) {
   600→            if (formKey == null || formKey.isBlank()) {
   601→                throw new BusinessException(400, type + " 类型数据源必须绑定表单 formKey");
   602→            }
   603→        } else if (TYPE_SYSTEM.equals(type) || TYPE_API.equals(type) || TYPE_SQL.equals(type)) {
   604→            if (sourceKey == null || sourceKey.isBlank()) {
   605→                throw new BusinessException(400, type + " 类型数据源必须填写 sourceKey");
   606→            }
   607→            if (TYPE_SYSTEM.equals(type) && !SYSTEM_SOURCE_KEYS.contains(sourceKey)) {
   608→                throw new BusinessException(400, "未注册的系统数据源: " + sourceKey);
   609→            }
   610→            if (TYPE_API.equals(type)) {
   611→                // LookupFetchConfig 契约：params 须为 JSON 对象且 action 必填
   612→                if (params == null || params.isBlank()) {
   613→                    throw new BusinessException(400, "API 数据源参数 params 必须包含 action（API 路径）");
   614→                }
   615→                try {
   616→                    JsonNode node = objectMapper.readTree(params);
   617→                    if (!node.isObject()) {
   618→                        throw new BusinessException(400, "API 数据源参数 params 必须是 JSON 对象");
   619→                    }
   620→                    JsonNode action = node.get("action");
   621→                    if (action == null || action.isNull() || action.asText().isBlank()) {
   622→                        throw new BusinessException(400, "API 数据源参数 params 必须包含 action（API 路径）");
   623→                    }
   624→                } catch (BusinessException e) {
   625→                    throw e;
   626→                } catch (JsonProcessingException e) {
   627→                    throw new BusinessException(400, "API 数据源参数 params 必须是合法 JSON: " + e.getOriginalMessage());
   628→                }
   629→            }
   630→            // SQL：仅要求 sourceKey（query 配置在 params 中，DRAFT 阶段可为空，启用后由适配器校验）
   631→        }
   632→    }

   634→    /** FORM 数据源启用/重绑前置校验：须存在已发布版本 */
   635→    private void requirePublishedForm(String tenantId, String formKey) {

   637→                .isEmpty()) {

   642→    /** WORKFLOW 数据源启用前置校验：表单存在、已发布且非 BUSINESS（业务表单无流程实例，无法跨实例聚合）。 */
   643→    private void requireWorkflowForm(String tenantId, String formKey) {
   644→        if (!formDefRepository.existsByTenantIdAndKey(tenantId, formKey)) {
   645→            throw new BusinessException(400, "表单不存在: " + formKey);
   646→        }
   647→        FormDefinition form = formDefRepository.findFirstByTenantIdAndKeyAndStatusOrderByVersionDesc(
   648→                        tenantId, formKey, "PUBLISHED")
   649→                .orElseThrow(() -> new BusinessException(400, "工作流表单必须先发布: " + formKey));
   650→        if ("BUSINESS".equals(form.getType())) {
   651→            throw new BusinessException(400, "业务表单不可配置为工作流表单数据源: " + formKey);
   652→        }
   653→    }
