    87→    /**
    88→     * 创建数据源（默认 DRAFT）。
    89→     * 校验：type 必填且合法；同租户 name 唯一；按类型必填项（FORM→formKey + 表单存在；
    90→     * SYSTEM/API→sourceKey；API→params 须为合法 JSON）。

    92→     * 注意：此方法仅供系统内部调用，用户不能直接创建数据源。
    93→     */
    94→    @Transactional
    95→    public DataSourceDefinition create(String name, String type, String formKey, String sourceKey, String params) {
    96→        String tenantId = tenantProvider.getTenantId();
    97→
    98→        if (type == null || type.isBlank()) {
    99→            throw new BusinessException(400, "数据源类型 type 必填");
   100→        }
   101→        if (!SUPPORTED_TYPES.contains(type)) {
   102→            throw new BusinessException(400, "不支持的数据源类型: " + type);
   103→        }
   104→        if (name == null || name.isBlank()) {
   105→            throw new BusinessException(400, "数据源名称不能为空");
   106→        }
   107→        if (dsRepository.existsByTenantIdAndName(tenantId, name)) {
   108→            throw new BusinessException(400, "数据源名称已存在: " + name);
   109→        }
   110→        // FORM/WORKFLOW：sourceKey 恒等于 formKey（formKey 权威）；其余类型以入参 sourceKey 为准
   111→        boolean formBound = TYPE_FORM.equals(type) || TYPE_WORKFLOW.equals(type);
   112→        String effSourceKey = formBound ? formKey : sourceKey;
   113→        validateRequiredFields(type, formKey, effSourceKey, params);
   114→        if (formBound && !formDefRepository.existsByTenantIdAndKey(tenantId, formKey)) {
   115→            throw new BusinessException(400, "绑定的表单不存在: " + formKey);
   116→        }
   117→        if (effSourceKey == null || effSourceKey.isBlank()) {
   118→            throw new BusinessException(400, "数据源必须填写 sourceKey");
   119→        }
   120→        if (dsRepository.existsByTenantIdAndSourceKey(tenantId, effSourceKey)) {
   121→            throw new BusinessException(400, "数据源标识 sourceKey 已存在: " + effSourceKey);
   122→        }
   123→
   124→        DataSourceDefinition ds = new DataSourceDefinition();
   125→        ds.setId(UUID.randomUUID().toString().replace("-", ""));
   126→        ds.setTenantId(tenantId);
   127→        ds.setName(name);
   128→        ds.setType(type);
   129→        ds.setFormKey(formKey);
   130→        ds.setSourceKey(effSourceKey);
   131→        if (TYPE_FORM.equals(type) || TYPE_SYSTEM.equals(type)) {
   132→            if (TYPE_FORM.equals(type) && hasQueryModeSegment(params)) {
   133→                // FORM 带 queryMode 配置段：校验后与自动生成端点合并保存
   134→                validateFormQueryConfig(params);
   135→                ds.setParams(mergeQueryConfig(generateParams(type, formKey, sourceKey), params));
   136→            } else {
   137→                ds.setParams(generateParams(type, formKey, sourceKey));
   138→            }
   139→        } else {
   140→            ds.setParams(params);
   141→        }
   142→        // API/SQL 为手动配置的数据源：创建即发布（ENABLED）；其余类型仍 DRAFT（防御：手动创建仅 API/SQL）
   143→        boolean manualPublish = TYPE_API.equals(type) || TYPE_SQL.equals(type);
   144→        ds.setStatus(manualPublish ? STATUS_ENABLED : STATUS_DRAFT);
   145→        return dsRepository.save(ds);
   146→    }

   148→    /**
   149→     * 原地更新数据源（name/type/formKey/sourceKey/params；null 表示不更新）。
   150→     * 若当前 ENABLED 且 type/formKey 变更，重新校验（FORM 须仍指向已发布表单）。
   151→     * 
   152→     * 注意：此方法仅供系统内部调用，用户不能直接编辑数据源。
   153→     */
   154→    @Transactional
   155→    public DataSourceDefinition update(String id, String name, String type, String formKey,
   156→                                       String sourceKey, String params) {
   157→        String tenantId = tenantProvider.getTenantId();
   158→        DataSourceDefinition ds = getById(id);
   159→        // 内建保护：tenant_id='system' 的预置行不允许修改（对齐 NodeJS requireNotBuiltIn）
   160→        requireNotBuiltIn(ds, "修改");
   161→
   162→        String newType = type == null || type.isBlank() ? ds.getType() : type;
   163→        if (type != null && !type.isBlank() && !SUPPORTED_TYPES.contains(newType)) {
   164→            throw new BusinessException(400, "不支持的数据源类型: " + newType);
   165→        }
   166→        if (name != null && !name.isBlank() && !name.equals(ds.getName())
   167→                && dsRepository.existsByTenantIdAndName(tenantId, name)) {
   168→            throw new BusinessException(400, "数据源名称已存在: " + name);
   169→        }
   170→
   171→        String newFormKey = formKey == null ? ds.getFormKey() : formKey;
   172→        String newSourceKey = sourceKey == null ? ds.getSourceKey() : sourceKey;
   173→        String newParams = params == null ? ds.getParams() : params;
   174→        // FORM/WORKFLOW：sourceKey 恒等于 formKey（formKey 权威），忽略入参 sourceKey 差异
   175→        boolean formBound = TYPE_FORM.equals(newType) || TYPE_WORKFLOW.equals(newType);
   176→        String effNewSourceKey = formBound ? newFormKey : newSourceKey;
   177→        validateRequiredFields(newType, newFormKey, effNewSourceKey, newParams);
   178→        if (formBound && !formDefRepository.existsByTenantIdAndKey(tenantId, newFormKey)) {
   179→            throw new BusinessException(400, "绑定的表单不存在: " + newFormKey);
   180→        }
   181→        if (TYPE_FORM.equals(newType)) {
   182→            // FORM 查询配置段（queryMode/joins/query/columns/params）保存校验
   183→            validateFormQueryConfig(newParams);
   184→        }
   185→        // sourceKey 变更（不等于当前值）时校验租户内唯一；保持不变则跳过（自身不算冲突）
   186→        if (!java.util.Objects.equals(effNewSourceKey, ds.getSourceKey())
   187→                && dsRepository.existsByTenantIdAndSourceKey(tenantId, effNewSourceKey)) {
   188→            throw new BusinessException(400, "数据源标识 sourceKey 已存在: " + effNewSourceKey);
   189→        }
   190→
   191→        // 已启用数据源若变更类型/绑定对象，须重新校验发布状态
   192→        boolean bindChanged = !TYPE_FORM.equals(ds.getType()) || (formKey != null && !formKey.equals(ds.getFormKey()));
   193→        if (STATUS_ENABLED.equals(ds.getStatus()) && bindChanged) {
   194→            if (TYPE_FORM.equals(newType)) {
   195→                requirePublishedForm(tenantId, newFormKey);
   196→            } else if (TYPE_WORKFLOW.equals(newType)) {
   197→                requireWorkflowForm(tenantId, newFormKey);
   198→            }
   199→        }
   200→
   201→        ds.setName(name == null ? ds.getName() : name);
   202→        ds.setType(newType);
   203→        ds.setFormKey(newFormKey);
   204→        ds.setSourceKey(effNewSourceKey);
   205→        ds.setParams(newParams);
   206→        return dsRepository.save(ds);
   207→    }

   223→            requireWorkflowForm(tenantId, ds.getFormKey());
   224→        }
   225→        ds.setStatus(STATUS_ENABLED);
   226→        if ((TYPE_FORM.equals(ds.getType()) || TYPE_SYSTEM.equals(ds.getType())) && ds.getParams() == null) {
   227→            ds.setParams(generateParams(ds.getType(), ds.getFormKey(), ds.getSourceKey()));
   228→        }
   229→        return dsRepository.save(ds);
   230→    }

   246→    /**
   247→     * 删除数据源：任意状态可删除（ENABLED/DISABLED 均可），但被页面引用时拒绝（400）。
   248→     * 引用统计覆盖两种绑定方式：
   249→     * 1) PageDefinition.dataSourceId 列（VIEW 新协议，迁移器回填）；
   250→     * 2) PAGE 类型页面 schema.dataSources[].refId（设计器 dataSources 声明）。
   251→     * 两者取并集，任一命中即拒绝删除。
   252→     * 
   253→     * 注意：此方法仅供系统内部调用，用户不能直接删除数据源。
   254→     */
   255→    @Transactional
   256→    public void delete(String id) {
   257→        DataSourceDefinition ds = getById(id);
   258→        // 内建保护：先于引用统计（对齐 NodeJS remove：requireNotBuiltIn → countRefs → delete）
   259→        requireNotBuiltIn(ds, "删除");
   260→        String tenantId = ds.getTenantId();
   261→        long refCount = countRefs(tenantId, id);
   262→        if (refCount > 0) {
   263→            throw new BusinessException(400, "数据源已被 " + refCount + " 个页面引用，无法删除");
   264→        }
   265→        dsRepository.delete(ds);
   266→    }

   268→    /**
   269→     * 统计当前租户内引用指定数据源的页面数（dataSourceId 列 + PAGE schema dataSources[].refId 并集）。
   270→     */
   271→    private long countRefs(String tenantId, String dataSourceId) {
   272→        long columnRefs = pageRepository.countByTenantIdAndDataSourceId(tenantId, dataSourceId);
   273→        if (columnRefs > 0) {
   274→            return columnRefs;
   275→        }
   276→        // PAGE 类型页面引用声明在 schema.dataSources[].refId（dataSourceId 列为空），需扫描；
   277→        // 页面软删除（ARCHIVED）后不再使用，其 schema 引用不阻塞删除
   278→        long schemaRefs = 0;
   279→        Page<PageDefinition> pages = pageRepository
   280→                .findByTenantIdAndTypeOrderByUpdatedAtDesc(tenantId, "PAGE", PageRequest.of(0, Integer.MAX_VALUE));
   281→        for (PageDefinition page : pages.getContent()) {
   282→            if ("ARCHIVED".equals(page.getStatus())) {
   283→                continue;
   284→            }
   285→            if (schemaRefsDataSource(page.getSchema(), dataSourceId)) {
   286→                schemaRefs++;
   287→            }
   288→        }
   289→        return schemaRefs;
   290→    }

   292→    /**
   293→     * 判断页面 schema 的 dataSources[].refId 是否指向指定数据源。
   294→     */
   295→    private boolean schemaRefsDataSource(String schema, String dataSourceId) {
   296→        if (schema == null || schema.isBlank()) {
   297→            return false;
   298→        }
   299→        try {
   300→            JsonNode root = objectMapper.readTree(schema);
   301→            JsonNode dataSources = root.path("dataSources");
   302→            if (!dataSources.isArray()) {
   303→                return false;
   304→            }
   305→            for (JsonNode ds : dataSources) {
   306→                String refId = ds.path("refId").asText(null);
   307→                if (dataSourceId.equals(refId)) {
   308→                    return true;
   309→                }
   310→            }
   311→        } catch (JsonProcessingException e) {
   312→            return false;
   313→        }
   314→        return false;
   315→    }

   401→    /** 按数据源类型找适配器（数据源须 ENABLED） */
   402→    private DataSourceAdapter adapterOf(String id) {
   403→        DataSourceDefinition ds = getById(id);
   404→        if (!STATUS_ENABLED.equals(ds.getStatus())) {
   405→            throw new BusinessException(400, "数据源未启用，无法访问: " + ds.getName());
   406→        }
   407→        for (DataSourceAdapter adapter : adapters) {
   408→            if (adapter.supports(ds.getType())) {
   409→                return adapter;
   410→            }
   411→        }
   412→        throw new BusinessException(400, "数据源类型未启用: " + ds.getType());
   413→    }

   415→    // ==================== 参数自动生成 ====================
   416→
   417→    /**
   418→     * 为 FORM/SYSTEM 数据源自动生成 params JSON（只读配置，UI 不可编辑）。
   419→     * - FORM：list/get/create/update/delete → /api/v1/biz-data/{formKey}[/{id}]
   420→     * - SYSTEM：list → /api/v1/internal/system/{internalKey}，internalKey 由 sourceKey 映射
   421→     */
   422→    private String generateParams(String type, String formKey, String sourceKey) {
   423→        ObjectNode params = objectMapper.getNodeFactory().objectNode();
   424→        if (TYPE_FORM.equals(type)) {
   425→            String base = "/api/v1/biz-data/" + formKey;
   426→            JsonNode list = params.putObject("list")
   427→                    .put("action", base).put("method", "GET")
   428→                    .put("parse", "records").put("totalParse", "total");
   429→            params.putObject("create").put("action", base).put("method", "POST");
   430→            params.putObject("get").put("action", base + "/{id}").put("method", "GET");
   431→            params.putObject("update").put("action", base + "/{id}").put("method", "PUT");
   432→            params.putObject("delete").put("action", base + "/{id}").put("method", "DELETE");
   433→        } else if (TYPE_SYSTEM.equals(type)) {
   434→            params.putObject("list")
   435→                    .put("action", "/api/v1/internal/system/" + mapSystemInternalKey(sourceKey))
   436→                    .put("method", "GET");
   437→        }
   438→        return params.toString();
   439→    }

   441→    /** sourceKey → internal API 路径 key 映射（唯一事实源见 BuiltInSystemSources.mapSystemInternalPath） */
   442→    private String mapSystemInternalKey(String sourceKey) {
   443→        String path = BuiltInSystemSources.mapSystemInternalPath(sourceKey);
   444→        if (path == null || path.isBlank()) {
   445→            throw new BusinessException(400, "未注册的系统数据源: " + sourceKey);
   446→        }
   447→        return path;
   448→    }

   465→    /**
   466→     * FORM 数据源查询配置段保存校验（Task 5；Task 51 扩展内建目标）。
   467→     * <ul>
   468→     *   <li>无 queryMode 段 → 向后兼容，跳过校验</li>
   469→     *   <li>config：joins 非空；alias 可缺省（运行时自动分配）；目标支持业务表单（存在性）
   470→     *       与内建数据源（JoinTargetCatalog 物理列白名单）；必填字段齐全、virtualKey 唯一</li>
   471→     *   <li>sql：query/columns 合法性复用 SqlTemplateEngine.validate（SELECT / :tenantId / 列匹配 / 参数白名单）</li>
   472→     *   <li>未知 queryMode → 400</li>
   473→     * </ul>
   474→     */
   475→    private void validateFormQueryConfig(String params) {
   476→        if (params == null || params.isBlank()) {
   477→            return;
   478→        }
   479→        JsonNode root;
   480→        try {
   481→            root = objectMapper.readTree(params);
   482→        } catch (JsonProcessingException e) {
   483→            throw new BusinessException(400, "数据源参数 params 必须是合法 JSON: " + e.getOriginalMessage());
   484→        }
   485→        if (root == null || !root.isObject()) {
   486→            throw new BusinessException(400, "数据源参数 params 必须是 JSON 对象");
   487→        }
   488→        JsonNode modeNode = root.get("queryMode");
   489→        if (modeNode == null || modeNode.isNull() || modeNode.asText().isBlank()) {
   490→            return; // 无 queryMode 段 → 单表查询，向后兼容
   491→        }
   492→        String mode = modeNode.asText();
   493→        if ("config".equals(mode)) {
   494→            validateConfigJoins(root.get("joins"));
   495→        } else if ("sql".equals(mode)) {
   496→            validateSqlConfig(root);
   497→        } else {
   498→            throw new BusinessException(400, "未知查询模式 queryMode: " + mode + "（支持 config / sql）");
   499→        }
   500→    }
   501→
   502→    /** config 模式：joins[] 结构校验（别名/目标表/字段/虚拟列唯一性）。
   503→     * <p>
   504→     * alias 可缺省：前端不录入，由运行时 {@code FormQueryConfig.parseJoins} 自动分配（j1/j2/...）；
   505→     * 传入则校验格式与唯一性。目标表支持内建数据源（JoinTargetCatalog 白名单）：
   506→     * SYSTEM 目标不查 form_def，且 foreignField/joinField 必须是目录内物理列。 */
   507→    private void validateConfigJoins(JsonNode joins) {
   508→        if (joins == null || !joins.isArray() || joins.isEmpty()) {
   509→            throw new BusinessException(400, "queryMode=config 时必须配置至少一个关联 joins");
   510→        }
   511→        String tenantId = tenantProvider.getTenantId();
   512→        Set<String> virtualKeys = new HashSet<>();
   513→        Set<String> aliases = new HashSet<>();
   514→        int idx = 0;
   515→        for (JsonNode j : joins) {
   516→            idx++;
   517→            if (j == null || !j.isObject()) {
   518→                throw new BusinessException(400, "joins 第 " + idx + " 项必须是对象");
   519→            }
   520→            String alias = text(j, "alias");
   521→            if (alias != null) {
   522→                if (!JOIN_ALIAS_PATTERN.matcher(alias).matches()) {
   523→                    throw new BusinessException(400, "joins 第 " + idx + " 项 alias 非法: " + alias);
   524→                }
   525→                if (!aliases.add(alias)) {
   526→                    throw new BusinessException(400, "joins 第 " + idx + " 项 alias 重复: " + alias);
   527→                }
   528→            }
   529→            String targetFormKey = text(j, "targetFormKey");
   530→            if (targetFormKey == null || targetFormKey.isBlank()) {
   531→                throw new BusinessException(400, "joins 第 " + idx + " 项必须指定目标表 targetFormKey");
   532→            }
   533→            Set<String> foreignCandidates = null;
   534→            if (JoinTargetCatalog.isJoinTargetSystemKey(targetFormKey)) {
   535→                // 内建目标：物理列白名单来自 JoinTargetCatalog（不走 form_def）
   536→                foreignCandidates = JoinTargetCatalog.systemColumnKeys(targetFormKey);
   537→            } else {
   538→                if (!FORM_KEY_PATTERN.matcher(targetFormKey).matches()) {
   539→                    throw new BusinessException(400, "非法关联目标: " + targetFormKey);
   540→                }
   541→                if (!formDefRepository.existsByTenantIdAndKey(tenantId, targetFormKey)) {
   542→                    throw new BusinessException(400, "目标表单不存在: " + targetFormKey);
   543→                }
   544→            }
   545→            String foreignField = text(j, "foreignField");
   546→            requireJoinField(j, "foreignField", "目标表关联字段", idx);
   547→            requireJoinField(j, "localField", "主表关联字段", idx);
   548→            requireJoinField(j, "joinField", "显示字段", idx);
   549→            if (foreignCandidates != null) {
   550→                if (!foreignCandidates.contains(foreignField)) {
   551→                    throw new BusinessException(400, "joins 第 " + idx
   552→                            + " 项目标表关联字段不在内建数据源物理列中: " + foreignField);
   553→                }
   554→                String joinFieldValue = text(j, "joinField");
   555→                if (joinFieldValue == null || !foreignCandidates.contains(joinFieldValue)) {
   556→                    throw new BusinessException(400, "joins 第 " + idx
   557→                            + " 项显示字段不在内建数据源物理列中: " + joinFieldValue);
   558→                }
   559→            }
   560→            requireJoinField(j, "label", "显示名称", idx);
   561→            String virtualKey = text(j, "virtualKey");
   562→            if (virtualKey == null || virtualKey.isBlank()) {
   563→                throw new BusinessException(400, "joins 第 " + idx + " 项必须指定虚拟列标识 virtualKey");
   564→            }
   565→            if (!virtualKeys.add(virtualKey)) {
   566→                throw new BusinessException(400, "虚拟列 virtualKey 重复: " + virtualKey);
   567→            }
   568→        }
   569→    }
   570→
   571→    private void requireJoinField(JsonNode j, String field, String label, int idx) {
   572→        String v = text(j, field);
   573→        if (v == null || v.isBlank()) {
   574→            throw new BusinessException(400, "joins 第 " + idx + " 项必须指定" + label + " " + field);
   575→        }
   576→    }

   578→    /** sql 模式：query/columns/参数白名单复用 SqlTemplateEngine.validate（IllegalArgumentException → 400） */
   579→    private void validateSqlConfig(JsonNode root) {
   580→        String query = text(root, "query");
   581→        List<JoinSqlGenerator.QueryColumn> columns = parseSqlColumns(root.get("columns"));
   582→        List<String> declaredParams = parseStringList(root.get("params"));
   583→        try {
   584→            SqlTemplateEngine.validate(query, columns, declaredParams);
   585→        } catch (IllegalArgumentException e) {
   586→            throw new BusinessException(400, e.getMessage());
   587→        }
   588→    }
   589→
   590→    private List<JoinSqlGenerator.QueryColumn> parseSqlColumns(JsonNode node) {
   591→        List<JoinSqlGenerator.QueryColumn> out = new ArrayList<>();
   592→        if (node == null || !node.isArray()) {
   593→            return out;
   594→        }
   595→        for (JsonNode n : node) {
   596→            if (n == null || !n.isObject()) {
   597→                continue;
   598→            }
   599→            out.add(new JoinSqlGenerator.QueryColumn(
   600→                    text(n, "key"), text(n, "key"), text(n, "columnType"),
   601→                    boolVal(n, "sortable"), boolVal(n, "filterable")));
   602→        }
   603→        return out;
   604→    }
   605→
   606→    private List<String> parseStringList(JsonNode node) {
   607→        List<String> out = new ArrayList<>();
   608→        if (node == null || !node.isArray()) {
   609→            return out;
   610→        }
   611→        for (JsonNode n : node) {
   612→            if (n != null && n.isTextual() && !n.asText().isBlank()) {
   613→                out.add(n.asText());
   614→            }
   615→        }
   616→        return out;
   617→    }

   619→    /** 合并：生成端点 params 之上叠加 queryMode 配置段（config 的 joins / sql 的 query+columns+params 白名单） */
   620→    private String mergeQueryConfig(String generated, String params) {
   621→        try {
   622→            ObjectNode out = (ObjectNode) objectMapper.readTree(generated);
   623→            JsonNode input = objectMapper.readTree(params);
   624→            for (String field : List.of("queryMode", "joins", "query", "columns", "params")) {
   625→                if (input.has(field)) {
   626→                    out.set(field, input.get(field));
   627→                }
   628→            }
   629→            return out.toString();
   630→        } catch (JsonProcessingException e) {
   631→            throw new BusinessException(400, "数据源参数 params 必须是合法 JSON: " + e.getOriginalMessage());
   632→        }
   633→    }

   636→        JsonNode v = node.get(field);
   637→        return v == null || v.isNull() ? null : v.asText();
   638→    }

   641→        JsonNode v = node.get(field);
   642→        return v != null && v.isBoolean() && v.asBoolean();
   643→    }

   657→    /** 按类型校验必填项：FORM/WORKFLOW→formKey（sourceKey 由 formKey 派生）；SYSTEM/API/SQL→sourceKey；API→params 合法 JSON */
   658→    private void validateRequiredFields(String type, String formKey, String sourceKey, String params) {
   659→        if (TYPE_FORM.equals(type) || TYPE_WORKFLOW.equals(type)) {
   660→            if (formKey == null || formKey.isBlank()) {
   661→                throw new BusinessException(400, type + " 类型数据源必须绑定表单 formKey");
   662→            }
   663→        } else if (TYPE_SYSTEM.equals(type) || TYPE_API.equals(type) || TYPE_SQL.equals(type)) {
   664→            if (sourceKey == null || sourceKey.isBlank()) {
   665→                throw new BusinessException(400, type + " 类型数据源必须填写 sourceKey");
   666→            }
   667→            if (TYPE_SYSTEM.equals(type) && !SYSTEM_SOURCE_KEYS.contains(sourceKey)) {
   668→                throw new BusinessException(400, "未注册的系统数据源: " + sourceKey);
   669→            }
   670→            if (TYPE_API.equals(type)) {
   671→                // LookupFetchConfig 契约：params 须为 JSON 对象且 action 必填
   672→                if (params == null || params.isBlank()) {
   673→                    throw new BusinessException(400, "API 数据源参数 params 必须包含 action（API 路径）");
   674→                }
   675→                try {
   676→                    JsonNode node = objectMapper.readTree(params);
   677→                    if (!node.isObject()) {
   678→                        throw new BusinessException(400, "API 数据源参数 params 必须是 JSON 对象");
   679→                    }
   680→                    JsonNode action = node.get("action");
   681→                    if (action == null || action.isNull() || action.asText().isBlank()) {
   682→                        throw new BusinessException(400, "API 数据源参数 params 必须包含 action（API 路径）");
   683→                    }
   684→                } catch (BusinessException e) {
   685→                    throw e;
   686→                } catch (JsonProcessingException e) {
   687→                    throw new BusinessException(400, "API 数据源参数 params 必须是合法 JSON: " + e.getOriginalMessage());
   688→                }
   689→            }
   690→            // SQL：仅要求 sourceKey（query 配置在 params 中，DRAFT 阶段可为空，启用后由适配器校验）
   691→        }
   692→    }

   703→    private void requireWorkflowForm(String tenantId, String formKey) {

   711→            throw new BusinessException(400, "业务表单不可配置为工作流表单数据源: " + formKey);
   712→        }
   713→    }
