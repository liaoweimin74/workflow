    27→/** 组织树节点（`SystemService.orgTree()` 的返回元素，只用到这四个字段）。 */
    28→interface OrgTreeNode {
    29→  id: number
    30→  parentId: number | null
    31→  label: string | null
    32→  code: string | null
    33→  children?: OrgTreeNode[] | null
    34→}
    35→
    36→/**
    37→ * 部门/用户列常量已收编到 `system-source-catalog.ts`（唯一事实源，与 seeder/
    38→ * router/写服务共享）；字段逐字段保持与 Java `DEPT_COLUMNS`/`USER_COLUMNS` 一致。
    39→ */
    40→
    41→/** 未知 sourceKey 的元数据列回退（保持旧行为：非 user-tree 一律部门列）。 */
    42→const DEPT_FALLBACK_COLUMNS = builtInSourceByKey('dept-tree')?.columns ?? []
    43→
    44→/**
    45→ * 复制列并强制 `sortable=false`（对齐 Java `copyWithSortableFalse`）。
    46→ *
    47→ * ⚠️ **只拷 key/label/columnType**，`length` 等其余字段保持 `newColumnConfig()` 的默认值
    48→ * （即 null）—— golden 实测 SYSTEM 元数据的列上 `length: null`，尽管常量里写了 64/128。
    49→ * 「顺手把 length 也拷过来」会立刻与 Java 分叉。
    50→ */
    51→function copyWithSortableFalse(src: ColumnConfig[]): ColumnConfig[] {
    52→  return src.map((column) => ({
    53→    ...newColumnConfig(),
    54→    key: column.key,
    55→    label: column.label,
    56→    columnType: column.columnType,
    57→    sortable: false,
    58→  }))
    59→}
    60→
    61→/**
    62→ * 追加 sql 模式管理员声明列到 metadata；与主表列 key 重复的声明列跳过
    63→ * （对齐 Java `appendDeclaredColumns`）。
    64→ *
    65→ * `label` 空白时回落 `key`。同样只拷 5 个字段。
    66→ */
    67→function appendDeclaredColumns(columns: ColumnConfig[], declared: ColumnConfig[]): void {
    68→  const existing = new Set(columns.map((column) => column.key))
    69→  for (const item of declared) {
    70→    const key = item.key
    71→    if (key === null || key.trim() === '' || existing.has(key)) continue
    72→    columns.push({
    73→      ...newColumnConfig(),
    74→      key,
    75→      label: item.label === null || item.label.trim() === '' ? key : item.label,
    76→      columnType: item.columnType,
    77→      sortable: item.sortable,
    78→      filterable: item.filterable,
    79→    })
    80→    existing.add(key)
    81→  }
    82→}

    84→/** 本适配器覆盖的数据源类型（与 Java `UnifiedDataSourceAdapter.supports` 完全一致）。 */
    85→const SUPPORTED_TYPES = ['FORM', 'SYSTEM', 'API', 'WORKFLOW', 'SQL']
    86→
    87→/**
    88→ * 统一数据源适配器（对齐 Java `com.workflow.engine.datasource.UnifiedDataSourceAdapter`）。
    89→ *
    90→ * **5 种类型的读路径全部已迁移且在契约网内**（FORM / SYSTEM / API / WORKFLOW / SQL）：
    91→ * 「数据源元数据与取数」「SYSTEM 数据源读路径」「API 数据源读路径」「WORKFLOW 数据源读路径」
    92→ * 「SQL 数据源（管理员模板）」「FORM 数据源 config 与 sql 查询模式」六个场景覆盖。
    93→ *
    94→ * `supports()` 刻意对全部 5 种类型都返回 true —— **选路逻辑必须与 Java 一致**，
    95→ * 否则一个 SYSTEM 数据源会得到「数据源类型未启用」这种**错误的**错误信息，
    96→ * 掩盖真正的原因。真正走到「与 Java 不一致的类型」时由 `notMigrated()` 显式抛错
    97→ * （当前只可能在数据库里出现未知 `type` 值时才触发）。
    98→ */
    99→@Injectable()
   100→export class UnifiedDataSourceAdapter implements DataSourceAdapter {
   101→  constructor(
   102→    private readonly bizDataService: BizDataService,
   103→    private readonly router: InternalDataSourceRouter,
   104→    private readonly systemService: SystemService,
   105→    private readonly httpExecutor: HttpLogicExecutor,
   106→    private readonly workflowQuery: WorkflowFormDataQueryService,
   107→    private readonly systemSourceQuery: SystemSourceQueryService,
   108→  ) {}
   109→
   110→  supports(type: string): boolean {
   111→    return SUPPORTED_TYPES.includes(type)
   112→  }

   114→  async metadata(dataSource: DataSourceRef): Promise<DataSourceMetadata> {
   115→    if (dataSource.type === 'SYSTEM') {
   116→      // ⚠️ SYSTEM 的列是**常量**（目录统一维护，源对齐 Java 的 DEPT/USER_COLUMNS），不查库；
   117→      //    且 `writable=false`（系统数据源只读）。
   118→      //    ⚠️ `copyWithSortableFalse` 只拷 key/label/columnType —— **不拷 length**，
   119→      //    golden 实测列上 `length: null`（尽管常量里写了 64/128）。
   120→      const source = builtInSourceByKey(dataSource.sourceKey ?? '')
   121→      const columns = copyWithSortableFalse(source === null ? DEPT_FALLBACK_COLUMNS : source.columns)
   122→      return { columns, writable: false, formKey: null }
   123→    }
   124→    if (dataSource.type === 'WORKFLOW') {
   125→      // 列来自最新 PUBLISHED 表单的 **schema**（WORKFLOW 表单的 columnConfig 为空），
   126→      // 只读，formKey 原样带出。表单未发布时降级为空列（对齐 FORM 分支与 Java 侧）。
   127→      let columns: ColumnConfig[]
   128→      try {
   129→        columns = await this.workflowQuery.columnsFor(requireFormKey(dataSource, 'metadata'))
   130→      } catch (e) {
   131→        if (e instanceof BusinessException && e.code === 404) {
   132→          return { columns: [], writable: false, formKey: dataSource.formKey }
   133→        }
   134→        throw e
   135→      }
   136→      resolveSortable(columns)
   137→      return { columns, writable: false, formKey: dataSource.formKey }
   138→    }
   139→    if (dataSource.type === 'API') {
   140→      return this.apiMetadata(dataSource, this.parseApiParams(dataSource))
   141→    }
   142→    if (dataSource.type === 'SQL') {
   143→      // 字段元数据 = params.columns（**单一来源**；探测/主表单覆盖/行内编辑由前端写入，
   144→      // 不与绑定表单的列做默认合并）。config 模式下声明列来自 joins，故为**空列表**。
   145→      const config = parseFormQueryConfig(dataSource.params, (message) => new BusinessException(400, message))
   146→      const columns = isConfigMode(config) ? [] : config.columns
   147→      const writable = dataSource.formKey !== null && dataSource.formKey.trim() !== ''
   148→      return {
   149→        columns,
   150→        writable,
   151→        formKey: writable ? dataSource.formKey : null,
   152→      }
   153→    }
   154→    if (dataSource.type !== 'FORM') {
   155→      throw notMigrated(dataSource, 'metadata')
   156→    }
   157→    const formKey = requireFormKey(dataSource, 'metadata')
   158→    let columns: ColumnConfig[]
   159→    try {
   160→      columns = cloneColumns(await this.bizDataService.loadColumns(formKey))
   161→    } catch (e) {
   162→      // 绑定表单尚未发布（或不存在）：字段元数据降级为空列而非 404，
   163→      // 不阻断数据源编辑（对齐 Java 侧同款降级；表单发布后此处自动展示列）
   164→      if (e instanceof BusinessException && e.code === 404) {
   165→        return { columns: [], writable: false, formKey }
   166→      }
   167→      throw e
   168→    }
   169→
   170→    // config / sql 两种 queryMode 会往列清单里追加 JOIN 虚拟列或声明列
   171→    // （对齐 Java `metadata` 的 FORM 分支：config → appendJoinColumns，sql → appendDeclaredColumns）
   172→    const config = parseFormQueryConfig(dataSource.params, (message) => new BusinessException(400, message))
   173→    if (isConfigMode(config)) {
   174→      await this.appendJoinColumns(columns, config.joins)
   175→    } else if (isSqlMode(config)) {
   176→      appendDeclaredColumns(columns, config.columns)
   177→    }
   178→
   179→    resolveSortable(columns)
   180→    return { columns, writable: true, formKey }
   181→  }
   182→
   183→  async query(dataSource: DataSourceRef, req: BizDataQueryRequest): Promise<BizDataPageVO> {
   184→    if (dataSource.type === 'SYSTEM') {
   185→      // 先过 router（Java 的 SYSTEM 分支里有 `router.resolve(ds, "list")`），再取数
   186→      this.router.resolve(dataSource, 'list')
   187→      return this.systemQuery(dataSource, req)
   188→    }
   189→    if (dataSource.type === 'WORKFLOW') {
   190→      return this.workflowQuery.query(String(dataSource.formKey), req)
   191→    }
   192→    if (dataSource.type === 'API') {
   193→      return this.apiQuery(dataSource, req)
   194→    }
   195→    if (dataSource.type === 'SQL') {
   196→      this.router.resolve(dataSource, 'list')
   197→      const config = parseFormQueryConfig(dataSource.params, (message) => new BusinessException(400, message))
   198→      if (isVisualMode(config) || isSqlMode(config)) {
   199→        // 绕过表单 covering handler：SQL 数据源执行管理员显式 SQL，不被绑定表单业务定制劫持
   200→        return this.bizDataService.querySqlRaw(dataSource.formKey, req, config)
   201→      }
   202→      throw new BusinessException(400, 'SQL 数据源缺少查询配置')
   203→    }
   204→    if (dataSource.type !== 'FORM') {
   205→      throw notMigrated(dataSource, 'query')
   206→    }
   207→    const formKey = requireFormKey(dataSource, 'query')
   208→    const config = parseFormQueryConfig(dataSource.params, (message) => new BusinessException(400, message))
   209→    if (isConfigMode(config)) {
   210→      return this.bizDataService.queryJoin(formKey, req, config.joins)
   211→    }
   212→    if (isSqlMode(config)) {
   213→      return this.bizDataService.querySql(formKey, req, config)
   214→    }
   215→    // visual 模式在 FORM 类型上**不**生效（Java 只判 config/sql），回退单表查询
   216→    return this.bizDataService.query(formKey, req)
   217→  }

   219→  async get(dataSource: DataSourceRef, id: string): Promise<BizDataVO> {
   220→    if (dataSource.type === 'SYSTEM') {
   221→      // Java `systemGet`：取**全部行**（忽略分页）再按 id 找，找不到 → 业务 404
   222→      const all = await this.systemQuery(dataSource, {
   223→        filter: null,
   224→        keyword: null,
   225→        keywordColumn: null,
   226→        sort: null,
   227→        order: null,
   228→        params: null,
   229→        page: 1,
   230→        size: 20,
   231→      })
   232→      const row = all.records.find((r) => r.id === id)
   233→      if (row === undefined) {
   234→        throw new BusinessException(404, `系统数据不存在: ${id}`)
   235→      }
   236→      return row
   237→    }
   238→    if (dataSource.type === 'WORKFLOW') {
   239→      return this.workflowQuery.getById(String(dataSource.formKey), id)
   240→    }
   241→    if (dataSource.type === 'API') {
   242→      return this.apiGet(dataSource, id)
   243→    }
   244→    if (dataSource.type === 'SQL') {
   245→      // SQL 类型行级操作**始终走绑定表单**（`get` 未绑定 → 400）
   246→      return this.bizDataService.getById(requireFormKey(dataSource, 'get'), id)
   247→    }
   248→    if (dataSource.type !== 'FORM') {
   249→      throw notMigrated(dataSource, 'get')
   250→    }
   251→    const formKey = requireFormKey(dataSource, 'get')
   252→    return this.bizDataService.getById(formKey, id)
   253→  }
   254→
   255→  // ==================== SYSTEM 取数（对齐 Java 的 systemQuery 两个分支） ====================
   256→
   257→  /**
   258→   * 系统数据源取数。
   259→   *
   260→   * ⚠️ 两个分支的**分页语义完全不同**（golden 实测）：
   261→   *   - `user-tree`：走用户分页（keyword + page/size 都生效），`{total, page, size}` 来自请求；
   262→   *   - 其余（含 `dept-tree`）：返回**全部扁平化节点**，外壳是 `page=0, size=rows.length`
   263→   *     —— **忽略请求里的分页**。照抄，不要"顺手"改成尊重分页。
   264→   *
   265→   * ⚠️ 空值一律给**空串**而不是 null（`parentId`/`code`/`nickname`/`orgId`/`orgName`），
   266→   *    这与「null 列不进 data」的 FORM 路径完全不同。
   267→   */
   268→  private async systemQuery(
   269→    dataSource: DataSourceRef,
   270→    req: BizDataQueryRequest,
   271→  ): Promise<BizDataPageVO> {
   272→    const sourceKey = dataSource.sourceKey ?? ''
   273→    if (sourceKey === 'user-tree') {
   274→      const page = Math.max(req.page, 1)
   275→      const result = await this.systemService.listUsersByUsername(req.keyword, page, req.size)
   276→      const records: BizDataVO[] = result.rows.map((row) => ({
   277→        id: String(row.id),
   278→        data: {
   279→          id: String(row.id),
   280→          username: row.username,
   281→          nickname: row.nickname ?? '',
   282→          orgId: row.orgId === null || row.orgId === undefined ? '' : String(row.orgId),
   283→          orgName: row.orgName ?? '',
   284→          status: row.status,
   285→        },
   286→        version: null,
   287→        createdAt: null,
   288→        updatedAt: null,
   289→      }))
   290→      return { records, total: result.total, page: result.page, size: result.size }
   291→    }
   292→
   293→    // 新 6 个内建系统数据源（菜单/角色/字典/流程定义/流程实例/待办任务）
   294→    // 统一委托 SystemSourceQueryService —— 与 SystemInternalController 的 REST
   295→    // 端点共享同一份实现，避免双实现漂移。
   296→    if (SystemSourceQueryService.handles(sourceKey)) {
   297→      return this.systemSourceQuery.query(sourceKey, req)
   298→    }
   299→
   300→    const nodes = await this.systemService.orgTree()
   301→    const records: BizDataVO[] = []
   302→    const collect = (node: OrgTreeNode): void => {
   303→      records.push({
   304→        id: String(node.id),
   305→        data: {
   306→          id: String(node.id),
   307→          parentId: node.parentId === null || node.parentId === undefined ? '' : String(node.parentId),
   308→          label: node.label ?? '',
   309→          code: node.code ?? '',
   310→        },
   311→        version: null,
   312→        createdAt: null,
   313→        updatedAt: null,
   314→      })
   315→      for (const child of node.children ?? []) collect(child)
   316→    }
   317→    for (const node of nodes) collect(node)
   318→    return { records, total: records.length, page: 0, size: records.length }
   319→  }

   321→  // ==================== 写路径（对齐 Java create / update / delete） ====================
   322→
   323→  /**
   324→   * 新增（对齐 Java `create`）。
   325→   *
   326→   * ⚠️ **每个类型走的是不同的分支，错误形态完全不同**，不能合并：
   327→   *    - FORM：先 `router.resolve`（不是 `requireFormKey`）→ `bizDataService.create`
   328→   *    - WORKFLOW：**只读**，恒 400
   329→   *    - API：外部 HTTP（见 `apiCreate`）
   330→   *    - SQL：`requireFormKey`（消息是 SQL 专属的）→ `bizDataService.create`
   331→   *    - SYSTEM 及未知类型：`default` → 「该数据源不支持create」
   332→   *    SYSTEM 在这里**不经过 router**（Java 的 switch 里没有 SYSTEM 分支），
   333→   *    所以 dept-tree/user-tree 都只报「不支持」，而不是先撞 allowlist。
   334→   */
   335→  async create(dataSource: DataSourceRef, data: Record<string, unknown> | null): Promise<string> {
   336→    switch (dataSource.type) {
   337→      case 'FORM': {
   338→        const formKey = this.formKeyAfterResolve(dataSource, 'create')
   339→        const created = await this.bizDataService.create(formKey, data)
   340→        return created.id
   341→      }
   342→      case 'WORKFLOW':
   343→        throw readOnlyWorkflow()
   344→      case 'API':
   345→        return this.apiCreate(dataSource, data)
   346→      case 'SQL': {
   347→        const formKey = requireFormKey(dataSource, 'create')
   348→        const created = await this.bizDataService.create(formKey, data)
   349→        return created.id
   350→      }
   351→      default:
   352→        throw unsupported(dataSource, 'create')
   353→    }
   354→  }
   355→
   356→  /** 更新（对齐 Java `update`）；分支与 `create` 一一对应。 */
   357→  async update(
   358→    dataSource: DataSourceRef,
   359→    id: string,
   360→    data: Record<string, unknown> | null,
   361→    version: number | null,
   362→  ): Promise<void> {
   363→    switch (dataSource.type) {
   364→      case 'FORM': {
   365→        const formKey = this.formKeyAfterResolve(dataSource, 'update')
   366→        await this.bizDataService.update(formKey, id, data, version)
   367→        return
   368→      }
   369→      case 'WORKFLOW':
   370→        throw readOnlyWorkflow()
   371→      case 'API':
   372→        // ⚠️ 必须 `return`/`await`：`apiUpdate` 是 async，写成「调用 + return」会让
   373→        //    拒绝变成一个**游离的 Promise**（既不抛也不等）—— 端点返回 200 而实际失败了。
   374→        return this.apiUpdate(dataSource, id, data)
   375→      case 'SQL': {
   376→        const formKey = requireFormKey(dataSource, 'update')
   377→        await this.bizDataService.update(formKey, id, data, version)
   378→        return
   379→      }
   380→      default:
   381→        throw unsupported(dataSource, 'update')
   382→    }
   383→  }
   384→
   385→  /**
   386→   * 删除（对齐 Java `delete`）。
   387→   *
   388→   * ⚠️ 与 create/update 的唯一结构差异：**SYSTEM 有显式分支**，它会先
   389→   *    `router.resolve(ds, "delete")` 再抛「不支持」。Java 注释写明了原因 ——
   390→   *    「SYSTEM is read-only at adapter level; router allows for audit」：
   391→   *    即先让 allowlist 审计一次这次删除尝试，再拒绝。
   392→   *    结果消息与 default 分支相同，但**未注册的 sourceKey 会先被拦下**。
   393→   */
   394→  async delete(dataSource: DataSourceRef, id: string): Promise<void> {
   395→    switch (dataSource.type) {
   396→      case 'FORM': {
   397→        const formKey = this.formKeyAfterResolve(dataSource, 'delete')
   398→        await this.bizDataService.remove(formKey, id)
   399→        return
   400→      }
   401→      case 'SYSTEM':
   402→        this.router.resolve(dataSource, 'delete')
   403→        throw unsupported(dataSource, 'delete')
   404→      case 'WORKFLOW':
   405→        throw readOnlyWorkflow()
   406→      case 'API':
   407→        // 同上：必须 return，否则拒绝变成游离 Promise、outbound 失败被吞掉
   408→        return this.apiDelete(dataSource, id)
   409→      case 'SQL': {
   410→        const formKey = requireFormKey(dataSource, 'delete')
   411→        await this.bizDataService.remove(formKey, id)
   412→        return
   413→      }
   414→      default:
   415→        throw unsupported(dataSource, 'delete')
   416→    }
   417→  }
   418→
   419→  // ==================== FORM helpers ====================
   420→
   421→  /**
   422→   * FORM 分支的 formKey：**先过 router 再取值**（顺序与 Java 一致）。
   423→   *
   424→   * `router.resolve` 已经拦掉了空 formKey，下面那次判断因此不可达 ——
   425→   * 它存在的意义只是让类型收窄，消息与 router 里的那条保持逐字一致。
   426→   */
   427→  private formKeyAfterResolve(dataSource: DataSourceRef, op: string): string {
   428→    this.router.resolve(dataSource, op)
   429→    if (dataSource.formKey === null || dataSource.formKey.trim() === '') {
   430→      throw new BusinessException(400, 'FORM 数据源缺少 formKey')
   431→    }
   432→    return dataSource.formKey
   433→  }
   434→
   435→  /**
   436→   * 追加 config 模式 JOIN 虚拟列到 metadata；与主表列 key 重复的 `virtualKey` 跳过
   437→   * （防御性去重，对齐 Java `appendJoinColumns`）。
   438→   *
   439→   * ⚠️ 这里**只拷 5 个字段**（key/label/columnType/sortable/filterable），
   440→   *    其余（`length`/`required`/`pickerConfig`…）保持默认值。与 SYSTEM 的
   441→   *    `copyWithSortableFalse` 是同一类坑：多拷一个字段就与 Java 分叉。
   442→   */
   443→  private async appendJoinColumns(
   444→    columns: ColumnConfig[],
   445→    joins: JoinConfig[],
   446→  ): Promise<void> {
   447→    const existing = new Set(columns.map((column) => column.key))
   448→    for (const join of joins) {
   449→      const virtualKey = join.virtualKey
   450→      if (
   451→        virtualKey === null ||
   452→        virtualKey.trim() === '' ||
   453→        existing.has(virtualKey)
   454→      ) {
   455→        continue
   456→      }
   457→      columns.push({
   458→        ...newColumnConfig(),
   459→        key: virtualKey,
   460→        label: join.label === null || join.label.trim() === '' ? virtualKey : join.label,
   461→        columnType: await this.resolveJoinColumnType(join),
   462→        sortable: join.sortable,
   463→        filterable: join.filterable,
   464→      })
   465→      existing.add(virtualKey)
   466→    }
   467→  }
   468→
   469→  /**
   470→   * 虚拟列 `columnType`：目标表单 `joinField` 的列类型；查不到 fallback `"VARCHAR"`。
   471→   *
   472→   * ⚠️ 与 `BizDataSupport.resolveJoinColumnType`（query 路径）**不是同一份实现**：
   473→   *    这里要求目标列类型**非空非空白**才采用（空白继续找下一列），
   474→   *    那边是「首个同名列，null 才 fallback」。两份差异照抄 Java，不要合并。
   475→   */
   476→  private async resolveJoinColumnType(join: JoinConfig): Promise<string> {
   477→    try {
   478→      const target = await this.bizDataService.loadColumns(String(join.targetFormKey))
   479→      for (const column of target) {
   480→        if (
   481→          join.joinField === column.key &&
   482→          column.columnType !== null &&
   483→          column.columnType.trim() !== ''
   484→        ) {
   485→          return column.columnType.toUpperCase()
   486→        }
   487→      }
   488→    } catch (error) {
   489→      // 目标表单不可解析时回退（对齐 Java 的 `catch (BusinessException ignored)`）
   490→      if (!(error instanceof BusinessException)) throw error
   491→    }
   492→    return 'VARCHAR'
   493→  }

   495→  // ==================== API helpers ====================
   496→
   497→  /**
   498→   * API 数据源 `params` 解析（对齐 Java `parseParams`）。
   499→   *
   500→   * 三种失败形态各有专属消息，且**都带数据源名**；`requireAction` 的消息反而
   501→   * **不带**名字 —— 逐字照抄，不要「统一美化」。
   502→   */
   503→  private parseApiParams(dataSource: DataSourceRef): Record<string, unknown> {
   504→    const json = dataSource.params
   505→    if (json === null || json.trim() === '') {
   506→      throw new BusinessException(400, `API 数据源缺少 params: ${dataSource.name}`)
   507→    }
   508→    let parsed: unknown
   509→    try {
   510→      parsed = JSON.parse(json)
   511→    } catch {
   512→      throw new BusinessException(400, `API 数据源 params 不是合法 JSON: ${dataSource.name}`)
   513→    }
   514→    if (parsed === null || typeof parsed !== 'object' || Array.isArray(parsed)) {
   515→      throw new BusinessException(400, `API 数据源 params 必须是 JSON 对象: ${dataSource.name}`)
   516→    }
   517→    return parsed as Record<string, unknown>
   518→  }
   519→
   520→  /**
   521→   * 取某个操作段的配置（对齐 Java `operation`）。
   522→   *
   523→   * `list` 有一条**兼容旧格式**的旁路：顶层直接放 `action`/`method`/`parse` 时
   524→   * 组装成与 `list` 子对象等价的映射。写路径的三个操作**没有**这条旁路 ——
   525→   * 所以顶层 `action` 对 create/update/delete 不生效（这正是 golden 里
   526→   * 「API 数据源未配置 create 操作」那条断言的来源）。
   527→   */
   528→  private apiOperation(
   529→    params: Record<string, unknown>,
   530→    name: 'list' | 'get' | 'create' | 'update' | 'delete',
   531→  ): Record<string, unknown> | null {
   532→    const op = params[name]
   533→    if (op !== null && typeof op === 'object' && !Array.isArray(op)) {
   534→      return op as Record<string, unknown>
   535→    }
   536→    if (name === 'list' && params.action !== null && params.action !== undefined) {
   537→      return {
   538→        action: params.action,
   539→        method: params.method,
   540→        parse: params.parse,
   541→        totalParse: params.totalParse,
   542→      }
   543→    }
   544→    return null
   545→  }
   546→
   547→  /** 操作段必须有非空 `action`（对齐 Java `requireAction`）。 */
   548→  private requireApiAction(op: Record<string, unknown>, name: string): string {
   549→    const action = op.action
   550→    if (action === null || action === undefined || String(action).trim() === '') {
   551→      throw new BusinessException(400, `API 数据源 ${name} 操作缺少 action`)
   552→    }
   553→    return String(action)
   554→  }
   555→
   556→  /**
   557→   * 出站调用（对齐 Java 的 `httpExecutor.execute(action, opMethod(op), headers(params), …, vars, 10000, 10000, 0)`）。
   558→   *
   559→   * ⚠️ 三个常量与 Java 一致：超时 10000ms、重试 0 次、query/body 映射为空
   560→   *    （API 数据源的参数**只通过 URL 模板变量**传递，不走 query/body 映射）。
   561→   */
   562→  private async apiExecute(
   563→    dataSource: DataSourceRef,
   564→    params: Record<string, unknown>,
   565→    op: Record<string, unknown>,
   566→    opName: string,
   567→    vars: Record<string, unknown>,
   568→  ): Promise<string> {
   569→    const action = this.requireApiAction(op, opName)
   570→    const method = op.method === null || op.method === undefined || String(op.method).trim() === ''
   571→      ? 'GET'
   572→      : String(op.method).toUpperCase()
   573→    const headers: Record<string, string> = {}
   574→    if (params.headers !== null && typeof params.headers === 'object' && !Array.isArray(params.headers)) {
   575→      for (const [key, value] of Object.entries(params.headers as Record<string, unknown>)) {
   576→        headers[key] = String(value)
   577→      }
   578→    }
   579→    return this.httpExecutor.execute(
   580→      action,
   581→      method,
   582→      headers,
   583→      [],
   584→      [],
   585→      vars,
   586→      DEFAULT_TIMEOUT_MS,
   587→      DEFAULT_TIMEOUT_MS,
   588→      DEFAULT_RETRY,
   589→    )
   590→  }

   592→  private async apiCreate(
   593→    dataSource: DataSourceRef,
   594→    data: Record<string, unknown> | null,
   595→  ): Promise<string> {
   596→    const params = this.parseApiParams(dataSource)
   597→    const op = this.apiOperation(params, 'create')
   598→    if (op === null) {
   599→      throw new BusinessException(400, `API 数据源未配置 create 操作: ${dataSource.name}`)
   600→    }
   601→    // Java：vars 直接就是整个 body（不是 body 映射）
   602→    const raw = await this.apiExecute(dataSource, params, op, 'create', { ...(data ?? {}) })
   603→    const id = asDataMap(raw).id
   604→    return id === null || id === undefined ? '' : String(id)
   605→  }

   629→  // ==================== API 读路径（对齐 Java 的 apiMetadata / apiQuery / apiGet） ====================
   630→
   631→  /**
   632→   * API 元数据（对齐 Java `apiMetadata` + `metadata` 里的 API 分支）。
   633→   *
   634→   * ⚠️ 列**来自 `params.columns`**（不查库、不探测），而且外层还会再套一次
   635→   *    `copyWithSortableFalse` —— golden 实测：即便 `params.columns` 里写了
   636→   *    `length: 64`，返回的列上 `length` 仍是 **null**、`sortable` 是 **false**。
   637→   *    只拷 key/label/columnType 这一点与 SYSTEM 分支同源。
   638→   * ⚠️ `writable` = 「配了 create / update / delete 任一」（`operation` 的旧格式旁路
   639→   *    **只对 list 生效**，所以顶层 action 不会让 writable 变 true）。
   640→   */
   641→  private apiMetadata(dataSource: DataSourceRef, params: Record<string, unknown>): DataSourceMetadata {
   642→    const columns: ColumnConfig[] = []
   643→    const colsNode = params.columns
   644→    if (Array.isArray(colsNode)) {
   645→      for (const entry of colsNode) {
   646→        if (entry === null || typeof entry !== 'object' || Array.isArray(entry)) continue
   647→        columns.push({ ...newColumnConfig(), ...(entry as Partial<ColumnConfig>) })
   648→      }
   649→    }
   650→    const writable =
   651→      this.apiOperation(params, 'create') !== null ||
   652→      this.apiOperation(params, 'update') !== null ||
   653→      this.apiOperation(params, 'delete') !== null
   654→    return {
   655→      columns: copyWithSortableFalse(columns),
   656→      writable,
   657→      formKey: dataSource.formKey,
   658→    }
   659→  }
   660→
   661→  /**
   662→   * API 取数（对齐 Java `apiQuery`）。
   663→   *
   664→   * 变量表与 Java 一致：`params.data` 的键值 + `page`/`size` + `keyword`（null → 空串），
   665→   * 以及「keyword 非空且配了 `searchParam`」时把 keyword 再塞进 `vars[searchParam]`。
   666→   */
   667→  private async apiQuery(
   668→    dataSource: DataSourceRef,
   669→    req: BizDataQueryRequest,
   670→  ): Promise<BizDataPageVO> {
   671→    const params = this.parseApiParams(dataSource)
   672→    const op = this.apiOperation(params, 'list')
   673→    if (op === null) {
   674→      throw new BusinessException(400, `API 数据源未配置 list 操作: ${dataSource.name}`)
   675→    }
   676→    const vars: Record<string, unknown> = {}
   677→    if (params.data !== null && typeof params.data === 'object' && !Array.isArray(params.data)) {
   678→      for (const [key, value] of Object.entries(params.data as Record<string, unknown>)) {
   679→        vars[key] = value
   680→      }
   681→    }
   682→    vars.page = req.page
   683→    vars.size = req.size
   684→    const keyword = req.keyword
   685→    vars.keyword = keyword === null || keyword === undefined ? '' : keyword
   686→    const searchParam = params.searchParam === null || params.searchParam === undefined
   687→      ? null
   688→      : String(params.searchParam)
   689→    if (
   690→      keyword !== null &&
   691→      keyword !== undefined &&
   692→      keyword.trim() !== '' &&
   693→      searchParam !== null &&
   694→      searchParam.trim() !== ''
   695→    ) {
   696→      vars[searchParam] = keyword
   697→    }
   698→    const raw = await this.apiExecute(dataSource, params, op, 'list', vars)
   699→    return toPageVO(raw, op, req)
   700→  }

   702→  /** API 单行取数（对齐 Java `apiGet`）：变量只有 `id`。 */
   703→  private async apiGet(dataSource: DataSourceRef, id: string): Promise<BizDataVO> {
   704→    const params = this.parseApiParams(dataSource)
   705→    const op = this.apiOperation(params, 'get')
   706→    if (op === null) {
   707→      throw new BusinessException(400, `API 数据源未配置 get 操作: ${dataSource.name}`)
   708→    }
   709→    const raw = await this.apiExecute(dataSource, params, op, 'get', { id })
   710→    return { id, data: asDataMap(raw), version: null, createdAt: null, updatedAt: null }
   711→  }
   712→}

   714→/** API 数据源的固定出站超时（对齐 Java `DEFAULT_TIMEOUT_MS`）。 */
   715→const DEFAULT_TIMEOUT_MS = 10000
   716→/** API 数据源的固定重试次数（对齐 Java `DEFAULT_RETRY`）。 */
   717→const DEFAULT_RETRY = 0
   718→
   719→/**
   720→ * 出站响应 → `{id, data}` 映射（对齐 Java `asDataMap`）。
   721→ *
   722→ * ⚠️ 字符串响应要**先当 JSON 解析**；解析失败时 Java 抛
   723→ * 「外部 API 响应不是合法 JSON」（400）。对象直接返回；其余（数组/标量/null）→ 空对象。
   724→ */
   725→export function asDataMap(raw: unknown): Record<string, unknown> {
   726→  if (raw === null || raw === undefined) return {}
   727→  if (typeof raw === 'string') {
   728→    let parsed: unknown
   729→    try {
   730→      parsed = JSON.parse(raw)
   731→    } catch {
   732→      throw new BusinessException(400, '外部 API 响应不是合法 JSON')
   733→    }
   734→    if (parsed !== null && typeof parsed === 'object' && !Array.isArray(parsed)) {
   735→      return parsed as Record<string, unknown>
   736→    }
   737→    return {}
   738→  }
   739→  if (typeof raw === 'object' && !Array.isArray(raw)) return raw as Record<string, unknown>
   740→  return {}
   741→}
   742→
   743→/**
   744→ * 出站响应 → 分页结果（对齐 Java `toPageVO`）。
   745→ *
   746→ * 三步：把响应解析成 JSON 树 → 按 `op.parse` 的路径 walk 到数组（walk 不到或不是数组 →
   747→ * **空页**，不是报错）→ 逐行取 `id` 字段；`total` 默认取行数，配了 `totalParse`
   748→ * 且该处是数字时以它为准。
   749→ *
   750→ * ⚠️ `page` 用 `Math.max(req.page, 1)` 归一，`size` 原样透传（与 Java 一致）。
   751→ */
   752→export function toPageVO(
   753→  raw: unknown,
   754→  op: Record<string, unknown>,
   755→  req: BizDataQueryRequest,
   756→): BizDataPageVO {
   757→  const page = Math.max(req.page, 1)
   758→  let root: unknown
   759→  try {
   760→    root = typeof raw === 'string' ? JSON.parse(raw) : raw
   761→  } catch {
   762→    throw new BusinessException(400, '外部 API 响应不是合法 JSON')
   763→  }
   764→
   765→  const parsePath = op.parse === null || op.parse === undefined ? null : String(op.parse)
   766→  const recordsNode = parsePath === null || parsePath.trim() === '' ? root : walkPath(root, parsePath)
   767→  if (!Array.isArray(recordsNode)) {
   768→    return { records: [], total: 0, page, size: req.size }
   769→  }
   770→
   771→  const records: BizDataVO[] = recordsNode.map((item) => {
   772→    const data =
   773→      item !== null && typeof item === 'object' && !Array.isArray(item)
   774→        ? (item as Record<string, unknown>)
   775→        : {}
   776→    const id = data.id === null || data.id === undefined ? '' : String(data.id)
   777→    return { id, data, version: null, createdAt: null, updatedAt: null }
   778→  })
   779→
   780→  let total = records.length
   781→  const totalParse = op.totalParse === null || op.totalParse === undefined ? null : String(op.totalParse)
   782→  if (totalParse !== null && totalParse.trim() !== '') {
   783→    const node = walkPath(root, totalParse)
   784→    if (typeof node === 'number') total = node
   785→  }
   786→  return { records, total, page, size: req.size }
   787→}
   788→
   789→/** 按 `.` 分段 walk JSON（对齐 Java `walkPath`：中途不是对象 → null）。 */
   790→function walkPath(root: unknown, path: string): unknown {
   791→  let current: unknown = root
   792→  for (const part of path.split('.')) {
   793→    if (current === null || typeof current !== 'object' || Array.isArray(current)) return null
   794→    current = (current as Record<string, unknown>)[part]
   795→  }
   796→  return current
   797→}
