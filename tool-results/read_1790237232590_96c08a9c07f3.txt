   978→/** 覆盖策略 C：命中 key 全属性覆盖（保留 key），表单多出的 key 追加，当前列保留 */
   979→function overlayFromFormColumns(formCols: ColumnConfigItem[]) {
   980→  const target = form.type === 'API' ? apiColumns.value : sqlConfig.declaredColumns
   981→  const byKey = new Map(target.map((c) => [c.key, c]))
   982→  for (const fc of formCols) {
   983→    if (byKey.has(fc.key)) {
   984→      Object.assign(byKey.get(fc.key)!, fc, { key: fc.key })
   985→    } else {
   986→      target.push({ ...fc })
   987→      byKey.set(fc.key, target[target.length - 1])
   988→    }
   989→  }
   990→}

  1329→  /** SQL 类型：组装 params JSON */

  1335→      params.visual = {

  1338→        joins: sqlConfig.visual.joins,

  1344→      params.query = generatePreviewSql()
  1345→    } else {
  1346→      // SQL 模式：直接使用手写 SQL
  1347→      params.query = sqlConfig.queryText
  1348→    }
  1349→    // 列声明（全字段序列化，字段元数据 tab 编辑结果单一来源）
  1350→    const columns = sqlConfig.declaredColumns.filter((c) => c.key && c.key.trim())
  1351→    if (columns.length > 0) {
  1352→      params.columns = columns.map(serializeColumnConfig)
  1353→    }
  1354→    // 运行时参数白名单
  1355→    if (sqlConfig.declaredParams.length > 0) {
  1356→      params.params = sqlConfig.declaredParams
  1357→    }
  1358→    return params
  1359→  }

  1361→  /** 可视化模式：前端生成预览 SQL（简化版，后端会重新生成） */
  1362→  function generatePreviewSql(): string {
  1363→    const v = sqlConfig.visual
  1364→    if (!v.mainTable) return ''
  1365→    let sql = `SELECT ${v.selectColumns.join(', ') || '*'}`
  1366→    sql += ` FROM ${v.mainTable} ${v.mainAlias || 'm'}`
  1367→    for (const j of v.joins) {
  1368→      if (j.targetTable && j.on) {
  1369→        sql += ` ${j.joinType} ${j.targetTable} ${j.alias} ON ${j.on}`
  1370→      }
  1371→    }
  1372→    sql += ` WHERE ${v.mainAlias || 'm'}.tenant_id = :tenantId`
  1373→    for (const w of v.where) {
  1374→      if (w.column && w.op) {
  1375→        sql += ` AND ${w.column} ${w.op} ?`
  1376→      }
  1377→    }
  1378→    if (v.orderBy.length > 0) {
  1379→      const parts = v.orderBy.filter((o) => o.column).map((o) => `${o.column} ${o.order || 'ASC'}`)
  1380→      if (parts.length > 0) sql += ` ORDER BY ${parts.join(', ')}`
  1381→    }
  1382→    return sql
  1383→  }

  1395→    if (form.type === 'WORKFLOW' && !form.formKey) {
  1396→      ElMessage.warning('请选择绑定的工作流表单')
  1397→      return
  1398→    }
  1399→   if (form.type === 'SYSTEM' && !form.sourceKey) {
  1400→     ElMessage.warning('请选择系统结构')
  1401→     return
  1402→   }
  1403→   if (form.type === 'API') {
  1404→     if (!form.sourceKey || !form.sourceKey.trim()) {
  1405→       ElMessage.warning('请输入接口标识')
  1406→       return
  1407→     }
  1408→     if (!apiOps.list.action || !apiOps.list.action.trim()) {
  1409→       ElMessage.warning('列表查询 (list) 接口路径必填')
  1410→       return
  1411→     }
  1412→   }
  1413→    if (form.type === 'SQL') {
  1414→      if (!form.sourceKey || !form.sourceKey.trim()) {
  1415→        ElMessage.warning('请输入数据源标识（sourceKey）')
  1416→        return
  1417→      }
  1418→      if (sqlConfig.queryMode === 'visual' && !sqlConfig.visual.mainTable.trim()) {
  1419→        ElMessage.warning('请配置主表')
  1420→        return
  1421→      }
  1422→      if (sqlConfig.queryMode === 'sql' && !sqlConfig.queryText.trim()) {
  1423→        ElMessage.warning('请输入 SQL 模板')
  1424→        return
  1425→      }
  1426→    }
  1427→    if (form.type === 'FORM' && formJoin.value.queryMode === 'config') {
  1428→      const incomplete = formJoin.value.joins.find(
  1429→        (j) => !j.targetFormKey || !j.localField || !j.foreignField || !j.joinField || !j.virtualKey,
  1430→      )
  1431→      if (incomplete) {
  1432→        ElMessage.warning('请完整配置关联（目标表单/关联字段/显示字段/虚拟列标识）')
  1433→        return
  1434→      }
  1435→    }
  1436→    if (form.type === 'FORM' && formJoin.value.queryMode === 'sql' && !(formJoin.value.query || '').trim()) {
  1437→      ElMessage.warning('请输入 SQL 模板')
  1438→      return
  1439→    }
  1440→    saving.value = true
  1441→    try {
  1442→      const payload = normalizePayload()
  1443→      if (editingId.value) {
  1444→        await dataSourceApi.updateDataSource(editingId.value, payload)
  1445→      } else {
  1446→        await dataSourceApi.createDataSource(payload)
  1447→      }
  1448→      ElMessage.success(editingId.value ? '保存成功' : '创建成功')
  1449→      inlineVisible.value = false
  1450→      tableRef.value?.fetchList()
  1451→    } catch {
  1452→      // http 拦截器已弹出错误消息
  1453→    } finally {
  1454→      saving.value = false
  1455→    }
  1456→ }
  1457→
  1458→  /** FORM 类型：保留原始端点段（list/get/create/update/delete）基础上叠加关联查询配置段（queryMode/joins/query/columns/params） */
  1459→  function buildFormParams(): Record<string, any> {
  1460→    const params: Record<string, any> = { ...formJoinBaseParams.value }
  1461→    if (formJoin.value.queryMode === 'config') {
  1462→      params.queryMode = 'config'
  1463→      params.joins = formJoin.value.joins.filter((j) => j.targetFormKey && j.virtualKey)
  1464→    } else if (formJoin.value.queryMode === 'sql') {
  1465→      params.queryMode = 'sql'
  1466→      if (formJoin.value.query) {
  1467→        params.query = formJoin.value.query
  1468→      }
  1469→      const cols = (formJoin.value.columns || []).filter((c) => c.key && c.key.trim())
  1470→      if (cols.length > 0) {
  1471→        params.columns = cols.map(serializeColumnConfig)
  1472→      }
  1473→      if (formJoin.value.params && formJoin.value.params.length > 0) {
  1474→        params.params = [...formJoin.value.params]
  1475→      }
  1476→    }
  1477→    return params
  1478→  }

  1480→  /** 按类型归一化提交载荷：所有类型均通过统一 API 编辑器，FORM/SYSTEM params 由前端自动生成 */

  1484→      type: form.type || 'FORM',

  1486→      sourceKey: form.type === 'SYSTEM' ? form.sourceKey || null : form.type === 'API' || form.type === 'SQL' ? form.sourceKey || null : null,
  1487→      params: form.type === 'SQL' ? JSON.stringify(buildSqlParams())
  1488→        : form.type === 'FORM' ? JSON.stringify(buildFormParams())
  1489→          : JSON.stringify(buildApiParams()),
  1490→    }
  1491→  }

  1493→  /** 生成统一 API 端点描述（FORM/SYSTEM 自动填充到 API 编辑器；WORKFLOW 经 SPI 按数据源 ID 访问） */
