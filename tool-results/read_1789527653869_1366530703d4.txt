     1→<script lang="ts">
     2→// ----- 可视化查询构建组件 -----
     3→// 属性：modelValue（VisualQueryConfig）、params（运行时参数）、tables（可选表列表）、tableFields（表名→字段名列表）、disabled
     4→// 事件：update:modelValue、update:params
     5→// 别名规则：主表固定 m；目标表取表名去 wf_(biz_)? 前缀首字母，冲突时追加序号（c→c1→c2，自关联/同表多次关联由别名区分）
     6→// 目标表下拉不排除主表/已选表（自关联与同表多次关联为合法场景），仅以「主表/已关联」提示占用情况
     7→// 选择列：由 主表字段多选 + 各 JOIN 目标表字段多选 驱动，效果写入 selectColumns（别名.字段），选择列区以 tag 展示可删除
     8→// 筛选/排序：表名下拉（主表 + JOIN 目标表）+ 字段名下拉，组装为 别名.字段 存于 column
     9→export interface JoinClause {
    10→  alias: string
    11→  targetTable: string
    12→  joinType: string
    13→  on: string
    14→  columns: string[]
    15→}
    16→export interface WhereCondition {
    17→  column: string
    18→  op: string
    19→  value: string
    20→}
    21→export interface OrderClause {
    22→  column: string
    23→  order: string
    24→}
    25→export interface VisualQueryConfig {
    26→  mainTable: string
    27→  mainAlias: string
    28→  joins: JoinClause[]
    29→  selectColumns: string[]
    30→  selectColumnsInput: string
    31→  where: WhereCondition[]
    32→  orderBy: OrderClause[]
    33→}
    34→
    35→/** 主表别名固定值 */
    36→export const MAIN_ALIAS = 'm'
    37→
    38→/** 目标表别名：表名去 wf_(biz_)? 前缀取首字母小写，与 used 冲突时追加序号 */
    39→export function deriveJoinAlias(targetTable: string, used: string[]): string {
    40→  const base = (targetTable || '').replace(/^wf_(biz_)?/, '').trim()
    41→  const first = base ? base[0].toLowerCase() : 'j'
    42→  let alias = first
    43→  let i = 1
    44→  while (used.includes(alias)) {
    45→    alias = first + i++
    46→  }
    47→  return alias
    48→}
    49→
    50→/** 组装 ON 条件：主表字段 = 目标表字段（均带别名限定） */
    51→export function buildJoinOn(mainAlias: string, targetAlias: string, mainField: string, targetField: string): string {
    52→  return `${mainAlias}.${mainField} = ${targetAlias}.${targetField}`
    53→}
    54→
    55→/** 某表在目标表下拉中的占用提示：主表 → 「主表」，其他 JOIN 行已选 → 「已关联」；不限制选择（自关联/同表多次关联是合法 SQL 场景） */
    56→export type JoinTableHint = '' | '主表' | '已关联'
    57→export function joinTableHint(table: string, mainTable: string, joins: JoinClause[], skipIdx: number): JoinTableHint {
    58→  if (!table) return ''
    59→  if (table === mainTable) return '主表'
    60→  return joins.some((j, i) => i !== skipIdx && j.targetTable === table) ? '已关联' : ''
    61→}
    62→
    63→/** 解析 ON 条件字符串 → { mainField, targetField }。无别名限定/无法识别时默认左侧为主表字段 */
    64→export function parseJoinOn(on: string, mainAlias: string): { mainField: string; targetField: string } {
    65→  const eq = (on || '').split('=').map((s) => s.trim())
    66→  if (eq.length !== 2) return { mainField: '', targetField: '' }
    67→  const [left, right] = eq
    68→  const bare = (side: string) => (side.includes('.') ? side.split('.').slice(1).join('.') : side)
    69→  const rightAlias = right.includes('.') ? right.split('.')[0] : ''
    70→  if (rightAlias === mainAlias) {
    71→    // 右侧带主表别名 → 右侧为主表字段（兼容历史格式 c.id = m.customer_id）
    72→    return { mainField: bare(right), targetField: bare(left) }
    73→  }
    74→  return { mainField: bare(left), targetField: bare(right) }
    75→}

    79→import { ref, computed, watch } from 'vue'

    82→const props = withDefaults(
    83→  defineProps<{
    84→    modelValue: VisualQueryConfig
    85→    params?: string[]
    86→    tables?: string[]
    87→    tableFields?: Record<string, string[]>
    88→    disabled?: boolean
    89→  }>(),
    90→  { params: () => [], tables: () => [], tableFields: () => ({}), disabled: false }
    91→)

    93→const emit = defineEmits<{
    94→  (e: 'update:modelValue', value: VisualQueryConfig): void
    95→  (e: 'update:params', value: string[]): void
    96→}>()
    97→
    98→const WHERE_OPS = ['=', '!=', '>', '>=', '<', '<=', 'LIKE', 'IN']

   100→function deepClone(v: VisualQueryConfig): VisualQueryConfig {
   101→  return JSON.parse(JSON.stringify(v)) as VisualQueryConfig
   102→}
   103→
   104→// 本地副本：props 变化时整体替换，本地修改后 emit 新对象
   105→const local = ref<VisualQueryConfig>(deepClone(props.modelValue))
   106→watch(
   107→  () => props.modelValue,
   108→  (v) => {
   109→    local.value = deepClone(v)
   110→  },
   111→  { deep: true }
   112→)
   113→
   114→/** 每行 JOIN 的 on 字段下拉值（不入模型，仅 UI 编辑态） */
   115→const rowFields = ref<{ mainField: string; targetField: string }[]>([])
   116→function syncRowFields() {
   117→  while (rowFields.value.length < local.value.joins.length) {
   118→    const on = local.value.joins[rowFields.value.length].on || ''
   119→    const parsed = parseJoinOn(on, MAIN_ALIAS)
   120→    rowFields.value.push({ mainField: parsed.mainField, targetField: parsed.targetField })
   121→  }
   122→  while (rowFields.value.length > local.value.joins.length) {
   123→    rowFields.value.pop()
   124→  }
   125→}
   126→watch(
   127→  () => local.value.joins,
   128→  () => syncRowFields(),
   129→  { deep: true, immediate: true }
   130→)
   131→
   132→// ----- 别名 ↔ 表名 映射（筛选/排序表名下拉用）-----
   133→function aliasOfTable(table: string): string {
   134→  if (!table) return ''
   135→  if (table === local.value.mainTable) return MAIN_ALIAS
   136→  const j = local.value.joins.find((jj) => jj.targetTable === table)
   137→  return j ? j.alias : ''
   138→}
   139→function tableOfAlias(alias: string): string {
   140→  if (!alias) return ''
   141→  if (alias === MAIN_ALIAS) return local.value.mainTable
   142→  const j = local.value.joins.find((jj) => jj.alias === alias)
   143→  return j ? j.targetTable : ''
   144→}
   145→function splitColumn(column: string): { alias: string; field: string } {
   146→  const i = (column || '').indexOf('.')
   147→  if (i < 0) return { alias: '', field: column || '' }
   148→  return { alias: column.slice(0, i), field: column.slice(i + 1) }
   149→}
   150→/** 表名下拉选项：主表 + 各 JOIN 目标表 */
   151→const tableOptions = computed(() => {
   152→  const opts: string[] = []
   153→  if (local.value.mainTable) opts.push(local.value.mainTable)
   154→  for (const j of local.value.joins) {
   155→    if (j.targetTable && !opts.includes(j.targetTable)) opts.push(j.targetTable)
   156→  }
   157→  return opts
   158→})
   159→
   160→// ----- 筛选 UI 状态：每行 where 的 表名/字段名（column 即 别名.字段）-----
   161→interface ColUI {
   162→  table: string
   163→  field: string
   164→}
   165→const whereUIs = ref<ColUI[]>([])
   166→const orderUIs = ref<ColUI[]>([])
   167→function syncColUIs(target: ColUI[], rows: Array<{ column: string }>) {
   168→  while (target.length < rows.length) {
   169→    const { alias, field } = splitColumn(rows[target.length].column)
   170→    target.push({ table: tableOfAlias(alias), field })
   171→  }
   172→  while (target.length > rows.length) target.pop()
   173→}
   174→watch(
   175→  () => local.value.where,
   176→  () => syncColUIs(whereUIs.value, local.value.where),
   177→  { deep: true, immediate: true }
   178→)
   179→watch(
   180→  () => local.value.orderBy,
   181→  () => syncColUIs(orderUIs.value, local.value.orderBy),
   182→  { deep: true, immediate: true }
   183→)
   184→
   185→/** 已占用别名集合：主表别名 + 其他 JOIN 的别名 */
   186→function usedAliases(skipIdx: number): string[] {
   187→  const used = [MAIN_ALIAS]
   188→  for (let i = 0; i < local.value.joins.length; i++) {
   189→    if (i !== skipIdx && local.value.joins[i].targetTable) {
   190→      used.push(local.value.joins[i].alias || deriveJoinAlias(local.value.joins[i].targetTable, used))
   191→    }
   192→  }
   193→  return used
   194→}
   195→
   196→const newParamName = ref('')
   197→
   198→function addJoin() {
   199→  local.value.joins.push({ alias: '', targetTable: '', joinType: 'LEFT JOIN', on: '', columns: [] })
   200→  syncRowFields()
   201→  emitModel()
   202→}
   203→function removeJoin(idx: number) {
   204→  local.value.joins.splice(idx, 1)
   205→  syncRowFields()
   206→  emitModel()
   207→}
   208→function addWhere() {
   209→  local.value.where.push({ column: '', op: '=', value: '' })
   210→  emitModel()
   211→}
   212→function removeWhere(idx: number) {
   213→  local.value.where.splice(idx, 1)
   214→  emitModel()
   215→}
   216→function addOrderBy() {
   217→  local.value.orderBy.push({ column: '', order: 'ASC' })
   218→  emitModel()
   219→}
   220→function removeOrderBy(idx: number) {
   221→  local.value.orderBy.splice(idx, 1)
   222→  emitModel()
   223→}
   224→function addParam() {
   225→  const name = newParamName.value.trim()
   226→  if (name && !props.params.includes(name)) {
   227→    emit('update:params', [...props.params, name])
   228→    newParamName.value = ''
   229→  }
   230→}
   231→function removeParam(idx: number) {
   232→  const next = [...props.params]
   233→  next.splice(idx, 1)
   234→  emit('update:params', next)
   235→}
   236→function emitModel() {
   237→  emit('update:modelValue', deepClone(local.value))
   238→}
   239→
   240→/** JOIN 目标表选项占用提示：主表/已选表仅提示不限制（自关联合法） */
   241→function joinOptionHint(idx: number, table: string): string {
   242→  return joinTableHint(table, local.value.mainTable, local.value.joins, idx)
   243→}
   244→
   245→/** 目标表变化：重算别名并同步（主表别名固定 m） */
   246→function onJoinTargetChange(idx: number) {
   247→  const j = local.value.joins[idx]
   248→  if (j.targetTable) {
   249→    j.alias = deriveJoinAlias(j.targetTable, usedAliases(idx))
   250→  }
   251→  emitModel()
   252→}
   253→
   254→/** on 字段下拉变化：组装 ON 条件字符串（两者都有值才生成） */
   255→function onJoinFieldChange(idx: number) {
   256→  const j = local.value.joins[idx]
   257→  const f = rowFields.value[idx]
   258→  if (f && f.mainField && f.targetField) {
   259→    j.alias = j.alias || deriveJoinAlias(j.targetTable, usedAliases(idx))
   260→    j.on = buildJoinOn(MAIN_ALIAS, j.alias, f.mainField, f.targetField)
   261→  } else {
   262→    j.on = ''
   263→  }
   264→  emitModel()
   265→}
   266→
   267→/** 主表变化：emit 同步（供父组件联动字段候选） */
   268→function onMainChange() {
   269→  emitModel()
   270→}
   271→
   272→/** 主表字段候选：tableFields[mainTable] */
   273→const mainFieldOptions = computed(() => props.tableFields[local.value.mainTable] || [])
   274→/** 指定 JOIN 的目标表字段候选 */
   275→function targetFieldOptions(targetTable: string): string[] {
   276→  return props.tableFields[targetTable] || []
   277→}
   278→/** 筛选/排序字段候选：tableFields[表名] */
   279→function fieldOptionsFor(table: string): string[] {
   280→  return props.tableFields[table] || []
   281→}
   282→
   283→// ----- 选择列：别名.字段 前缀集合操作 -----
   284→function colsWithPrefix(alias: string): string[] {
   285→  const p = alias + '.'
   286→  return local.value.selectColumns.filter((c) => c.startsWith(p)).map((c) => c.slice(p.length))
   287→}
   288→function setColsWithPrefix(alias: string, fields: string[]) {
   289→  const p = alias + '.'
   290→  const others = local.value.selectColumns.filter((c) => !c.startsWith(p))
   291→  local.value.selectColumns = [...others, ...fields.map((f) => p + f)]
   292→}
   293→/** 主表字段多选选中集 */
   294→const mainSelectedCols = computed(() => colsWithPrefix(MAIN_ALIAS))
   295→function onMainColsChange(v: string[]) {
   296→  setColsWithPrefix(MAIN_ALIAS, v)
   297→  emitModel()
   298→}
   299→/** JOIN 目标表字段多选选中集（别名未生成时即时推导，不写回） */
   300→function joinColsValue(idx: number): string[] {
   301→  const j = local.value.joins[idx]
   302→  if (!j || !j.targetTable) return []
   303→  const alias = j.alias || deriveJoinAlias(j.targetTable, usedAliases(idx))
   304→  return colsWithPrefix(alias)
   305→}
   306→function onJoinColsChange(idx: number, v: string[]) {
   307→  const j = local.value.joins[idx]
   308→  if (!j || !j.targetTable) return
   309→  const alias = j.alias || deriveJoinAlias(j.targetTable, usedAliases(idx))
   310→  if (!j.alias) j.alias = alias
   311→  setColsWithPrefix(alias, v)
   312→  emitModel()
   313→}
   314→
   315→// ----- 筛选/排序：表名/字段名变化 → 组装 column -----
   316→function rebuildWhereColumn(idx: number) {
   317→  const ui = whereUIs.value[idx]
   318→  const w = local.value.where[idx]
   319→  if (!ui || !w) return
   320→  const alias = ui.table ? aliasOfTable(ui.table) : ''
   321→  w.column = ui.field && alias ? `${alias}.${ui.field}` : ''
   322→  emitModel()
   323→}
   324→function onWhereTableChange(idx: number, table: string) {
   325→  const ui = whereUIs.value[idx]
   326→  if (!ui) return
   327→  ui.table = table
   328→  ui.field = ''
   329→  rebuildWhereColumn(idx)
   330→}
   331→function onWhereFieldChange(idx: number, field: string) {
   332→  const ui = whereUIs.value[idx]
   333→  if (!ui) return
   334→  ui.field = field
   335→  rebuildWhereColumn(idx)
   336→}
   337→function rebuildOrderColumn(idx: number) {
   338→  const ui = orderUIs.value[idx]
   339→  const o = local.value.orderBy[idx]
   340→  if (!ui || !o) return
   341→  const alias = ui.table ? aliasOfTable(ui.table) : ''
   342→  o.column = ui.field && alias ? `${alias}.${ui.field}` : ''
   343→  emitModel()
   344→}
   345→function onOrderTableChange(idx: number, table: string) {
   346→  const ui = orderUIs.value[idx]
   347→  if (!ui) return
   348→  ui.table = table
   349→  ui.field = ''
   350→  rebuildOrderColumn(idx)
   351→}
   352→function onOrderFieldChange(idx: number, field: string) {
   353→  const ui = orderUIs.value[idx]
   354→  if (!ui) return
   355→  ui.field = field
   356→  rebuildOrderColumn(idx)
   357→}

   359→// 只读 SQL 预览：与后端 VisualSqlGenerator 输出一致（主表别名固定 m）
   360→const previewSql = computed(() => {
   361→  const v = local.value
   362→  if (!v.mainTable) return ''
   363→  const cols = v.selectColumns || []
   364→  let sql = `SELECT ${cols.join(', ') || '*'}`
   365→  sql += ` FROM ${v.mainTable} ${MAIN_ALIAS}`
   366→  for (const j of v.joins) {
   367→    if (j.targetTable && j.on) {
   368→      const alias = j.alias || deriveJoinAlias(j.targetTable, usedAliases(0))
   369→      sql += ` ${j.joinType} ${j.targetTable} ${alias} ON ${j.on}`
   370→    }
   371→  }
   372→  sql += ` WHERE ${MAIN_ALIAS}.tenant_id = :tenantId`
   373→  for (const w of v.where) {
   374→    if (w.column && w.op) {
   375→      sql += ` AND ${w.column} ${w.op} ?`
   376→    }
   377→  }
   378→  if (v.orderBy.length > 0) {
   379→    const parts = v.orderBy.filter((o) => o.column).map((o) => `${o.column} ${o.order || 'ASC'}`)
   380→    if (parts.length > 0) sql += ` ORDER BY ${parts.join(', ')}`
   381→  }
   382→  return sql
   383→})
