    64→/** params 解析（非法 JSON / 非对象 → 400） */
    65→function parseParams(paramsJson: string | null | undefined, allowNull = false): Data | null {
    66→  if (!paramsJson || paramsJson.trim() === "") {
    67→    if (allowNull) return null;
    68→    throw new BusinessException("数据源参数 params 必须是 JSON 对象", 400);
    69→  }
    70→  let node: unknown;
    71→  try {
    72→    node = JSON.parse(paramsJson);
    73→  } catch (e) {
    74→    throw new BusinessException("数据源参数 params 必须是合法 JSON: " + (e instanceof Error ? e.message : String(e)), 400);
    75→  }
    76→  if (!node || typeof node !== "object" || Array.isArray(node)) {
    77→    throw new BusinessException("数据源参数 params 必须是 JSON 对象", 400);
    78→  }
    79→  return node as Data;
    80→}

    84→function generateParams(type: string, formKey: string | null, sourceKey: string | null): string {
    85→  const params: Data = {};

    90→    params.get = { action: base + "/{id}", method: "GET" };
    91→    params.update = { action: base + "/{id}", method: "PUT" };
    92→    params.delete = { action: base + "/{id}", method: "DELETE" };
    93→  } else if (type === "SYSTEM") {
    94→    params.list = { action: "/api/v1/internal/system/" + mapSystemInternalKey(sourceKey!), method: "GET" };
    95→  }
    96→  return JSON.stringify(params);

   163→function resolveSortable(columns: ColumnConfig[]): ColumnConfig[] {
   164→  for (const c of columns) {
   165→    if (c.sortable == null) {
   166→      c.sortable = isSortable(c);
   167→    }
   168→  }
   169→  return columns;
   170→}
   171→
   172→function isSortable(c: ColumnConfig): boolean {
   173→  if (Array.isArray(c.subColumns) && c.subColumns.length > 0) return false;
   174→  const type = (c.columnType ?? "").toUpperCase();
   175→  if (type === "JSON" || type === "TEXT") return false;
   176→  return !c.componentType || c.componentType !== "colorPicker";
   177→}
   178→
   179→function copySortableFalse(columns: ColumnConfig[]): ColumnConfig[] {
   180→  return (columns ?? []).map((c) => ({ ...c, sortable: false }));
   181→}

   200→function dsMetadata(tenantId: string, ds: Row): Row {
   201→  const params = parseFormQueryConfig((ds.params as string | null) ?? "{}");
   202→  const type = ds.type as string;
   203→  if (type === "FORM") {
   204→    const cols = [...getBusinessColumnsByKey(tenantId, ds.formKey as string)];
   205→    if (params.queryMode === "config" && params.joins.length > 0) {
   206→      const mainKeys = new Set(cols.map((c) => c.key));
   207→      for (const j of params.joins) {
   208→        if (!mainKeys.has(j.virtualKey)) {
   209→          cols.push({ key: j.virtualKey, label: j.label ?? j.virtualKey, columnType: "VARCHAR", sortable: j.sortable, filterable: j.filterable });
   210→        }
   211→      }
   212→    } else if (params.queryMode === "sql" && params.query) {
   213→      const mainKeys = new Set(cols.map((c) => c.key));
   214→      for (const c of params.columns) if (!mainKeys.has(c.key)) cols.push(c);
   215→    }
   216→    resolveSortable(cols);
   217→    return { columns: cols, writable: true, formKey: ds.formKey };
   218→  }
   219→  if (type === "WORKFLOW") {
   220→    const cols = workflowColumnsFor(tenantId, ds.formKey as string);
   221→    resolveSortable(cols);
   222→    return { columns: cols, writable: false, formKey: ds.formKey };
   223→  }
   224→  if (type === "SYSTEM") {
   225→    return { columns: copySortableFalse(ds.sourceKey === "user-tree" ? USER_COLUMNS : DEPT_COLUMNS), writable: false, formKey: null };
   226→  }
   227→  if (type === "API") {
   228→    const p = parseParams(ds.params as string, true);
   229→    const declared = Array.isArray(p?.columns) ? (p!.columns as ColumnConfig[]) : [];
   230→    const writable = ["create", "update", "delete"].some((op) => op in (p ?? {}));
   231→    return { columns: copySortableFalse(declared), writable, formKey: null };
   232→  }
   233→  if (type === "SQL") {
   234→    const cols = params.queryMode === "config" ? [] : params.columns;
   235→    const writable = !!(ds.formKey && String(ds.formKey).trim() !== "");
   236→    return { columns: cols, writable, formKey: writable ? ds.formKey : null };
   237→  }
   238→  throw new BusinessException("数据源类型未启用: " + type, 400);
   239→}

   245→function workflowColumnsFor(tenantId: string, formKey: string): ColumnConfig[] {
   246→  const latest = queryRows(
   247→    'SELECT * FROM WF_FORM_DEF WHERE TENANT_ID = ? AND "key" = ? AND STATUS = ? ORDER BY VERSION DESC LIMIT 1',
   248→    [tenantId, formKey, "PUBLISHED"],
   249→  );
   250→  if (!latest.length) throw new BusinessException("表单不存在或未发布: " + formKey, 404);
   251→  const def = latest[0];
   252→  const extracted = def.type === "WORKFLOW" ? extractSchemaColumns(def.schema as string) : parseColumnConfig(def.columnConfig as string);
   253→  const map = new Map<string, ColumnConfig>();
   254→  for (const c of extracted) {
   255→    if (!c.key || !/^[a-zA-Z][a-zA-Z0-9_]{0,63}$/.test(c.key)) continue;
   256→    if (!map.has(c.key)) map.set(c.key, c);
   257→  }
   258→  return [...map.values()];
   259→}

   268→function workflowQuery(tenantId: string, formKey: string, req: BizQueryRequest): Row {
   269→  const bizCols = workflowColumnsFor(tenantId, formKey);
   270→  const filters: Array<[string, string]> = [];
   271→  if (req.filter) {
   272→    let parsed: Data;
   273→    try {
   274→      parsed = JSON.parse(req.filter);
   275→    } catch {
   276→      throw new BusinessException("筛选参数 filter 格式非法，应为 JSON 对象", 400);
   277→    }
   278→    if (parsed && typeof parsed === "object" && !Array.isArray(parsed)) {
   279→      // WORKFLOW 查询仅支持旧格式 {col:v} 等值（对齐 Java parseFilters）
   280→      if (Array.isArray(parsed.conditions)) {
   281→        for (const c of parsed.conditions as Data[]) {
   282→          const col = String(c.column ?? "");
   283→          if (!bizCols.some((b) => b.key === col)) throw new BusinessException("非法筛选字段: " + col, 400);
   284→          if (c.value != null) filters.push([col, String(c.value)]);
   285→        }
   286→      } else {
   287→        for (const [col, v] of Object.entries(parsed)) {
   288→          if (!bizCols.some((b) => b.key === col)) throw new BusinessException("非法筛选字段: " + col, 400);
   289→          if (v != null) filters.push([col, String(v)]);
   290→        }
   291→      }
   292→    }
   293→  }
   294→  let keywordColumn: string | null = null;
   295→  if (req.keyword && req.keywordColumn) {
   296→    for (const col of req.keywordColumn.split(",").map((s) => s.trim())) {
   297→      if (bizCols.some((b) => b.key === col)) {
   298→        keywordColumn = col;
   299→        break;
   300→      }
   301→    }
   302→    if (!keywordColumn) throw new BusinessException("关键词匹配列不在表单字段中", 400);
   303→  }
   304→
   305→  const ids = versionIds(tenantId, formKey);
   306→  const size = Math.max(1, req.size);
   307→  const page = Math.max(req.page, 1);
   308→  if (ids.length === 0) return bizPage([], 0, page, size);
   309→
   310→  const where: string[] = [
   311→    " FROM WF_FORM_DATA f LEFT JOIN WF_PROC_INST p ON p.ID = f.PROCESS_INSTANCE_ID",
   312→    " WHERE f.TENANT_ID = ? AND f.FORM_DEF_ID IN (" + ids.map(() => "?").join(",") + ") AND f.IS_SNAPSHOT = 0 AND f.PROCESS_INSTANCE_ID IS NOT NULL",
   313→  ];
   314→  const params: unknown[] = [tenantId, ...ids];
   315→  for (const [col, v] of filters) {
   316→    where.push(" AND JSON_UNQUOTE(JSON_EXTRACT(f.DATA_JSON, '$." + col + "')) = ?");
   317→    params.push(v);
   318→  }
   319→  if (keywordColumn) {
   320→    where.push(" AND JSON_UNQUOTE(JSON_EXTRACT(f.DATA_JSON, '$." + keywordColumn + "')) LIKE CONCAT('%', ?, '%')");
   321→    params.push(req.keyword!.trim());
   322→  }
   323→
   324→  const countSql = translateSql("SELECT COUNT(*)" + where.join(""));
   325→  const total = Number(queryOne<{ c: number }>(countSql, params)?.c ?? 0);
   326→
   327→  // 排序（startTime→START_TIME；派生列不可排；数值列先 CAST）
   328→  let orderBy: string;
   329→  if (!req.sort || req.sort.trim() === "") {
   330→    orderBy = " ORDER BY COALESCE(p.START_TIME, f.CREATED_AT) DESC";
   331→  } else if (req.sort === "startTime") {
   332→    orderBy = " ORDER BY p.START_TIME " + (String(req.order).toLowerCase() === "asc" ? "ASC" : "DESC");
   333→  } else if (WF_SYSTEM_KEYS.has(req.sort)) {
   334→    throw new BusinessException("排序字段不可排序: " + req.sort, 400);
   335→  } else {
   336→    const col = bizCols.find((b) => b.key === req.sort);
   337→    if (!col || !/^[a-zA-Z][a-zA-Z0-9_]{0,63}$/.test(req.sort)) throw new BusinessException("排序字段不在表单字段中: " + req.sort, 400);
   338→    const type = (col.columnType ?? "VARCHAR").toUpperCase();
   339→    const dir = String(req.order ?? "").toLowerCase() === "asc" ? "ASC" : "DESC";
   340→    if (type.includes("INT") || type.includes("DECIMAL")) {
   341→      const cast = type.includes("DECIMAL") ? "REAL" : "INTEGER";
   342→      orderBy = " ORDER BY CAST(JSON_UNQUOTE(JSON_EXTRACT(f.DATA_JSON, '$." + req.sort + "')) AS " + cast + ") " + dir;
   343→    } else {
   344→      orderBy = " ORDER BY JSON_UNQUOTE(JSON_EXTRACT(f.DATA_JSON, '$." + req.sort + "')) " + dir;
   345→    }
   346→  }
   347→
   348→  const rowsSql = translateSql(
   349→    "SELECT f.ID AS FID, f.DATA_JSON, f.PROCESS_INSTANCE_ID, p.START_TIME, p.START_USER, p.END_TIME, p.STATUS AS PSTATUS" +
   350→      where.join("") + orderBy + " LIMIT ? OFFSET ?",
   351→  );
   352→  const rawRows = query(rowsSql, [...params, size, (page - 1) * size]) as Row[];
   353→
   354→  const records = rawRows.map((r) => {
   355→    let raw: Data = {};
   356→    try {
   357→      raw = r.DATA_JSON ? JSON.parse(String(r.DATA_JSON)) : {};
   358→    } catch {
   359→      raw = {};
   360→    }
   361→    const pid = r.PROCESS_INSTANCE_ID == null ? "" : String(r.PROCESS_INSTANCE_ID);
   362→    const startUser = r.START_USER == null ? null : String(r.START_USER);
   363→    let initiatorName: string | null = null;
   364→    if (startUser != null && /^\d+$/.test(startUser)) {
   365→      const u = queryOne<{ NICKNAME: string | null; USERNAME: string }>("SELECT NICKNAME, USERNAME FROM SYS_USER WHERE ID = ?", [Number(startUser)]);
   366→      initiatorName = u ? (u.NICKNAME || u.USERNAME) : null;
   367→    }
   368→    let currentNodeName: string | null = null;
   369→    if (r.END_TIME == null && pid) {
   370→      const tasks = query("SELECT NODE_NAME FROM WF_TASK_INST WHERE PROC_INST_ID = ? AND END_TIME IS NULL AND NODE_NAME IS NOT NULL", [pid]) as Array<{ NODE_NAME: string }>;
   371→      const names = [...new Set(tasks.map((t) => t.NODE_NAME).filter((n) => n && n.trim() !== ""))].join("、");
   372→      if (names) currentNodeName = names;
   373→    }
   374→    const data: Data = {
   375→      instanceId: pid === "" ? null : pid,
   376→      processStatus: r.END_TIME == null ? (r.PSTATUS === "suspended" ? "suspended" : "running") : "completed",
   377→      initiatorName,
   378→      startTime: normTs(r.START_TIME),
   379→      currentNodeName,
   380→    };
   381→    for (const b of bizCols) data[b.key] = raw[b.key] ?? null;
   382→    return { id: String(r.FID), data, version: null, createdAt: null, updatedAt: null };
   383→  });
   384→  return bizPage(records, total, page, size);
   385→}

   387→function workflowGetById(tenantId: string, formKey: string, id: string): Row {
   388→  const ids = versionIds(tenantId, formKey);
   389→  if (!ids.length) throw new BusinessException("数据不存在: " + id, 404);
   390→  const exists = queryOne(
   391→    "SELECT ID FROM WF_FORM_DATA WHERE ID = ? AND TENANT_ID = ? AND FORM_DEF_ID IN (" + ids.map(() => "?").join(",") + ") AND IS_SNAPSHOT = 0 AND PROCESS_INSTANCE_ID IS NOT NULL",
   392→    [id, tenantId, ...ids],
   393→  );
   394→  if (!exists) throw new BusinessException("数据不存在: " + id, 404);
   395→  const page = workflowQuery(tenantId, formKey, { filter: null, keyword: null, keywordColumn: null, sort: null, order: null, params: null, page: 1, size: 100000 });
   396→  const hit = (page.records as Row[]).find((r) => String(r.id) === id);
   397→  if (!hit) throw new BusinessException("数据不存在: " + id, 404);
   398→  return hit;
   399→}

   528→function pageRefCount(tenantId: string, dsId: string): number {
   529→  const byColumn = queryOne<{ c: number }>(
   530→    "SELECT COUNT(1) AS c FROM WF_PAGE_DEF WHERE TENANT_ID = ? AND DATA_SOURCE_ID = ? AND STATUS <> ?",
   531→    [tenantId, dsId, "ARCHIVED"],
   532→  )?.c ?? 0;
   533→  let bySchema = 0;
   534→  const pages = queryRows(
   535→    "WF_PAGE_DEF",
   536→    "SELECT \"schema\" AS SCHEMA_JSON FROM WF_PAGE_DEF WHERE TENANT_ID = ? AND TYPE = ? AND STATUS <> ? AND DATA_SOURCE_ID IS NULL",
   537→    [tenantId, "PAGE", "ARCHIVED"],
   538→  );
   539→  for (const p of pages) {
   540→    try {
   541→      const schema = JSON.parse(String(p.SCHEMA_JSON ?? "{}"));
   542→      const sources = Array.isArray(schema?.dataSources) ? schema.dataSources : [];
   543→      if (sources.some((s: Data) => String(s?.refId ?? "") === dsId)) bySchema++;
   544→    } catch {
   545→      // 非法 schema 跳过
   546→    }
   547→  }
   548→  return Number(byColumn) + bySchema;
   549→}

   603→/** FORM 查询配置段保存校验（config→joins 校验；sql→SqlTemplateEngine.validate） */
   604→function validateFormQueryConfig(tenantId: string, params: string): void {
   605→  let root: Data;
   606→  try {
   607→    root = JSON.parse(params);
   608→  } catch (e) {
   609→    throw new BusinessException("数据源参数 params 必须是合法 JSON: " + (e instanceof Error ? e.message : String(e)), 400);
   610→  }
   611→  if (!root || typeof root !== "object" || Array.isArray(root)) {
   612→    throw new BusinessException("数据源参数 params 必须是 JSON 对象", 400);
   613→  }
   614→  const mode = root.queryMode == null ? null : String(root.queryMode);
   615→  if (mode === "config") {
   616→    const joins = Array.isArray(root.joins) ? (root.joins as Data[]) : [];
   617→    if (joins.length === 0) throw new BusinessException("queryMode=config 时必须配置至少一个关联 joins", 400);
   618→    const aliases = new Set<string>();
   619→    const virtualKeys = new Set<string>();
   620→    joins.forEach((j, idx) => {
   621→      const alias = String(j.alias ?? "");
   622→      if (!/^[a-zA-Z][a-zA-Z0-9_]{0,63}$/.test(alias)) throw new BusinessException(`joins 第 ${idx + 1} 项 alias 非法: ${alias}`, 400);
   623→      if (aliases.has(alias)) throw new BusinessException("关联别名重复: " + alias, 400);
   624→      aliases.add(alias);
   625→      const targetFormKey = String(j.targetFormKey ?? "");
   626→      if (!targetFormKey) throw new BusinessException(`joins 第 ${idx + 1} 项必须指定目标表单 targetFormKey`, 400);
   627→      if (!queryOne('SELECT ID FROM WF_FORM_DEF WHERE TENANT_ID = ? AND "key" = ? LIMIT 1', [tenantId, targetFormKey])) {
   628→        throw new BusinessException("目标表单不存在: " + targetFormKey, 400);
   629→      }
   630→      if (!tableExists("WF_BIZ_" + targetFormKey.toUpperCase())) throw new BusinessException("关联表单不存在: " + targetFormKey, 400);
   631→      for (const f of ["localField", "foreignField", "joinField", "virtualKey"]) {
   632→        if (!String(j[f] ?? "").trim()) throw new BusinessException(`joins 第 ${idx + 1} 项必须指定${f} ${f}`, 400);
   633→      }
   634→      const vkey = String(j.virtualKey ?? "");
   635→      if (virtualKeys.has(vkey)) throw new BusinessException("虚拟列 virtualKey 重复: " + vkey, 400);
   636→      virtualKeys.add(vkey);
   637→    });
   638→    return;
   639→  }
   640→  if (mode === "sql" || mode === "visual") {
   641→    try {
   642→      // 复用 bizdata 的模板校验语义（经 querySqlTemplate 的 validate 路径）
   643→      const cfg = parseFormQueryConfig(params);
   644→      const cols = cfg.columns.map((c) => ({ key: c.key, ref: c.key, columnType: (c.columnType ?? "").toUpperCase(), sortable: c.sortable === true, filterable: c.filterable === true }));
   645→      validateTemplatePublic(cfg.query, cols, cfg.declaredParams);
   646→    } catch (e) {
   647→      if (e instanceof Error) throw new BusinessException(e.message, 400);
   648→      throw e;
   649→    }
   650→  }
   651→}
