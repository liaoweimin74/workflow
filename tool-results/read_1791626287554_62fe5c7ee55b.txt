   494→    // 合并排序状态（服务器端排序；order 已归一化为 asc/desc）
   495→    if (sortState.value) {
   496→      params.sort = sortState.value.prop
   497→      params.order = sortState.value.order
   498→    }
   499→    const res = await props.fetchApi(params)
   500→    list.value = Array.isArray(res.rows) ? res.rows : []
   501→    total.value = Number(res.total) || 0
   502→  } finally {
   503→    loading.value = false
   504→  }
   505→}

   507→/** 排序变化：记录排序状态并重新请求；同时照常转发 sort-change 事件（不破坏事件总线动作） */
   508→function handleSortChange(args: { column: any; prop: string; order: string }) {
   509→  sortState.value = args.order
   510→    ? { prop: args.prop, order: args.order === 'ascending' ? 'asc' : 'desc' }
   511→    : null
   512→  emit('sort-change', args)
   513→  fetchList()
   514→}

   533→async function handleCreate(initialValues?: Record<string, any>) {
   534→  await props.formConfig?.onFormOpen?.()
   535→  if (props.formConfig?.beforeCreate) {
   536→    const ok = await props.formConfig.beforeCreate()

   539→  isEdit.value = false
   540→  editId.value = 0
   541→  editingRow.value = null
   542→  const configInitialValues = props.formConfig?.initialValues || {}
   543→  dialogInitialValues.value = { ...configInitialValues, ...(initialValues || {}) }
   544→  dialogTitle.value = props.formConfig?.dialogTitle?.create || '新增'
   545→  dialogFormKey.value++ // 强制重建 FormRenderer，加载最新 initialValues
   546→  dialogVisible.value = true
   547→}

   549→async function handleEdit(row: any) {
   550→  await props.formConfig?.onFormOpen?.()
   551→  if (props.formConfig?.beforeEdit) {
   552→    const ok = await props.formConfig.beforeEdit(row)
   553→    if (ok === false) return
   554→  }
   555→  isEdit.value = true
   556→  editId.value = row.id
   557→  editingRow.value = row
   558→  dialogTitle.value = props.formConfig?.dialogTitle?.edit || '编辑'
   559→
   560→  if (props.formConfig?.getApi) {
   561→    formLoading.value = true
   562→    try {
   563→      const res = await props.formConfig.getApi(row.id)
   564→      dialogInitialValues.value = { ...res }
   565→    } finally {
   566→      formLoading.value = false
   567→    }
   568→  } else {
   569→    dialogInitialValues.value = { ...row }
   570→  }
   571→  dialogFormKey.value++ // 强制重建 FormRenderer，加载最新 initialValues
   572→  dialogVisible.value = true
   573→}

   575→async function handleDelete(row: any) {
   576→  if (props.formConfig?.beforeDelete) {
   577→    const ok = await props.formConfig.beforeDelete(row)
   578→    if (ok === false) return
   579→  }
   580→  const msg = props.deleteConfirm ? props.deleteConfirm(row) : '确定删除该记录吗？'
   581→  try {
   582→    await ElMessageBox.confirm(msg, '删除确认', { type: 'warning' })
   583→  } catch {
   584→    return
   585→  }
   586→  try {
   587→    await props.formConfig?.deleteApi?.(row.id, row)
   588→    ElMessage.success('删除成功')
   589→    props.formConfig?.afterDelete?.()
   590→    fetchList()
   591→  } catch {
   592→    // http 拦截器统一处理
   593→  }
   594→}

   596→async function handleDialogSubmit() {
   597→  const formData = formRendererRef.value?.getFormData() || {}
   598→
   599→  formLoading.value = true
   600→  try {
   601→    if (isEdit.value) {
   602→      await props.formConfig?.updateApi?.(editId.value, formData, editingRow.value)
   603→      props.formConfig?.afterUpdate?.(formData)
   604→    } else {
   605→      await props.formConfig?.createApi?.(formData)
   606→      props.formConfig?.afterCreate?.(formData)
   607→    }
   608→    ElMessage.success(isEdit.value ? '更新成功' : '创建成功')
   609→    dialogVisible.value = false
   610→    fetchList()
   611→  } finally {
   612→    formLoading.value = false
   613→  }
   614→}
