    69→    69→     * 加载表单运行时上下文（表名 + 列 + 子表元数据）。
    70→    70→     */
    71→    71→    public BizDataContext loadContext(String formKey) {
    72→    72→        if (formKey == null || !FORM_KEY_PATTERN.matcher(formKey).matches()) {
    73→    73→            throw new BusinessException(400, "非法表单 key: " + formKey);
    74→    74→        }
    75→    75→        String tableName = "wf_biz_" + formKey;
    76→    76→        if (!tableManager.tableExists(tableName)) {
    77→    77→            throw new BusinessException(404, "业务表单数据表不存在: " + formKey);
    78→    78→        }
    79→    79→        List<ColumnConfig> columns = formDefService.getBusinessColumnsByKey(formKey);
    80→    80→        // 主表列：排除子表字段（子表字段映射独立物理表，非主表列）
    81→    81→        List<String> keys = columns.stream()
    82→    82→                .filter(c -> c.getSubColumns() == null || c.getSubColumns().isEmpty())
    83→    83→                .map(ColumnConfig::getKey)
    84→    84→                .toList();
    85→    85→        Map<String, BizDataContext.SubTableDef> subTables = new LinkedHashMap<>();
    86→    86→        for (ColumnConfig c : columns) {
    87→    87→            if (c.getSubColumns() != null && !c.getSubColumns().isEmpty()) {
    88→    88→                String mode = c.getSubMode() == null || c.getSubMode().isBlank() ? "embedded" : c.getSubMode();
    89→    89→                List<String> subKeys = c.getSubColumns().stream().map(ColumnConfig::getKey).toList();
    90→    90→                subTables.put(c.getKey(), new BizDataContext.SubTableDef(
    91→    91→                        "wf_biz_" + formKey + "_" + c.getKey(), mode, c.getSubColumns(), subKeys));
    92→    92→            }
    93→    93→        }
    94→    94→        return new BizDataContext(tableName, formKey, columns, keys, subTables);
    95→    95→    }

   165→   165→     * 批量解析被引用记录的显示文本（id → displayField 值）。
   166→   166→     */
   167→   167→    public Map<String, String> resolveDisplayTexts(String sourceFormKey, List<String> ids, String displayField) {
   168→   168→        if (sourceFormKey == null || !FORM_KEY_PATTERN.matcher(sourceFormKey).matches()) {
   169→   169→            throw new BusinessException(400, "非法目标表单 key: " + sourceFormKey);
   170→   170→        }
   171→   171→        if (ids == null || ids.isEmpty()) {
   172→   172→            return Map.of();
   173→   173→        }
   174→   174→        if (displayField == null || !displayField.matches("^[a-zA-Z][a-zA-Z0-9_]{0,63}$")) {
   175→   175→            throw new BusinessException(400, "非法显示字段: " + displayField);
   176→   176→        }
   177→   177→        String table = "wf_biz_" + sourceFormKey;
   178→   178→        String placeholders = String.join(",", java.util.Collections.nCopies(ids.size(), "?"));
   179→   179→        String sql = "SELECT id, " + displayField + " FROM " + table
   180→   180→                + " WHERE tenant_id = ? AND id IN (" + placeholders + ")";
   181→   181→        List<Object> params = new ArrayList<>();
   182→   182→        params.add(tenantProvider.getTenantId());
   183→   183→        params.addAll(ids);
   184→   184→
   185→   185→        Map<String, String> result = new LinkedHashMap<>();
   186→   186→        for (Map<String, Object> row : jdbcTemplate.queryForList(sql, params.toArray())) {
   187→   187→            Object v = row.get(displayField);
   188→   188→            result.put(String.valueOf(row.get("id")), v == null ? "" : String.valueOf(v));
   189→   189→        }
   190→   190→        return result;
   191→   191→    }

   235→   235→        String bizId = insertedId(insert);

   237→   237→            Object raw = data.get(e.getKey());
   238→   238→            if (raw instanceof List<?> rows) {
   239→   239→                writeSubRows(e.getValue(), bizId, rows);
   240→   240→            }
   241→   241→        }
   242→   242→
   243→   243→        // 新行 id 由 buildInsert 内部生成，查询返回
   244→   244→        return findById(ctx.tableName(), tenantId, ctx, bizId);

   251→   251→        String tenantId = tenantProvider.getTenantId();

   253→   253→
   254→   254→        Map<String, Object> filters = parseFilter(req.getFilter());

   256→   256→        Map<String, String> columnTypeOf = ctx.columns().stream()
   257→   257→                .collect(Collectors.toMap(ColumnConfig::getKey,
   258→   258→                        c -> c.getColumnType() == null ? "" : c.getColumnType().toUpperCase(), (a, b) -> a));
   259→   259→        int page = Math.max(req.getPage(), 1);

   261→   261→        int size = req.getSize() <= 0 ? req.getSize() : Math.min(Math.max(req.getSize(), 1), 100);
   262→   262→
   263→   263→        try {
   264→   264→            BizDataQueryBuilder.SqlAndParams count = BizDataQueryBuilder.buildCount(
   265→   265→                    ctx.tableName(), ctx.columnKeys(), columnTypeOf, tenantId, filters, req.getKeyword(), req.getKeywordColumn());
   266→   266→
   267→   267→            BizDataQueryBuilder.SqlAndParams select = BizDataQueryBuilder.buildSelect(
   268→   268→                    ctx.tableName(), ctx.columnKeys(), columnTypeOf, tenantId, filters,
   269→   269→                    req.getKeyword(), req.getKeywordColumn(), req.getSort(), req.getOrder(), page - 1, size);
   270→   270→
   271→   271→            return sqlQueryEngine.execPage(page, size, count, select, row -> toVO(ctx, row));
   272→   272→        } catch (IllegalArgumentException e) {

   274→   274→        }
   275→   275→    }

   309→   309→    public BizDataPageVO querySqlTemplate(String formKey, BizDataQueryRequest req, FormQueryConfig cfg) {
   310→   310→        // SQL 类型无 formKey 时跳过校验（管理员 SQL 独立定义，不依赖表单）
   311→   311→        if (formKey != null && !FORM_KEY_PATTERN.matcher(formKey).matches()) {

   313→   313→        }

   315→   315→        List<JoinSqlGenerator.QueryColumn> columns = toQueryColumns(cfg.columns());
   316→   316→        Map<String, Object> filters = parseFilter(req.getFilter());
   317→   317→        Map<String, Object> runtimeParams = parseRuntimeParams(req.getParams());
   318→   318→        int page = Math.max(req.getPage(), 1);
   319→   319→        // size <= 0 表示不分页取全部；正数沿用原钳制上限
   320→   320→        int size = req.getSize() <= 0 ? req.getSize() : Math.min(Math.max(req.getSize(), 1), 100);
   321→   321→
   322→   322→        try {
   323→   323→            SqlQueryEngine.WrappedQuery wq = SqlTemplateEngine.wrap(
   324→   324→                    cfg.query(), tenantId, columns, filters, req.getKeyword(), req.getKeywordColumn(),
   325→   325→                    req.getSort(), req.getOrder(), page, size, cfg.declaredParams(), runtimeParams);
   326→   326→            return sqlQueryEngine.execPage(page, size, wq.count(), wq.select(),
   327→   327→                    row -> toSqlVO(cfg.columns(), row));
   328→   328→        } catch (IllegalArgumentException e) {
   329→   329→            throw new BusinessException(400, e.getMessage());
   330→   330→        }

   333→   333→    /** 构建查询列映射：主表列（ref="m."+key，默认全可排可筛）+ 虚拟列（ref=alias+"."+joinField，能力取 join 声明） */
   334→   334→    private List<JoinSqlGenerator.QueryColumn> buildJoinColumns(BizDataContext ctx,
   335→   335→                                                                List<JoinSqlGenerator.JoinConfig> joins) {
   336→   336→        Map<String, String> typeOf = new HashMap<>();
   337→   337→        for (ColumnConfig c : ctx.columns()) {
   338→   338→            typeOf.put(c.getKey(), c.getColumnType() == null ? "" : c.getColumnType().toUpperCase());
   339→   339→        }

   341→   341→        for (String key : ctx.columnKeys()) {
   342→   342→            columns.add(new JoinSqlGenerator.QueryColumn(key, "m." + key, typeOf.getOrDefault(key, ""), true, true));
   343→   343→        }
   344→   344→        for (JoinSqlGenerator.JoinConfig j : joins) {
   345→   345→            columns.add(new JoinSqlGenerator.QueryColumn(j.virtualKey(), j.alias() + "." + j.joinField(),
   346→   346→                    resolveJoinColumnType(j), j.sortable(), j.filterable()));
   347→   347→        }
   348→   348→        return columns;
   349→   349→    }

   355→   355→            if (targetCols != null) {

   363→   363→            // 目标表单不可解析时 fallback 类型

   396→   396→    @Transactional
   397→   397→    public BizDataVO updateGeneric(String formKey, String id, Map<String, Object> data, Integer version) {
   398→   398→        String tenantId = tenantProvider.getTenantId();
   399→   399→        BizDataContext ctx = loadContext(formKey);
   400→   400→
   401→   401→        validateRequired(ctx.columns(), data);
   402→   402→        int currentVersion = version == null ? 1 : version;
   403→   403→
   404→   404→        Map<String, Object> merged = serializeJsonColumns(data);
   405→   405→        merged.putAll(resolvePickerValues(ctx, merged));
   406→   406→
   407→   407→        BizDataQueryBuilder.SqlAndParams update = BizDataQueryBuilder.buildUpdate(
   408→   408→                ctx.tableName(), ctx.columnKeys(), merged, tenantId, id, currentVersion);
   409→   409→        int affected = jdbcTemplate.update(update.sql(), update.params().toArray());
   410→   410→
   411→   411→        if (affected == 0) {
   412→   412→            // 区分"记录不存在"（404）与"版本冲突"（409）
   413→   413→            List<Map<String, Object>> exists = jdbcTemplate.queryForList(
   414→   414→                    "SELECT id, version FROM " + ctx.tableName() + " WHERE id = ? AND tenant_id = ?", id, tenantId);
   415→   415→            if (exists.isEmpty()) {
   416→   416→                throw new BusinessException(404, "业务数据不存在: " + id);
   417→   417→            }
   418→   418→            throw new BusinessException(409, "数据已被他人修改，请刷新后重试");
   419→   419→        }
   420→   420→
   421→   421→        // 子表行增量 diff（仅当请求携带子表字段时）
   422→   422→        for (Map.Entry<String, BizDataContext.SubTableDef> e : ctx.subTables().entrySet()) {
   423→   423→            Object raw = data.get(e.getKey());
   424→   424→            if (raw instanceof List<?> rows) {
   425→   425→                diffSubRows(e.getValue(), id, rows);
   426→   426→            }
   427→   427→        }
   428→   428→
   429→   429→        return findById(ctx.tableName(), tenantId, ctx, id);
