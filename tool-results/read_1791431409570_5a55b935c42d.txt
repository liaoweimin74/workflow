     1→/**
     2→ * 逻辑流 DSL ↔ vue-flow 画布数据 转换工具。
     3→ *
     4→ * 契约（worklog Task 2 + SUBFLOW/inputVars 增量 + DATA_UPDATE）：
     5→ * - nodes[]: { id, type: START|END|HTTP|BEAN|SCRIPT|CONDITION|BATCH|SUBFLOW|DATA_UPDATE, name, x, y,
     6→ *              config{...}, resultVar?, errorAction?: FAIL_FLOW|IGNORE_CONTINUE }
     7→ *   - HTTP   config = { url, method, headers{...}, queryParams[{source,target}],
     8→ *                       bodyParams[{source,target}], connTimeoutMs, readTimeoutMs, retryCount }
     9→ *   - BEAN   config = { beanName, methodName, params[{source,target}] }
    10→ *   - SCRIPT config = { language: 'groovy', source }
    11→ *   - CONDITION config = { variable, operator: EQ|NE|GT|LT|GTE|LTE|EMPTY|NOT_EMPTY, value? }
    12→ *   - BATCH  config = { collection, itemVar?, indexVar?, body?: BatchBodyNode[](循环体链，
    13→ *                       按序逐项执行；节点可选 x/y 画布绝对坐标——仅拖离默认排布位才
    14→ *                       写入，缺省时回显自动居中排布)，stopOnError?, maxItems? }
    15→ *                       （legacy 兼容：actionType: HTTP|SCRIPT|BEAN + actionConfig 单动作，
    16→ *                         读取时自动合成为单节点循环体，保存后升级为 body 形态）
    17→ *   - SUBFLOW config = { flowId, passAllVars=true, varsMapping[{source,target}] }
    18→ *   - DATA_UPDATE config = { table, setOps[{column, mode: SET|ADD|SUB, value}],
    19→ *                            where[{column, op: EQ|NE|GT|GTE|LT|LTE|IS_NULL|NOT_NULL, value?}] }
    20→ *       值支持字面量或 {{var}}/{{formData.xxx}} 点路径；受影响行数写回 resultVar
    21→ * - edges[]: { id?, source, target, branch?: 'true'|'false'（仅 CONDITION 出边用） }
    22→ *   - BATCH 循环体边（画布 data.loop=true）为设计器内部结构，序列化时剔除；
    23→ *     循环体内容编入 BATCH.config.body[]
    24→ * - 顶层 inputVars[]: 入参声明（可选）：[{ name, type: string|number|boolean|json, required?, desc? }]
    25→ *
    26→ * vue-flow 侧：
    27→ * - node.type 固定为 'logic'（自定义节点渲染槽 #node-logic），
    28→ *   节点业务类型存 node.data.nodeType
    29→ * - position 需要 {x,y}；契约的顶层 x/y 与之对应
    30→ * - branch 存到 edge.data.branch
    31→ * - 序列化时剥离 vue-flow 内部字段（selected/handles/dimensions...），只留契约字段
    32→ */

   276→    target,
   277→    sourceHandle,
   278→    targetHandle,
   279→    type: source === target ? 'loop' : 'smoothstep',
   280→    markerEnd: 'arrowclosed',
   281→    class: 'lf-edge-loop',

   359→/** 短随机码（节点 id 兜底去重用） */
   360→function shortCode(): string {
   361→  return Math.random().toString(36).slice(2, 6)
   362→}

   364→/** 生成不与现有节点冲突的节点 id：http_a1b2 形式 */
   365→export function createNodeId(type: LogicNodeType, existingIds: Iterable<string>): string {
   366→  const taken = new Set(existingIds)
   367→  const prefix = type.toLowerCase()
   368→  let id = `${prefix}_${shortCode()}`
   369→  while (taken.has(id)) {
   370→    id = `${prefix}_${shortCode()}${Math.floor(Math.random() * 10)}`
   371→  }
   372→  return id
   373→}

   388→/** 循环体类型归一：非法类型返回 null（调用方跳过该节点） */
   389→function toBatchBodyType(raw: unknown): BatchBodyType | null {
   390→  const upper = String(raw ?? '').toUpperCase()
   391→  return (BATCH_BODY_TYPES as string[]).includes(upper) ? (upper as BatchBodyType) : null
   392→}

   394→/**
   395→ * 为每个 BATCH 节点合成循环体画布结构与闭合循环连线：
   396→ * - config.body 非空 → 循环体节点沿链 loop_start→b0→…→loop_end；
   397→ *   节点已持久化 x/y（曾拖离默认位）→ 用绝对坐标回显，位置不再丢失；
   398→ *   未持久化 → 横排在批处理下方居中排布（默认位）；
   399→ * - legacy（无 body 有 actionType/actionConfig）→ 合成单个循环体节点；
   400→ * - 两者皆无 → loop_start 直连 loop_end（空循环虚线）。
   401→ * 循环边均为 deletable:false + data.loop=true（设计器内部结构，serialize 时剔除）。
   402→ */
   403→function synthesizeBatchLoops(nodes: FlowNode[], edges: FlowEdge[]): void {
   404→  for (const batch of nodes) {
   405→    if (batch.data.nodeType !== 'BATCH') continue
   406→    const cfg = (batch.data.config ?? {}) as Partial<BatchNodeConfig>
   407→    const rawBody = Array.isArray(cfg.body) ? cfg.body : []
   408→
   409→    // legacy 单动作 → 合成为单节点循环体
   410→    let bodyItems = rawBody
   411→    if (!bodyItems.length && cfg.actionType) {
   412→      const legacyType = toBatchBodyType(cfg.actionType)
   413→      if (legacyType) {
   414→        bodyItems = [
   415→          {
   416→            type: legacyType,
   417→            name: defaultNodeName(legacyType as LogicNodeType),
   418→            config: cfg.actionConfig as NodeConfig | undefined,
   419→          },
   420→        ]
   421→      }
   422→    }
   423→
   424→    const valid = bodyItems
   425→      .map((bn) => {
   426→        const type = toBatchBodyType(bn?.type)
   427→        return type ? { bn, type } : null
   428→      })
   429→      .filter((x): x is { bn: BatchBodyNode; type: BatchBodyType } => x !== null)
   430→
   431→    if (valid.length) {
   432→      const count = valid.length
   433→      const bodyNodes: FlowNode[] = valid.map(({ bn, type }, i) => ({
   434→        id: bn.id || `${batch.id}__b${i}`,
   435→        type: FLOW_NODE_TYPE,
   436→        position: {
   437→          // 已持久化的绝对坐标优先（拖动位置不丢失）；否则按默认位：批处理下方居中横排
   438→          x:
   439→            Number.isFinite(bn.x) && Number.isFinite(bn.y)
   440→              ? Math.round(bn.x as number)
   441→              : Math.round(batch.position.x + (i - (count - 1) / 2) * 180),
   442→          y:
   443→            Number.isFinite(bn.x) && Number.isFinite(bn.y)
   444→              ? Math.round(bn.y as number)
   445→              : Math.round(batch.position.y + 130),
   446→        },
   447→        data: {
   448→          nodeType: type as LogicNodeType,
   449→          name: bn.name || defaultNodeName(type as LogicNodeType),
   450→          config: bn.config,
   451→          resultVar: bn.resultVar || undefined,
   452→          errorAction: bn.errorAction || undefined,
   453→        },
   454→      }))
   455→      nodes.push(...bodyNodes)
   456→      edges.push(createLoopEdge(batch.id, bodyNodes[0].id, LOOP_HANDLE_START, 'in'))
   457→      for (let i = 1; i < bodyNodes.length; i++) {
   458→        edges.push(createLoopEdge(bodyNodes[i - 1].id, bodyNodes[i].id, 'out', 'in'))
   459→      }
   460→      edges.push(
   461→        createLoopEdge(bodyNodes[bodyNodes.length - 1].id, batch.id, 'out', LOOP_HANDLE_END)
   462→      )
   463→    } else {
   464→      edges.push(createLoopEdge(batch.id, batch.id, LOOP_HANDLE_START, LOOP_HANDLE_END))
   465→    }
   466→  }
   467→}

   469→/** 解析 DSL 字符串为 vue-flow 画布数据。非法 JSON / 结构不符时抛错。 */
   470→export function parseDsl(dsl: string): FlowGraph {
   471→  let raw: unknown
   472→  try {
   473→    raw = JSON.parse(dsl)
   474→  } catch (err) {
   475→    throw new Error(`DSL 不是合法的 JSON：${err instanceof Error ? err.message : String(err)}`)
   476→  }
   477→  if (raw == null || typeof raw !== 'object' || Array.isArray(raw)) {
   478→    throw new Error('DSL 结构不符：顶层应为对象（含 nodes/edges）')
   479→  }
   480→  const obj = raw as { nodes?: unknown; edges?: unknown; inputVars?: unknown }
   481→  const rawNodes = Array.isArray(obj.nodes) ? obj.nodes : []
   482→  const rawEdges = Array.isArray(obj.edges) ? obj.edges : []
   483→  const rawInputVars = Array.isArray(obj.inputVars) ? obj.inputVars : []
   484→
   485→  const inputVars: InputVarDef[] = rawInputVars
   486→    .map((item): InputVarDef | null => {
   487→      const v = (item ?? {}) as Record<string, unknown>
   488→      const name = String(v.name ?? '').trim()
   489→      if (!name) return null
   490→      const type = ['string', 'number', 'boolean', 'json'].includes(String(v.type))
   491→        ? (String(v.type) as InputVarDef['type'])
   492→        : 'string'
   493→      return {
   494→        name,
   495→        type,
   496→        required: Boolean(v.required),
   497→        desc: v.desc ? String(v.desc) : undefined,
   498→      }
   499→    })
   500→    .filter((v): v is InputVarDef => v !== null)
   501→
   502→  const nodes: FlowNode[] = rawNodes.map((item) => {
   503→    const n = (item ?? {}) as Record<string, unknown>
   504→    const type = normalizeType(n.type)
   505→    return {
   506→      id: String(n.id ?? '') || createNodeId(type, []),
   507→      type: FLOW_NODE_TYPE,
   508→      position: { x: toNum(n.x, 0), y: toNum(n.y, 0) },
   509→      data: {
   510→        nodeType: type,
   511→        name: String(n.name ?? '') || defaultNodeName(type),
   512→        config: (n.config ?? undefined) as NodeConfig | undefined,
   513→        resultVar: n.resultVar ? String(n.resultVar) : undefined,
   514→        errorAction: n.errorAction ? (String(n.errorAction) as ErrorAction) : undefined,
   515→      },
   516→    }
   517→  })
   518→
   519→  const edges: FlowEdge[] = rawEdges
   520→    .map((item, index) => {
   521→      const e = (item ?? {}) as Record<string, unknown>
   522→      const branch =
   523→        e.branch === 'true' || e.branch === 'false' ? (e.branch as 'true' | 'false') : undefined
   524→      return {
   525→        id: e.id != null ? String(e.id) : `e_${index}_${shortCode()}`,
   526→        source: String(e.source ?? ''),
   527→        target: String(e.target ?? ''),
   528→        // 默认渲染：折线 + 箭头（契约不存 type，回显统一补齐）
   529→        type: 'smoothstep',
   530→        markerEnd: 'arrowclosed',
   531→        // branch → sourceHandle：vue-flow 依 sourceHandle 定位出发点；
   532→        // CONDITION 有 true/false 两个 source handle，缺省时渲染会兜底连到第一个（true），
   533→        // 导致「否」连线重开后错显在「是」上（bug: 条件分支回显丢失）
   534→        sourceHandle: branch,
   535→        data: branch ? { branch } : undefined,
   536→      }
   537→    })
   538→    .filter((e) => e.source && e.target)
   539→
   540→  synthesizeBatchLoops(nodes, edges)
   541→
   542→  return { nodes, edges, inputVars }
   543→}

   547→/** START/END 无 resultVar / errorAction（契约：业务执行节点有；CONDITION 例外仅运行结果布尔） */
   548→function hasExecutionMeta(type: LogicNodeType): boolean {
   549→  return (
   550→    type === 'HTTP' ||
   551→    type === 'BEAN' ||
   552→    type === 'SCRIPT' ||
   553→    type === 'BATCH' ||
   554→    type === 'SUBFLOW' ||
   555→    type === 'DATA_UPDATE'
   556→  )
   557→}
   558→
   559→/** 画布节点 → 契约节点（剥离 vue-flow 内部字段） */
   560→function toDslNode(node: FlowNode): DslNode | null {
   561→  const data = node?.data
   562→  if (!data) return null
   563→  const type = normalizeType(data.nodeType)
   564→  const out: DslNode = {
   565→    id: node.id,
   566→    type,
   567→    name: String(data.name ?? '') || defaultNodeName(type),
   568→    x: Math.round(node.position?.x ?? 0),
   569→    y: Math.round(node.position?.y ?? 0),
   570→  }
   571→  if (data.config && typeof data.config === 'object') {
   572→    out.config = data.config
   573→  }
   574→  if (hasExecutionMeta(type)) {
   575→    if (data.resultVar) out.resultVar = data.resultVar
   576→    out.errorAction = data.errorAction === 'IGNORE_CONTINUE' ? 'IGNORE_CONTINUE' : 'FAIL_FLOW'
   577→  }
   578→  return out
   579→}
   580→
   581→/** 画布边 → 契约边（只留 id/source/target/branch；循环边不进契约） */
   582→function toDslEdge(edge: FlowEdge): DslEdge | null {
   583→  if (!edge?.source || !edge?.target) return null
   584→  if (isLoopEdge(edge)) return null
   585→  const out: DslEdge = { source: edge.source, target: edge.target }
   586→  if (edge.id) out.id = edge.id
   587→  // branch 以 sourceHandle 为准（连线/重连时与 handle 同步），data.branch 作后备
   588→  const fromHandle = edge.sourceHandle
   589→  const branch =
   590→    fromHandle === 'true' || fromHandle === 'false'
   591→      ? fromHandle
   592→      : edge.data?.branch === 'true' || edge.data?.branch === 'false'
   593→        ? edge.data.branch
   594→        : undefined
   595→  if (branch) out.branch = branch
   596→  return out
   597→}
   598→
   599→/** 画布节点 → 循环体节点（剥离 vue-flow 内部字段；非循环体类型返回 null）。
   600→ *  x/y 总是写入，由调用方按「默认排布位」判定后剥离，避免旧 DSL 被误标脏 */
   601→function toDslBodyNode(node: FlowNode): BatchBodyNode | null {
   602→  const data = node?.data
   603→  if (!data) return null
   604→  const type = toBatchBodyType(data.nodeType)
   605→  if (!type) return null
   606→  const out: BatchBodyNode = {
   607→    id: node.id,
   608→    type,
   609→    name: String(data.name ?? '') || defaultNodeName(data.nodeType),
   610→    x: Math.round(node.position?.x ?? 0),
   611→    y: Math.round(node.position?.y ?? 0),
   612→  }
   613→  if (data.config && typeof data.config === 'object') out.config = data.config
   614→  if (data.resultVar) out.resultVar = data.resultVar
   615→  out.errorAction = data.errorAction === 'IGNORE_CONTINUE' ? 'IGNORE_CONTINUE' : 'FAIL_FLOW'
   616→  return out
   617→}
   618→
   619→/**
   620→ * 从 loop 边集合提取某 BATCH 的循环体链：loop_start 出发沿链行走至无后继
   621→ * （或成环/步数耗尽）。body 只收合法循环体类型节点；链上节点 id 由调用方
   622→ * 从顶层画布剔除（孤段节点保留顶层，由发布校验兜底报「节点无出边」）。
   623→ * 链上遇 BATCH（嵌套子批处理）即入 body 并终止本链行走 ——
   624→ * 子批处理的循环体链由 serializeDsl 顶层遍历单独提取，避免被父链吞并。
   625→ */
   626→function extractBatchBody(
   627→  batchId: string,
   628→  loopEdges: FlowEdge[],
   629→  nodeById: Map<string, FlowNode>
   630→): { body: BatchBodyNode[]; chainNodeIds: Set<string> } {
   631→  const body: BatchBodyNode[] = []
   632→  const chainNodeIds = new Set<string>()
   633→  const startEdge = loopEdges.find(
   634→    (e) => e.source === batchId && e.sourceHandle === LOOP_HANDLE_START
   635→  )
   636→  if (!startEdge) return { body, chainNodeIds }
   637→  const visited = new Set<string>([batchId])
   638→  let currentId = startEdge.target
   639→  let guard = 0
   640→  while (currentId && !visited.has(currentId) && guard++ < 100) {
   641→    visited.add(currentId)
   642→    const node = nodeById.get(currentId)
   643→    if (!node) break
   644→    const bodyNode = toDslBodyNode(node)
   645→    if (bodyNode) {
   646→      body.push(bodyNode)
   647→      chainNodeIds.add(currentId)
   648→      // 嵌套子批处理：入 body 后停止沿父链行走（子的循环体链独立提取）
   649→      if (bodyNode.type === 'BATCH') break
   650→    }
   651→    const nextEdge = loopEdges.find((e) => e.source === currentId)
   652→    if (!nextEdge) break
   653→    currentId = nextEdge.target
   654→  }
   655→
   656→  // 位置持久化（仅拖离默认位才写入）：与 synthesizeBatchLoops 的默认公式互逆——
   657→  // 仍在默认位的节点剔除 x/y，旧 DSL / 未移动场景往返稳定（isDslEqual 脏检测不误报）
   658→  const batch = nodeById.get(batchId)
   659→  if (batch && body.length) {
   660→    const count = body.length
   661→    body.forEach((bn, i) => {
   662→      const defX = Math.round(batch.position.x + (i - (count - 1) / 2) * 180)
   663→      const defY = Math.round(batch.position.y + 130)
   664→      if (bn.x === defX && bn.y === defY) {
   665→        delete bn.x
   666→        delete bn.y
   667→      }
   668→    })
   669→  }
   670→
   671→  return { body, chainNodeIds }
   672→}

   674→/** 序列化画布为 DSL JSON 字符串（空画布 → 空节点/空边结构；inputVars 非空时带上） */
   675→export function serializeDsl(
   676→  nodes: FlowNode[],
   677→  edges: FlowEdge[],
   678→  inputVars?: InputVarDef[]
   679→): string {
   680→  const safeNodes = nodes || []
   681→  const safeEdges = edges || []
   682→  const loopEdges = safeEdges.filter(isLoopEdge)
   683→  const nodeById = new Map(safeNodes.map((n) => [n.id, n]))
   684→
   685→  // BATCH 循环体提取：链上节点剔出顶层、编入 config.body（替换的是序列化副本，不改画布）
   686→  const stripNodeIds = new Set<string>()
   687→  const batchBodies = new Map<string, BatchBodyNode[]>()
   688→  for (const node of safeNodes) {
   689→    if (node.data?.nodeType !== 'BATCH') continue
   690→    const { body, chainNodeIds } = extractBatchBody(node.id, loopEdges, nodeById)
   691→    if (body.length) {
   692→      batchBodies.set(node.id, body)
   693→      chainNodeIds.forEach((id) => stripNodeIds.add(id))
   694→    }
   695→  }
   696→
   697→  const dsl: LogicFlowDsl = {
   698→    nodes: safeNodes
   699→      .filter((n) => !stripNodeIds.has(n.id))
   700→      .map(toDslNode)
   701→      .filter((n): n is DslNode => n !== null),
   702→    edges: safeEdges.map(toDslEdge).filter((e): e is DslEdge => e !== null),
   703→  }
   704→
   705→  // 循环体写回对应 BATCH config（浅拷贝替换，legacy actionType 升级后移除）。
   706→  // 嵌套批处理：子 BATCH 被剔出顶层（不在 dsl.nodes），需从父 body 中递归下钻写回，
   707→  // 否则子批处理的循环体链会因「无顶层节点可写」而丢失。
   708→  if (batchBodies.size) {
   709→    const applyBody = (node: DslNode): void => {
   710→      if (node.type !== 'BATCH') return
   711→      const body = batchBodies.get(node.id)
   712→      if (!body) return
   713→      const cfg = { ...((node.config ?? {}) as Record<string, unknown>) }
   714→      cfg.body = body
   715→      delete cfg.actionType
   716→      delete cfg.actionConfig
   717→      node.config = cfg as unknown as NodeConfig
   718→      for (const step of body) {
   719→        if (step.type === 'BATCH') applyBody(step as unknown as DslNode)
   720→      }
   721→    }
   722→    for (const node of dsl.nodes) applyBody(node)
   723→  }
   724→
   725→  const declared = (inputVars || []).filter((v) => v.name && v.name.trim())
   726→  if (declared.length) dsl.inputVars = declared
   727→  return JSON.stringify(dsl)
   728→}

   730→/** 规范化 JSON 序列化：对象键按字典序排列（数组保持原序），用于键序无关的深比较 */
   731→function stableStringify(value: unknown): string {
   732→  if (value === null || typeof value !== 'object') {
   733→    return JSON.stringify(value) ?? 'undefined'
   734→  }
   735→  if (Array.isArray(value)) {
   736→    return `[${value.map((item) => stableStringify(item)).join(',')}]`
   737→  }
   738→  const entries = Object.entries(value as Record<string, unknown>)
   739→    .filter(([, v]) => v !== undefined)
   740→    .sort(([a], [b]) => (a < b ? -1 : a > b ? 1 : 0))
   741→  return `{${entries.map(([k, v]) => `${JSON.stringify(k)}:${stableStringify(v)}`).join(',')}}`
   742→}
   743→
   744→/** 深比较两份 DSL 字符串是否等价（对象键序无关），用于脏检测 */
   745→export function isDslEqual(a: string, b: string): boolean {
   746→  if (a === b) return true
   747→  try {
   748→    return stableStringify(JSON.parse(a)) === stableStringify(JSON.parse(b))
   749→  } catch {
   750→    return false
   751→  }
   752→}

   754→// ==================== 输入变量扫描（运行测试建议用） ====================
   755→
   756→/**
   757→ * 扫描画布图，提取「被引用但未在本流产出」的输入变量候选名。
   758→ * - 深度遍历所有节点 config 字符串值中的 {{var}} 占位符；
   759→ * - 结构化引用：BEAN params[].source、HTTP queryParams/bodyParams[].source、
   760→ *   SUBFLOW varsMapping[].source、CONDITION variable、BATCH collection；
   761→ * - 剔除本流产出：各节点 resultVar + BATCH itemVar/indexVar。
   762→ */
   763→export function collectReferencedVars(graph: FlowGraph): string[] {
   764→  const referenced = new Set<string>()
   765→  const produced = new Set<string>()
   766→
   767→  const scanPlaceholders = (value: unknown): void => {
   768→    if (typeof value === 'string') {
   769→      for (const m of value.matchAll(/\{\{\s*([a-zA-Z_][\w]*)\s*\}\}/g)) {
   770→        referenced.add(m[1])
   771→      }
   772→    } else if (Array.isArray(value)) {
   773→      value.forEach(scanPlaceholders)
   774→    } else if (value && typeof value === 'object') {
   775→      Object.values(value as Record<string, unknown>).forEach(scanPlaceholders)
   776→    }
   777→  }
   778→
   779→  for (const node of graph.nodes) {
   780→    const cfg = node.data.config as Record<string, unknown> | undefined
   781→    if (cfg) scanPlaceholders(cfg)
   782→    const type = node.data.nodeType
   783→    if (node.data.resultVar) produced.add(node.data.resultVar)
   784→    if (type === 'BEAN' && Array.isArray(cfg?.params)) {
   785→      for (const p of cfg.params as { source?: string }[]) {
   786→        if (p?.source) referenced.add(p.source)
   787→      }
   788→    }
   789→    if (type === 'HTTP') {
   790→      for (const key of ['queryParams', 'bodyParams'] as const) {
   791→        if (Array.isArray(cfg?.[key])) {
   792→          for (const p of cfg[key] as { source?: string }[]) {
   793→            if (p?.source) referenced.add(p.source)
   794→          }
   795→        }
   796→      }
   797→    }
   798→    if (type === 'CONDITION' && typeof cfg?.variable === 'string' && cfg.variable) {
   799→      referenced.add(cfg.variable.replace(/^\{\{\s*|\s*\}\}$/g, ''))
   800→    }
   801→    if (type === 'BATCH') {
   802→      const collection = String(cfg?.collection ?? '')
   803→      const m = collection.match(/^\{\{\s*([a-zA-Z_][\w]*)\s*\}\}$/)
   804→      if (m) referenced.add(m[1])
   805→      const itemVar = String(cfg?.itemVar ?? 'item')
   806→      const indexVar = String(cfg?.indexVar ?? 'index')
   807→      if (itemVar) produced.add(itemVar)
   808→      if (indexVar) produced.add(indexVar)
   809→      // 循环体链：body 节点 resultVar 计入产出，结构化 source 引用计入引用
   810→      const body = Array.isArray(cfg?.body) ? (cfg?.body as BatchBodyNode[]) : []
   811→      for (const bn of body) {
   812→        if (!bn) continue
   813→        if (bn.resultVar) produced.add(bn.resultVar)
   814→        const bc = bn.config as Record<string, unknown> | undefined
   815→        if (!bc) continue
   816→        if (bn.type === 'BEAN' && Array.isArray(bc.params)) {
   817→          for (const p of bc.params as { source?: string }[]) {
   818→            if (p?.source) referenced.add(p.source)
   819→          }
   820→        }
   821→        if (bn.type === 'HTTP') {
   822→          for (const key of ['queryParams', 'bodyParams'] as const) {
   823→            if (Array.isArray(bc[key])) {
   824→              for (const p of bc[key] as { source?: string }[]) {
   825→                if (p?.source) referenced.add(p.source)
   826→              }
   827→            }
   828→          }
   829→        }
   830→        if (bn.type === 'SUBFLOW' && Array.isArray(bc.varsMapping)) {
   831→          for (const p of bc.varsMapping as { source?: string }[]) {
   832→            if (p?.source) referenced.add(p.source)
   833→          }
   834→        }
   835→      }
   836→    }
   837→    if (type === 'SUBFLOW' && Array.isArray(cfg?.varsMapping)) {
   838→      for (const p of cfg.varsMapping as { source?: string }[]) {
   839→        if (p?.source) referenced.add(p.source)
   840→      }
   841→    }
   842→  }
   843→
   844→  return [...referenced].filter((name) => !produced.has(name)).sort()
   845→}
