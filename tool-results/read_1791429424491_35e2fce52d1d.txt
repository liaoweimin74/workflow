     1→     1→/**
     2→     2→ * 逻辑流 DSL ↔ vue-flow 画布数据 转换工具。
     3→     3→ *
     4→     4→ * 契约（worklog Task 2 + SUBFLOW/inputVars 增量 + DATA_UPDATE）：
     5→     5→ * - nodes[]: { id, type: START|END|HTTP|BEAN|SCRIPT|CONDITION|BATCH|SUBFLOW|DATA_UPDATE, name, x, y,
     6→     6→ *              config{...}, resultVar?, errorAction?: FAIL_FLOW|IGNORE_CONTINUE }
     7→     7→ *   - HTTP   config = { url, method, headers{...}, queryParams[{source,target}],
     8→     8→ *                       bodyParams[{source,target}], connTimeoutMs, readTimeoutMs, retryCount }
     9→     9→ *   - BEAN   config = { beanName, methodName, params[{source,target}] }
    10→    10→ *   - SCRIPT config = { language: 'groovy', source }
    11→    11→ *   - CONDITION config = { variable, operator: EQ|NE|GT|LT|GTE|LTE|EMPTY|NOT_EMPTY, value? }
    12→    12→ *   - BATCH  config = { collection, itemVar?, indexVar?, body?: BatchBodyNode[](循环体链，
    13→    13→ *                       按序逐项执行)，stopOnError?, maxItems? }
    14→    14→ *                       （legacy 兼容：actionType: HTTP|SCRIPT|BEAN + actionConfig 单动作，
    15→    15→ *                         读取时自动合成为单节点循环体，保存后升级为 body 形态）
    16→    16→ *   - SUBFLOW config = { flowId, passAllVars=true, varsMapping[{source,target}] }
    17→    17→ *   - DATA_UPDATE config = { table, setOps[{column, mode: SET|ADD|SUB, value}],
    18→    18→ *                            where[{column, op: EQ|NE|GT|GTE|LT|LTE|IS_NULL|NOT_NULL, value?}] }
    19→    19→ *       值支持字面量或 {{var}}/{{formData.xxx}} 点路径；受影响行数写回 resultVar
    20→    20→ * - edges[]: { id?, source, target, branch?: 'true'|'false'（仅 CONDITION 出边用） }
    21→    21→ *   - BATCH 循环体边（画布 data.loop=true）为设计器内部结构，序列化时剔除；
    22→    22→ *     循环体内容编入 BATCH.config.body[]
    23→    23→ * - 顶层 inputVars[]: 入参声明（可选）：[{ name, type: string|number|boolean|json, required?, desc? }]
    24→    24→ *
    25→    25→ * vue-flow 侧：
    26→    26→ * - node.type 固定为 'logic'（自定义节点渲染槽 #node-logic），
    27→    27→ *   节点业务类型存 node.data.nodeType
    28→    28→ * - position 需要 {x,y}；契约的顶层 x/y 与之对应
    29→    29→ * - branch 存到 edge.data.branch
    30→    30→ * - 序列化时剥离 vue-flow 内部字段（selected/handles/dimensions...），只留契约字段
    31→    31→ */

   364→   364→  while (taken.has(id)) {
   365→   365→    id = `${prefix}_${shortCode()}${Math.floor(Math.random() * 10)}`
   366→   366→  }
   367→   367→  return id

   388→   388→
   389→   389→/**
   390→   390→ * 为每个 BATCH 节点合成循环体画布结构与闭合循环连线：
   391→   391→ * - config.body 非空 → 循环体节点横排在批处理下方，沿链 loop_start→b0→…→loop_end；
   392→   392→ * - legacy（无 body 有 actionType/actionConfig）→ 合成单个循环体节点；
   393→   393→ * - 两者皆无 → loop_start 直连 loop_end（空循环虚线）。
   394→   394→ * 循环边均为 deletable:false + data.loop=true（设计器内部结构，serialize 时剔除）。
   395→   395→ */
   396→   396→function synthesizeBatchLoops(nodes: FlowNode[], edges: FlowEdge[]): void {
   397→   397→  for (const batch of nodes) {
   398→   398→    if (batch.data.nodeType !== 'BATCH') continue
   399→   399→    const cfg = (batch.data.config ?? {}) as Partial<BatchNodeConfig>
   400→   400→    const rawBody = Array.isArray(cfg.body) ? cfg.body : []
   401→   401→
   402→   402→    // legacy 单动作 → 合成为单节点循环体
   403→   403→    let bodyItems = rawBody
   404→   404→    if (!bodyItems.length && cfg.actionType) {
   405→   405→      const legacyType = toBatchBodyType(cfg.actionType)
   406→   406→      if (legacyType) {
   407→   407→        bodyItems = [
   408→   408→          {
   409→   409→            type: legacyType,
   410→   410→            name: defaultNodeName(legacyType as LogicNodeType),
   411→   411→            config: cfg.actionConfig as NodeConfig | undefined,
   412→   412→          },
   413→   413→        ]
   414→   414→      }
   415→   415→    }
   416→   416→
   417→   417→    const valid = bodyItems
   418→   418→      .map((bn) => {
   419→   419→        const type = toBatchBodyType(bn?.type)
   420→   420→        return type ? { bn, type } : null
   421→   421→      })
   422→   422→      .filter((x): x is { bn: BatchBodyNode; type: BatchBodyType } => x !== null)
   423→   423→
   424→   424→    if (valid.length) {
   425→   425→      const count = valid.length
   426→   426→      const bodyNodes: FlowNode[] = valid.map(({ bn, type }, i) => ({
   427→   427→        id: bn.id || `${batch.id}__b${i}`,
   428→   428→        type: FLOW_NODE_TYPE,
   429→   429→        position: {
   430→   430→          x: Math.round(batch.position.x + (i - (count - 1) / 2) * 180),
   431→   431→          y: Math.round(batch.position.y + 130),
   432→   432→        },
   433→   433→        data: {
   434→   434→          nodeType: type as LogicNodeType,
   435→   435→          name: bn.name || defaultNodeName(type as LogicNodeType),
   436→   436→          config: bn.config,
   437→   437→          resultVar: bn.resultVar || undefined,
   438→   438→          errorAction: bn.errorAction || undefined,
   439→   439→        },
   440→   440→      }))
   441→   441→      nodes.push(...bodyNodes)
   442→   442→      edges.push(createLoopEdge(batch.id, bodyNodes[0].id, LOOP_HANDLE_START, 'in'))
   443→   443→      for (let i = 1; i < bodyNodes.length; i++) {
   444→   444→        edges.push(createLoopEdge(bodyNodes[i - 1].id, bodyNodes[i].id, 'out', 'in'))
   445→   445→      }
   446→   446→      edges.push(
   447→   447→        createLoopEdge(bodyNodes[bodyNodes.length - 1].id, batch.id, 'out', LOOP_HANDLE_END)
   448→   448→      )
   449→   449→    } else {
   450→   450→      edges.push(createLoopEdge(batch.id, batch.id, LOOP_HANDLE_START, LOOP_HANDLE_END))
   451→   451→    }
   452→   452→  }
   453→   453→}
   454→   454→
   455→   455→/** 解析 DSL 字符串为 vue-flow 画布数据。非法 JSON / 结构不符时抛错。 */
   456→   456→export function parseDsl(dsl: string): FlowGraph {
   457→   457→  let raw: unknown
   458→   458→  try {
   459→   459→    raw = JSON.parse(dsl)
   460→   460→  } catch (err) {
   461→   461→    throw new Error(`DSL 不是合法的 JSON：${err instanceof Error ? err.message : String(err)}`)
   462→   462→  }
   463→   463→  if (raw == null || typeof raw !== 'object' || Array.isArray(raw)) {
   464→   464→    throw new Error('DSL 结构不符：顶层应为对象（含 nodes/edges）')
   465→   465→  }
   466→   466→  const obj = raw as { nodes?: unknown; edges?: unknown; inputVars?: unknown }
   467→   467→  const rawNodes = Array.isArray(obj.nodes) ? obj.nodes : []
   468→   468→  const rawEdges = Array.isArray(obj.edges) ? obj.edges : []
   469→   469→  const rawInputVars = Array.isArray(obj.inputVars) ? obj.inputVars : []
   470→   470→
   471→   471→  const inputVars: InputVarDef[] = rawInputVars
   472→   472→    .map((item): InputVarDef | null => {
   473→   473→      const v = (item ?? {}) as Record<string, unknown>
   474→   474→      const name = String(v.name ?? '').trim()
   475→   475→      if (!name) return null
   476→   476→      const type = ['string', 'number', 'boolean', 'json'].includes(String(v.type))
   477→   477→        ? (String(v.type) as InputVarDef['type'])
   478→   478→        : 'string'
   479→   479→      return {
   480→   480→        name,
   481→   481→        type,
   482→   482→        required: Boolean(v.required),
   483→   483→        desc: v.desc ? String(v.desc) : undefined,
   484→   484→      }
   485→   485→    })
   486→   486→    .filter((v): v is InputVarDef => v !== null)
   487→   487→
   488→   488→  const nodes: FlowNode[] = rawNodes.map((item) => {
   489→   489→    const n = (item ?? {}) as Record<string, unknown>
   490→   490→    const type = normalizeType(n.type)
   491→   491→    return {
   492→   492→      id: String(n.id ?? '') || createNodeId(type, []),
   493→   493→      type: FLOW_NODE_TYPE,
   494→   494→      position: { x: toNum(n.x, 0), y: toNum(n.y, 0) },
   495→   495→      data: {
   496→   496→        nodeType: type,
   497→   497→        name: String(n.name ?? '') || defaultNodeName(type),
   498→   498→        config: (n.config ?? undefined) as NodeConfig | undefined,
   499→   499→        resultVar: n.resultVar ? String(n.resultVar) : undefined,
   500→   500→        errorAction: n.errorAction ? (String(n.errorAction) as ErrorAction) : undefined,
   501→   501→      },
   502→   502→    }
   503→   503→  })
   504→   504→
   505→   505→  const edges: FlowEdge[] = rawEdges
   506→   506→    .map((item, index) => {
   507→   507→      const e = (item ?? {}) as Record<string, unknown>
   508→   508→      const branch =
   509→   509→        e.branch === 'true' || e.branch === 'false' ? (e.branch as 'true' | 'false') : undefined
   510→   510→      return {
   511→   511→        id: e.id != null ? String(e.id) : `e_${index}_${shortCode()}`,
   512→   512→        source: String(e.source ?? ''),
   513→   513→        target: String(e.target ?? ''),
   514→   514→        // 默认渲染：折线 + 箭头（契约不存 type，回显统一补齐）
   515→   515→        type: 'smoothstep',
   516→   516→        markerEnd: 'arrowclosed',
   517→   517→        // branch → sourceHandle：vue-flow 依 sourceHandle 定位出发点；
   518→   518→        // CONDITION 有 true/false 两个 source handle，缺省时渲染会兜底连到第一个（true），
   519→   519→        // 导致「否」连线重开后错显在「是」上（bug: 条件分支回显丢失）
   520→   520→        sourceHandle: branch,
   521→   521→        data: branch ? { branch } : undefined,
   522→   522→      }
   523→   523→    })
   524→   524→    .filter((e) => e.source && e.target)
   525→   525→
   526→   526→  synthesizeBatchLoops(nodes, edges)
   527→   527→
   528→   528→  return { nodes, edges, inputVars }
   529→   529→}
   530→   530→
   531→   531→// ==================== 序列化：画布 → DSL 字符串 ====================
   532→   532→
   533→   533→/** START/END 无 resultVar / errorAction（契约：业务执行节点有；CONDITION 例外仅运行结果布尔） */
   534→   534→function hasExecutionMeta(type: LogicNodeType): boolean {
   535→   535→  return (
   536→   536→    type === 'HTTP' ||
   537→   537→    type === 'BEAN' ||
   538→   538→    type === 'SCRIPT' ||
   539→   539→    type === 'BATCH' ||
   540→   540→    type === 'SUBFLOW' ||
   541→   541→    type === 'DATA_UPDATE'
   542→   542→  )
   543→   543→}
   544→   544→
   545→   545→/** 画布节点 → 契约节点（剥离 vue-flow 内部字段） */
   546→   546→function toDslNode(node: FlowNode): DslNode | null {
   547→   547→  const data = node?.data
   548→   548→  if (!data) return null
   549→   549→  const type = normalizeType(data.nodeType)
   550→   550→  const out: DslNode = {
   551→   551→    id: node.id,
   552→   552→    type,
   553→   553→    name: String(data.name ?? '') || defaultNodeName(type),
   554→   554→    x: Math.round(node.position?.x ?? 0),
   555→   555→    y: Math.round(node.position?.y ?? 0),
   556→   556→  }
   557→   557→  if (data.config && typeof data.config === 'object') {
   558→   558→    out.config = data.config
   559→   559→  }
   560→   560→  if (hasExecutionMeta(type)) {
   561→   561→    if (data.resultVar) out.resultVar = data.resultVar
   562→   562→    out.errorAction = data.errorAction === 'IGNORE_CONTINUE' ? 'IGNORE_CONTINUE' : 'FAIL_FLOW'
   563→   563→  }
   564→   564→  return out
   565→   565→}
   566→   566→
   567→   567→/** 画布边 → 契约边（只留 id/source/target/branch；循环边不进契约） */
   568→   568→function toDslEdge(edge: FlowEdge): DslEdge | null {
   569→   569→  if (!edge?.source || !edge?.target) return null
   570→   570→  if (isLoopEdge(edge)) return null
   571→   571→  const out: DslEdge = { source: edge.source, target: edge.target }
   572→   572→  if (edge.id) out.id = edge.id
   573→   573→  // branch 以 sourceHandle 为准（连线/重连时与 handle 同步），data.branch 作后备
   574→   574→  const fromHandle = edge.sourceHandle
   575→   575→  const branch =
   576→   576→    fromHandle === 'true' || fromHandle === 'false'
   577→   577→      ? fromHandle
   578→   578→      : edge.data?.branch === 'true' || edge.data?.branch === 'false'
   579→   579→        ? edge.data.branch
   580→   580→        : undefined
   581→   581→  if (branch) out.branch = branch
   582→   582→  return out
   583→   583→}
   584→   584→
   585→   585→/** 画布节点 → 循环体节点（剥离 vue-flow 内部字段；非循环体类型返回 null） */
   586→   586→function toDslBodyNode(node: FlowNode): BatchBodyNode | null {
   587→   587→  const data = node?.data
   588→   588→  if (!data) return null
   589→   589→  const type = toBatchBodyType(data.nodeType)
   590→   590→  if (!type) return null
   591→   591→  const out: BatchBodyNode = {
   592→   592→    id: node.id,
   593→   593→    type,
   594→   594→    name: String(data.name ?? '') || defaultNodeName(data.nodeType),
   595→   595→  }
   596→   596→  if (data.config && typeof data.config === 'object') out.config = data.config
   597→   597→  if (data.resultVar) out.resultVar = data.resultVar
   598→   598→  out.errorAction = data.errorAction === 'IGNORE_CONTINUE' ? 'IGNORE_CONTINUE' : 'FAIL_FLOW'
   599→   599→  return out
   600→   600→}
   601→   601→
   602→   602→/**
   603→   603→ * 从 loop 边集合提取某 BATCH 的循环体链：loop_start 出发沿链行走至无后继
   604→   604→ * （或成环/步数耗尽）。body 只收合法循环体类型节点；链上节点 id 由调用方
   605→   605→ * 从顶层画布剔除（孤段节点保留顶层，由发布校验兜底报「节点无出边」）。
   606→   606→ * 链上遇 BATCH（嵌套子批处理）即入 body 并终止本链行走 ——
   607→   607→ * 子批处理的循环体链由 serializeDsl 顶层遍历单独提取，避免被父链吞并。
   608→   608→ */
   609→   609→function extractBatchBody(
   610→   610→  batchId: string,
   611→   611→  loopEdges: FlowEdge[],
   612→   612→  nodeById: Map<string, FlowNode>
   613→   613→): { body: BatchBodyNode[]; chainNodeIds: Set<string> } {
   614→   614→  const body: BatchBodyNode[] = []
   615→   615→  const chainNodeIds = new Set<string>()
   616→   616→  const startEdge = loopEdges.find(
   617→   617→    (e) => e.source === batchId && e.sourceHandle === LOOP_HANDLE_START
   618→   618→  )
   619→   619→  if (!startEdge) return { body, chainNodeIds }
   620→   620→  const visited = new Set<string>([batchId])
   621→   621→  let currentId = startEdge.target
   622→   622→  let guard = 0
   623→   623→  while (currentId && !visited.has(currentId) && guard++ < 100) {
   624→   624→    visited.add(currentId)
   625→   625→    const node = nodeById.get(currentId)
   626→   626→    if (!node) break
   627→   627→    const bodyNode = toDslBodyNode(node)
   628→   628→    if (bodyNode) {
   629→   629→      body.push(bodyNode)
   630→   630→      chainNodeIds.add(currentId)
   631→   631→      // 嵌套子批处理：入 body 后停止沿父链行走（子的循环体链独立提取）
   632→   632→      if (bodyNode.type === 'BATCH') break
   633→   633→    }
   634→   634→    const nextEdge = loopEdges.find((e) => e.source === currentId)
   635→   635→    if (!nextEdge) break
   636→   636→    currentId = nextEdge.target
   637→   637→  }
   638→   638→  return { body, chainNodeIds }
   639→   639→}
   640→   640→
   641→   641→/** 序列化画布为 DSL JSON 字符串（空画布 → 空节点/空边结构；inputVars 非空时带上） */
   642→   642→export function serializeDsl(
   643→   643→  nodes: FlowNode[],
   644→   644→  edges: FlowEdge[],
   645→   645→  inputVars?: InputVarDef[]
   646→   646→): string {
   647→   647→  const safeNodes = nodes || []
   648→   648→  const safeEdges = edges || []
   649→   649→  const loopEdges = safeEdges.filter(isLoopEdge)
   650→   650→  const nodeById = new Map(safeNodes.map((n) => [n.id, n]))
   651→   651→
   652→   652→  // BATCH 循环体提取：链上节点剔出顶层、编入 config.body（替换的是序列化副本，不改画布）
   653→   653→  const stripNodeIds = new Set<string>()
   654→   654→  const batchBodies = new Map<string, BatchBodyNode[]>()
   655→   655→  for (const node of safeNodes) {
   656→   656→    if (node.data?.nodeType !== 'BATCH') continue
   657→   657→    const { body, chainNodeIds } = extractBatchBody(node.id, loopEdges, nodeById)
   658→   658→    if (body.length) {
   659→   659→      batchBodies.set(node.id, body)
   660→   660→      chainNodeIds.forEach((id) => stripNodeIds.add(id))
   661→   661→    }
   662→   662→  }
   663→   663→
   664→   664→  const dsl: LogicFlowDsl = {
   665→   665→    nodes: safeNodes
   666→   666→      .filter((n) => !stripNodeIds.has(n.id))
   667→   667→      .map(toDslNode)
   668→   668→      .filter((n): n is DslNode => n !== null),
   669→   669→    edges: safeEdges.map(toDslEdge).filter((e): e is DslEdge => e !== null),
   670→   670→  }
   671→   671→
   672→   672→  // 循环体写回对应 BATCH config（浅拷贝替换，legacy actionType 升级后移除）。
   673→   673→  // 嵌套批处理：子 BATCH 被剔出顶层（不在 dsl.nodes），需从父 body 中递归下钻写回，
   674→   674→  // 否则子批处理的循环体链会因「无顶层节点可写」而丢失。
   675→   675→  if (batchBodies.size) {
   676→   676→    const applyBody = (node: DslNode): void => {
   677→   677→      if (node.type !== 'BATCH') return
   678→   678→      const body = batchBodies.get(node.id)
   679→   679→      if (!body) return
   680→   680→      const cfg = { ...((node.config ?? {}) as Record<string, unknown>) }
   681→   681→      cfg.body = body
   682→   682→      delete cfg.actionType
   683→   683→      delete cfg.actionConfig
   684→   684→      node.config = cfg as unknown as NodeConfig
   685→   685→      for (const step of body) {
   686→   686→        if (step.type === 'BATCH') applyBody(step as unknown as DslNode)
   687→   687→      }
   688→   688→    }
   689→   689→    for (const node of dsl.nodes) applyBody(node)
   690→   690→  }
   691→   691→
   692→   692→  const declared = (inputVars || []).filter((v) => v.name && v.name.trim())
   693→   693→  if (declared.length) dsl.inputVars = declared
   694→   694→  return JSON.stringify(dsl)
   695→   695→}
   696→   696→
   697→   697→/** 规范化 JSON 序列化：对象键按字典序排列（数组保持原序），用于键序无关的深比较 */
   698→   698→function stableStringify(value: unknown): string {
   699→   699→  if (value === null || typeof value !== 'object') {
   700→   700→    return JSON.stringify(value) ?? 'undefined'
   701→   701→  }
   702→   702→  if (Array.isArray(value)) {
   703→   703→    return `[${value.map((item) => stableStringify(item)).join(',')}]`
   704→   704→  }
   705→   705→  const entries = Object.entries(value as Record<string, unknown>)
   706→   706→    .filter(([, v]) => v !== undefined)
   707→   707→    .sort(([a], [b]) => (a < b ? -1 : a > b ? 1 : 0))
   708→   708→  return `{${entries.map(([k, v]) => `${JSON.stringify(k)}:${stableStringify(v)}`).join(',')}}`
   709→   709→}
   710→   710→
   711→   711→/** 深比较两份 DSL 字符串是否等价（对象键序无关），用于脏检测 */
   712→   712→export function isDslEqual(a: string, b: string): boolean {
   713→   713→  if (a === b) return true
   714→   714→  try {
   715→   715→    return stableStringify(JSON.parse(a)) === stableStringify(JSON.parse(b))
   716→   716→  } catch {
   717→   717→    return false
   718→   718→  }
   719→   719→}
   720→   720→
   721→   721→// ==================== 输入变量扫描（运行测试建议用） ====================
   722→   722→
   723→   723→/**
   724→   724→ * 扫描画布图，提取「被引用但未在本流产出」的输入变量候选名。
   725→   725→ * - 深度遍历所有节点 config 字符串值中的 {{var}} 占位符；
   726→   726→ * - 结构化引用：BEAN params[].source、HTTP queryParams/bodyParams[].source、
   727→   727→ *   SUBFLOW varsMapping[].source、CONDITION variable、BATCH collection；
   728→   728→ * - 剔除本流产出：各节点 resultVar + BATCH itemVar/indexVar。
   729→   729→ */
   730→   730→export function collectReferencedVars(graph: FlowGraph): string[] {
   731→   731→  const referenced = new Set<string>()
   732→   732→  const produced = new Set<string>()
   733→   733→
   734→   734→  const scanPlaceholders = (value: unknown): void => {
   735→   735→    if (typeof value === 'string') {
   736→   736→      for (const m of value.matchAll(/\{\{\s*([a-zA-Z_][\w]*)\s*\}\}/g)) {
   737→   737→        referenced.add(m[1])
   738→   738→      }
   739→   739→    } else if (Array.isArray(value)) {
   740→   740→      value.forEach(scanPlaceholders)
   741→   741→    } else if (value && typeof value === 'object') {
   742→   742→      Object.values(value as Record<string, unknown>).forEach(scanPlaceholders)
   743→   743→    }
   744→   744→  }
   745→   745→
   746→   746→  for (const node of graph.nodes) {
   747→   747→    const cfg = node.data.config as Record<string, unknown> | undefined
   748→   748→    if (cfg) scanPlaceholders(cfg)
   749→   749→    const type = node.data.nodeType
   750→   750→    if (node.data.resultVar) produced.add(node.data.resultVar)
   751→   751→    if (type === 'BEAN' && Array.isArray(cfg?.params)) {
   752→   752→      for (const p of cfg.params as { source?: string }[]) {
   753→   753→        if (p?.source) referenced.add(p.source)
   754→   754→      }
   755→   755→    }
   756→   756→    if (type === 'HTTP') {
   757→   757→      for (const key of ['queryParams', 'bodyParams'] as const) {
   758→   758→        if (Array.isArray(cfg?.[key])) {
   759→   759→          for (const p of cfg[key] as { source?: string }[]) {
   760→   760→            if (p?.source) referenced.add(p.source)
   761→   761→          }
   762→   762→        }
   763→   763→      }
   764→   764→    }
   765→   765→    if (type === 'CONDITION' && typeof cfg?.variable === 'string' && cfg.variable) {
   766→   766→      referenced.add(cfg.variable.replace(/^\{\{\s*|\s*\}\}$/g, ''))
   767→   767→    }
   768→   768→    if (type === 'BATCH') {
   769→   769→      const collection = String(cfg?.collection ?? '')
   770→   770→      const m = collection.match(/^\{\{\s*([a-zA-Z_][\w]*)\s*\}\}$/)
   771→   771→      if (m) referenced.add(m[1])
   772→   772→      const itemVar = String(cfg?.itemVar ?? 'item')
   773→   773→      const indexVar = String(cfg?.indexVar ?? 'index')
   774→   774→      if (itemVar) produced.add(itemVar)
   775→   775→      if (indexVar) produced.add(indexVar)
   776→   776→      // 循环体链：body 节点 resultVar 计入产出，结构化 source 引用计入引用
   777→   777→      const body = Array.isArray(cfg?.body) ? (cfg?.body as BatchBodyNode[]) : []
   778→   778→      for (const bn of body) {
   779→   779→        if (!bn) continue
   780→   780→        if (bn.resultVar) produced.add(bn.resultVar)
   781→   781→        const bc = bn.config as Record<string, unknown> | undefined
   782→   782→        if (!bc) continue
   783→   783→        if (bn.type === 'BEAN' && Array.isArray(bc.params)) {
   784→   784→          for (const p of bc.params as { source?: string }[]) {
   785→   785→            if (p?.source) referenced.add(p.source)
   786→   786→          }
   787→   787→        }
   788→   788→        if (bn.type === 'HTTP') {
   789→   789→          for (const key of ['queryParams', 'bodyParams'] as const) {
   790→   790→            if (Array.isArray(bc[key])) {
   791→   791→              for (const p of bc[key] as { source?: string }[]) {
   792→   792→                if (p?.source) referenced.add(p.source)
   793→   793→              }
   794→   794→            }
   795→   795→          }
   796→   796→        }
   797→   797→        if (bn.type === 'SUBFLOW' && Array.isArray(bc.varsMapping)) {
   798→   798→          for (const p of bc.varsMapping as { source?: string }[]) {
   799→   799→            if (p?.source) referenced.add(p.source)
   800→   800→          }
   801→   801→        }
   802→   802→      }
   803→   803→    }
   804→   804→    if (type === 'SUBFLOW' && Array.isArray(cfg?.varsMapping)) {
   805→   805→      for (const p of cfg.varsMapping as { source?: string }[]) {
   806→   806→        if (p?.source) referenced.add(p.source)
   807→   807→      }
   808→   808→    }
   809→   809→  }
   810→   810→
   811→   811→  return [...referenced].filter((name) => !produced.has(name)).sort()
   812→   812→}
   813→   813→
