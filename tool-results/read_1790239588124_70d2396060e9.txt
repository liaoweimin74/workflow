    93→    @Override
    94→    public DataSourceMetadata metadata(DataSourceDefinition ds) {
    95→        return switch (ds.getType()) {
    96→            case "FORM" -> {
    97→                List<ColumnConfig> cols;
    98→                try {
    99→                    cols = new ArrayList<>(formDefService.getBusinessColumnsByKey(ds.getFormKey()));
   100→                } catch (BusinessException e) {
   101→                    if (e.getCode() == 404) {
   102→                        // 绑定表单尚未发布（或不存在）：字段元数据降级为空列而非 404，
   103→                        // 不阻断数据源编辑（对齐 NestJS 侧同款降级；表单发布后自动展示）
   104→                        DataSourceMetadata empty = new DataSourceMetadata(new ArrayList<>(), false);
   105→                        empty.setFormKey(ds.getFormKey());
   106→                        yield empty;
   107→                    }
   108→                    throw e;
   109→                }
   110→                FormQueryConfig cfg = FormQueryConfig.parse(ds.getParams(), objectMapper);
   111→                if (cfg.isConfigMode()) {
   112→                    appendJoinColumns(cols, cfg.joins());
   113→                } else if (cfg.isSqlMode()) {
   114→                    appendDeclaredColumns(cols, cfg.columns());
   115→                }
   116→                SortableResolver.resolve(cols);
   117→                DataSourceMetadata m = new DataSourceMetadata(cols, true);
   118→                m.setFormKey(ds.getFormKey());
   119→                yield m;
   120→            }
   121→            case "WORKFLOW" -> {
   122→                List<ColumnConfig> cols;
   123→                try {
   124→                    cols = workflowQueryService.columnsFor(ds.getFormKey());
   125→                } catch (BusinessException e) {
   126→                    if (e.getCode() == 404) {
   127→                        // 绑定表单尚未发布（或不存在）：字段元数据降级为空列（对齐 FORM 分支与 NestJS 侧）
   128→                        DataSourceMetadata empty = new DataSourceMetadata(new ArrayList<>(), false);
   129→                        empty.setFormKey(ds.getFormKey());
   130→                        yield empty;
   131→                    }
   132→                    throw e;
   133→                }
   134→                SortableResolver.resolve(cols);
   135→                DataSourceMetadata m = new DataSourceMetadata(cols, false);
   136→                m.setFormKey(ds.getFormKey());
   137→                yield m;
   138→            }
   139→            case "SYSTEM" -> {
   140→                List<ColumnConfig> cols = "user-tree".equals(ds.getSourceKey())
   141→                        ? copyWithSortableFalse(USER_COLUMNS)
   142→                        : copyWithSortableFalse(DEPT_COLUMNS);
   143→                yield new DataSourceMetadata(cols, false);
   144→            }
   145→            case "API" -> {
   146→                DataSourceMetadata m = apiMetadata(ds);
   147→                DataSourceMetadata copy = new DataSourceMetadata(copyWithSortableFalse(m.getColumns()), m.isWritable());
   148→                copy.setFormKey(m.getFormKey());
   149→                yield copy;
   150→            }
   151→            case "SQL" -> {
   152→                // 字段元数据 = params.columns（单一来源；探测/主表单覆盖/行内编辑由前端写入，无表单列默认合并）
   153→                FormQueryConfig cfg = FormQueryConfig.parse(ds.getParams(), objectMapper);
   154→                List<ColumnConfig> cols = cfg.isConfigMode() ? List.of() : cfg.columns();
   155→                boolean writable = ds.getFormKey() != null && !ds.getFormKey().isBlank();
   156→                DataSourceMetadata m = new DataSourceMetadata(cols, writable);
   157→                if (ds.getFormKey() != null && !ds.getFormKey().isBlank()) {
   158→                    m.setFormKey(ds.getFormKey());
   159→                }
   160→                yield m;
   161→            }
   162→            default -> throw unsupported(ds, "metadata");
   163→        };
   164→    }

   166→    /** 复制列并强制标记 sortable=false（SYSTEM/API 数据源不可排序；避免就地污染共享常量列）。 */
   167→    private List<ColumnConfig> copyWithSortableFalse(List<ColumnConfig> src) {
   168→        List<ColumnConfig> out = new ArrayList<>();
   169→        if (src != null) {
   170→            for (ColumnConfig c : src) {
   171→                ColumnConfig copy = new ColumnConfig();
   172→                copy.setKey(c.getKey());
   173→                copy.setLabel(c.getLabel());
   174→                copy.setColumnType(c.getColumnType());
   175→                copy.setSortable(false);
   176→                out.add(copy);
   177→            }
   178→        }
   179→        return out;
   180→    }

   183→    public BizDataPageVO query(DataSourceDefinition ds, BizDataQueryRequest req) {
   184→        return switch (ds.getType()) {
   185→            case "FORM" -> {
   186→                router.resolve(ds, "list");
   187→                FormQueryConfig cfg = FormQueryConfig.parse(ds.getParams(), objectMapper);
   188→                if (cfg.isConfigMode()) {
   189→                    yield bizDataService.queryJoin(ds.getFormKey(), req, cfg.joins());
   190→                }
   191→                if (cfg.isSqlMode()) {
   192→                    yield bizDataService.querySql(ds.getFormKey(), req, cfg);
   193→                }
   194→                yield bizDataService.query(ds.getFormKey(), req);
   195→            }
   196→            case "WORKFLOW" -> workflowQueryService.query(ds.getFormKey(), req);
   197→            case "SYSTEM" -> {
   198→                router.resolve(ds, "list");
   199→                yield systemQuery(ds, req);
   200→            }
   201→            case "API" -> apiQuery(ds, req);
   202→            case "SQL" -> {
   203→                router.resolve(ds, "list");
   204→                FormQueryConfig cfg = FormQueryConfig.parse(ds.getParams(), objectMapper);
   205→                if (cfg.isVisualMode() || cfg.isSqlMode()) {
   206→                    // 绕过表单 covering handler：SQL 数据源执行管理员显式 SQL，不被绑定表单业务定制劫持
   207→                    yield bizDataService.querySqlRaw(ds.getFormKey(), req, cfg);
   208→                }
   209→                throw new BusinessException(400, "SQL 数据源缺少查询配置");
   210→            }
   211→            default -> throw unsupported(ds, "query");
   212→        };
   213→    }

   299→    /** 追加 config 模式 JOIN 虚拟列到 metadata；与主表列 key 重复的 virtualKey 跳过（防御性去重）。 */
   300→    private void appendJoinColumns(List<ColumnConfig> cols, List<JoinConfig> joins) {
   301→        Set<String> existing = cols.stream().map(ColumnConfig::getKey).collect(Collectors.toSet());
   302→        for (JoinConfig j : joins) {
   303→            if (j.virtualKey() == null || j.virtualKey().isBlank() || existing.contains(j.virtualKey())) {
   304→                continue;
   305→            }
   306→            List<ColumnConfig> targetCols = resolveJoinTargets(j);
   307→            ColumnConfig c = new ColumnConfig();
   308→            c.setKey(j.virtualKey());
   309→            c.setLabel(j.label() == null || j.label().isBlank() ? j.virtualKey() : j.label());
   310→            c.setColumnType(findJoinColumnType(targetCols, j.joinField()));
   311→            c.setSortable(j.sortable());
   312→            c.setFilterable(j.filterable());
   313→            cols.add(c);
   314→            existing.add(c.getKey());
   315→
   316→            // 目标列为 dataPicker 引用列（pickerConfig 非空）：透传 pickerConfig，前端按引用列渲染显示文本
   317→            ColumnConfig target = findJoinTarget(targetCols, j.joinField());
   318→            if (target != null && target.getPickerConfig() != null && !target.getPickerConfig().isBlank()) {
   319→                c.setPickerConfig(target.getPickerConfig());
   320→            }
   321→            // 目标表含 <joinField>_text 冗余列（dataPicker 两列映射）→ 追加隐藏冗余列，供前端引用渲染取值
   322→            if (findJoinTarget(targetCols, j.joinField() + "_text") != null) {
   323→                ColumnConfig text = new ColumnConfig();
   324→                text.setKey(j.virtualKey() + "_text");
   325→                text.setLabel(c.getLabel() + "（显示）");
   326→                text.setColumnType(findJoinColumnType(targetCols, j.joinField() + "_text"));
   327→                text.setHidden(true);
   328→                text.setSortable(false);
   329→                text.setFilterable(false);
   330→                text.setComponentType("dataPickerText");
   331→                cols.add(text);
   332→                existing.add(text.getKey());
   333→            }
   334→        }
   335→    }

   337→    /** 追加 sql 模式管理员声明列到 metadata；与主表列 key 重复的声明列跳过。 */
   338→    private void appendDeclaredColumns(List<ColumnConfig> cols, List<ColumnConfig> declared) {
   339→        Set<String> existing = cols.stream().map(ColumnConfig::getKey).collect(Collectors.toSet());
   340→        for (ColumnConfig d : declared) {
   341→            if (d.getKey() == null || d.getKey().isBlank() || existing.contains(d.getKey())) {
   342→                continue;
   343→            }
   344→            ColumnConfig c = new ColumnConfig();
   345→            c.setKey(d.getKey());
   346→            c.setLabel(d.getLabel() == null || d.getLabel().isBlank() ? d.getKey() : d.getLabel());
   347→            c.setColumnType(d.getColumnType());
   348→            c.setSortable(d.getSortable());
   349→            c.setFilterable(d.getFilterable());
   350→            cols.add(c);
   351→            existing.add(c.getKey());
   352→        }
   353→    }

   355→    /** 目标表单列列表；解析失败/不存在 → 空列表（调用方统一按"查不到"处理）。 */
   356→    private List<ColumnConfig> resolveJoinTargets(JoinConfig j) {
   357→        try {
   358→            List<ColumnConfig> target = formDefService.getBusinessColumnsByKey(j.targetFormKey());
   359→            return target == null ? List.of() : target;
   360→        } catch (BusinessException ignored) {
   361→            // 目标表单不可解析时回退
   362→            return List.of();
   363→        }
   364→    }

   366→    /** 目标列查找：按 key 匹配；查不到返回 null。 */
   367→    private static ColumnConfig findJoinTarget(List<ColumnConfig> cols, String key) {
   368→        if (cols == null) {
   369→            return null;
   370→        }
   371→        for (ColumnConfig c : cols) {
   372→            if (key.equals(c.getKey())) {
   373→                return c;
   374→            }
   375→        }
   376→        return null;
   377→    }

   379→    /** 目标列 columnType：查不到 fallback "VARCHAR"。 */
   380→    private static String findJoinColumnType(List<ColumnConfig> cols, String key) {
   381→        ColumnConfig c = findJoinTarget(cols, key);
   382→        return c != null && c.getColumnType() != null && !c.getColumnType().isBlank()
   383→                ? c.getColumnType().toUpperCase()
   384→                : "VARCHAR";
   385→    }

   404→    private BizDataPageVO queryDeptTree() {
   405→        List<BizDataVO> rows = new ArrayList<>();
   406→        for (TreeNode node : organizationService.tree()) {
   407→            collectNode(node, rows);
   408→        }
   409→        return new BizDataPageVO(rows, rows.size(), 0, rows.size());

   412→    private void collectNode(TreeNode node, List<BizDataVO> out) {
   413→        Map<String, Object> data = new LinkedHashMap<>();
   414→        data.put("id", String.valueOf(node.id()));
   415→        data.put("parentId", node.parentId() == null ? "" : String.valueOf(node.parentId()));
   416→        data.put("label", node.label() == null ? "" : node.label());
   417→        data.put("code", node.code() == null ? "" : node.code());
   418→        out.add(new BizDataVO(String.valueOf(node.id()), data, null, null, null));
   419→        if (node.children() != null) {
   420→            for (TreeNode child : node.children()) {
   421→                collectNode(child, out);
   422→            }
   423→        }
   424→    }

   426→    private BizDataPageVO queryUsers(BizDataQueryRequest req) {
   427→        int userServicePage = Math.max(req.getPage(), 1);
   428→        UserQueryRequest query = new UserQueryRequest(
   429→                req.getKeyword(), null, null, null, null, null, userServicePage, req.getSize());
   430→        PageResult<UserVO> page = userService.list(query);
   431→        List<BizDataVO> rows = new ArrayList<>();
   432→        for (UserVO u : page.getRows() == null ? List.<UserVO>of() : page.getRows()) {
   433→            Map<String, Object> data = new LinkedHashMap<>();
   434→            data.put("id", String.valueOf(u.id()));
   435→            data.put("username", u.username());
   436→            data.put("nickname", u.nickname() == null ? "" : u.nickname());
   437→            data.put("orgId", u.orgId() == null ? "" : String.valueOf(u.orgId()));
   438→            data.put("orgName", u.orgName() == null ? "" : u.orgName());
   439→            data.put("status", u.status());
   440→            rows.add(new BizDataVO(String.valueOf(u.id()), data, null, null, null));
   441→        }
   442→        return new BizDataPageVO(rows, page.getTotal(), userServicePage, req.getSize());
   443→    }

   447→    private DataSourceMetadata apiMetadata(DataSourceDefinition ds) {
   448→        Map<String, Object> params = parseParams(ds);
   449→        List<ColumnConfig> columns = new ArrayList<>();
   450→        JsonNode colsNode = jsonNode(params, "columns");
   451→        if (colsNode != null && colsNode.isArray()) {
   452→            for (JsonNode node : colsNode) {
   453→                columns.add(objectMapper.convertValue(node, ColumnConfig.class));
   454→            }
   455→        }
   456→        DataSourceMetadata m = new DataSourceMetadata(columns, writable(params));
   457→        m.setFormKey(ds.getFormKey());
   458→        return m;
   459→    }

   461→    private BizDataPageVO apiQuery(DataSourceDefinition ds, BizDataQueryRequest req) {
   462→        Map<String, Object> params = parseParams(ds);
   463→        Map<String, Object> op = operation(params, "list");
   464→        if (op == null) {
   465→            throw new BusinessException(400, "API 数据源未配置 list 操作: " + ds.getName());
   466→        }
   467→        String action = requireAction(op, "list");
   468→        String method = opMethod(op);
   469→        Map<String, Object> vars = new HashMap<>();
   470→        if (params.get("data") instanceof Map<?, ?> data) {
   471→            for (Map.Entry<?, ?> e : data.entrySet()) {
   472→                vars.put(String.valueOf(e.getKey()), e.getValue());
   473→            }
   474→        }
   475→        vars.put("page", req.getPage());
   476→        vars.put("size", req.getSize());
   477→        String kw = req.getKeyword();
   478→        vars.put("keyword", kw == null ? "" : kw);
   479→        String searchParam = str(params.get("searchParam"));
   480→        if (kw != null && !kw.isBlank() && searchParam != null && !searchParam.isBlank()) {
   481→            vars.put(searchParam, kw);
   482→        }
   483→        Object raw = httpExecutor.execute(action, method, headers(params), List.of(), List.of(),
   484→                vars, DEFAULT_TIMEOUT_MS, DEFAULT_TIMEOUT_MS, DEFAULT_RETRY);
   485→        return toPageVO(raw, op, req);
   486→    }

   488→    private BizDataVO apiGet(DataSourceDefinition ds, String id) {

   492→            throw new BusinessException(400, "API 数据源未配置 get 操作: " + ds.getName());

   494→        String action = requireAction(op, "get");
   495→        Map<String, Object> vars = Map.of("id", id);
   496→        Object raw = httpExecutor.execute(action, opMethod(op), headers(params), List.of(), List.of(),
   497→                vars, DEFAULT_TIMEOUT_MS, DEFAULT_TIMEOUT_MS, DEFAULT_RETRY);
   498→        return new BizDataVO(id, asDataMap(raw), null, null, null);
   499→    }

   540→    @SuppressWarnings("unchecked")
   541→    private Map<String, Object> parseParams(DataSourceDefinition ds) {
   542→        String json = ds.getParams();
   543→        if (json == null || json.isBlank()) {
   544→            throw new BusinessException(400, "API 数据源缺少 params: " + ds.getName());
   545→        }
   546→        try {
   547→            JsonNode node = objectMapper.readTree(json);
   548→            if (!node.isObject()) {
   549→                throw new BusinessException(400, "API 数据源 params 必须是 JSON 对象: " + ds.getName());
   550→            }
   551→            return objectMapper.convertValue(node, Map.class);
   552→        } catch (JsonProcessingException e) {
   553→            throw new BusinessException(400, "API 数据源 params 不是合法 JSON: " + ds.getName());
   554→        }
   555→    }

   557→    @SuppressWarnings("unchecked")
   558→    private Map<String, Object> operation(Map<String, Object> params, String name) {
   559→        Object op = params.get(name);
   560→        if (op instanceof Map<?, ?> m) return (Map<String, Object>) m;
   561→        if ("list".equals(name) && params.get("action") != null) {
   562→            Map<String, Object> legacy = new LinkedHashMap<>();
   563→            legacy.put("action", params.get("action"));
   564→            legacy.put("method", params.get("method"));
   565→            legacy.put("parse", params.get("parse"));
   566→            legacy.put("totalParse", params.get("totalParse"));
   567→            return legacy;
   568→        }
   569→        return null;
   570→    }

   575→            throw new BusinessException(400, "API 数据源 " + name + " 操作缺少 action");
   576→        }
   577→        return action;
   578→    }

   586→    private Map<String, String> headers(Map<String, Object> params) {
   587→        Map<String, String> headers = new LinkedHashMap<>();
   588→        Object h = params.get("headers");
   589→        if (h instanceof Map<?, ?> hm) {
   590→            for (Map.Entry<?, ?> e : hm.entrySet()) {

   593→        }
   594→        return headers;

   603→    private JsonNode jsonNode(Map<String, Object> params, String key) {
   604→        Object v = params.get(key);
   605→        if (v == null) return null;
   606→        if (v instanceof JsonNode jn) return jn;
   607→        try { return objectMapper.valueToTree(v); } catch (IllegalArgumentException e) { return null; }
   608→    }

   610→    private JsonNode toJsonNode(Object raw) {
   611→        if (raw instanceof String s) {
   612→            try { return objectMapper.readTree(s); }
   613→            catch (JsonProcessingException e) { throw new BusinessException(400, "外部 API 响应不是合法 JSON"); }
   614→        }
   615→        return objectMapper.valueToTree(raw);
   616→    }

   618→    private JsonNode walkPath(JsonNode root, String path) {
   619→        JsonNode cur = root;
   620→        for (String part : path.split("\\.")) {
   621→            if (cur == null || cur.isMissingNode() || !cur.isObject()) return null;
   622→            cur = cur.get(part);
   623→        }
   624→        return cur;
   625→    }

   627→    private BizDataPageVO toPageVO(Object raw, Map<String, Object> op, BizDataQueryRequest req) {
   628→        int normalizedPage = Math.max(req.getPage(), 1);
   629→        JsonNode root = toJsonNode(raw);
   630→        JsonNode recordsNode = root;
   631→        String parse = str(op.get("parse"));
   632→        if (parse != null && !parse.isBlank()) {
   633→            recordsNode = walkPath(root, parse);
   634→        }
   635→        if (recordsNode == null || !recordsNode.isArray()) {
   636→            return new BizDataPageVO(List.of(), 0, normalizedPage, req.getSize());
   637→        }
   638→        List<BizDataVO> records = new ArrayList<>();
   639→        for (JsonNode item : recordsNode) {
   640→            Map<String, Object> data = objectMapper.convertValue(item, Map.class);
   641→            Object id = item.has("id") ? item.get("id").asText() : null;
   642→            records.add(new BizDataVO(id == null ? "" : String.valueOf(id), data, null, null, null));
   643→        }
   644→        long total = records.size();
   645→        String totalParse = str(op.get("totalParse"));
   646→        if (totalParse != null && !totalParse.isBlank()) {
   647→            JsonNode totalNode = walkPath(root, totalParse);
   648→            if (totalNode != null && totalNode.isNumber()) total = totalNode.asLong();
   649→        }
   650→        return new BizDataPageVO(records, total, normalizedPage, req.getSize());
   651→    }

   657→        return objectMapper.convertValue(toJsonNode(raw), Map.class);
   658→    }
