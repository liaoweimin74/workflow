    80→    80→     * 创建数据源（默认 DRAFT）。
    81→    81→     * 校验：type 必填且合法；同租户 name 唯一；按类型必填项（FORM→formKey + 表单存在；
    82→    82→     * SYSTEM/API→sourceKey；API→params 须为合法 JSON）。

    86→    86→    @Transactional
    87→    87→    public DataSourceDefinition create(String name, String type, String formKey, String sourceKey, String params) {
    88→    88→        String tenantId = tenantProvider.getTenantId();
    89→    89→
    90→    90→        if (type == null || type.isBlank()) {
    91→    91→            throw new BusinessException(400, "数据源类型 type 必填");
    92→    92→        }
    93→    93→        if (!SUPPORTED_TYPES.contains(type)) {
    94→    94→            throw new BusinessException(400, "不支持的数据源类型: " + type);
    95→    95→        }
    96→    96→        if (name == null || name.isBlank()) {
    97→    97→            throw new BusinessException(400, "数据源名称不能为空");
    98→    98→        }
    99→    99→        if (dsRepository.existsByTenantIdAndName(tenantId, name)) {
   100→   100→            throw new BusinessException(400, "数据源名称已存在: " + name);
   101→   101→        }
   102→   102→        // FORM/WORKFLOW：sourceKey 恒等于 formKey（formKey 权威）；其余类型以入参 sourceKey 为准
   103→   103→        boolean formBound = TYPE_FORM.equals(type) || TYPE_WORKFLOW.equals(type);
   104→   104→        String effSourceKey = formBound ? formKey : sourceKey;
   105→   105→        validateRequiredFields(type, formKey, effSourceKey, params);
   106→   106→        if (formBound && !formDefRepository.existsByTenantIdAndKey(tenantId, formKey)) {
   107→   107→            throw new BusinessException(400, "绑定的表单不存在: " + formKey);
   108→   108→        }
   109→   109→        if (effSourceKey == null || effSourceKey.isBlank()) {
   110→   110→            throw new BusinessException(400, "数据源必须填写 sourceKey");
   111→   111→        }
   112→   112→        if (dsRepository.existsByTenantIdAndSourceKey(tenantId, effSourceKey)) {
   113→   113→            throw new BusinessException(400, "数据源标识 sourceKey 已存在: " + effSourceKey);
   114→   114→        }
   115→   115→
   116→   116→        DataSourceDefinition ds = new DataSourceDefinition();
   117→   117→        ds.setId(UUID.randomUUID().toString().replace("-", ""));
   118→   118→        ds.setTenantId(tenantId);
   119→   119→        ds.setName(name);
   120→   120→        ds.setType(type);
   121→   121→        ds.setFormKey(formKey);
   122→   122→        ds.setSourceKey(effSourceKey);
   123→   123→        if (TYPE_FORM.equals(type) || TYPE_SYSTEM.equals(type)) {
   124→   124→            if (TYPE_FORM.equals(type) && hasQueryModeSegment(params)) {
   125→   125→                // FORM 带 queryMode 配置段：校验后与自动生成端点合并保存
   126→   126→                validateFormQueryConfig(params);
   127→   127→                ds.setParams(mergeQueryConfig(generateParams(type, formKey, sourceKey), params));
   128→   128→            } else {
   129→   129→                ds.setParams(generateParams(type, formKey, sourceKey));
   130→   130→            }
   131→   131→        } else {
   132→   132→            ds.setParams(params);
   133→   133→        }
   134→   134→        // API/SQL 为手动配置的数据源：创建即发布（ENABLED）；其余类型仍 DRAFT（防御：手动创建仅 API/SQL）
   135→   135→        boolean manualPublish = TYPE_API.equals(type) || TYPE_SQL.equals(type);
   136→   136→        ds.setStatus(manualPublish ? STATUS_ENABLED : STATUS_DRAFT);
   137→   137→        return dsRepository.save(ds);
   138→   138→    }

   140→   140→    /**
   141→   141→     * 原地更新数据源（name/type/formKey/sourceKey/params；null 表示不更新）。
   142→   142→     * 若当前 ENABLED 且 type/formKey 变更，重新校验（FORM 须仍指向已发布表单）。
   143→   143→     * 
   144→   144→     * 注意：此方法仅供系统内部调用，用户不能直接编辑数据源。
   145→   145→     */
   146→   146→    @Transactional
   147→   147→    public DataSourceDefinition update(String id, String name, String type, String formKey,
   148→   148→                                       String sourceKey, String params) {
   149→   149→        String tenantId = tenantProvider.getTenantId();
   150→   150→        DataSourceDefinition ds = getById(id);
   151→   151→
   152→   152→        String newType = type == null || type.isBlank() ? ds.getType() : type;
   153→   153→        if (type != null && !type.isBlank() && !SUPPORTED_TYPES.contains(newType)) {
   154→   154→            throw new BusinessException(400, "不支持的数据源类型: " + newType);
   155→   155→        }
   156→   156→        if (name != null && !name.isBlank() && !name.equals(ds.getName())
   157→   157→                && dsRepository.existsByTenantIdAndName(tenantId, name)) {
   158→   158→            throw new BusinessException(400, "数据源名称已存在: " + name);
   159→   159→        }
   160→   160→
   161→   161→        String newFormKey = formKey == null ? ds.getFormKey() : formKey;
   162→   162→        String newSourceKey = sourceKey == null ? ds.getSourceKey() : sourceKey;
   163→   163→        String newParams = params == null ? ds.getParams() : params;
   164→   164→        // FORM/WORKFLOW：sourceKey 恒等于 formKey（formKey 权威），忽略入参 sourceKey 差异
   165→   165→        boolean formBound = TYPE_FORM.equals(newType) || TYPE_WORKFLOW.equals(newType);
   166→   166→        String effNewSourceKey = formBound ? newFormKey : newSourceKey;
   167→   167→        validateRequiredFields(newType, newFormKey, effNewSourceKey, newParams);
   168→   168→        if (formBound && !formDefRepository.existsByTenantIdAndKey(tenantId, newFormKey)) {
   169→   169→            throw new BusinessException(400, "绑定的表单不存在: " + newFormKey);
   170→   170→        }
   171→   171→        if (TYPE_FORM.equals(newType)) {
   172→   172→            // FORM 查询配置段（queryMode/joins/query/columns/params）保存校验
   173→   173→            validateFormQueryConfig(newParams);
   174→   174→        }
   175→   175→        // sourceKey 变更（不等于当前值）时校验租户内唯一；保持不变则跳过（自身不算冲突）
   176→   176→        if (!java.util.Objects.equals(effNewSourceKey, ds.getSourceKey())
   177→   177→                && dsRepository.existsByTenantIdAndSourceKey(tenantId, effNewSourceKey)) {
   178→   178→            throw new BusinessException(400, "数据源标识 sourceKey 已存在: " + effNewSourceKey);
   179→   179→        }
   180→   180→
   181→   181→        // 已启用数据源若变更类型/绑定对象，须重新校验发布状态
   182→   182→        boolean bindChanged = !TYPE_FORM.equals(ds.getType()) || (formKey != null && !formKey.equals(ds.getFormKey()));
   183→   183→        if (STATUS_ENABLED.equals(ds.getStatus()) && bindChanged) {
   184→   184→            if (TYPE_FORM.equals(newType)) {
   185→   185→                requirePublishedForm(tenantId, newFormKey);
   186→   186→            } else if (TYPE_WORKFLOW.equals(newType)) {
   187→   187→                requireWorkflowForm(tenantId, newFormKey);
   188→   188→            }
   189→   189→        }
   190→   190→
   191→   191→        ds.setName(name == null ? ds.getName() : name);
   192→   192→        ds.setType(newType);
   193→   193→        ds.setFormKey(newFormKey);
   194→   194→        ds.setSourceKey(effNewSourceKey);
   195→   195→        ds.setParams(newParams);
   196→   196→        return dsRepository.save(ds);
   197→   197→    }

   234→   234→    /**
   235→   235→     * 删除数据源：任意状态可删除（ENABLED/DISABLED 均可），但被页面引用时拒绝（400）。
   236→   236→     * 引用统计覆盖两种绑定方式：
   237→   237→     * 1) PageDefinition.dataSourceId 列（VIEW 新协议，迁移器回填）；
   238→   238→     * 2) PAGE 类型页面 schema.dataSources[].refId（设计器 dataSources 声明）。
   239→   239→     * 两者取并集，任一命中即拒绝删除。
   240→   240→     * 
   241→   241→     * 注意：此方法仅供系统内部调用，用户不能直接删除数据源。
   242→   242→     */
   243→   243→    @Transactional
   244→   244→    public void delete(String id) {
   245→   245→        DataSourceDefinition ds = getById(id);
   246→   246→        String tenantId = ds.getTenantId();
   247→   247→        long refCount = countRefs(tenantId, id);
   248→   248→        if (refCount > 0) {
   249→   249→            throw new BusinessException(400, "数据源已被 " + refCount + " 个页面引用，无法删除");
   250→   250→        }
   251→   251→        dsRepository.delete(ds);
   252→   252→    }
   253→   253→
   254→   254→    /**
   255→   255→     * 统计当前租户内引用指定数据源的页面数（dataSourceId 列 + PAGE schema dataSources[].refId 并集）。
   256→   256→     */
   257→   257→    private long countRefs(String tenantId, String dataSourceId) {
   258→   258→        long columnRefs = pageRepository.countByTenantIdAndDataSourceId(tenantId, dataSourceId);
   259→   259→        if (columnRefs > 0) {
   260→   260→            return columnRefs;
   261→   261→        }
   262→   262→        // PAGE 类型页面引用声明在 schema.dataSources[].refId（dataSourceId 列为空），需扫描；
   263→   263→        // 页面软删除（ARCHIVED）后不再使用，其 schema 引用不阻塞删除
   264→   264→        long schemaRefs = 0;
   265→   265→        Page<PageDefinition> pages = pageRepository
   266→   266→                .findByTenantIdAndTypeOrderByUpdatedAtDesc(tenantId, "PAGE", PageRequest.of(0, Integer.MAX_VALUE));
   267→   267→        for (PageDefinition page : pages.getContent()) {
   268→   268→            if ("ARCHIVED".equals(page.getStatus())) {
   269→   269→                continue;
   270→   270→            }
   271→   271→            if (schemaRefsDataSource(page.getSchema(), dataSourceId)) {
   272→   272→                schemaRefs++;
   273→   273→            }
   274→   274→        }
   275→   275→        return schemaRefs;
   276→   276→    }

   278→   278→    /**
   279→   279→     * 判断页面 schema 的 dataSources[].refId 是否指向指定数据源。
   280→   280→     */
   281→   281→    private boolean schemaRefsDataSource(String schema, String dataSourceId) {
   282→   282→        if (schema == null || schema.isBlank()) {
   283→   283→            return false;
   284→   284→        }
   285→   285→        try {
   286→   286→            JsonNode root = objectMapper.readTree(schema);
   287→   287→            JsonNode dataSources = root.path("dataSources");
   288→   288→            if (!dataSources.isArray()) {
   289→   289→                return false;
   290→   290→            }
   291→   291→            for (JsonNode ds : dataSources) {
   292→   292→                String refId = ds.path("refId").asText(null);
   293→   293→                if (dataSourceId.equals(refId)) {
   294→   294→                    return true;
   295→   295→                }
   296→   296→            }
   297→   297→        } catch (JsonProcessingException e) {
   298→   298→            return false;
   299→   299→        }
   300→   300→        return false;
   301→   301→    }

   387→   387→    /** 按数据源类型找适配器（数据源须 ENABLED） */
   388→   388→    private DataSourceAdapter adapterOf(String id) {
   389→   389→        DataSourceDefinition ds = getById(id);
   390→   390→        if (!STATUS_ENABLED.equals(ds.getStatus())) {
   391→   391→            throw new BusinessException(400, "数据源未启用，无法访问: " + ds.getName());
   392→   392→        }
   393→   393→        for (DataSourceAdapter adapter : adapters) {
   394→   394→            if (adapter.supports(ds.getType())) {
   395→   395→                return adapter;
   396→   396→            }
   397→   397→        }
   398→   398→        throw new BusinessException(400, "数据源类型未启用: " + ds.getType());
   399→   399→    }

   408→   408→    private String generateParams(String type, String formKey, String sourceKey) {

   413→   413→                    .put("action", base).put("method", "GET")

   416→   416→            params.putObject("get").put("action", base + "/{id}").put("method", "GET");
   417→   417→            params.putObject("update").put("action", base + "/{id}").put("method", "PUT");

   420→   420→            params.putObject("list")
   421→   421→                    .put("action", "/api/v1/internal/system/" + mapSystemInternalKey(sourceKey))
   422→   422→                    .put("method", "GET");
   423→   423→        }
   424→   424→        return params.toString();
   425→   425→    }

   436→   436→    // ==================== FORM 查询配置段（queryMode）校验 ====================

   438→   438→    /** 判断 params 是否含 queryMode 配置段（非合法 JSON 视为无，由后续校验处理） */
   439→   439→    private boolean hasQueryModeSegment(String params) {
   440→   440→        if (params == null || params.isBlank()) {
   441→   441→            return false;
   442→   442→        }
   443→   443→        try {
   444→   444→            JsonNode root = objectMapper.readTree(params);
   445→   445→            return root != null && root.isObject() && root.has("queryMode");
   446→   446→        } catch (JsonProcessingException e) {
   447→   447→            return false;
   448→   448→        }
   449→   449→    }
   450→   450→
   451→   451→    /**
   452→   452→     * FORM 数据源查询配置段保存校验（Task 5）。
   453→   453→     * <ul>
   454→   454→     *   <li>无 queryMode 段 → 向后兼容，跳过校验</li>
   455→   455→     *   <li>config：joins 非空；每项 alias 合法、targetFormKey 表单存在、必填字段齐全、virtualKey 唯一</li>
   456→   456→     *   <li>sql：query/columns 合法性复用 SqlTemplateEngine.validate（SELECT / :tenantId / 列匹配 / 参数白名单）</li>
   457→   457→     *   <li>未知 queryMode → 400</li>
   458→   458→     * </ul>
   459→   459→     */
   460→   460→    private void validateFormQueryConfig(String params) {
   461→   461→        if (params == null || params.isBlank()) {
   462→   462→            return;
   463→   463→        }
   464→   464→        JsonNode root;
   465→   465→        try {
   466→   466→            root = objectMapper.readTree(params);
   467→   467→        } catch (JsonProcessingException e) {
   468→   468→            throw new BusinessException(400, "数据源参数 params 必须是合法 JSON: " + e.getOriginalMessage());
   469→   469→        }
   470→   470→        if (root == null || !root.isObject()) {
   471→   471→            throw new BusinessException(400, "数据源参数 params 必须是 JSON 对象");
   472→   472→        }
   473→   473→        JsonNode modeNode = root.get("queryMode");
   474→   474→        if (modeNode == null || modeNode.isNull() || modeNode.asText().isBlank()) {
   475→   475→            return; // 无 queryMode 段 → 单表查询，向后兼容
   476→   476→        }
   477→   477→        String mode = modeNode.asText();
   478→   478→        if ("config".equals(mode)) {
   479→   479→            validateConfigJoins(root.get("joins"));
   480→   480→        } else if ("sql".equals(mode)) {
   481→   481→            validateSqlConfig(root);
   482→   482→        } else {
   483→   483→            throw new BusinessException(400, "未知查询模式 queryMode: " + mode + "（支持 config / sql）");
   484→   484→        }
   485→   485→    }
   486→   486→
   487→   487→    /** config 模式：joins[] 结构校验（别名/目标表单/字段/虚拟列唯一性） */
   488→   488→    private void validateConfigJoins(JsonNode joins) {
   489→   489→        if (joins == null || !joins.isArray() || joins.isEmpty()) {
   490→   490→            throw new BusinessException(400, "queryMode=config 时必须配置至少一个关联 joins");
   491→   491→        }
   492→   492→        String tenantId = tenantProvider.getTenantId();
   493→   493→        Set<String> virtualKeys = new HashSet<>();
   494→   494→        int idx = 0;
   495→   495→        for (JsonNode j : joins) {
   496→   496→            idx++;
   497→   497→            if (j == null || !j.isObject()) {
   498→   498→                throw new BusinessException(400, "joins 第 " + idx + " 项必须是对象");
   499→   499→            }
   500→   500→            String alias = text(j, "alias");
   501→   501→            if (alias == null || !alias.matches("[a-zA-Z_][a-zA-Z0-9_]*")) {
   502→   502→                throw new BusinessException(400, "joins 第 " + idx + " 项 alias 非法: " + alias);
   503→   503→            }
   504→   504→            String targetFormKey = text(j, "targetFormKey");
   505→   505→            if (targetFormKey == null || targetFormKey.isBlank()) {
   506→   506→                throw new BusinessException(400, "joins 第 " + idx + " 项必须指定目标表单 targetFormKey");
   507→   507→            }
   508→   508→            if (!formDefRepository.existsByTenantIdAndKey(tenantId, targetFormKey)) {
   509→   509→                throw new BusinessException(400, "目标表单不存在: " + targetFormKey);
   510→   510→            }
   511→   511→            requireJoinField(j, "localField", "主表关联字段", idx);
   512→   512→            requireJoinField(j, "foreignField", "目标表关联字段", idx);
   513→   513→            requireJoinField(j, "joinField", "显示字段", idx);
   514→   514→            requireJoinField(j, "label", "显示名称", idx);
   515→   515→            String virtualKey = text(j, "virtualKey");
   516→   516→            if (virtualKey == null || virtualKey.isBlank()) {
   517→   517→                throw new BusinessException(400, "joins 第 " + idx + " 项必须指定虚拟列标识 virtualKey");
   518→   518→            }
   519→   519→            if (!virtualKeys.add(virtualKey)) {
   520→   520→                throw new BusinessException(400, "虚拟列 virtualKey 重复: " + virtualKey);
   521→   521→            }
   522→   522→        }
   523→   523→    }
   524→   524→
   525→   525→    private void requireJoinField(JsonNode j, String field, String label, int idx) {
   526→   526→        String v = text(j, field);
   527→   527→        if (v == null || v.isBlank()) {
   528→   528→            throw new BusinessException(400, "joins 第 " + idx + " 项必须指定" + label + " " + field);
   529→   529→        }
   530→   530→    }

   532→   532→    /** sql 模式：query/columns/参数白名单复用 SqlTemplateEngine.validate（IllegalArgumentException → 400） */
   533→   533→    private void validateSqlConfig(JsonNode root) {
   534→   534→        String query = text(root, "query");
   535→   535→        List<JoinSqlGenerator.QueryColumn> columns = parseSqlColumns(root.get("columns"));
   536→   536→        List<String> declaredParams = parseStringList(root.get("params"));
   537→   537→        try {
   538→   538→            SqlTemplateEngine.validate(query, columns, declaredParams);
   539→   539→        } catch (IllegalArgumentException e) {
   540→   540→            throw new BusinessException(400, e.getMessage());
   541→   541→        }
   542→   542→    }
   543→   543→
   544→   544→    private List<JoinSqlGenerator.QueryColumn> parseSqlColumns(JsonNode node) {
   545→   545→        List<JoinSqlGenerator.QueryColumn> out = new ArrayList<>();
   546→   546→        if (node == null || !node.isArray()) {
   547→   547→            return out;
   548→   548→        }
   549→   549→        for (JsonNode n : node) {
   550→   550→            if (n == null || !n.isObject()) {
   551→   551→                continue;
   552→   552→            }
   553→   553→            out.add(new JoinSqlGenerator.QueryColumn(
   554→   554→                    text(n, "key"), text(n, "key"), text(n, "columnType"),
   555→   555→                    boolVal(n, "sortable"), boolVal(n, "filterable")));
   556→   556→        }
   557→   557→        return out;
   558→   558→    }
   559→   559→
   560→   560→    private List<String> parseStringList(JsonNode node) {
   561→   561→        List<String> out = new ArrayList<>();
   562→   562→        if (node == null || !node.isArray()) {
   563→   563→            return out;
   564→   564→        }
   565→   565→        for (JsonNode n : node) {
   566→   566→            if (n != null && n.isTextual() && !n.asText().isBlank()) {
   567→   567→                out.add(n.asText());
   568→   568→            }
   569→   569→        }
   570→   570→        return out;
   571→   571→    }

   573→   573→    /** 合并：生成端点 params 之上叠加 queryMode 配置段（config 的 joins / sql 的 query+columns+params 白名单） */
   574→   574→    private String mergeQueryConfig(String generated, String params) {
   575→   575→        try {
   576→   576→            ObjectNode out = (ObjectNode) objectMapper.readTree(generated);
   577→   577→            JsonNode input = objectMapper.readTree(params);
   578→   578→            for (String field : List.of("queryMode", "joins", "query", "columns", "params")) {
   579→   579→                if (input.has(field)) {
   580→   580→                    out.set(field, input.get(field));
   581→   581→                }
   582→   582→            }
   583→   583→            return out.toString();
   584→   584→        } catch (JsonProcessingException e) {
   585→   585→            throw new BusinessException(400, "数据源参数 params 必须是合法 JSON: " + e.getOriginalMessage());
   586→   586→        }
   587→   587→    }

   590→   590→        JsonNode v = node.get(field);
   591→   591→        return v == null || v.isNull() ? null : v.asText();
   592→   592→    }

   601→   601→    /** 按类型校验必填项：FORM/WORKFLOW→formKey（sourceKey 由 formKey 派生）；SYSTEM/API/SQL→sourceKey；API→params 合法 JSON */
   602→   602→    private void validateRequiredFields(String type, String formKey, String sourceKey, String params) {
   603→   603→        if (TYPE_FORM.equals(type) || TYPE_WORKFLOW.equals(type)) {
   604→   604→            if (formKey == null || formKey.isBlank()) {
   605→   605→                throw new BusinessException(400, type + " 类型数据源必须绑定表单 formKey");
   606→   606→            }
   607→   607→        } else if (TYPE_SYSTEM.equals(type) || TYPE_API.equals(type) || TYPE_SQL.equals(type)) {
   608→   608→            if (sourceKey == null || sourceKey.isBlank()) {
   609→   609→                throw new BusinessException(400, type + " 类型数据源必须填写 sourceKey");
   610→   610→            }
   611→   611→            if (TYPE_SYSTEM.equals(type) && !SYSTEM_SOURCE_KEYS.contains(sourceKey)) {
   612→   612→                throw new BusinessException(400, "未注册的系统数据源: " + sourceKey);
   613→   613→            }
   614→   614→            if (TYPE_API.equals(type)) {
   615→   615→                // LookupFetchConfig 契约：params 须为 JSON 对象且 action 必填
   616→   616→                if (params == null || params.isBlank()) {
   617→   617→                    throw new BusinessException(400, "API 数据源参数 params 必须包含 action（API 路径）");
   618→   618→                }
   619→   619→                try {
   620→   620→                    JsonNode node = objectMapper.readTree(params);
   621→   621→                    if (!node.isObject()) {
   622→   622→                        throw new BusinessException(400, "API 数据源参数 params 必须是 JSON 对象");
   623→   623→                    }
   624→   624→                    JsonNode action = node.get("action");
   625→   625→                    if (action == null || action.isNull() || action.asText().isBlank()) {
   626→   626→                        throw new BusinessException(400, "API 数据源参数 params 必须包含 action（API 路径）");
   627→   627→                    }
   628→   628→                } catch (BusinessException e) {
   629→   629→                    throw e;
   630→   630→                } catch (JsonProcessingException e) {
   631→   631→                    throw new BusinessException(400, "API 数据源参数 params 必须是合法 JSON: " + e.getOriginalMessage());
   632→   632→                }
   633→   633→            }
   634→   634→            // SQL：仅要求 sourceKey（query 配置在 params 中，DRAFT 阶段可为空，启用后由适配器校验）
   635→   635→        }
   636→   636→    }
