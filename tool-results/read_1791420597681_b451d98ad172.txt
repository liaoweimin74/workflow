   545→function loadHeaderRows() {

   554→function syncHeaders() {
   555→  const map: Record<string, string> = {}
   556→  for (const row of headerRows.value) {
   557→    const key = row.key.trim()

   559→  }
   560→  httpCfg.value.headers = map
   561→}
