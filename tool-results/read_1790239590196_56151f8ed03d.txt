    94→    94→    public DataSourceMetadata metadata(DataSourceDefinition ds) {
    95→    95→        return switch (ds.getType()) {
    96→    96→            case "FORM" -> {
    97→    97→                List<ColumnConfig> cols;
    98→    98→                try {
    99→    99→                    cols = new ArrayList<>(formDefService.getBusinessColumnsByKey(ds.getFormKey()));
   100→   100→                } catch (BusinessException e) {
   101→   101→                    if (e.getCode() == 404) {
   102→   102→                        // 绑定表单尚未发布（或不存在）：字段元数据降级为空列而非 404，
   103→   103→                        // 不阻断数据源编辑（对齐 NestJS 侧同款降级；表单发布后自动展示）
   104→   104→                        DataSourceMetadata empty = new DataSourceMetadata(new ArrayList<>(), false);
   105→   105→                        empty.setFormKey(ds.getFormKey());
   106→   106→                        yield empty;
   107→   107→                    }
   108→   108→                    throw e;
   109→   109→                }
   110→   110→                FormQueryConfig cfg = FormQueryConfig.parse(ds.getParams(), objectMapper);
   111→   111→                if (cfg.isConfigMode()) {
   112→   112→                    appendJoinColumns(cols, cfg.joins());
   113→   113→                } else if (cfg.isSqlMode()) {
   114→   114→                    appendDeclaredColumns(cols, cfg.columns());
   115→   115→                }
   116→   116→                SortableResolver.resolve(cols);
   117→   117→                DataSourceMetadata m = new DataSourceMetadata(cols, true);
   118→   118→                m.setFormKey(ds.getFormKey());
   119→   119→                yield m;
   120→   120→            }
   121→   121→            case "WORKFLOW" -> {
   122→   122→                List<ColumnConfig> cols;
   123→   123→                try {
   124→   124→                    cols = workflowQueryService.columnsFor(ds.getFormKey());
   125→   125→                } catch (BusinessException e) {
   126→   126→                    if (e.getCode() == 404) {
   127→   127→                        // 绑定表单尚未发布（或不存在）：字段元数据降级为空列（对齐 FORM 分支与 NestJS 侧）
   128→   128→                        DataSourceMetadata empty = new DataSourceMetadata(new ArrayList<>(), false);
   129→   129→                        empty.setFormKey(ds.getFormKey());
   130→   130→                        yield empty;
   131→   131→                    }
   132→   132→                    throw e;
   133→   133→                }
   134→   134→                SortableResolver.resolve(cols);
   135→   135→                DataSourceMetadata m = new DataSourceMetadata(cols, false);
   136→   136→                m.setFormKey(ds.getFormKey());
   137→   137→                yield m;
   138→   138→            }
   139→   139→            case "SYSTEM" -> {
   140→   140→                List<ColumnConfig> cols = "user-tree".equals(ds.getSourceKey())
   141→   141→                        ? copyWithSortableFalse(USER_COLUMNS)
   142→   142→                        : copyWithSortableFalse(DEPT_COLUMNS);
   143→   143→                yield new DataSourceMetadata(cols, false);
   144→   144→            }
   145→   145→            case "API" -> {
   146→   146→                DataSourceMetadata m = apiMetadata(ds);
   147→   147→                DataSourceMetadata copy = new DataSourceMetadata(copyWithSortableFalse(m.getColumns()), m.isWritable());
   148→   148→                copy.setFormKey(m.getFormKey());
   149→   149→                yield copy;
   150→   150→            }
   151→   151→            case "SQL" -> {
   152→   152→                // 字段元数据 = params.columns（单一来源；探测/主表单覆盖/行内编辑由前端写入，无表单列默认合并）
   153→   153→                FormQueryConfig cfg = FormQueryConfig.parse(ds.getParams(), objectMapper);
   154→   154→                List<ColumnConfig> cols = cfg.isConfigMode() ? List.of() : cfg.columns();
   155→   155→                boolean writable = ds.getFormKey() != null && !ds.getFormKey().isBlank();
   156→   156→                DataSourceMetadata m = new DataSourceMetadata(cols, writable);
   157→   157→                if (ds.getFormKey() != null && !ds.getFormKey().isBlank()) {
   158→   158→                    m.setFormKey(ds.getFormKey());
   159→   159→                }
   160→   160→                yield m;
   161→   161→            }
   162→   162→            default -> throw unsupported(ds, "metadata");
   163→   163→        };
   164→   164→    }

   299→   299→    /** 追加 config 模式 JOIN 虚拟列到 metadata；与主表列 key 重复的 virtualKey 跳过（防御性去重）。 */
   300→   300→    private void appendJoinColumns(List<ColumnConfig> cols, List<JoinConfig> joins) {
   301→   301→        Set<String> existing = cols.stream().map(ColumnConfig::getKey).collect(Collectors.toSet());
   302→   302→        for (JoinConfig j : joins) {
   303→   303→            if (j.virtualKey() == null || j.virtualKey().isBlank() || existing.contains(j.virtualKey())) {
   304→   304→                continue;
   305→   305→            }
   306→   306→            List<ColumnConfig> targetCols = resolveJoinTargets(j);
   307→   307→            ColumnConfig c = new ColumnConfig();
   308→   308→            c.setKey(j.virtualKey());
   309→   309→            c.setLabel(j.label() == null || j.label().isBlank() ? j.virtualKey() : j.label());
   310→   310→            c.setColumnType(findJoinColumnType(targetCols, j.joinField()));
   311→   311→            c.setSortable(j.sortable());
   312→   312→            c.setFilterable(j.filterable());
   313→   313→            cols.add(c);
   314→   314→            existing.add(c.getKey());
   315→   315→
   316→   316→            // 目标列为 dataPicker 引用列（pickerConfig 非空）：透传 pickerConfig，前端按引用列渲染显示文本
   317→   317→            ColumnConfig target = findJoinTarget(targetCols, j.joinField());
   318→   318→            if (target != null && target.getPickerConfig() != null && !target.getPickerConfig().isBlank()) {
   319→   319→                c.setPickerConfig(target.getPickerConfig());
   320→   320→            }
   321→   321→            // 目标表含 <joinField>_text 冗余列（dataPicker 两列映射）→ 追加隐藏冗余列，供前端引用渲染取值
   322→   322→            if (findJoinTarget(targetCols, j.joinField() + "_text") != null) {
   323→   323→                ColumnConfig text = new ColumnConfig();
   324→   324→                text.setKey(j.virtualKey() + "_text");
   325→   325→                text.setLabel(c.getLabel() + "（显示）");
   326→   326→                text.setColumnType(findJoinColumnType(targetCols, j.joinField() + "_text"));
   327→   327→                text.setHidden(true);
   328→   328→                text.setSortable(false);
   329→   329→                text.setFilterable(false);
   330→   330→                text.setComponentType("dataPickerText");
   331→   331→                cols.add(text);
   332→   332→                existing.add(text.getKey());
   333→   333→            }
   334→   334→        }
   335→   335→    }

   337→   337→    /** 追加 sql 模式管理员声明列到 metadata；与主表列 key 重复的声明列跳过。 */
   338→   338→    private void appendDeclaredColumns(List<ColumnConfig> cols, List<ColumnConfig> declared) {
   339→   339→        Set<String> existing = cols.stream().map(ColumnConfig::getKey).collect(Collectors.toSet());
   340→   340→        for (ColumnConfig d : declared) {
   341→   341→            if (d.getKey() == null || d.getKey().isBlank() || existing.contains(d.getKey())) {
   342→   342→                continue;
   343→   343→            }
   344→   344→            ColumnConfig c = new ColumnConfig();
   345→   345→            c.setKey(d.getKey());
   346→   346→            c.setLabel(d.getLabel() == null || d.getLabel().isBlank() ? d.getKey() : d.getLabel());
   347→   347→            c.setColumnType(d.getColumnType());
   348→   348→            c.setSortable(d.getSortable());
   349→   349→            c.setFilterable(d.getFilterable());
   350→   350→            cols.add(c);
   351→   351→            existing.add(c.getKey());
   352→   352→        }
   353→   353→    }

   356→   356→    private List<ColumnConfig> resolveJoinTargets(JoinConfig j) {
   357→   357→        try {
   358→   358→            List<ColumnConfig> target = formDefService.getBusinessColumnsByKey(j.targetFormKey());
   359→   359→            return target == null ? List.of() : target;
   360→   360→        } catch (BusinessException ignored) {
   361→   361→            // 目标表单不可解析时回退
   362→   362→            return List.of();
   363→   363→        }
   364→   364→    }

   366→   366→    /** 目标列查找：按 key 匹配；查不到返回 null。 */
   367→   367→    private static ColumnConfig findJoinTarget(List<ColumnConfig> cols, String key) {
   368→   368→        if (cols == null) {
   369→   369→            return null;
   370→   370→        }
   371→   371→        for (ColumnConfig c : cols) {
   372→   372→            if (key.equals(c.getKey())) {
   373→   373→                return c;
   374→   374→            }
   375→   375→        }
   376→   376→        return null;
   377→   377→    }

   618→   618→    private JsonNode walkPath(JsonNode root, String path) {
   619→   619→        JsonNode cur = root;
   620→   620→        for (String part : path.split("\\.")) {
   621→   621→            if (cur == null || cur.isMissingNode() || !cur.isObject()) return null;
   622→   622→            cur = cur.get(part);
   623→   623→        }
   624→   624→        return cur;

   627→   627→    private BizDataPageVO toPageVO(Object raw, Map<String, Object> op, BizDataQueryRequest req) {
   628→   628→        int normalizedPage = Math.max(req.getPage(), 1);
   629→   629→        JsonNode root = toJsonNode(raw);
   630→   630→        JsonNode recordsNode = root;
   631→   631→        String parse = str(op.get("parse"));
   632→   632→        if (parse != null && !parse.isBlank()) {
   633→   633→            recordsNode = walkPath(root, parse);
   634→   634→        }
   635→   635→        if (recordsNode == null || !recordsNode.isArray()) {
   636→   636→            return new BizDataPageVO(List.of(), 0, normalizedPage, req.getSize());
   637→   637→        }
   638→   638→        List<BizDataVO> records = new ArrayList<>();
   639→   639→        for (JsonNode item : recordsNode) {
   640→   640→            Map<String, Object> data = objectMapper.convertValue(item, Map.class);
   641→   641→            Object id = item.has("id") ? item.get("id").asText() : null;
   642→   642→            records.add(new BizDataVO(id == null ? "" : String.valueOf(id), data, null, null, null));
   643→   643→        }
   644→   644→        long total = records.size();
   645→   645→        String totalParse = str(op.get("totalParse"));
   646→   646→        if (totalParse != null && !totalParse.isBlank()) {
   647→   647→            JsonNode totalNode = walkPath(root, totalParse);
   648→   648→            if (totalNode != null && totalNode.isNumber()) total = totalNode.asLong();
   649→   649→        }
   650→   650→        return new BizDataPageVO(records, total, normalizedPage, req.getSize());
   651→   651→    }
