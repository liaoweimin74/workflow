   188→/** procDefId（Node 契约 `{key}:{version}:{rowId}`）解析 */
   189→function parseProcDefId(id: string): { key: string; version: number; rowId: number } | null {
   190→  const i1 = id.lastIndexOf(':');
   191→  if (i1 <= 0) return null;
   192→  const rowId = Number(id.slice(i1 + 1));
   193→  if (!Number.isInteger(rowId) || rowId <= 0) return null;
   194→  const i2 = id.lastIndexOf(':', i1 - 1);
   195→  if (i2 < 0) return null;
   196→  const version = Number(id.slice(i2 + 1, i1));
   197→  if (!Number.isInteger(version) || version <= 0) return null;
   198→  return { key: id.slice(0, i2), version, rowId };
   199→}

   259→/** ProcessDefinitionController#extractFormConfig 移植 */
   260→function extractFormConfig(configJson: string): FormConfig | null {
   261→  let json: Record<string, unknown>;
   262→  try {
   263→    json = JSON.parse(configJson) as Record<string, unknown>;
   264→  } catch {
   265→    return null;
   266→  }
   267→  if (json == null || typeof json !== 'object') return null;
   268→  const form = json['form'];
   269→  if (form == null || typeof form !== 'object' || Array.isArray(form)) return null;
   270→  const f = form as Record<string, unknown>;
   271→  if (f['formDefId'] == null) return null;
   272→  const val = typeof f['formDefId'] === 'string' ? f['formDefId'] : String(f['formDefId']);
   273→  if (val === '') return null;
   274→  const permNode = f['fieldPermissions'];
   275→  let permissions: Record<string, string> | null = null;
   276→  if (permNode != null && typeof permNode === 'object' && !Array.isArray(permNode)) {
   277→    permissions = {};
   278→    for (const [k, v] of Object.entries(permNode as Record<string, unknown>)) {
   279→      permissions[k] = typeof v === 'string' ? v : String(v);
   280→    }
   281→    permissions = javaHashMapOrdered(permissions) as Record<string, string>;
   282→  }
   283→  return { formDefId: val, fieldPermissions: permissions };
   284→}

   286→/**
   287→ * ProcessDefinitionController#resolveFormDefIds 移植：
   288→ * 优先级 发起人节点表单 > 流程默认表单（表单与字段权限同层取，不跨层合并）。
   289→ * 返回插入序：formDefId, fieldPermissions, initiatorFormDefId, processFormDefId。
   290→ */
   291→function resolveFormDefIds(procDefId: string, deployRow: Row | null): Record<string, unknown> {
   292→  const out: Record<string, unknown> = {
   293→    formDefId: null,
   294→    fieldPermissions: null,
   295→    initiatorFormDefId: null,
   296→    processFormDefId: null,
   297→  };
   298→  const configs = snapshotConfigRows(procDefId);
   299→  let processCfg: FormConfig | null = null;
   300→  let initiatorCfg: FormConfig | null = null;
   301→
   302→  for (const nc of configs) {
   303→    try {
   304→      const cfg = extractFormConfig(String(nc['config_json'] ?? ''));
   305→      if (cfg == null) continue;
   306→      if (String(nc['node_id']) === '__PROCESS__') {
   307→        processCfg = cfg;
   308→      }
   309→    } catch {
   310→      // 忽略解析错误
   311→    }
   312→  }
   313→
   314→  const xml = deployedXmlOf(procDefId, deployRow);
   315→  const initiatorNodeId = xml == null ? null : extractInitiatorNodeId(parseBpmnToIr(xml));
   316→  if (initiatorNodeId != null) {
   317→    for (const nc of configs) {
   318→      if (String(nc['node_id']) === initiatorNodeId) {
   319→        try {
   320→          initiatorCfg = extractFormConfig(String(nc['config_json'] ?? ''));
   321→        } catch {
   322→          // 忽略解析错误
   323→        }
   324→        break;
   325→      }
   326→    }
   327→  }
   328→
   329→  const effective = initiatorCfg != null ? initiatorCfg : processCfg;
   330→  out['formDefId'] = effective != null ? effective.formDefId : null;
   331→  out['fieldPermissions'] = effective != null ? effective.fieldPermissions : null;
   332→  out['initiatorFormDefId'] = initiatorCfg != null ? initiatorCfg.formDefId : null;
   333→  out['processFormDefId'] = processCfg != null ? processCfg.formDefId : null;
   334→  return out;
   335→}

   339→function isSameAsLastDeployment(draft: Row, effectiveBpmnXml: string, nodeConfigMap: Record<string, string>): boolean {
   340→  if (trimToNull(draft['deployed_xml'] == null ? null : String(draft['deployed_xml'])) !== trimToNull(effectiveBpmnXml)) {
   341→    return false; // XML 变化
   342→  }
   343→  const procDefId = draft['process_definition_id'] == null ? null : String(draft['process_definition_id']);
   344→  if (procDefId == null) {
   345→    return true; // 无历史部署记录且 XML 相同，视为未变化
   346→  }
   347→  const snapshotMap = configMap(all(`SELECT "node_id", "config_json" FROM wf_node_config WHERE "process_def_id" = ? AND "process_definition_id" = ?`, [
   348→    String(draft['id']),
   349→    procDefId,
   350→  ]));
   351→  return mapEquals(snapshotMap, nodeConfigMap);
   352→}

   354→function doDeploy(tenant: string, draftId: string): Record<string, unknown> {
   355→  const draft = draftById(draftId, tenant);
   356→
   357→  // 当前编辑中配置 + MultiInstanceBpmnRewriter 改写（会签/或签 → MI；单实例 → assignee/candidateUsers）
   358→  const nodeConfigMap = configMap(currentConfigRows(draftId));
   359→  let effectiveBpmnXml = rewriteMultiInstance(draft['bpmn_xml'] == null ? null : String(draft['bpmn_xml']), nodeConfigMap);
   360→  // 注入 StartEvent/EndEvent 默认名称
   361→  effectiveBpmnXml = injectEventNames(effectiveBpmnXml);
   362→
   363→  // 变化检测：hash 优先，历史数据（deployed_config_hash 为空）降级比较
   364→  const currentHash = deployHash(effectiveBpmnXml, nodeConfigMap);
   365→  const storedHash = trimToNull(draft['deployed_config_hash'] == null ? null : String(draft['deployed_config_hash']));
   366→  const unchanged = storedHash != null ? storedHash === currentHash : isSameAsLastDeployment(draft, effectiveBpmnXml, nodeConfigMap);
   367→  if (unchanged) {
   368→    throw new BusinessException('流程数据未变化，无需部署', 400);
   369→  }
   370→
   371→  // Flowable 引擎校验/解析对位：XML 解析失败 → 部署失败（消息含 BPMN → 流程定义不完整）
   372→  const key = String(draft['process_key']);
   373→  let ir;
   374→  try {
   375→    ir = parseBpmnToIr(effectiveBpmnXml, { nodeConfigs: nodeConfigMap });
   376→  } catch (e) {
   377→    const msg = e instanceof Error ? e.message : String(e);
   378→    if (msg.includes('BPMN')) {
   379→      throw new BusinessException(`流程定义不完整：${msg}`, 400);
   380→    }
   381→    throw new BusinessException(`部署失败：${msg || '未知错误'}`, 400);
   382→  }
   383→
   384→  const now = nowText();
   385→  const resourceName = `${key}.bpmn20.xml`;
   386→  const diagramResourceName = hasDiagramInfo(effectiveBpmnXml) ? `${key}.${key}.png` : null;
   387→  // Flowable ProcessDefinition.category = BPMN targetNamespace（非 deployment category）
   388→  const category = extractTargetNamespace(effectiveBpmnXml);
   389→
   390→  run(
   391→    `INSERT INTO WF_PROC_DEPLOY (NAME, KEY_, CATEGORY, VERSION, DEPLOY_TIME, RESOURCE_NAME, DIAGRAM_RESOURCE_NAME, TENANT_ID, SUSPENSION_STATE, BPMN_XML, CONFIG_JSON)
   392→     VALUES (?, ?, ?, ?, ?, ?, ?, ?, 1, ?, ?)`,
   393→    [String(draft['name']), key, category, 0, now, resourceName, diagramResourceName, tenant, effectiveBpmnXml, JSON.stringify(ir)],
   394→  );
   395→  const rowId = Number(one(`SELECT last_insert_rowid() AS ID`)!['ID']);
   396→
   397→  let procDefId: string | null = null;
   398→  let version = Number(draft['version'] ?? 0);
   399→
   400→  if (ir.procKey != null) {
   401→    // version = 同 key+租户 已部署最大版本 + 1（Flowable 语义，首版 = 1）
