   330→/** 扁平选项 label 列表（select 类查询：label=value；树/级联/穿梭从 props.data/props.options 取） */
   331→function formOptionLabelItems(rule: any): { label: string; value: string }[] {
   332→  const list: any[] = rule?.options ?? rule?.props?.options ?? rule?.props?.data ?? []
   333→  const labels: string[] = []
   334→  const collect = (items: any[]) => {
   335→    for (const it of items) {
   336→      if (it && it.label !== undefined) labels.push(String(it.label))
   337→      if (Array.isArray(it.children)) collect(it.children)
   338→    }
   339→  }
   340→  collect(list)
   341→  return labels.map((l) => ({ label: l, value: l }))
   342→}

   420→/** JSON 数组 → 逗号拼接；非数组（旧逗号串/字符串）原样返回 */
   421→function formatArrayValue(v: unknown): unknown {
   422→  if (Array.isArray(v)) return v.join(', ')
   423→  if (typeof v === 'string') {
   424→    try {
   425→      const parsed = JSON.parse(v)
   426→      if (Array.isArray(parsed)) return parsed.join(', ')
   427→    } catch {
   428→      // 旧逗号串或普通字符串，原样
   429→    }
   430→    return v
   431→  }
   432→  return v

   675→/** visible 表达式求值（$row.xxx → 行值，$param.xxx → 路由参数） */
   676→function evalVisible(expr: string, row: any): boolean {
   677→  try {
   678→    const resolved = expr
   679→      .replace(/\$row\.([\w]+)/g, (_: string, k: string) => {
   680→        const v = getNestedValue(row, k)
   681→        return v === null || v === undefined ? 'undefined' : JSON.stringify(v)
   682→      })
   683→      .replace(/\$param\.([\w]+)/g, (_: string, k: string) => {
   684→        const v = (window as any).__routeParams?.[k]
   685→        return v === null || v === undefined ? 'undefined' : JSON.stringify(v)
   686→      })
   687→    if (/[a-zA-Z_$]\s*\(/.test(resolved)) return false
   688→    return !!Function('"use strict"; return (' + resolved + ')')()
   689→  } catch {
   690→    return true
   691→  }
   692→}

   694→function getNestedValue(obj: any, path: string): any {
   695→  return path.split('.').reduce((o, k) => o?.[k], obj)
   696→}

   731→const fetchApi = async (params: { page: number; size: number; [key: string]: any }) => {
   732→  const dsId = resolvedRefId.value
   733→  if (!dsId) return { rows: [], total: 0 }
   734→
   735→  // 组装筛选条件：静态筛选（DsBindingConfigDialog 配置）+ 动作总线 set-filter + 搜索栏条件
   736→  const filterConditions: any[] = []
   737→  // 1. 静态筛选（组件级配置，始终生效；从 tableFilterStore 按页面内 dataSourceId 读取）
   738→  let staticFilter: any = props.dataSourceId ? tableFilterStore[props.dataSourceId] : undefined
   739→  if (staticFilter?.conditions) {
   740→    for (const c of staticFilter.conditions) {
   741→      if (c.column) {
   742→        // DsBindingConfigDialog 输出 fixedValue，兼容 value
   743→        const val = c.fixedValue ?? c.value ?? ''
   744→        filterConditions.push({ column: c.column, op: c.op || 'eq', value: val })
   745→      }
   746→    }
   747→  }
   748→  // 2. 动作总线 set-filter 注入的条件
   749→  if (currentFilter.value) {
   750→    for (const [column, value] of Object.entries(currentFilter.value)) {
   751→      if (value === '' || value === null || value === undefined) continue
   752→      filterConditions.push({ column, op: 'eq', value })
   753→    }
   754→  }
   755→  // 2.5 表头筛选条件（Task ⑤：漏斗按列筛选；多选→in / 区间→range；与其它条件 AND 合并）
   756→  for (const [hfKey, hfValue] of Object.entries(headerFilters.value)) {
   757→    const cond = buildHeaderFilterCondition(hfKey, hfValue)
   758→    if (cond) filterConditions.push(cond)
   759→  }
   760→  // 3. 搜索栏条件（数组值组件主列 → 映射 <key>_text 显示列查询 label；级联路径 label 数组 join('/') 匹配 _text 全路径）
   761→  for (const field of resolvedSearchFields.value) {
   762→    const raw = params[field.prop]
   763→    if (raw === '' || raw === null || raw === undefined) continue
   764→    const v = Array.isArray(raw) ? raw.join('/') : raw
   765→    filterConditions.push({ column: resolveSearchColumn(field.prop), op: 'like', value: v })
   766→  }
   767→  const query: Record<string, any> = props.designMode
   768→    // 设计态预览固定取首页且最多 10 条（含"不分页"场景，避免 size:-1 拉全量）
   769→    ? { page: 1, size: Math.min(params.size || 10, 10) }
   770→    : props.pagination === false
   771→      // 不分页：请求全部数据（后端 size<=0 跳过 LIMIT），保留排序
   772→      ? { size: -1 }
   773→          : { page: Math.max(1, params.page), size: params.size }
   774→  // 排序状态透传（SearchTable 内部维护，服务器端排序）
   775→  if (params.sort) query.sort = params.sort
   776→  if (params.order) query.order = params.order
   777→  if (filterConditions.length > 0) {
   778→    query.filter = JSON.stringify({ logic: 'AND', conditions: filterConditions })
   779→  }
   780→  const res: any = await dataSourceApi.queryData(dsId, query)
   781→  const rows = (res?.data?.records || []).map((r: any) => ({ ...(r.data || {}), id: r.id, version: r.version }))
   782→  records.value = rows
   783→  emit('loaded', rows)
   784→  return { rows, total: res?.data?.total || 0 }
   785→}

   806→function handleActionClick(btn: any, row: any) {
   807→  // 有绑定事件 → 优先执行事件链（内建/自定义都一样，对齐 PageRenderer）
   808→  if (btn.events?.length) {
   809→    for (const ev of btn.events) {
   810→      for (const action of ev.actions || []) {
   811→        void dispatchButtonAction(action, row)
   812→      }
   813→    }
   814→    return
   815→  }
   816→  // 表格-容器联动触发器：派发到页面动作总线（动作链消费时跳过默认行为，避免双弹窗）
   817→  const linkageTrigger = LINKAGE_TRIGGERS[btn.key]
   818→  if (linkageTrigger && actionBus) {
   819→    const consumed = actionBus.dispatch(
   820→      linkageTrigger,
   821→      row ? { node: row, row, source: props.dataSourceId } : { source: props.dataSourceId },
   822→    )
   823→    if (consumed) return
   824→  }
   825→  // 无事件 → 默认行为
   826→  // 存在同数据源的数据容器 → 打开容器表单（自动关联，无需动作链）；否则本地默认表单
   827→  const dsId = props.dataSourceId
   828→  const linked = actionBus?.hasLinkedContainer?.(dsId)
   829→  if (btn.key === 'create') {
   830→    if (linked && dsId) actionBus?.openLinkedContainer?.(dsId, 'create', null)
   831→    else openLocalForm('create')
   832→  } else if (btn.key === 'edit') {
   833→    if (linked && dsId) actionBus?.openLinkedContainer?.(dsId, 'edit', row)
   834→    else openLocalForm('edit', row)
   835→  } else if (btn.key === 'delete') {
   836→    handleDelete(row)
   837→  } else if (btn.key === 'view') {
   838→    if (linked && dsId) actionBus?.openLinkedContainer?.(dsId, 'view', row)
   839→    else openDetail(row)
   840→  }
   841→}

   844→/** 模板变量替换：$row.字段（当前行）/ $param.参数（路由参数） */
   845→function resolveTemplate(tpl: string, row: any): string {
   846→  return tpl
   847→    .replace(/\$row\.([\w]+)/g, (_: string, k: string) => {
   848→      const v = row?.[k]
   849→      return v === null || v === undefined ? '' : String(v)
   850→    })
   851→    .replace(/\$param\.([\w]+)/g, (_: string, k: string) => (route.query?.[k] == null ? '' : String(route.query[k])))
   852→}

   855→function resolveParams(params: { key: string; value: string }[], row: any): Record<string, any> {
   856→  const out: Record<string, any> = {}
   857→  for (const p of params || []) {
   858→    out[p.key] = resolveTemplate(p.value, row)

   864→function exportData() {
   865→  const rows = records.value
   866→  const blob = new Blob([JSON.stringify(rows, null, 2)], { type: 'application/json' })
   867→  const url = URL.createObjectURL(blob)
   868→  const a = document.createElement('a')
   869→  a.href = url
   870→  a.download = `${props.pageKey || 'page'}-data.json`
   871→  a.click()
   872→  URL.revokeObjectURL(url)
   873→}

   875→/** 执行单个事件链动作（ViewDesigner 风格：type/params） */
   876→async function dispatchButtonAction(action: { type: string; params?: { key: string; value: string }[] }, row: any) {
   877→  const resolved = resolveParams(action.params || [], row)
   878→  switch (action.type) {
   879→    case 'open-detail':
   880→      openDetail(row)
   881→      break
   882→    case 'open-link':
   883→      if (resolved.url) router.push(resolved.url)
   884→      break
   885→    case 'open-create':
   886→      openLocalForm('create')
   887→      break
   888→    case 'edit':
   889→      if (row) openLocalForm('edit', row)
   890→      break
   891→    case 'delete':
   892→      if (row) await handleDelete(row)
   893→      break
   894→    case 'refresh':
   895→      tableRef.value?.fetchList()
   896→      break
   897→    case 'export':
   898→      exportData()
   899→      break
   900→    case 'message':
   901→      ElMessage({
   902→        message: resolved.text || resolved.message || '提示',
   903→        type: (resolved.type as any) || 'info',
   904→      })
   905→      break
   906→    case 'set-filter':
   907→      setFilter(resolved)
   908→      break
   909→    case 'set-page': {
   910→      const page = parseInt(resolved.page, 10)
   911→      if (!isNaN(page) && page > 0) {
   912→        tableRef.value?.setQuery({ page }, false)
   913→      }
   914→      break
   915→    }
   916→    case 'set-sort':
   917→      tableRef.value?.sort(resolved.field || resolved.prop || '', resolved.order || 'ascending')
   918→      break
   919→    case 'clear-selection':
   920→      tableRef.value?.clearSelection()
   921→      break
   922→    case 'script': {
   923→      const source: string = resolved.source || ''
   924→      if (!source) {
   925→        console.warn('[page-table] 脚本动作缺少 source 参数')
   926→        break
   927→      }
   928→      if (!isScriptEventEnabled()) {
   929→        console.warn('[page-table] 脚本事件未启用（设置 VITE_PAGE_SCRIPT_ENABLED=true 开启）')
   930→        break
   931→      }
   932→      await executeScript(source, {
   933→        row,
   934→        params: route.query || {},
   935→        selectedRows: row ? [row] : [],
   936→        ds: {
   937→          query: (filter?: Record<string, any>) => {
   938→            if (filter) setFilter(filter)
   939→            else refresh()
   940→          },
   941→          detail: (id: string) => (resolvedRefId.value ? dataSourceApi.getData(resolvedRefId.value, id) : Promise.reject(new Error('数据源未绑定'))),
   942→          create: (data: Record<string, unknown>) =>
   943→            resolvedRefId.value ? dataSourceApi.createData(resolvedRefId.value, data) : Promise.reject(new Error('数据源未绑定')),
   944→          update: (id: string, data: Record<string, unknown>) =>
   945→            resolvedRefId.value ? dataSourceApi.updateData(resolvedRefId.value, id, data, row?.version) : Promise.reject(new Error('数据源未绑定')),
   946→          remove: (id: string) =>
   947→            resolvedRefId.value ? dataSourceApi.deleteData(resolvedRefId.value, id) : Promise.reject(new Error('数据源未绑定')),
   948→        },
   949→        api: { pageKey: props.pageKey || '', dataSourceId: resolvedRefId.value || '' },
   950→        actions: {
   951→          refresh: () => refresh(),
   952→          openDetail: () => openDetail(row),
   953→          openCreate: () => openLocalForm('create'),
   954→          openEdit: () => (row ? openLocalForm('edit', row) : undefined),
   955→          remove: (id: string) =>
   956→            resolvedRefId.value ? dataSourceApi.deleteData(resolvedRefId.value, id) : Promise.reject(new Error('数据源未绑定')),
   957→        },
   958→        $: { message: (msg: string, type = 'info') => ElMessage({ message: msg, type: type as any }) },
   959→      })
   960→      break
   961→    }
   962→    default:
   963→      console.warn('[page-table] 未知动作类型:', action.type)
   964→  }
   965→}

   967→/** 详情：popup → 弹窗；drawer/inline → 本地表单只读查看（按 formMode 展示） */

   980→/** 删除：确认后调用数据源删除 API，并刷新列表 */
   981→async function handleDelete(row: any) {
   982→  if (!resolvedRefId.value) return
   983→  try {
   984→    await ElMessageBox.confirm('确定要删除该记录吗？', '删除确认', { type: 'warning' })
   985→  } catch {
   986→    return
   987→  }
   988→  try {
   989→    await dataSourceApi.deleteData(resolvedRefId.value, row.id)
   990→    ElMessage.success('删除成功')
   991→    tableRef.value?.fetchList()
   992→  } catch {
   993→    // 拦截器已弹错误
   994→  }
   995→}

   997→// ==================== 批量操作（Task ⑤） ====================
   998→const batchEnabled = computed(() => featureEnabled(props.batch))
   999→const batchDeleteOn = computed(() => batchDeleteEnabled(props.batch))
  1000→const batchDeleting = ref(false)
  1001→/** 已选行集合（批量动作挂载点：经 defineExpose 暴露，事件链/脚本/外部可读取） */
  1002→const selectedRows = ref<any[]>([])

  1004→/** 批量删除：确认后逐行调用单行删除同款 API（dataSourceApi.deleteData），完成后清空选择并刷新 */
  1005→async function handleBatchDelete() {
  1006→  const rows = selectedRows.value
  1007→  if (!rows.length || !resolvedRefId.value) return
  1008→  try {
  1009→    await ElMessageBox.confirm(`确定删除选中的 ${rows.length} 条记录吗？`, '批量删除', { type: 'warning' })
  1010→  } catch {
  1011→    return
  1012→  }
  1013→  batchDeleting.value = true
  1014→  try {
  1015→    const ids = selectedRowIds(rows)
  1016→    await Promise.all(ids.map((id: any) => dataSourceApi.deleteData(resolvedRefId.value, id)))
  1017→    ElMessage.success(`已删除 ${ids.length} 条记录`)
  1018→    selectedRows.value = []
  1019→    tableRef.value?.clearSelection()
  1020→    tableRef.value?.fetchList()
  1021→  } catch {
  1022→    // 拦截器已弹错误
  1023→  } finally {
  1024→    batchDeleting.value = false
  1025→  }
  1026→}

  1034→/** 触发 viewEvents 中匹配触发器的动作（动作由本组件自执行，对齐 PageRenderer triggerEvents） */
  1035→function triggerViewEvents(trigger: string, target: string, ctx: { row?: any; column?: any; selectedRows?: any[]; prop?: string; order?: string }) {
  1036→  for (const ev of props.viewEvents || []) {
  1037→    if (ev.trigger !== trigger) continue
  1038→    if (ev.target && ev.target !== target) continue
  1039→    for (const action of ev.actions || []) {
  1040→      void dispatchButtonAction(action, ctx.row)
  1041→    }
  1042→  }
