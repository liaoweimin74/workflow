    52→    52→/**
    53→    53→ * 复制列并强制 `sortable=false`（对齐 Java `copyWithSortableFalse`）。
    54→    54→ *
    55→    55→ * ⚠️ **只拷 key/label/columnType**，`length` 等其余字段保持 `newColumnConfig()` 的默认值
    56→    56→ * （即 null）—— golden 实测 SYSTEM 元数据的列上 `length: null`，尽管常量里写了 64/128。
    57→    57→ * 「顺手把 length 也拷过来」会立刻与 Java 分叉。
    58→    58→ */
    59→    59→function copyWithSortableFalse(src: ColumnConfig[]): ColumnConfig[] {
    60→    60→  return src.map((column) => ({
    61→    61→    ...newColumnConfig(),
    62→    62→    key: column.key,
    63→    63→    label: column.label,
    64→    64→    columnType: column.columnType,
    65→    65→    sortable: false,
    66→    66→  }))
    67→    67→}
    68→    68→
    69→    69→/**
    70→    70→ * 追加 sql 模式管理员声明列到 metadata；与主表列 key 重复的声明列跳过
    71→    71→ * （对齐 Java `appendDeclaredColumns`）。
    72→    72→ *
    73→    73→ * `label` 空白时回落 `key`。同样只拷 5 个字段。
    74→    74→ */
    75→    75→function appendDeclaredColumns(columns: ColumnConfig[], declared: ColumnConfig[]): void {
    76→    76→  const existing = new Set(columns.map((column) => column.key))
    77→    77→  for (const item of declared) {
    78→    78→    const key = item.key
    79→    79→    if (key === null || key.trim() === '' || existing.has(key)) continue
    80→    80→    columns.push({
    81→    81→      ...newColumnConfig(),
    82→    82→      key,
    83→    83→      label: item.label === null || item.label.trim() === '' ? key : item.label,
    84→    84→      columnType: item.columnType,
    85→    85→      sortable: item.sortable,
    86→    86→      filterable: item.filterable,
    87→    87→    })
    88→    88→    existing.add(key)
    89→    89→  }
    90→    90→}

    92→    92→/** 本适配器覆盖的数据源类型（与 Java `UnifiedDataSourceAdapter.supports` 完全一致）。 */
    93→    93→const SUPPORTED_TYPES = ['FORM', 'SYSTEM', 'API', 'WORKFLOW', 'SQL']

    95→    95→/**
    96→    96→ * 统一数据源适配器（对齐 Java `com.workflow.engine.datasource.UnifiedDataSourceAdapter`）。
    97→    97→ *
    98→    98→ * **5 种类型的读路径全部已迁移且在契约网内**（FORM / SYSTEM / API / WORKFLOW / SQL）：
    99→    99→ * 「数据源元数据与取数」「SYSTEM 数据源读路径」「API 数据源读路径」「WORKFLOW 数据源读路径」
   100→   100→ * 「SQL 数据源（管理员模板）」「FORM 数据源 config 与 sql 查询模式」六个场景覆盖。
   101→   101→ *
   102→   102→ * `supports()` 刻意对全部 5 种类型都返回 true —— **选路逻辑必须与 Java 一致**，
   103→   103→ * 否则一个 SYSTEM 数据源会得到「数据源类型未启用」这种**错误的**错误信息，
   104→   104→ * 掩盖真正的原因。真正走到「与 Java 不一致的类型」时由 `notMigrated()` 显式抛错

   121→   121→  async metadata(dataSource: DataSourceRef): Promise<DataSourceMetadata> {
   122→   122→    if (dataSource.type === 'SYSTEM') {
   123→   123→      // ⚠️ SYSTEM 的列是**常量**（Java 的 DEPT_COLUMNS / USER_COLUMNS），不查库；
   124→   124→      //    且 `writable=false`（系统数据源只读）。
   125→   125→      //    ⚠️ `copyWithSortableFalse` 只拷 key/label/columnType —— **不拷 length**，
   126→   126→      //    golden 实测列上 `length: null`（尽管常量里写了 64/128）。
   127→   127→      const columns = copyWithSortableFalse(
   128→   128→        dataSource.sourceKey === 'user-tree' ? USER_COLUMNS : DEPT_COLUMNS,
   129→   129→      )
   130→   130→      return { columns, writable: false, formKey: null }
   131→   131→    }
   132→   132→    if (dataSource.type === 'WORKFLOW') {
   133→   133→      // 列来自最新 PUBLISHED 表单的 **schema**（WORKFLOW 表单的 columnConfig 为空），
   134→   134→      // 只读，formKey 原样带出。表单未发布时降级为空列（对齐 FORM 分支与 Java 侧）。
   135→   135→      let columns: ColumnConfig[]
   136→   136→      try {
   137→   137→        columns = await this.workflowQuery.columnsFor(requireFormKey(dataSource, 'metadata'))
   138→   138→      } catch (e) {
   139→   139→        if (e instanceof BusinessException && e.code === 404) {
   140→   140→          return { columns: [], writable: false, formKey: dataSource.formKey }
   141→   141→        }
   142→   142→        throw e
   143→   143→      }
   144→   144→      resolveSortable(columns)
   145→   145→      return { columns, writable: false, formKey: dataSource.formKey }
   146→   146→    }
   147→   147→    if (dataSource.type === 'API') {
   148→   148→      return this.apiMetadata(dataSource, this.parseApiParams(dataSource))
   149→   149→    }
   150→   150→    if (dataSource.type === 'SQL') {
   151→   151→      // 字段元数据 = params.columns（**单一来源**；探测/主表单覆盖/行内编辑由前端写入，
   152→   152→      // 不与绑定表单的列做默认合并）。config 模式下声明列来自 joins，故为**空列表**。
   153→   153→      const config = parseFormQueryConfig(dataSource.params, (message) => new BusinessException(400, message))
   154→   154→      const columns = isConfigMode(config) ? [] : config.columns
   155→   155→      const writable = dataSource.formKey !== null && dataSource.formKey.trim() !== ''
   156→   156→      return {
   157→   157→        columns,
   158→   158→        writable,
   159→   159→        formKey: writable ? dataSource.formKey : null,
   160→   160→      }
   161→   161→    }
   162→   162→    if (dataSource.type !== 'FORM') {
   163→   163→      throw notMigrated(dataSource, 'metadata')
   164→   164→    }
   165→   165→    const formKey = requireFormKey(dataSource, 'metadata')
   166→   166→    let columns: ColumnConfig[]
   167→   167→    try {
   168→   168→      columns = cloneColumns(await this.bizDataService.loadColumns(formKey))
   169→   169→    } catch (e) {
   170→   170→      // 绑定表单尚未发布（或不存在）：字段元数据降级为空列而非 404，
   171→   171→      // 不阻断数据源编辑（对齐 Java 侧同款降级；表单发布后此处自动展示列）
   172→   172→      if (e instanceof BusinessException && e.code === 404) {
   173→   173→        return { columns: [], writable: false, formKey }
   174→   174→      }
   175→   175→      throw e
   176→   176→    }
   177→   177→
   178→   178→    // config / sql 两种 queryMode 会往列清单里追加 JOIN 虚拟列或声明列
   179→   179→    // （对齐 Java `metadata` 的 FORM 分支：config → appendJoinColumns，sql → appendDeclaredColumns）
   180→   180→    const config = parseFormQueryConfig(dataSource.params, (message) => new BusinessException(400, message))
   181→   181→    if (isConfigMode(config)) {
   182→   182→      await this.appendJoinColumns(columns, config.joins)
   183→   183→    } else if (isSqlMode(config)) {
   184→   184→      appendDeclaredColumns(columns, config.columns)
   185→   185→    }
   186→   186→
   187→   187→    resolveSortable(columns)
   188→   188→    return { columns, writable: true, formKey }
   189→   189→  }

   191→   191→  async query(dataSource: DataSourceRef, req: BizDataQueryRequest): Promise<BizDataPageVO> {
   192→   192→    if (dataSource.type === 'SYSTEM') {
   193→   193→      // 先过 router（Java 的 SYSTEM 分支里有 `router.resolve(ds, "list")`），再取数
   194→   194→      this.router.resolve(dataSource, 'list')
   195→   195→      return this.systemQuery(dataSource, req)
   196→   196→    }
   197→   197→    if (dataSource.type === 'WORKFLOW') {
   198→   198→      return this.workflowQuery.query(String(dataSource.formKey), req)
   199→   199→    }
   200→   200→    if (dataSource.type === 'API') {
   201→   201→      return this.apiQuery(dataSource, req)
   202→   202→    }
   203→   203→    if (dataSource.type === 'SQL') {
   204→   204→      this.router.resolve(dataSource, 'list')
   205→   205→      const config = parseFormQueryConfig(dataSource.params, (message) => new BusinessException(400, message))
   206→   206→      if (isVisualMode(config) || isSqlMode(config)) {
   207→   207→        // 绕过表单 covering handler：SQL 数据源执行管理员显式 SQL，不被绑定表单业务定制劫持
   208→   208→        return this.bizDataService.querySqlRaw(dataSource.formKey, req, config)
   209→   209→      }
   210→   210→      throw new BusinessException(400, 'SQL 数据源缺少查询配置')
   211→   211→    }
   212→   212→    if (dataSource.type !== 'FORM') {
   213→   213→      throw notMigrated(dataSource, 'query')
   214→   214→    }
   215→   215→    const formKey = requireFormKey(dataSource, 'query')
   216→   216→    const config = parseFormQueryConfig(dataSource.params, (message) => new BusinessException(400, message))
   217→   217→    if (isConfigMode(config)) {
   218→   218→      return this.bizDataService.queryJoin(formKey, req, config.joins)
   219→   219→    }
   220→   220→    if (isSqlMode(config)) {
   221→   221→      return this.bizDataService.querySql(formKey, req, config)
   222→   222→    }
   223→   223→    // visual 模式在 FORM 类型上**不**生效（Java 只判 config/sql），回退单表查询
   224→   224→    return this.bizDataService.query(formKey, req)
   225→   225→  }

   227→   227→  async get(dataSource: DataSourceRef, id: string): Promise<BizDataVO> {
   228→   228→    if (dataSource.type === 'SYSTEM') {
   229→   229→      // Java `systemGet`：取**全部行**（忽略分页）再按 id 找，找不到 → 业务 404
   230→   230→      const all = await this.systemQuery(dataSource, {
   231→   231→        filter: null,
   232→   232→        keyword: null,
   233→   233→        keywordColumn: null,
   234→   234→        sort: null,
   235→   235→        order: null,
   236→   236→        params: null,
   237→   237→        page: 1,
   238→   238→        size: 20,
   239→   239→      })
   240→   240→      const row = all.records.find((r) => r.id === id)
   241→   241→      if (row === undefined) {
   242→   242→        throw new BusinessException(404, `系统数据不存在: ${id}`)
   243→   243→      }
   244→   244→      return row
   245→   245→    }
   246→   246→    if (dataSource.type === 'WORKFLOW') {
   247→   247→      return this.workflowQuery.getById(String(dataSource.formKey), id)
   248→   248→    }
   249→   249→    if (dataSource.type === 'API') {
   250→   250→      return this.apiGet(dataSource, id)
   251→   251→    }
   252→   252→    if (dataSource.type === 'SQL') {
   253→   253→      // SQL 类型行级操作**始终走绑定表单**（`get` 未绑定 → 400）
   254→   254→      return this.bizDataService.getById(requireFormKey(dataSource, 'get'), id)
   255→   255→    }
   256→   256→    if (dataSource.type !== 'FORM') {
   257→   257→      throw notMigrated(dataSource, 'get')
   258→   258→    }
   259→   259→    const formKey = requireFormKey(dataSource, 'get')
   260→   260→    return this.bizDataService.getById(formKey, id)
   261→   261→  }

   263→   263→  // ==================== SYSTEM 取数（对齐 Java 的 systemQuery 两个分支） ====================
   264→   264→
   265→   265→  /**
   266→   266→   * 系统数据源取数。
   267→   267→   *
   268→   268→   * ⚠️ 两个分支的**分页语义完全不同**（golden 实测）：
   269→   269→   *   - `user-tree`：走用户分页（keyword + page/size 都生效），`{total, page, size}` 来自请求；
   270→   270→   *   - 其余（含 `dept-tree`）：返回**全部扁平化节点**，外壳是 `page=0, size=rows.length`
   271→   271→   *     —— **忽略请求里的分页**。照抄，不要"顺手"改成尊重分页。
   272→   272→   *
   273→   273→   * ⚠️ 空值一律给**空串**而不是 null（`parentId`/`code`/`nickname`/`orgId`/`orgName`），
   274→   274→   *    这与「null 列不进 data」的 FORM 路径完全不同。
   275→   275→   */
   276→   276→  private async systemQuery(
   277→   277→    dataSource: DataSourceRef,
   278→   278→    req: BizDataQueryRequest,
   279→   279→  ): Promise<BizDataPageVO> {
   280→   280→    if (dataSource.sourceKey === 'user-tree') {
   281→   281→      const page = Math.max(req.page, 1)
   282→   282→      const result = await this.systemService.listUsersByUsername(req.keyword, page, req.size)
   283→   283→      const records: BizDataVO[] = result.rows.map((row) => ({
   284→   284→        id: String(row.id),
   285→   285→        data: {
   286→   286→          id: String(row.id),
   287→   287→          username: row.username,
   288→   288→          nickname: row.nickname ?? '',
   289→   289→          orgId: row.orgId === null || row.orgId === undefined ? '' : String(row.orgId),
   290→   290→          orgName: row.orgName ?? '',
   291→   291→          status: row.status,
   292→   292→        },
   293→   293→        version: null,
   294→   294→        createdAt: null,
   295→   295→        updatedAt: null,
   296→   296→      }))
   297→   297→      return { records, total: result.total, page: result.page, size: result.size }
   298→   298→    }
   299→   299→
   300→   300→    const nodes = await this.systemService.orgTree()
   301→   301→    const records: BizDataVO[] = []
   302→   302→    const collect = (node: OrgTreeNode): void => {
   303→   303→      records.push({
   304→   304→        id: String(node.id),
   305→   305→        data: {
   306→   306→          id: String(node.id),
   307→   307→          parentId: node.parentId === null || node.parentId === undefined ? '' : String(node.parentId),
   308→   308→          label: node.label ?? '',
   309→   309→          code: node.code ?? '',
   310→   310→        },
   311→   311→        version: null,
   312→   312→        createdAt: null,
   313→   313→        updatedAt: null,
   314→   314→      })
   315→   315→      for (const child of node.children ?? []) collect(child)
   316→   316→    }
   317→   317→    for (const node of nodes) collect(node)
   318→   318→    return { records, total: records.length, page: 0, size: records.length }
   319→   319→  }

   321→   321→  // ==================== 写路径（对齐 Java create / update / delete） ====================
   322→   322→
   323→   323→  /**
   324→   324→   * 新增（对齐 Java `create`）。
   325→   325→   *
   326→   326→   * ⚠️ **每个类型走的是不同的分支，错误形态完全不同**，不能合并：
   327→   327→   *    - FORM：先 `router.resolve`（不是 `requireFormKey`）→ `bizDataService.create`
   328→   328→   *    - WORKFLOW：**只读**，恒 400
   329→   329→   *    - API：外部 HTTP（见 `apiCreate`）
   330→   330→   *    - SQL：`requireFormKey`（消息是 SQL 专属的）→ `bizDataService.create`
   331→   331→   *    - SYSTEM 及未知类型：`default` → 「该数据源不支持create」
   332→   332→   *    SYSTEM 在这里**不经过 router**（Java 的 switch 里没有 SYSTEM 分支），
   333→   333→   *    所以 dept-tree/user-tree 都只报「不支持」，而不是先撞 allowlist。
   334→   334→   */
   335→   335→  async create(dataSource: DataSourceRef, data: Record<string, unknown> | null): Promise<string> {
   336→   336→    switch (dataSource.type) {
   337→   337→      case 'FORM': {
   338→   338→        const formKey = this.formKeyAfterResolve(dataSource, 'create')
   339→   339→        const created = await this.bizDataService.create(formKey, data)
   340→   340→        return created.id
   341→   341→      }
   342→   342→      case 'WORKFLOW':
   343→   343→        throw readOnlyWorkflow()
   344→   344→      case 'API':
   345→   345→        return this.apiCreate(dataSource, data)
   346→   346→      case 'SQL': {
   347→   347→        const formKey = requireFormKey(dataSource, 'create')
   348→   348→        const created = await this.bizDataService.create(formKey, data)
   349→   349→        return created.id
   350→   350→      }
   351→   351→      default:
   352→   352→        throw unsupported(dataSource, 'create')
   353→   353→    }
   354→   354→  }

   356→   356→  /** 更新（对齐 Java `update`）；分支与 `create` 一一对应。 */
   357→   357→  async update(
   358→   358→    dataSource: DataSourceRef,
   359→   359→    id: string,
   360→   360→    data: Record<string, unknown> | null,
   361→   361→    version: number | null,
   362→   362→  ): Promise<void> {
   363→   363→    switch (dataSource.type) {
   364→   364→      case 'FORM': {
   365→   365→        const formKey = this.formKeyAfterResolve(dataSource, 'update')
   366→   366→        await this.bizDataService.update(formKey, id, data, version)
   367→   367→        return
   368→   368→      }
   369→   369→      case 'WORKFLOW':
   370→   370→        throw readOnlyWorkflow()
   371→   371→      case 'API':
   372→   372→        // ⚠️ 必须 `return`/`await`：`apiUpdate` 是 async，写成「调用 + return」会让
   373→   373→        //    拒绝变成一个**游离的 Promise**（既不抛也不等）—— 端点返回 200 而实际失败了。
   374→   374→        return this.apiUpdate(dataSource, id, data)
   375→   375→      case 'SQL': {
   376→   376→        const formKey = requireFormKey(dataSource, 'update')
   377→   377→        await this.bizDataService.update(formKey, id, data, version)
   378→   378→        return
   379→   379→      }
   380→   380→      default:
   381→   381→        throw unsupported(dataSource, 'update')
   382→   382→    }
   383→   383→  }
   384→   384→
   385→   385→  /**
   386→   386→   * 删除（对齐 Java `delete`）。
   387→   387→   *
   388→   388→   * ⚠️ 与 create/update 的唯一结构差异：**SYSTEM 有显式分支**，它会先
   389→   389→   *    `router.resolve(ds, "delete")` 再抛「不支持」。Java 注释写明了原因 ——
   390→   390→   *    「SYSTEM is read-only at adapter level; router allows for audit」：
   391→   391→   *    即先让 allowlist 审计一次这次删除尝试，再拒绝。
   392→   392→   *    结果消息与 default 分支相同，但**未注册的 sourceKey 会先被拦下**。
   393→   393→   */
   394→   394→  async delete(dataSource: DataSourceRef, id: string): Promise<void> {
   395→   395→    switch (dataSource.type) {
   396→   396→      case 'FORM': {
   397→   397→        const formKey = this.formKeyAfterResolve(dataSource, 'delete')
   398→   398→        await this.bizDataService.remove(formKey, id)
   399→   399→        return
   400→   400→      }
   401→   401→      case 'SYSTEM':
   402→   402→        this.router.resolve(dataSource, 'delete')
   403→   403→        throw unsupported(dataSource, 'delete')
   404→   404→      case 'WORKFLOW':
   405→   405→        throw readOnlyWorkflow()
   406→   406→      case 'API':
   407→   407→        // 同上：必须 return，否则拒绝变成游离 Promise、outbound 失败被吞掉
   408→   408→        return this.apiDelete(dataSource, id)
   409→   409→      case 'SQL': {
   410→   410→        const formKey = requireFormKey(dataSource, 'delete')
   411→   411→        await this.bizDataService.remove(formKey, id)
   412→   412→        return
   413→   413→      }
   414→   414→      default:
   415→   415→        throw unsupported(dataSource, 'delete')
   416→   416→    }
   417→   417→  }
   418→   418→
   419→   419→  // ==================== FORM helpers ====================
   420→   420→
   421→   421→  /**
   422→   422→   * FORM 分支的 formKey：**先过 router 再取值**（顺序与 Java 一致）。
   423→   423→   *
   424→   424→   * `router.resolve` 已经拦掉了空 formKey，下面那次判断因此不可达 ——
   425→   425→   * 它存在的意义只是让类型收窄，消息与 router 里的那条保持逐字一致。
   426→   426→   */
   427→   427→  private formKeyAfterResolve(dataSource: DataSourceRef, op: string): string {
   428→   428→    this.router.resolve(dataSource, op)
   429→   429→    if (dataSource.formKey === null || dataSource.formKey.trim() === '') {
   430→   430→      throw new BusinessException(400, 'FORM 数据源缺少 formKey')
   431→   431→    }
   432→   432→    return dataSource.formKey
   433→   433→  }
   434→   434→
   435→   435→  /**
   436→   436→   * 追加 config 模式 JOIN 虚拟列到 metadata；与主表列 key 重复的 `virtualKey` 跳过
   437→   437→   * （防御性去重，对齐 Java `appendJoinColumns`）。
   438→   438→   *
   439→   439→   * ⚠️ 这里**只拷 5 个字段**（key/label/columnType/sortable/filterable），
   440→   440→   *    其余（`length`/`required`/`pickerConfig`…）保持默认值。与 SYSTEM 的
   441→   441→   *    `copyWithSortableFalse` 是同一类坑：多拷一个字段就与 Java 分叉。
   442→   442→   */
   443→   443→  private async appendJoinColumns(
   444→   444→    columns: ColumnConfig[],
   445→   445→    joins: JoinConfig[],
   446→   446→  ): Promise<void> {
   447→   447→    const existing = new Set(columns.map((column) => column.key))
   448→   448→    for (const join of joins) {
   449→   449→      const virtualKey = join.virtualKey
   450→   450→      if (
   451→   451→        virtualKey === null ||
   452→   452→        virtualKey.trim() === '' ||
   453→   453→        existing.has(virtualKey)
   454→   454→      ) {
   455→   455→        continue
   456→   456→      }
   457→   457→      columns.push({
   458→   458→        ...newColumnConfig(),
   459→   459→        key: virtualKey,
   460→   460→        label: join.label === null || join.label.trim() === '' ? virtualKey : join.label,
   461→   461→        columnType: await this.resolveJoinColumnType(join),
   462→   462→        sortable: join.sortable,
   463→   463→        filterable: join.filterable,
   464→   464→      })
   465→   465→      existing.add(virtualKey)
   466→   466→    }
   467→   467→  }
   468→   468→
   469→   469→  /**
   470→   470→   * 虚拟列 `columnType`：目标表单 `joinField` 的列类型；查不到 fallback `"VARCHAR"`。
   471→   471→   *
   472→   472→   * ⚠️ 与 `BizDataSupport.resolveJoinColumnType`（query 路径）**不是同一份实现**：
   473→   473→   *    这里要求目标列类型**非空非空白**才采用（空白继续找下一列），
   474→   474→   *    那边是「首个同名列，null 才 fallback」。两份差异照抄 Java，不要合并。
   475→   475→   */
   476→   476→  private async resolveJoinColumnType(join: JoinConfig): Promise<string> {
   477→   477→    try {
   478→   478→      const target = await this.bizDataService.loadColumns(String(join.targetFormKey))
   479→   479→      for (const column of target) {
   480→   480→        if (
   481→   481→          join.joinField === column.key &&
   482→   482→          column.columnType !== null &&
   483→   483→          column.columnType.trim() !== ''
   484→   484→        ) {
   485→   485→          return column.columnType.toUpperCase()
   486→   486→        }
   487→   487→      }
   488→   488→    } catch (error) {
   489→   489→      // 目标表单不可解析时回退（对齐 Java 的 `catch (BusinessException ignored)`）
   490→   490→      if (!(error instanceof BusinessException)) throw error
   491→   491→    }
   492→   492→    return 'VARCHAR'
   493→   493→  }
   494→   494→
   495→   495→  // ==================== API helpers ====================
   496→   496→
   497→   497→  /**
   498→   498→   * API 数据源 `params` 解析（对齐 Java `parseParams`）。
   499→   499→   *
   500→   500→   * 三种失败形态各有专属消息，且**都带数据源名**；`requireAction` 的消息反而
   501→   501→   * **不带**名字 —— 逐字照抄，不要「统一美化」。
   502→   502→   */
   503→   503→  private parseApiParams(dataSource: DataSourceRef): Record<string, unknown> {
   504→   504→    const json = dataSource.params
   505→   505→    if (json === null || json.trim() === '') {
   506→   506→      throw new BusinessException(400, `API 数据源缺少 params: ${dataSource.name}`)
   507→   507→    }
   508→   508→    let parsed: unknown
   509→   509→    try {
   510→   510→      parsed = JSON.parse(json)
   511→   511→    } catch {
   512→   512→      throw new BusinessException(400, `API 数据源 params 不是合法 JSON: ${dataSource.name}`)
   513→   513→    }
   514→   514→    if (parsed === null || typeof parsed !== 'object' || Array.isArray(parsed)) {
   515→   515→      throw new BusinessException(400, `API 数据源 params 必须是 JSON 对象: ${dataSource.name}`)
   516→   516→    }
   517→   517→    return parsed as Record<string, unknown>
   518→   518→  }
   519→   519→
   520→   520→  /**
   521→   521→   * 取某个操作段的配置（对齐 Java `operation`）。
   522→   522→   *
   523→   523→   * `list` 有一条**兼容旧格式**的旁路：顶层直接放 `action`/`method`/`parse` 时
   524→   524→   * 组装成与 `list` 子对象等价的映射。写路径的三个操作**没有**这条旁路 ——
   525→   525→   * 所以顶层 `action` 对 create/update/delete 不生效（这正是 golden 里
   526→   526→   * 「API 数据源未配置 create 操作」那条断言的来源）。
   527→   527→   */
   528→   528→  private apiOperation(
   529→   529→    params: Record<string, unknown>,
   530→   530→    name: 'list' | 'get' | 'create' | 'update' | 'delete',
   531→   531→  ): Record<string, unknown> | null {
   532→   532→    const op = params[name]
   533→   533→    if (op !== null && typeof op === 'object' && !Array.isArray(op)) {
   534→   534→      return op as Record<string, unknown>
   535→   535→    }
   536→   536→    if (name === 'list' && params.action !== null && params.action !== undefined) {
   537→   537→      return {
   538→   538→        action: params.action,
   539→   539→        method: params.method,
   540→   540→        parse: params.parse,
   541→   541→        totalParse: params.totalParse,
   542→   542→      }
   543→   543→    }
   544→   544→    return null
   545→   545→  }
   546→   546→
   547→   547→  /** 操作段必须有非空 `action`（对齐 Java `requireAction`）。 */
   548→   548→  private requireApiAction(op: Record<string, unknown>, name: string): string {
   549→   549→    const action = op.action
   550→   550→    if (action === null || action === undefined || String(action).trim() === '') {
   551→   551→      throw new BusinessException(400, `API 数据源 ${name} 操作缺少 action`)
   552→   552→    }
   553→   553→    return String(action)
   554→   554→  }
   555→   555→
   556→   556→  /**
   557→   557→   * 出站调用（对齐 Java 的 `httpExecutor.execute(action, opMethod(op), headers(params), …, vars, 10000, 10000, 0)`）。
   558→   558→   *
   559→   559→   * ⚠️ 三个常量与 Java 一致：超时 10000ms、重试 0 次、query/body 映射为空
   560→   560→   *    （API 数据源的参数**只通过 URL 模板变量**传递，不走 query/body 映射）。
   561→   561→   */
   562→   562→  private async apiExecute(
   563→   563→    dataSource: DataSourceRef,
   564→   564→    params: Record<string, unknown>,
   565→   565→    op: Record<string, unknown>,
   566→   566→    opName: string,
   567→   567→    vars: Record<string, unknown>,
   568→   568→  ): Promise<string> {
   569→   569→    const action = this.requireApiAction(op, opName)
   570→   570→    const method = op.method === null || op.method === undefined || String(op.method).trim() === ''
   571→   571→      ? 'GET'
   572→   572→      : String(op.method).toUpperCase()
   573→   573→    const headers: Record<string, string> = {}
   574→   574→    if (params.headers !== null && typeof params.headers === 'object' && !Array.isArray(params.headers)) {
   575→   575→      for (const [key, value] of Object.entries(params.headers as Record<string, unknown>)) {
   576→   576→        headers[key] = String(value)
   577→   577→      }
   578→   578→    }
   579→   579→    return this.httpExecutor.execute(
   580→   580→      action,
   581→   581→      method,
   582→   582→      headers,
   583→   583→      [],
   584→   584→      [],
   585→   585→      vars,
   586→   586→      DEFAULT_TIMEOUT_MS,
   587→   587→      DEFAULT_TIMEOUT_MS,
   588→   588→      DEFAULT_RETRY,
   589→   589→    )
   590→   590→  }
   591→   591→
   592→   592→  private async apiCreate(
   593→   593→    dataSource: DataSourceRef,
   594→   594→    data: Record<string, unknown> | null,
   595→   595→  ): Promise<string> {
   596→   596→    const params = this.parseApiParams(dataSource)
   597→   597→    const op = this.apiOperation(params, 'create')
   598→   598→    if (op === null) {
   599→   599→      throw new BusinessException(400, `API 数据源未配置 create 操作: ${dataSource.name}`)
   600→   600→    }
   601→   601→    // Java：vars 直接就是整个 body（不是 body 映射）
   602→   602→    const raw = await this.apiExecute(dataSource, params, op, 'create', { ...(data ?? {}) })
   603→   603→    const id = asDataMap(raw).id
   604→   604→    return id === null || id === undefined ? '' : String(id)
   605→   605→  }

   613→   613→    const op = this.apiOperation(params, 'update')

   615→   615→      throw new BusinessException(400, `API 数据源未配置 update 操作: ${dataSource.name}`)

   620→   620→  private async apiDelete(dataSource: DataSourceRef, id: string): Promise<void> {
   621→   621→    const params = this.parseApiParams(dataSource)
   622→   622→    const op = this.apiOperation(params, 'delete')
   623→   623→    if (op === null) {
   624→   624→      throw new BusinessException(400, `API 数据源未配置 delete 操作: ${dataSource.name}`)
   625→   625→    }
   626→   626→    await this.apiExecute(dataSource, params, op, 'delete', { id })
   627→   627→  }
   628→   628→
   629→   629→  // ==================== API 读路径（对齐 Java 的 apiMetadata / apiQuery / apiGet） ====================
   630→   630→
   631→   631→  /**
   632→   632→   * API 元数据（对齐 Java `apiMetadata` + `metadata` 里的 API 分支）。
   633→   633→   *
   634→   634→   * ⚠️ 列**来自 `params.columns`**（不查库、不探测），而且外层还会再套一次
   635→   635→   *    `copyWithSortableFalse` —— golden 实测：即便 `params.columns` 里写了
   636→   636→   *    `length: 64`，返回的列上 `length` 仍是 **null**、`sortable` 是 **false**。
   637→   637→   *    只拷 key/label/columnType 这一点与 SYSTEM 分支同源。
   638→   638→   * ⚠️ `writable` = 「配了 create / update / delete 任一」（`operation` 的旧格式旁路
   639→   639→   *    **只对 list 生效**，所以顶层 action 不会让 writable 变 true）。
   640→   640→   */
   641→   641→  private apiMetadata(dataSource: DataSourceRef, params: Record<string, unknown>): DataSourceMetadata {
   642→   642→    const columns: ColumnConfig[] = []
   643→   643→    const colsNode = params.columns
   644→   644→    if (Array.isArray(colsNode)) {
   645→   645→      for (const entry of colsNode) {
   646→   646→        if (entry === null || typeof entry !== 'object' || Array.isArray(entry)) continue
   647→   647→        columns.push({ ...newColumnConfig(), ...(entry as Partial<ColumnConfig>) })
   648→   648→      }
   649→   649→    }
   650→   650→    const writable =
   651→   651→      this.apiOperation(params, 'create') !== null ||
   652→   652→      this.apiOperation(params, 'update') !== null ||
   653→   653→      this.apiOperation(params, 'delete') !== null
   654→   654→    return {
   655→   655→      columns: copyWithSortableFalse(columns),
   656→   656→      writable,
   657→   657→      formKey: dataSource.formKey,
   658→   658→    }
   659→   659→  }
   660→   660→
   661→   661→  /**
   662→   662→   * API 取数（对齐 Java `apiQuery`）。
   663→   663→   *
   664→   664→   * 变量表与 Java 一致：`params.data` 的键值 + `page`/`size` + `keyword`（null → 空串），
   665→   665→   * 以及「keyword 非空且配了 `searchParam`」时把 keyword 再塞进 `vars[searchParam]`。
   666→   666→   */
   667→   667→  private async apiQuery(
   668→   668→    dataSource: DataSourceRef,
   669→   669→    req: BizDataQueryRequest,
   670→   670→  ): Promise<BizDataPageVO> {
   671→   671→    const params = this.parseApiParams(dataSource)
   672→   672→    const op = this.apiOperation(params, 'list')
   673→   673→    if (op === null) {
   674→   674→      throw new BusinessException(400, `API 数据源未配置 list 操作: ${dataSource.name}`)
   675→   675→    }
   676→   676→    const vars: Record<string, unknown> = {}
   677→   677→    if (params.data !== null && typeof params.data === 'object' && !Array.isArray(params.data)) {
   678→   678→      for (const [key, value] of Object.entries(params.data as Record<string, unknown>)) {
   679→   679→        vars[key] = value
   680→   680→      }
   681→   681→    }
   682→   682→    vars.page = req.page
   683→   683→    vars.size = req.size
   684→   684→    const keyword = req.keyword
   685→   685→    vars.keyword = keyword === null || keyword === undefined ? '' : keyword
   686→   686→    const searchParam = params.searchParam === null || params.searchParam === undefined
   687→   687→      ? null
   688→   688→      : String(params.searchParam)
   689→   689→    if (
   690→   690→      keyword !== null &&
   691→   691→      keyword !== undefined &&
   692→   692→      keyword.trim() !== '' &&
   693→   693→      searchParam !== null &&
   694→   694→      searchParam.trim() !== ''
   695→   695→    ) {
   696→   696→      vars[searchParam] = keyword
   697→   697→    }
   698→   698→    const raw = await this.apiExecute(dataSource, params, op, 'list', vars)
   699→   699→    return toPageVO(raw, op, req)
   700→   700→  }

   702→   702→  /** API 单行取数（对齐 Java `apiGet`）：变量只有 `id`。 */
   703→   703→  private async apiGet(dataSource: DataSourceRef, id: string): Promise<BizDataVO> {
   704→   704→    const params = this.parseApiParams(dataSource)
   705→   705→    const op = this.apiOperation(params, 'get')
   706→   706→    if (op === null) {
   707→   707→      throw new BusinessException(400, `API 数据源未配置 get 操作: ${dataSource.name}`)
   708→   708→    }
   709→   709→    const raw = await this.apiExecute(dataSource, params, op, 'get', { id })
   710→   710→    return { id, data: asDataMap(raw), version: null, createdAt: null, updatedAt: null }
   711→   711→  }
   712→   712→}
   713→   713→
   714→   714→/** API 数据源的固定出站超时（对齐 Java `DEFAULT_TIMEOUT_MS`）。 */
   715→   715→const DEFAULT_TIMEOUT_MS = 10000
   716→   716→/** API 数据源的固定重试次数（对齐 Java `DEFAULT_RETRY`）。 */
   717→   717→const DEFAULT_RETRY = 0
   718→   718→
   719→   719→/**
   720→   720→ * 出站响应 → `{id, data}` 映射（对齐 Java `asDataMap`）。
   721→   721→ *
   722→   722→ * ⚠️ 字符串响应要**先当 JSON 解析**；解析失败时 Java 抛
   723→   723→ * 「外部 API 响应不是合法 JSON」（400）。对象直接返回；其余（数组/标量/null）→ 空对象。
   724→   724→ */
   725→   725→export function asDataMap(raw: unknown): Record<string, unknown> {
   726→   726→  if (raw === null || raw === undefined) return {}
   727→   727→  if (typeof raw === 'string') {
   728→   728→    let parsed: unknown
   729→   729→    try {
   730→   730→      parsed = JSON.parse(raw)
   731→   731→    } catch {
   732→   732→      throw new BusinessException(400, '外部 API 响应不是合法 JSON')
   733→   733→    }
   734→   734→    if (parsed !== null && typeof parsed === 'object' && !Array.isArray(parsed)) {
   735→   735→      return parsed as Record<string, unknown>
   736→   736→    }
   737→   737→    return {}
   738→   738→  }
   739→   739→  if (typeof raw === 'object' && !Array.isArray(raw)) return raw as Record<string, unknown>
   740→   740→  return {}
   741→   741→}
   742→   742→
   743→   743→/**
   744→   744→ * 出站响应 → 分页结果（对齐 Java `toPageVO`）。
   745→   745→ *
   746→   746→ * 三步：把响应解析成 JSON 树 → 按 `op.parse` 的路径 walk 到数组（walk 不到或不是数组 →
   747→   747→ * **空页**，不是报错）→ 逐行取 `id` 字段；`total` 默认取行数，配了 `totalParse`
   748→   748→ * 且该处是数字时以它为准。
   749→   749→ *
   750→   750→ * ⚠️ `page` 用 `Math.max(req.page, 1)` 归一，`size` 原样透传（与 Java 一致）。
   751→   751→ */
   752→   752→export function toPageVO(
   753→   753→  raw: unknown,
   754→   754→  op: Record<string, unknown>,
   755→   755→  req: BizDataQueryRequest,
   756→   756→): BizDataPageVO {
   757→   757→  const page = Math.max(req.page, 1)
   758→   758→  let root: unknown
   759→   759→  try {
   760→   760→    root = typeof raw === 'string' ? JSON.parse(raw) : raw
   761→   761→  } catch {
   762→   762→    throw new BusinessException(400, '外部 API 响应不是合法 JSON')
   763→   763→  }
   764→   764→
   765→   765→  const parsePath = op.parse === null || op.parse === undefined ? null : String(op.parse)
   766→   766→  const recordsNode = parsePath === null || parsePath.trim() === '' ? root : walkPath(root, parsePath)
   767→   767→  if (!Array.isArray(recordsNode)) {
   768→   768→    return { records: [], total: 0, page, size: req.size }
   769→   769→  }
   770→   770→
   771→   771→  const records: BizDataVO[] = recordsNode.map((item) => {
   772→   772→    const data =
   773→   773→      item !== null && typeof item === 'object' && !Array.isArray(item)
   774→   774→        ? (item as Record<string, unknown>)
   775→   775→        : {}
   776→   776→    const id = data.id === null || data.id === undefined ? '' : String(data.id)
   777→   777→    return { id, data, version: null, createdAt: null, updatedAt: null }
   778→   778→  })
   779→   779→
   780→   780→  let total = records.length
   781→   781→  const totalParse = op.totalParse === null || op.totalParse === undefined ? null : String(op.totalParse)
   782→   782→  if (totalParse !== null && totalParse.trim() !== '') {
   783→   783→    const node = walkPath(root, totalParse)
   784→   784→    if (typeof node === 'number') total = node
   785→   785→  }
   786→   786→  return { records, total, page, size: req.size }
   787→   787→}

   789→   789→/** 按 `.` 分段 walk JSON（对齐 Java `walkPath`：中途不是对象 → null）。 */
   790→   790→function walkPath(root: unknown, path: string): unknown {
   791→   791→  let current: unknown = root
   792→   792→  for (const part of path.split('.')) {
   793→   793→    if (current === null || typeof current !== 'object' || Array.isArray(current)) return null
   794→   794→    current = (current as Record<string, unknown>)[part]
   795→   795→  }
   796→   796→  return current
   797→   797→}
