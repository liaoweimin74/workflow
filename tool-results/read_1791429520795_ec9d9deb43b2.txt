   437→function addNode(type: LogicNodeType, position: { x: number; y: number }): FlowNodeModel | null {
   438→  if (type === 'START' && allNodes().some((n) => n.data.nodeType === 'START')) {
   439→    ElMessage.warning('一个逻辑流只能有一个开始节点')
   440→    return null
   441→  }

   443→    id: createNodeId(type, allNodes().map((n) => n.id)),
   444→    type: 'logic',
   445→    position,
   446→    data: {
   447→      nodeType: type,
   448→      name: defaultNodeName(type),
   449→      config: defaultConfig(type),
   450→      errorAction: type === 'HTTP' || type === 'BEAN' || type === 'SCRIPT' ? 'FAIL_FLOW' : undefined,
   451→    },
   452→  }
   453→  addNodes([node])
   454→  if (type === 'BATCH') {
   455→    addEdges([createLoopEdge(node.id, node.id, LOOP_HANDLE_START, LOOP_HANDLE_END)])
   456→  }

   492→function onConnect(connection: Connection) {
   493→  const { source, target, sourceHandle, targetHandle } = connection
   494→  if (!source || !target) return
   495→  if (source === target) {
   496→    ElMessage.warning('不能连接到自身')
   497→    return
   498→  }
   499→  if (
   500→    sourceHandle === LOOP_HANDLE_START ||
   501→    sourceHandle === LOOP_HANDLE_END ||
   502→    targetHandle === LOOP_HANDLE_START ||
   503→    targetHandle === LOOP_HANDLE_END
   504→  ) {
   505→    ElMessage.warning('循环体请把动作节点拖到循环虚线上自动接入，不支持手动连线')
   506→    return
   507→  }
   508→  const branch = sourceHandle === 'true' || sourceHandle === 'false' ? sourceHandle : undefined
   509→  addEdges([
   510→    {
   511→      source,
   512→      target,
   513→      sourceHandle: sourceHandle ?? undefined,
   514→      targetHandle: targetHandle ?? undefined,
   515→      data: branch ? { branch } : undefined,
   516→    },
   517→  ])
   518→}

   544→// ===== 节点拖到连线自动吸附接入（普通连线 / 批处理循环虚线） =====
   545→
   546→/** 吸附判定距离（画布坐标 px）：主流实线 */
   547→const SNAP_THRESHOLD = 32
   548→
   549→/** 循环虚线独立吸附阈值：与 U 形外凸宽度（offset 56）对齐，视觉上「碰到虚线即吸」更跟手 */
   550→const LOOP_SNAP_THRESHOLD = 56
   551→
   552→/** vue-flow 未回填 dimensions 时的节点近似尺寸兜底 */
   553→const NODE_FALLBACK_SIZE = { w: 160, h: 54 }
   554→
   555→function nodeSize(node: FlowNodeModel): { w: number; h: number } {
   556→  const dims = (node as unknown as { dimensions?: { width?: number; height?: number } })
   557→    ?.dimensions
   558→  return {
   559→    w: Number(dims?.width) || NODE_FALLBACK_SIZE.w,
   560→    h: Number(dims?.height) || NODE_FALLBACK_SIZE.h,
   561→  }
   562→}
   563→
   564→function nodeCenter(node: FlowNodeModel): { x: number; y: number } {
   565→  const size = nodeSize(node)
   566→  return { x: node.position.x + size.w / 2, y: node.position.y + size.h / 2 }
   567→}
   568→
   569→/** 两节点间正交折线近似（中位水平线），与 smoothstep 走向一致 */
   570→function approxPointsBetween(
   571→  source: FlowNodeModel,
   572→  target: FlowNodeModel
   573→): { x: number; y: number }[] {
   574→  const sc = nodeCenter(source)
   575→  const tc = nodeCenter(target)
   576→  const midY = (sc.y + tc.y) / 2
   577→  return [sc, { x: sc.x, y: midY }, { x: tc.x, y: midY }, tc]
   578→}
   579→
   580→/** 自环 U 形近似：节点右侧外凸（与 LoopEdge.vue 渲染 offset 56 同侧同距） */
   581→function selfLoopApproxPoints(batch: FlowNodeModel): { x: number; y: number }[] {
   582→  const size = nodeSize(batch)
   583→  const x = batch.position.x + size.w
   584→  const y1 = batch.position.y + size.h * 0.3
   585→  const y2 = batch.position.y + size.h * 0.74
   586→  const bulge = 56
   587→  return [
   588→    { x, y: y1 },
   589→    { x: x + bulge, y: y1 },
   590→    { x: x + bulge, y: y2 },
   591→    { x, y: y2 },
   592→  ]
   593→}
   594→
   595→/** 边路径近似：源/目标节点中心的正交折线（中位水平线），与 smoothstep 走向一致 */
   596→function edgeApproxPoints(edge: FlowEdge): { x: number; y: number }[] | null {
   597→  const source = allNodes().find((n) => n.id === edge.source)
   598→  const target = allNodes().find((n) => n.id === edge.target)
   599→  if (!source || !target) return null
   600→  // 循环直连边（batch→batch 自环）：右侧 U 形近似（loop_start 30% → 外凸 → loop_end 74%）
   601→  if (isLoopEdge(edge) && edge.source === edge.target) {
   602→    return selfLoopApproxPoints(source)
   603→  }
   604→  return approxPointsBetween(source, target)
   605→}
   606→
   607→function distToSegment(
   608→  p: { x: number; y: number },
   609→  a: { x: number; y: number },
   610→  b: { x: number; y: number }
   611→): number {
   612→  const dx = b.x - a.x
   613→  const dy = b.y - a.y
   614→  const len2 = dx * dx + dy * dy
   615→  const t = len2 === 0 ? 0 : Math.max(0, Math.min(1, ((p.x - a.x) * dx + (p.y - a.y) * dy) / len2))
   616→  return Math.hypot(p.x - (a.x + t * dx), p.y - (a.y + t * dy))
   617→}
   618→
   619→function distToPolyline(p: { x: number; y: number }, pts: { x: number; y: number }[]): number {
   620→  let min = Infinity
   621→  for (let i = 1; i < pts.length; i++) {
   622→    min = Math.min(min, distToSegment(p, pts[i - 1], pts[i]))
   623→  }
   624→  return min
   625→}
   626→
   627→/** 节点是否已挂在某条循环体链上（循环体节点同时只能属于一条循环链）。
   628→ * 按 nodeId 所在端分别判定：batch→链首 段的 sourceHandle 恰为 loop_start、
   629→ * 链尾→batch 段的 targetHandle 恰为 loop_end（合法链段），故不能要求
   630→ * 一条边两端同时为普通口 —— 旧写法会把单节点循环链的成员误判为未挂链，
   631→ * 导致其拖出失效、且可被重复吸附（bug：循环拖出 E2E 暴露） */
   632→function nodeOnLoopChain(nodeId: string): boolean {
   633→  return allEdges().some(
   634→    (e) =>
   635→      isLoopEdge(e) &&
   636→      ((e.source === nodeId && e.sourceHandle !== LOOP_HANDLE_START) ||
   637→        (e.target === nodeId && e.targetHandle !== LOOP_HANDLE_END))
   638→  )
   639→}
   640→
   641→interface SnapCandidate {
   642→  edge: FlowEdge
   643→  isLoop: boolean
   644→  dist: number
   645→}
   646→
   647→/** 某条循环边所属的批处理（链源头）：链首段/尾段按 handle 直判，中段沿入链上溯 */
   648→function loopChainOwner(edge: FlowEdge): string | null {
   649→  const loopEdges = allEdges().filter(isLoopEdge)
   650→  if (edge.sourceHandle === LOOP_HANDLE_START) return edge.source
   651→  if (edge.targetHandle === LOOP_HANDLE_END) return edge.target
   652→  let cur = edge.source
   653→  for (let i = 0; i < 100; i++) {
   654→    const up = loopEdges.find((e) => e.target === cur)
   655→    if (!up) return null
   656→    if (up.sourceHandle === LOOP_HANDLE_START) return up.source
   657→    cur = up.source
   658→  }
   659→  return null
   660→}
   661→
   662→/** 批处理的循环体后代（含直接与嵌套子孙 batch）：沿 loop 链递归收集，防环用 */
   663→function batchDescendants(batchId: string, visited = new Set<string>()): string[] {
   664→  const out: string[] = []
   665→  if (visited.has(batchId)) return out
   666→  visited.add(batchId)
   667→  const loopEdges = allEdges().filter(isLoopEdge)
   668→  const start = loopEdges.find(
   669→    (e) => e.source === batchId && e.sourceHandle === LOOP_HANDLE_START
   670→  )
   671→  if (!start) return out
   672→  const seen = new Set<string>([batchId])
   673→  let cur: string | undefined = start.target
   674→  let guard = 0
   675→  while (cur && !seen.has(cur) && guard++ < 100) {
   676→    seen.add(cur)
   677→    const node = allNodes().find((n) => n.id === cur)
   678→    if (node?.data?.nodeType === 'BATCH') {
   679→      out.push(cur)
   680→      out.push(...batchDescendants(cur, visited))
   681→    }
   682→    const next = loopEdges.find((e) => e.source === cur)
   683→    if (!next) break
   684→    cur = next.target
   685→  }
   686→  return out
   687→}
   688→
   689→/** 嵌套/类型不可入提示节流（画布坐标拖动中每 4s 至多一次） */
   690→let loopHintLastAt = 0
   691→let loopHintText = ''
   692→
   693→/** 找被拖节点附近的边（最近且距离达标）；不满足类型/归属约束的边跳过 */
   694→function findSnapEdge(node: FlowNodeModel): SnapCandidate | null {
   695→  const nodeType = node.data?.nodeType
   696→  if (!nodeType || nodeType === 'START') return null
   697→  const center = nodeCenter(node)
   698→  let best: SnapCandidate | null = null
   699→  let loopHintNearby = false
   700→  loopHintText = ''
   701→  for (const edge of allEdges()) {
   702→    if (edge.source === node.id || edge.target === node.id) continue
   703→    const isLoop = isLoopEdge(edge)
   704→    if (isLoop) {
   705→      // 循环体仅收业务执行六型（START/END/CONDITION 不可入；BATCH=嵌套，须防环）
   706→      if (!(BATCH_BODY_TYPES as string[]).includes(nodeType)) {
   707→        if (distToPolyline(center, edgeApproxPoints(edge) ?? []) <= LOOP_SNAP_THRESHOLD) {
   708→          loopHintNearby = true
   709→          loopHintText = '该节点不可加入循环体（仅 HTTP/Bean/脚本/数据更新/子流程/批处理）'
   710→        }
   711→        continue
   712→      }
   713→      if (nodeOnLoopChain(node.id)) continue
   714→      // 嵌套批处理防环：BATCH 不能入自身/自身后代的循环体链（否则运行时无限递归）
   715→      if (nodeType === 'BATCH') {
   716→        const owner = loopChainOwner(edge)
   717→        if (owner && (owner === node.id || batchDescendants(node.id).includes(owner))) {
   718→          loopHintNearby = true
   719→          loopHintText = owner === node.id ? '批处理不能嵌套自身' : '批处理不能嵌套形成循环引用'
   720→          continue
   721→        }
   722→      }
   723→    } else if (nodeOnLoopChain(node.id)) {
   724→      // 已在循环链上的节点不可再接入主流（序列化时会被剔出顶层导致悬空边）
   725→      continue
   726→    }
   727→    const pts = edgeApproxPoints(edge)
   728→    if (!pts) continue
   729→    const dist = distToPolyline(center, pts)
   730→    const threshold = isLoop ? LOOP_SNAP_THRESHOLD : SNAP_THRESHOLD
   731→    if (dist <= threshold && (!best || dist < best.dist)) {
   732→      best = { edge, isLoop, dist }
   733→    }
   734→  }
   735→  if (!best && loopHintNearby) {
   736→    // 拖到虚线附近但不可入/成环：节流提示（无候选吸附时才弹，避免拖动中刷屏）
   737→    const now = Date.now()
   738→    if (now - loopHintLastAt > 4000) {
   739→      loopHintLastAt = now
   740→      ElMessage.warning(loopHintText)
   741→    }
   742→  }
   743→  return best
   744→}
   745→
   746→/** 吸附接入：原边拆成两段穿过该节点（循环边拆出的两段仍是循环边）。
   747→ * 顺序：先加新边再删旧边 —— vue-flow removeEdges 会跳过 deletable:false 的边，
   748→ * 且删旧边触发的自愈（nextTick）必须在链完整后执行才能正确判 complete。
   749→ * 已在链上的节点接入新线前先断开旧链（拖换位置/换链一步到位）。 */
   750→function snapInsertNode(node: FlowNodeModel, candidate: SnapCandidate): void {
   751→  detachNodeFromChains(node)
   752→  const edge = candidate.edge
   753→  const isLoop = candidate.isLoop
   754→  const data = isLoop ? { loop: true } : edge.data ? { ...edge.data } : undefined
   755→  const segBase = isLoop
   756→    ? { type: 'smoothstep', markerEnd: 'arrowclosed', class: 'lf-edge-loop' }
   757→    : { type: 'smoothstep', markerEnd: 'arrowclosed' }
   758→  addEdges([
   759→    {
   760→      source: edge.source,
   761→      target: node.id,
   762→      sourceHandle: edge.sourceHandle,
   763→      targetHandle: 'in',
   764→      data,
   765→      ...segBase,
   766→    },
   767→    {
   768→      source: node.id,
   769→      target: edge.target,
   770→      sourceHandle: 'out',
   771→      targetHandle: edge.targetHandle,
   772→      data: isLoop ? { loop: true } : undefined,
   773→      ...segBase,
   774→    },
   775→  ])
   776→  removeEdges([edge])
   777→  ElMessage.success(isLoop ? '已加入循环体（按链序逐项执行）' : '已自动接入连线')
   778→}

   780→/** 吸附统一入口：新节点落位 / 画布内拖动结束共用 */
   781→function trySnapNode(node: FlowNodeModel) {
   782→  const candidate = findSnapEdge(node)
   783→  if (candidate) snapInsertNode(node, candidate)
   784→}

   786→// ===== 节点解绑：按住 Shift 拖动 / 双击链上节点 → 断开并缝合原链 =====
   787→
   788→interface ChainEdges {
   789→  ins: FlowEdge[]
   790→  outs: FlowEdge[]
   791→}
   792→
   793→/** 节点的非循环连边 */
   794→function mainEdgesOf(nodeId: string): ChainEdges {
   795→  const ins: FlowEdge[] = []
   796→  const outs: FlowEdge[] = []
   797→  for (const e of allEdges()) {
   798→    if (isLoopEdge(e)) continue
   799→    if (e.target === nodeId) ins.push(e)
   800→    if (e.source === nodeId) outs.push(e)
   801→  }
   802→  return { ins, outs }
   803→}
   804→
   805→/** 节点在循环链上的连边（排除 batch 自身的 loop_start/loop_end 自环端） */
   806→function loopEdgesOf(nodeId: string): ChainEdges {
   807→  const ins: FlowEdge[] = []
   808→  const outs: FlowEdge[] = []
   809→  for (const e of allEdges()) {
   810→    if (!isLoopEdge(e)) continue
   811→    if (e.target === nodeId && e.targetHandle !== LOOP_HANDLE_END) ins.push(e)
   812→    if (e.source === nodeId && e.sourceHandle !== LOOP_HANDLE_START) outs.push(e)
   813→  }
   814→  return { ins, outs }
   815→}

   817→/** 主流链中段判定：恰一进一出（START 无入/END 无出/分支出边>1 天然排除） */
   818→function isMainChainInternal(nodeId: string): boolean {
   819→  const { ins, outs } = mainEdgesOf(nodeId)
   820→  return ins.length === 1 && outs.length === 1
   821→}
   822→
   823→/** 主流缝合直连：删 N 的两断边并补 src→tgt（保留 CONDITION 分支 sourceHandle） */
   824→function stitchMainChain(node: FlowNodeModel): void {
   825→  const { ins, outs } = mainEdgesOf(node.id)
   826→  const inE = ins[0]
   827→  const outE = outs[0]
   828→  const branch =
   829→    inE.sourceHandle === 'true' || inE.sourceHandle === 'false' ? inE.sourceHandle : undefined
   830→  addEdges([
   831→    {
   832→      source: inE.source,
   833→      target: outE.target,
   834→      sourceHandle: inE.sourceHandle ?? undefined,
   835→      targetHandle: outE.targetHandle ?? undefined,
   836→      data: branch ? { branch } : undefined,
   837→    },
   838→  ])
   839→  removeEdges([inE, outE])
   840→}
   841→
   842→/**
   843→ * 断开节点与既有链的连接（接入新线前置 / 双击移出共用）：
   844→ * - 主流中段：缝合前后直连；
   845→ * - 循环链：删全部循环连边，链自愈（nextTick healBatchLoops）缝合。
   846→ * 返回是否发生了断开。
   847→ */
   848→function detachNodeFromChains(node: FlowNodeModel): boolean {
   849→  if (isMainChainInternal(node.id)) {
   850→    stitchMainChain(node)
   851→    return true
   852→  }
   853→  if (nodeOnLoopChain(node.id)) {
   854→    const { ins, outs } = loopEdgesOf(node.id)
   855→    if (ins.length || outs.length) {
   856→      removeEdges([...ins, ...outs])
   857→      return true
   858→    }
   859→  }
   860→  return false
   861→}

   863→/** 解绑统一入口（Shift 拖动 / 双击链上节点共用）：解绑成功则弹消息 */
   864→function detachWithMessage(node: FlowNodeModel): void {
   865→  // 先判链归属再断开（断开后边已删，isMainChainInternal 恒为 false）
   866→  const wasMain = isMainChainInternal(node.id)
   867→  if (detachNodeFromChains(node)) {
   868→    ElMessage.success(wasMain ? '已从连线解绑，前后节点已重新连接' : '已移出循环体，循环链已重新缝合')
   869→  }
   870→}

   872→function onNodeDragStop({ event, node }: { event: { shiftKey?: boolean }; node: FlowNodeModel }) {
   873→  if (!node?.id) return
   874→  // 需求②：按住 Shift 移动节点即解绑连线（不按 shift 时不再做距离自动断开，避免误触）
   875→  if (event?.shiftKey) {
   876→    detachWithMessage(node)
   877→    return
   878→  }

   880→  if (candidate) {
   881→    snapInsertNode(node, candidate)
   882→  }
   883→}

   899→/** 对画布上每个 BATCH 检查循环链完整性并缝合断口（读 store 实时边集，非 v-model 滞后副本） */
   900→function healBatchLoops() {
   901→  for (const batch of storeNodes.value) {
   902→    if (batch.data?.nodeType === 'BATCH') healOneBatch(batch.id)
   903→  }
   904→}
   905→
   906→function healOneBatch(batchId: string) {
   907→  // store 边为 vue-flow 内部 Edge 类型，按本设计器 FlowEdge 形状读取（字段子集兼容）
   908→  const loopEdges = (storeEdges.value as unknown as FlowEdge[]).filter(isLoopEdge)
   909→  const directExists = loopEdges.some(
   910→    (e) =>
   911→      e.source === batchId &&
   912→      e.sourceHandle === LOOP_HANDLE_START &&
   913→      e.target === batchId &&
   914→      e.targetHandle === LOOP_HANDLE_END
   915→  )
   916→
   917→  // 正向：loop_start 出发沿链，until 回到 loop_end（完整）或无出边（断口在 tail 之后）。
   918→  // 嵌套批处理：后续跳只走链段边（sourceHandle='out'），跳过链上子 BATCH
   919→  // 自己的自环（sourceHandle=loop_start）——否则子自环被误当链路下一跳，
   920→  // 走环保护 break 导致误判断链而重复补链（bug：嵌套 E2E 暴露）。
   921→  let tail = batchId
   922→  let forwardComplete = false
   923→  {
   924→    const visited = new Set([batchId])
   925→    let cur = loopEdges.find((e) => e.source === batchId && e.sourceHandle === LOOP_HANDLE_START)
   926→    let guard = 0
   927→    while (cur && guard++ < 100) {
   928→      if (cur.target === batchId && cur.targetHandle === LOOP_HANDLE_END) {
   929→        forwardComplete = true
   930→        break
   931→      }
   932→      if (visited.has(cur.target)) break
   933→      visited.add(cur.target)
   934→      tail = cur.target
   935→      cur = loopEdges.find(
   936→        (e) => e.source === cur!.target && e.sourceHandle !== LOOP_HANDLE_START
   937→      )
   938→    }
   939→  }
   940→  if (forwardComplete) return
   941→
   942→  // 反向：loop_end 回溯链段头部（后续跳只走链段边 targetHandle='in'，同上跳过子 BATCH 自环）
   943→  let head = batchId
   944→  {
   945→    const visited = new Set([batchId])
   946→    let cur = loopEdges.find((e) => e.target === batchId && e.targetHandle === LOOP_HANDLE_END)
   947→    let guard = 0
   948→    while (cur && guard++ < 100) {
   949→      if (visited.has(cur.source)) break
   950→      visited.add(cur.source)
   951→      head = cur.source
   952→      cur = loopEdges.find(
   953→        (e) =>
   954→          e.target === cur!.source && e.source !== batchId && e.targetHandle !== LOOP_HANDLE_END
   955→      )
   956→    }
   957→  }
   958→
   959→  const patch: FlowEdge[] = []
   960→  if (tail === batchId && head === batchId) {
   961→    // 链完全为空：确保 loop_start→loop_end 直连存在
   962→    if (!directExists) {
   963→      patch.push(createLoopEdge(batchId, batchId, LOOP_HANDLE_START, LOOP_HANDLE_END))
   964→    }
   965→  } else if (tail !== batchId && head !== batchId) {
   966→    if (tail === head) {
   967→      // 仅剩单个中段节点：两端各接一条
   968→      patch.push(createLoopEdge(batchId, tail, LOOP_HANDLE_START, 'in'))
   969→      patch.push(createLoopEdge(tail, batchId, 'out', LOOP_HANDLE_END))
   970→    } else {
   971→      patch.push(createLoopEdge(tail, head, 'out', 'in'))
   972→    }
   973→  } else if (tail !== batchId) {
   974→    // 正向链存在但 loop_end 断开
   975→    patch.push(createLoopEdge(tail, batchId, 'out', LOOP_HANDLE_END))
   976→  } else {
   977→    // loop_start 无出链但 loop_end 有前驱段
   978→    patch.push(createLoopEdge(batchId, head, LOOP_HANDLE_START, 'in'))
   979→  }
   980→  if (patch.length) addEdges(patch)
   981→}

  1018→function pushHistory(dsl: string) {
  1019→  // 丢弃撤销后的前进分支，追加快照（上限 100 防内存膨胀）
  1020→  historyStack.value = historyStack.value.slice(0, historyIndex.value + 1)

  1026→function applySnapshot(dsl: string) {
  1027→  applyingHistory.value = true
  1028→  try {
  1029→    const graph = parseDsl(dsl)
  1030→    nodes.value = graph.nodes
  1031→    edges.value = graph.edges

  1034→    // 快照损坏不应发生（均为本组件序列化产物），忽略
  1035→  } finally {
  1036→    setTimeout(() => {
  1037→      applyingHistory.value = false
  1038→    }, 0)
  1039→  }
  1040→}

  1042→function undo() {
  1043→  if (!canUndo.value) return
  1044→  historyIndex.value -= 1

  1049→  if (!canRedo.value) return

  1071→// ===== 自动整理布局：从 START 做 Kahn 分层（环上节点兜底），层级自上而下、同层横向铺开 =====
  1072→function handleAutoLayout() {
  1073→  if (!nodes.value.length) return
  1074→  const leftX = 80
  1075→  const colGapX = 230
  1076→  const topY = 60
  1077→  const layerGapY = 160
  1078→
  1079→  const idSet = new Set(nodes.value.map((n) => n.id))
  1080→  const depth = new Map<string, number>()
  1081→  const indegree = new Map<string, number>()
  1082→  const out = new Map<string, string[]>()
  1083→  nodes.value.forEach((n) => {
  1084→    indegree.set(n.id, 0)
  1085→    out.set(n.id, [])
  1086→  })
  1087→  allEdges().forEach((e) => {
  1088→    if (idSet.has(e.source) && idSet.has(e.target)) {
  1089→      out.get(e.source)!.push(e.target)
  1090→      indegree.set(e.target, (indegree.get(e.target) ?? 0) + 1)
  1091→    }
  1092→  })
  1093→
  1094→  // START（或入度 0 者）为第 0 层；拓扑序传播最长路径深度
  1095→  const queue = nodes.value
  1096→    .filter((n) => (indegree.get(n.id) ?? 0) === 0)
  1097→    .map((n) => n.id)
  1098→  const startId = nodes.value.find((n) => n.data.nodeType === 'START')?.id
  1099→  if (startId && (indegree.get(startId) ?? 0) > 0) queue.unshift(startId)
  1100→  queue.forEach((id) => {
  1101→    if (!depth.has(id)) depth.set(id, 0)
  1102→  })
  1103→  let maxDepth = 0
  1104→  while (queue.length) {
  1105→    const id = queue.shift()!
  1106→    const current = depth.get(id) ?? 0
  1107→    maxDepth = Math.max(maxDepth, current)
  1108→    for (const next of out.get(id) ?? []) {
  1109→      const candidate = current + 1
  1110→      if (candidate > (depth.get(next) ?? -1)) depth.set(next, candidate)
  1111→      const remaining = (indegree.get(next) ?? 1) - 1
  1112→      indegree.set(next, remaining)
  1113→      if (remaining === 0) queue.push(next)
  1114→    }
  1115→  }
  1116→  // 环上残余节点：接到最深层后面
  1117→  nodes.value.forEach((n) => {
  1118→    if (!depth.has(n.id)) depth.set(n.id, ++maxDepth)
  1119→  })
  1120→
  1121→  // 分层内按原 x 排序减少跳动；同层横向铺开，层级沿 Y 轴自上而下
  1122→  const byLayer = new Map<number, FlowNodeModel[]>()
  1123→  nodes.value.forEach((n) => {
  1124→    const layer = depth.get(n.id) ?? 0
  1125→    if (!byLayer.has(layer)) byLayer.set(layer, [])
  1126→    byLayer.get(layer)!.push(n)
  1127→  })
  1128→  ;[...byLayer.entries()]
  1129→    .sort(([a], [b]) => a - b)
  1130→    .forEach(([layer, list]) => {
  1131→      list.sort((a, b) => a.position.x - b.position.x)
  1132→      list.forEach((n, index) => {
  1133→        n.position = { x: leftX + index * colGapX, y: topY + layer * layerGapY }
  1134→      })
  1135→    })
  1136→  setTimeout(() => fitView({ padding: 0.15, maxZoom: 1, duration: 300 }), 50)
  1137→}
