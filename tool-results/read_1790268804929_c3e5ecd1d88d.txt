   967→        formCols = JSON.parse(cfg) as ColumnConfigItem[]
   968→      } catch {
   969→        formCols = []
   970→      }
   971→    } else if (Array.isArray(cfg)) {

   987→/** 覆盖策略 C：命中 key 全属性覆盖（保留 key），表单多出的 key 追加，当前列保留 */
   988→function overlayFromFormColumns(formCols: ColumnConfigItem[]) {
   989→  const target = form.type === 'API' ? apiColumns.value : sqlConfig.declaredColumns
   990→  const byKey = new Map(target.map((c) => [c.key, c]))
   991→  for (const fc of formCols) {
   992→    if (byKey.has(fc.key)) {
   993→      Object.assign(byKey.get(fc.key)!, fc, { key: fc.key })
   994→    } else {
   995→      target.push({ ...fc })
   996→      byKey.set(fc.key, target[target.length - 1])
   997→    }
   998→  }
   999→}

  1307→ /** API 类型：组装多操作 params（未配置的操作省略；列定义仅写入非空 key 行） */

  1315→       if (op === 'list') {

  1317→         if (cfg.totalParse && cfg.totalParse.trim()) item.totalParse = cfg.totalParse.trim()
  1318→       }

  1323→    const columns = apiColumns.value.filter((c) => c.key && c.key.trim())
  1324→    if (columns.length > 0) {

  1329→   if (form.keywordColumn && form.keywordColumn.trim()) params.keywordColumn = form.keywordColumn.trim()
  1330→   if (form.pageBase === 0 || form.pageBase === 1) params.pageBase = form.pageBase
  1331→   const dataObj = parseParamsJson(form.data)
  1332→   if (dataObj) params.data = dataObj
  1333→   const headersObj = parseParamsJson(form.headers)
  1334→   if (headersObj) params.headers = headersObj

  1370→  /** 可视化模式：前端生成预览 SQL（简化版，后端会重新生成） */
  1371→  function generatePreviewSql(): string {
  1372→    const v = sqlConfig.visual
  1373→    if (!v.mainTable) return ''
  1374→    let sql = `SELECT ${v.selectColumns.join(', ') || '*'}`
  1375→    sql += ` FROM ${v.mainTable} ${v.mainAlias || 'm'}`
  1376→    for (const j of v.joins) {
  1377→      if (j.targetTable && j.on) {
  1378→        sql += ` ${j.joinType} ${j.targetTable} ${j.alias} ON ${j.on}`
  1379→      }
  1380→    }
  1381→    sql += ` WHERE ${v.mainAlias || 'm'}.tenant_id = :tenantId`
  1382→    for (const w of v.where) {
  1383→      if (w.column && w.op) {
  1384→        sql += ` AND ${w.column} ${w.op} ?`
  1385→      }
  1386→    }
  1387→    if (v.orderBy.length > 0) {
  1388→      const parts = v.orderBy.filter((o) => o.column).map((o) => `${o.column} ${o.order || 'ASC'}`)
  1389→      if (parts.length > 0) sql += ` ORDER BY ${parts.join(', ')}`
  1390→    }
  1391→    return sql
  1392→  }

  1394→  /** 校验并保存 */
  1395→ async function handleSave() {
  1396→   if (!form.name || !form.name.trim()) {
  1397→     ElMessage.warning('请输入数据源名称')
  1398→     return
  1399→   }
  1400→    if (form.type === 'FORM' && !form.formKey) {
  1401→      ElMessage.warning('请选择绑定的业务表单')
  1402→      return
  1403→    }
  1404→    if (form.type === 'WORKFLOW' && !form.formKey) {
  1405→      ElMessage.warning('请选择绑定的工作流表单')
  1406→      return
  1407→    }
  1408→   if (form.type === 'SYSTEM' && !form.sourceKey) {
  1409→     ElMessage.warning('请选择系统结构')
  1410→     return
  1411→   }
  1412→   if (form.type === 'API') {
  1413→     if (!form.sourceKey || !form.sourceKey.trim()) {
  1414→       ElMessage.warning('请输入接口标识')
  1415→       return
  1416→     }
  1417→     if (!apiOps.list.action || !apiOps.list.action.trim()) {
  1418→       ElMessage.warning('列表查询 (list) 接口路径必填')
  1419→       return
  1420→     }
  1421→   }
  1422→    if (form.type === 'SQL') {
  1423→      if (!form.sourceKey || !form.sourceKey.trim()) {
  1424→        ElMessage.warning('请输入数据源标识（sourceKey）')
  1425→        return
  1426→      }
  1427→      if (sqlConfig.queryMode === 'visual' && !sqlConfig.visual.mainTable.trim()) {
  1428→        ElMessage.warning('请配置主表')
  1429→        return
  1430→      }
  1431→      if (sqlConfig.queryMode === 'sql' && !sqlConfig.queryText.trim()) {
  1432→        ElMessage.warning('请输入 SQL 模板')
  1433→        return
  1434→      }
  1435→    }
  1436→    if (form.type === 'FORM' && formJoin.value.queryMode === 'config') {
  1437→      const incomplete = formJoin.value.joins.find(
  1438→        (j) => !j.targetFormKey || !j.localField || !j.foreignField || !j.joinField || !j.virtualKey,
  1439→      )
  1440→      if (incomplete) {
  1441→        ElMessage.warning('请完整配置关联（目标表/关联字段/显示字段/虚拟列标识）')
  1442→        return
  1443→      }
  1444→    }
  1445→    if (form.type === 'FORM' && formJoin.value.queryMode === 'sql' && !(formJoin.value.query || '').trim()) {
  1446→      ElMessage.warning('请输入 SQL 模板')
  1447→      return
  1448→    }
  1449→    saving.value = true
  1450→    try {
  1451→      const payload = normalizePayload()
  1452→      if (editingId.value) {
  1453→        await dataSourceApi.updateDataSource(editingId.value, payload)
  1454→      } else {
  1455→        await dataSourceApi.createDataSource(payload)
  1456→      }
  1457→      ElMessage.success(editingId.value ? '保存成功' : '创建成功')
  1458→      // 清数据源 GET 缓存（含 metadata 30s TTL）：否则新配的 JOIN 虚拟列在元数据/编辑弹窗里最长 30s 不可见
  1459→      clearHttpCache('/v1/data-sources')
  1460→      inlineVisible.value = false
  1461→      tableRef.value?.fetchList()
  1462→    } catch {
  1463→      // http 拦截器已弹出错误消息
  1464→    } finally {
  1465→      saving.value = false
  1466→    }
  1467→ }

  1469→  /** FORM 类型：保留原始端点段（list/get/create/update/delete）基础上叠加关联查询配置段（queryMode/joins/query/columns/params） */
  1470→  function buildFormParams(): Record<string, any> {
  1471→    const params: Record<string, any> = { ...formJoinBaseParams.value }
  1472→    if (formJoin.value.queryMode === 'config') {
  1473→      params.queryMode = 'config'
  1474→      params.joins = formJoin.value.joins.filter((j) => j.targetFormKey && j.virtualKey)
  1475→    } else if (formJoin.value.queryMode === 'sql') {
  1476→      params.queryMode = 'sql'

  1478→        params.query = formJoin.value.query

  1481→      if (cols.length > 0) {

  1483→      }
  1484→      if (formJoin.value.params && formJoin.value.params.length > 0) {
  1485→        params.params = [...formJoin.value.params]
  1486→      }
  1487→    }
  1488→    return params
  1489→  }
