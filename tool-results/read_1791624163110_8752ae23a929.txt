   470→    id: createNodeId(type, allNodes().map((n) => n.id)),
   471→    type: 'logic',
   472→    position,
   473→    data: {
   474→      nodeType: type,
   475→      name: defaultNodeName(type),
   476→      config: defaultConfig(type),
   477→      errorAction: type === 'HTTP' || type === 'BEAN' || type === 'SCRIPT' ? 'FAIL_FLOW' : undefined,
   478→    },
   479→  }
   480→  addNodes([node])
   481→  if (type === 'BATCH') {
   482→    addEdges([createLoopEdge(node.id, node.id, LOOP_HANDLE_START, LOOP_HANDLE_END)])
   483→  }
   484→  selectedNodeId.value = node.id
   485→  return node
   486→}

   571→// ===== 节点拖到连线自动吸附接入（普通连线 / 批处理循环虚线） =====
   572→
   573→/** 吸附判定距离（画布坐标 px）：主流实线 */
   574→const SNAP_THRESHOLD = 32
   575→
   576→/** 循环虚线独立吸附阈值：与 U 形外凸宽度（offset 56）对齐，视觉上「碰到虚线即吸」更跟手 */
   577→const LOOP_SNAP_THRESHOLD = 56
   578→
   579→/** vue-flow 未回填 dimensions 时的节点近似尺寸兜底 */
   580→const NODE_FALLBACK_SIZE = { w: 160, h: 54 }
   581→
   582→function nodeSize(node: FlowNodeModel): { w: number; h: number } {
   583→  const dims = (node as unknown as { dimensions?: { width?: number; height?: number } })
   584→    ?.dimensions
   585→  return {
   586→    w: Number(dims?.width) || NODE_FALLBACK_SIZE.w,
   587→    h: Number(dims?.height) || NODE_FALLBACK_SIZE.h,
   588→  }
   589→}
   590→
   591→function nodeCenter(node: FlowNodeModel): { x: number; y: number } {
   592→  const size = nodeSize(node)
   593→  return { x: node.position.x + size.w / 2, y: node.position.y + size.h / 2 }
   594→}
   595→
   596→/** 两节点间正交折线近似（中位水平线），与 smoothstep 走向一致 */
   597→function approxPointsBetween(
   598→  source: FlowNodeModel,
   599→  target: FlowNodeModel
   600→): { x: number; y: number }[] {
   601→  const sc = nodeCenter(source)
   602→  const tc = nodeCenter(target)
   603→  const midY = (sc.y + tc.y) / 2
   604→  return [sc, { x: sc.x, y: midY }, { x: tc.x, y: midY }, tc]
   605→}
   606→
   607→/** 自环 U 形近似：节点右侧外凸（与 LoopEdge.vue 渲染 offset 56 同侧同距） */
   608→function selfLoopApproxPoints(batch: FlowNodeModel): { x: number; y: number }[] {
   609→  const size = nodeSize(batch)
   610→  const x = batch.position.x + size.w
   611→  const y1 = batch.position.y + size.h * 0.3
   612→  const y2 = batch.position.y + size.h * 0.74
   613→  const bulge = 56
   614→  return [
   615→    { x, y: y1 },
   616→    { x: x + bulge, y: y1 },
   617→    { x: x + bulge, y: y2 },
   618→    { x, y: y2 },
   619→  ]
   620→}
   621→
   622→/** 边路径近似：源/目标节点中心的正交折线（中位水平线），与 smoothstep 走向一致 */
   623→function edgeApproxPoints(edge: FlowEdge): { x: number; y: number }[] | null {
   624→  const source = allNodes().find((n) => n.id === edge.source)
   625→  const target = allNodes().find((n) => n.id === edge.target)
   626→  if (!source || !target) return null
   627→  // 循环直连边（batch→batch 自环）：右侧 U 形近似（loop_start 30% → 外凸 → loop_end 74%）
   628→  if (isLoopEdge(edge) && edge.source === edge.target) {
   629→    return selfLoopApproxPoints(source)
   630→  }
   631→  return approxPointsBetween(source, target)
   632→}
   633→
   634→function distToSegment(
   635→  p: { x: number; y: number },
   636→  a: { x: number; y: number },
   637→  b: { x: number; y: number }
   638→): number {
   639→  const dx = b.x - a.x
   640→  const dy = b.y - a.y
   641→  const len2 = dx * dx + dy * dy
   642→  const t = len2 === 0 ? 0 : Math.max(0, Math.min(1, ((p.x - a.x) * dx + (p.y - a.y) * dy) / len2))
   643→  return Math.hypot(p.x - (a.x + t * dx), p.y - (a.y + t * dy))
   644→}
   645→
   646→function distToPolyline(p: { x: number; y: number }, pts: { x: number; y: number }[]): number {
   647→  let min = Infinity
   648→  for (let i = 1; i < pts.length; i++) {
   649→    min = Math.min(min, distToSegment(p, pts[i - 1], pts[i]))
   650→  }
   651→  return min
   652→}
   653→
   654→/** 节点是否已挂在某条循环体链上（循环体节点同时只能属于一条循环链）。
   655→ * 按 nodeId 所在端分别判定：batch→链首 段的 sourceHandle 恰为 loop_start、
   656→ * 链尾→batch 段的 targetHandle 恰为 loop_end（合法链段），故不能要求
   657→ * 一条边两端同时为普通口 —— 旧写法会把单节点循环链的成员误判为未挂链，
   658→ * 导致其拖出失效、且可被重复吸附（bug：循环拖出 E2E 暴露） */
   659→function nodeOnLoopChain(nodeId: string): boolean {
   660→  return allEdges().some(
   661→    (e) =>
   662→      isLoopEdge(e) &&
   663→      ((e.source === nodeId && e.sourceHandle !== LOOP_HANDLE_START) ||
   664→        (e.target === nodeId && e.targetHandle !== LOOP_HANDLE_END))
   665→  )
   666→}
   667→
   668→interface SnapCandidate {
   669→  edge: FlowEdge
   670→  isLoop: boolean
   671→  dist: number
   672→}
   673→
   674→/** 某条循环边所属的批处理（链源头）：链首段/尾段按 handle 直判，中段沿入链上溯 */
   675→function loopChainOwner(edge: FlowEdge): string | null {
   676→  const loopEdges = allEdges().filter(isLoopEdge)
   677→  if (edge.sourceHandle === LOOP_HANDLE_START) return edge.source
   678→  if (edge.targetHandle === LOOP_HANDLE_END) return edge.target
   679→  let cur = edge.source
   680→  for (let i = 0; i < 100; i++) {
   681→    const up = loopEdges.find((e) => e.target === cur)
   682→    if (!up) return null
   683→    if (up.sourceHandle === LOOP_HANDLE_START) return up.source
   684→    cur = up.source
   685→  }
   686→  return null
   687→}
   688→
   689→/** 批处理的循环体后代（含直接与嵌套子孙 batch）：沿 loop 链递归收集，防环用 */
   690→function batchDescendants(batchId: string, visited = new Set<string>()): string[] {
   691→  const out: string[] = []
   692→  if (visited.has(batchId)) return out
   693→  visited.add(batchId)
   694→  const loopEdges = allEdges().filter(isLoopEdge)
   695→  const start = loopEdges.find(
   696→    (e) => e.source === batchId && e.sourceHandle === LOOP_HANDLE_START
   697→  )
   698→  if (!start) return out
   699→  const seen = new Set<string>([batchId])
   700→  let cur: string | undefined = start.target
   701→  let guard = 0
   702→  while (cur && !seen.has(cur) && guard++ < 100) {
   703→    seen.add(cur)
   704→    const node = allNodes().find((n) => n.id === cur)
   705→    if (node?.data?.nodeType === 'BATCH') {
   706→      out.push(cur)
   707→      out.push(...batchDescendants(cur, visited))
   708→    }
   709→    const next = loopEdges.find((e) => e.source === cur)
   710→    if (!next) break
   711→    cur = next.target
   712→  }
   713→  return out
   714→}
   715→
   716→/** 嵌套/类型不可入提示节流（画布坐标拖动中每 4s 至多一次） */
   717→let loopHintLastAt = 0
   718→let loopHintText = ''
   719→
   720→/** 找被拖节点附近的边（最近且距离达标）；不满足类型/归属约束的边跳过 */
   721→function findSnapEdge(node: FlowNodeModel): SnapCandidate | null {
   722→  const nodeType = node.data?.nodeType
   723→  if (!nodeType || nodeType === 'START') return null
   724→  const center = nodeCenter(node)
   725→  let best: SnapCandidate | null = null
   726→  let loopHintNearby = false
   727→  loopHintText = ''
   728→  for (const edge of allEdges()) {
   729→    if (edge.source === node.id || edge.target === node.id) continue
   730→    const isLoop = isLoopEdge(edge)
   731→    if (isLoop) {
   732→      // 循环体仅收业务执行六型（START/END/CONDITION 不可入；BATCH=嵌套，须防环）
   733→      if (!(BATCH_BODY_TYPES as string[]).includes(nodeType)) {
   734→        if (distToPolyline(center, edgeApproxPoints(edge) ?? []) <= LOOP_SNAP_THRESHOLD) {
   735→          loopHintNearby = true
   736→          loopHintText = '该节点不可加入循环体（仅 HTTP/Bean/脚本/数据更新/子流程/批处理）'
   737→        }
   738→        continue
   739→      }
   740→      if (nodeOnLoopChain(node.id)) continue
   741→      // 嵌套批处理防环：BATCH 不能入自身/自身后代的循环体链（否则运行时无限递归）
   742→      if (nodeType === 'BATCH') {
   743→        const owner = loopChainOwner(edge)
   744→        if (owner && (owner === node.id || batchDescendants(node.id).includes(owner))) {
   745→          loopHintNearby = true
   746→          loopHintText = owner === node.id ? '批处理不能嵌套自身' : '批处理不能嵌套形成循环引用'
   747→          continue
   748→        }
   749→      }
   750→    } else if (nodeOnLoopChain(node.id)) {
   751→      // 已在循环链上的节点不可再接入主流（序列化时会被剔出顶层导致悬空边）
   752→      continue
   753→    }
   754→    const pts = edgeApproxPoints(edge)
   755→    if (!pts) continue
   756→    const dist = distToPolyline(center, pts)
   757→    const threshold = isLoop ? LOOP_SNAP_THRESHOLD : SNAP_THRESHOLD
   758→    if (dist <= threshold && (!best || dist < best.dist)) {
   759→      best = { edge, isLoop, dist }
   760→    }
   761→  }
   762→  if (!best && loopHintNearby) {
   763→    // 拖到虚线附近但不可入/成环：节流提示（无候选吸附时才弹，避免拖动中刷屏）
   764→    const now = Date.now()
   765→    if (now - loopHintLastAt > 4000) {
   766→      loopHintLastAt = now
   767→      ElMessage.warning(loopHintText)
   768→    }
   769→  }
   770→  return best
   771→}
   772→
   773→/** 吸附接入：原边拆成两段穿过该节点（循环边拆出的两段仍是循环边）。
   774→ * 顺序：先加新边再删旧边 —— vue-flow removeEdges 会跳过 deletable:false 的边，
   775→ * 且删旧边触发的自愈（nextTick）必须在链完整后执行才能正确判 complete。
   776→ * 已在链上的节点接入新线前先断开旧链（拖换位置/换链一步到位）。 */
   777→function snapInsertNode(node: FlowNodeModel, candidate: SnapCandidate): void {
   778→  detachNodeFromChains(node)
   779→  const edge = candidate.edge
   780→  const isLoop = candidate.isLoop
   781→  const data = isLoop ? { loop: true } : edge.data ? { ...edge.data } : undefined
   782→  const segBase = isLoop
   783→    ? { type: 'smoothstep', markerEnd: 'arrowclosed', class: 'lf-edge-loop' }
   784→    : { type: 'smoothstep', markerEnd: 'arrowclosed' }
   785→  addEdges([
   786→    {
   787→      source: edge.source,
   788→      target: node.id,
   789→      sourceHandle: edge.sourceHandle,
   790→      targetHandle: 'in',
   791→      data,
   792→      ...segBase,
   793→    },
   794→    {
   795→      source: node.id,
   796→      target: edge.target,
   797→      sourceHandle: 'out',
   798→      targetHandle: edge.targetHandle,
   799→      data: isLoop ? { loop: true } : undefined,
   800→      ...segBase,
   801→    },
   802→  ])
   803→  removeEdges([edge])
   804→  ElMessage.success(isLoop ? '已加入循环体（按链序逐项执行）' : '已自动接入连线')
   805→}
   806→
   807→/** 吸附统一入口：新节点落位 / 画布内拖动结束共用 */
   808→function trySnapNode(node: FlowNodeModel) {
   809→  const candidate = findSnapEdge(node)
   810→  if (candidate) snapInsertNode(node, candidate)
   811→}

   813→// ===== 节点解绑：按住 Shift 拖动 / 双击链上节点 → 断开并缝合原链 =====
   814→
   815→interface ChainEdges {
   816→  ins: FlowEdge[]
   817→  outs: FlowEdge[]
   818→}
   819→
   820→/** 节点的非循环连边 */
   821→function mainEdgesOf(nodeId: string): ChainEdges {
   822→  const ins: FlowEdge[] = []
   823→  const outs: FlowEdge[] = []
   824→  for (const e of allEdges()) {
   825→    if (isLoopEdge(e)) continue
   826→    if (e.target === nodeId) ins.push(e)
   827→    if (e.source === nodeId) outs.push(e)
   828→  }
   829→  return { ins, outs }
   830→}
   831→
   832→/** 节点在循环链上的连边（排除 batch 自身的 loop_start/loop_end 自环端） */
   833→function loopEdgesOf(nodeId: string): ChainEdges {
   834→  const ins: FlowEdge[] = []
   835→  const outs: FlowEdge[] = []
   836→  for (const e of allEdges()) {
   837→    if (!isLoopEdge(e)) continue
   838→    if (e.target === nodeId && e.targetHandle !== LOOP_HANDLE_END) ins.push(e)
   839→    if (e.source === nodeId && e.sourceHandle !== LOOP_HANDLE_START) outs.push(e)
   840→  }
   841→  return { ins, outs }
   842→}
   843→
   844→/** 主流链中段判定：恰一进一出（START 无入/END 无出/分支出边>1 天然排除） */
   845→function isMainChainInternal(nodeId: string): boolean {
   846→  const { ins, outs } = mainEdgesOf(nodeId)
   847→  return ins.length === 1 && outs.length === 1
   848→}
   849→
   850→/** 主流缝合直连：删 N 的两断边并补 src→tgt（保留 CONDITION 分支 sourceHandle） */
   851→function stitchMainChain(node: FlowNodeModel): void {
   852→  const { ins, outs } = mainEdgesOf(node.id)
   853→  const inE = ins[0]
   854→  const outE = outs[0]
   855→  const branch =
   856→    inE.sourceHandle === 'true' || inE.sourceHandle === 'false' ? inE.sourceHandle : undefined
   857→  addEdges([
   858→    {
   859→      source: inE.source,
   860→      target: outE.target,
   861→      sourceHandle: inE.sourceHandle ?? undefined,
   862→      targetHandle: outE.targetHandle ?? undefined,
   863→      data: branch ? { branch } : undefined,
   864→    },
   865→  ])
   866→  removeEdges([inE, outE])
   867→}
   868→
   869→/**
   870→ * 断开节点与既有链的连接（接入新线前置 / 双击移出共用）：
   871→ * - 主流中段：缝合前后直连；
   872→ * - 循环链：删全部循环连边，链自愈（nextTick healBatchLoops）缝合。
   873→ * 返回是否发生了断开。
   874→ */
   875→function detachNodeFromChains(node: FlowNodeModel): boolean {
   876→  if (isMainChainInternal(node.id)) {
   877→    stitchMainChain(node)
   878→    return true
   879→  }
   880→  if (nodeOnLoopChain(node.id)) {
   881→    const { ins, outs } = loopEdgesOf(node.id)
   882→    if (ins.length || outs.length) {
   883→      removeEdges([...ins, ...outs])
   884→      return true
   885→    }
   886→  }
   887→  return false
   888→}
   889→
   890→/** 解绑统一入口（Shift 拖动 / 双击链上节点共用）：解绑成功则弹消息 */
   891→function detachWithMessage(node: FlowNodeModel): void {
   892→  // 先判链归属再断开（断开后边已删，isMainChainInternal 恒为 false）
   893→  const wasMain = isMainChainInternal(node.id)
   894→  if (detachNodeFromChains(node)) {
   895→    ElMessage.success(wasMain ? '已从连线解绑，前后节点已重新连接' : '已移出循环体，循环链已重新缝合')
   896→  }
   897→}

   899→function onNodeDragStop({ event, node }: { event: { shiftKey?: boolean }; node: FlowNodeModel }) {
   900→  if (!node?.id) return
   901→  // 需求②：按住 Shift 移动节点即解绑连线（不按 shift 时不再做距离自动断开，避免误触）
   902→  if (event?.shiftKey) {
   903→    detachWithMessage(node)
   904→    return
   905→  }
   906→  const candidate = findSnapEdge(node)
   907→  if (candidate) {
   908→    snapInsertNode(node, candidate)
   909→  }
   910→}

   914→function onNodesChange(changes: NodeChange[]) {
   915→  if (changes.some((c) => c.type === 'remove')) {
   916→    nextTick(() => healBatchLoops())
   917→  }

   919→

   933→function healOneBatch(batchId: string) {
   934→  // store 边为 vue-flow 内部 Edge 类型，按本设计器 FlowEdge 形状读取（字段子集兼容）
   935→  const loopEdges = (storeEdges.value as unknown as FlowEdge[]).filter(isLoopEdge)
   936→  const directExists = loopEdges.some(
   937→    (e) =>
   938→      e.source === batchId &&
   939→      e.sourceHandle === LOOP_HANDLE_START &&
   940→      e.target === batchId &&
   941→      e.targetHandle === LOOP_HANDLE_END
   942→  )
   943→
   944→  // 正向：loop_start 出发沿链，until 回到 loop_end（完整）或无出边（断口在 tail 之后）。
   945→  // 嵌套批处理：后续跳只走链段边（sourceHandle='out'），跳过链上子 BATCH
   946→  // 自己的自环（sourceHandle=loop_start）——否则子自环被误当链路下一跳，
   947→  // 走环保护 break 导致误判断链而重复补链（bug：嵌套 E2E 暴露）。
   948→  let tail = batchId
   949→  let forwardComplete = false
   950→  {
   951→    const visited = new Set([batchId])
   952→    let cur = loopEdges.find((e) => e.source === batchId && e.sourceHandle === LOOP_HANDLE_START)
   953→    let guard = 0
   954→    while (cur && guard++ < 100) {
   955→      if (cur.target === batchId && cur.targetHandle === LOOP_HANDLE_END) {
   956→        forwardComplete = true
   957→        break
   958→      }
   959→      if (visited.has(cur.target)) break
   960→      visited.add(cur.target)
   961→      tail = cur.target
   962→      cur = loopEdges.find(
   963→        (e) => e.source === cur!.target && e.sourceHandle !== LOOP_HANDLE_START
   964→      )
   965→    }
   966→  }
   967→  if (forwardComplete) return
   968→
   969→  // 反向：loop_end 回溯链段头部（后续跳只走链段边 targetHandle='in'，同上跳过子 BATCH 自环）
   970→  let head = batchId
   971→  {
   972→    const visited = new Set([batchId])
   973→    let cur = loopEdges.find((e) => e.target === batchId && e.targetHandle === LOOP_HANDLE_END)
   974→    let guard = 0
   975→    while (cur && guard++ < 100) {
   976→      if (visited.has(cur.source)) break
   977→      visited.add(cur.source)
   978→      head = cur.source
   979→      cur = loopEdges.find(
   980→        (e) =>
   981→          e.target === cur!.source && e.source !== batchId && e.targetHandle !== LOOP_HANDLE_END
   982→      )
   983→    }
   984→  }
   985→
   986→  const patch: FlowEdge[] = []
   987→  if (tail === batchId && head === batchId) {
   988→    // 链完全为空：确保 loop_start→loop_end 直连存在
   989→    if (!directExists) {
   990→      patch.push(createLoopEdge(batchId, batchId, LOOP_HANDLE_START, LOOP_HANDLE_END))
   991→    }
   992→  } else if (tail !== batchId && head !== batchId) {
   993→    if (tail === head) {
   994→      // 仅剩单个中段节点：两端各接一条
   995→      patch.push(createLoopEdge(batchId, tail, LOOP_HANDLE_START, 'in'))
   996→      patch.push(createLoopEdge(tail, batchId, 'out', LOOP_HANDLE_END))
   997→    } else {
   998→      patch.push(createLoopEdge(tail, head, 'out', 'in'))
   999→    }
  1000→  } else if (tail !== batchId) {
  1001→    // 正向链存在但 loop_end 断开
  1002→    patch.push(createLoopEdge(tail, batchId, 'out', LOOP_HANDLE_END))
  1003→  } else {
  1004→    // loop_start 无出链但 loop_end 有前驱段
  1005→    patch.push(createLoopEdge(batchId, head, LOOP_HANDLE_START, 'in'))
  1006→  }
  1007→  if (patch.length) addEdges(patch)
  1008→}

  1045→function pushHistory(dsl: string) {
  1046→  // 丢弃撤销后的前进分支，追加快照（上限 100 防内存膨胀）
  1047→  historyStack.value = historyStack.value.slice(0, historyIndex.value + 1)
  1048→  historyStack.value.push(dsl)
  1049→  if (historyStack.value.length > 100) historyStack.value.shift()
  1050→  historyIndex.value = historyStack.value.length - 1
  1051→}

  1053→function applySnapshot(dsl: string) {
  1054→  applyingHistory.value = true
  1055→  try {
  1056→    const graph = parseDsl(dsl)
  1057→    nodes.value = graph.nodes
  1058→    edges.value = graph.edges
  1059→    selectedNodeId.value = null
  1060→  } catch {
  1061→    // 快照损坏不应发生（均为本组件序列化产物），忽略
  1062→  } finally {
  1063→    setTimeout(() => {
  1064→      applyingHistory.value = false
  1065→    }, 0)
  1066→  }
  1067→}

  1069→function undo() {
  1070→  if (!canUndo.value) return
  1071→  historyIndex.value -= 1
  1072→  applySnapshot(historyStack.value[historyIndex.value])
  1073→}
  1074→
  1075→function redo() {
  1076→  if (!canRedo.value) return
  1077→  historyIndex.value += 1
  1078→  applySnapshot(historyStack.value[historyIndex.value])
  1079→}

  1098→// ===== 自动整理布局（v4）：主流边 Kahn 分层（自上而下、同层横向铺开，间距按节点
  1099→// 实际尺寸自适应），同层主流节点共享「层中线」垂直居中对齐。BATCH 循环体链不参与
  1100→// 主流分层：v4 按链序在宿主 BATCH 右侧纵向排列——循环链成员连线走底部 out → 顶部
  1101→// in 手柄，纵向排布时链边为竖直直线（v3 横向成行迫使链边走 S 形弯）；列首锚定
  1102→// loop_start 线（卡高 30%）下方，链走完自末节点回 loop_end（74% 线），闭环呈
  1103→// 「右出 → 纵向下行 → 右回」顺时针回路；列内节点共享「列中轴」水平居中对齐。
  1104→// 嵌套 BATCH 的循环列以宿主为锚继续右移下探，嵌套层次从外到内自左向右展开
  1105→// （子列左缘避开宿主卡、宿主列最宽成员与前序嵌套兄弟子树，纵向并行的列互不
  1106→// 重叠）；同层横向铺开时为 BATCH 的整棵循环子树让位，循环列不压右侧兄弟卡片。
  1107→// 修复：循环链边（loop_start→链→loop_end→BATCH）构成图环，旧版把环上节点全部堆到
  1108→// 兜底层导致「加入批处理节点后整理布局混乱」；层高按居中后的块底缘精确累计。 =====
  1109→
  1110→/** loop_start 手柄纵锚（卡高比例，与 FlowNode.vue 右侧手柄 top:30% 对齐） */
  1111→const LOOP_START_ANCHOR = 0.3
  1112→/** 循环列与宿主 BATCH 右缘的间距（与自环 U 形外凸 offset 56 视觉对齐） */
  1113→const BODY_COL_GAP_X = 56
  1114→/** 循环列内相邻成员的纵向间距（竖直连线段呼吸感，与主流层距同量级） */
  1115→const BODY_COL_GAP_Y = 56
  1116→/** 列首成员顶缘位于 loop_start 线（卡高 30%）下方的落差：入边经 smoothstep 20px
  1117→ * 手柄偏移平滑落入顶部 in 口，避免与 loop_start 线等高产生回折 */
  1118→const BODY_COL_ANCHOR_DROP = 24
  1119→/** 嵌套子循环列左缘相对宿主列最宽成员右缘的让位余量（防纵向并行的列间重叠） */
  1120→const BODY_COL_NEST_MARGIN = 24
  1121→
  1122→/** 收集某 BATCH 的循环体链（画布节点 id 序列，保序）：loop_start 出边沿链行走至
  1123→ * 回环 loop_end，visited 防环。嵌套 BATCH 是链中普通一员（照常收进、布局时递归
  1124→ * 展开其子列），链经其 'out' 口继续——旧版在嵌套 BATCH 处截断，会使其后的外层
  1125→ * 链成员（嵌套BATCH→尾节点→loop_end）失去归属、混入主流分层被甩到顶层 */
  1126→function collectBodyChain(batchId: string): string[] {
  1127→  const chain: string[] = []
  1128→  const loopEdges = allEdges().filter(isLoopEdge)
  1129→  const startEdge = loopEdges.find(
  1130→    (e) => e.source === batchId && e.sourceHandle === LOOP_HANDLE_START
  1131→  )
  1132→  if (!startEdge) return chain
  1133→  const nodesById = new Map(allNodes().map((n) => [n.id, n]))
  1134→  const seen = new Set<string>([batchId])
  1135→  let cur = startEdge.target
  1136→  let guard = 0
  1137→  while (cur && !seen.has(cur) && guard++ < 100) {
  1138→    seen.add(cur)
  1139→    const node = nodesById.get(cur)
  1140→    if (!node) break
  1141→    chain.push(cur)
  1142→    const next = loopEdges.find(
  1143→      (e) => e.source === cur && e.sourceHandle !== LOOP_HANDLE_START
  1144→    )
  1145→    if (!next) break
  1146→    cur = next.target
  1147→  }
  1148→  return chain
  1149→}

  1151→function handleAutoLayout() {
  1152→  if (!nodes.value.length) return
  1153→  const leftX = 80
  1154→  const colGapX = 56
  1155→  const topY = 60
  1156→  const layerGapY = 64
  1157→
  1158→  const nodesById = new Map(allNodes().map((n) => [n.id, n]))
  1159→
  1160→  // 1) 循环体归属：无条件收集每个 BATCH 的 body 链（嵌套 BATCH 亦然——若因遍历
  1161→  //    顺序被宿主链先行占名而跳过收集，placeChain 递归将找不到其链，其成员会滞留
  1162→  //    原位不参与布局）；成员从主流分层名单剔除（成员只有 loop 边，主流入度恒 0，
  1163→  //    混进分层会把它们误当第 0 层起点）
  1164→  const bodyOwner = new Map<string, string>()
  1165→  const bodyChains = new Map<string, string[]>()
  1166→  for (const n of allNodes()) {
  1167→    if (n.data.nodeType !== 'BATCH') continue
  1168→    const chain = collectBodyChain(n.id)
  1169→    if (!chain.length) continue
  1170→    bodyChains.set(n.id, chain)
  1171→    chain.forEach((id) => bodyOwner.set(id, n.id))
  1172→  }
  1173→
  1174→  // 2) 主流边（排除循环链边）Kahn 分层：START（或入度 0 者）为第 0 层，拓扑序传播最长路径深度
  1175→  const mainIds = allNodes().filter((n) => !bodyOwner.has(n.id)).map((n) => n.id)
  1176→  const idSet = new Set(mainIds)
  1177→  const depth = new Map<string, number>()
  1178→  const indegree = new Map<string, number>()
  1179→  const out = new Map<string, string[]>()
  1180→  mainIds.forEach((id) => {
  1181→    indegree.set(id, 0)
  1182→    out.set(id, [])
  1183→  })
  1184→  allEdges().forEach((e) => {
  1185→    if (isLoopEdge(e)) return
  1186→    if (idSet.has(e.source) && idSet.has(e.target)) {
  1187→      out.get(e.source)!.push(e.target)
  1188→      indegree.set(e.target, (indegree.get(e.target) ?? 0) + 1)
  1189→    }
  1190→  })
  1191→  const queue = mainIds.filter((id) => (indegree.get(id) ?? 0) === 0)
  1192→  const startId = allNodes().find((n) => n.data.nodeType === 'START')?.id
  1193→  if (startId && idSet.has(startId) && (indegree.get(startId) ?? 0) > 0) queue.unshift(startId)
  1194→  queue.forEach((id) => {
  1195→    if (!depth.has(id)) depth.set(id, 0)
  1196→  })
  1197→  let maxDepth = 0
  1198→  while (queue.length) {
  1199→    const id = queue.shift()!
  1200→    const current = depth.get(id) ?? 0
  1201→    maxDepth = Math.max(maxDepth, current)
  1202→    for (const next of out.get(id) ?? []) {
  1203→      if (current + 1 > (depth.get(next) ?? -1)) depth.set(next, current + 1)
  1204→      const remaining = (indegree.get(next) ?? 1) - 1
  1205→      indegree.set(next, remaining)
  1206→      if (remaining === 0) queue.push(next)
  1207→    }
  1208→  }
  1209→  // 主流环上残余节点：接到最深层后面
  1210→  mainIds.forEach((id) => {
  1211→    if (!depth.has(id)) depth.set(id, ++maxDepth)
  1212→  })
  1213→
  1214→  // 3) 子树纵向块高：普通节点=自身卡高；BATCH=卡片顶缘到 max(卡底, 循环列最深
  1215→  //    底缘)——列首锚 = 卡高 30% + 落差 24，成员自上而下逐个累计，嵌套 BATCH 成员
  1216→  //    的块底缘 = 其卡片顶纵距 + 子块块高。供层内块底缘与层高精确累计用
  1217→  const blockH = new Map<string, number>()
  1218→  function blockHeight(id: string, stack: Set<string> = new Set()): number {
  1219→    if (blockH.has(id)) return blockH.get(id)!
  1220→    const self = nodeSize(nodesById.get(id)!).h
  1221→    if (stack.has(id)) return self
  1222→    stack.add(id)
  1223→    let h = self
  1224→    const chain = bodyChains.get(id)
  1225→    if (chain?.length) {
  1226→      const colTop = self * LOOP_START_ANCHOR + BODY_COL_ANCHOR_DROP
  1227→      let offset = 0
  1228→      let deepest = 0
  1229→      for (const cid of chain) {
  1230→        const cn = nodesById.get(cid)
  1231→        const cardH = cn ? nodeSize(cn).h : 0
  1232→        deepest = Math.max(
  1233→          deepest,
  1234→          offset + (cn?.data.nodeType === 'BATCH' ? blockHeight(cid, stack) : cardH)
  1235→        )
  1236→        offset += cardH + BODY_COL_GAP_Y
  1237→      }
  1238→      h = Math.max(self, colTop + deepest)
  1239→    }
  1240→    stack.delete(id)
  1241→    blockH.set(id, h)
  1242→    return h
  1243→  }
  1244→  mainIds.forEach((id) => blockHeight(id))
  1245→
  1246→  // 3b) 循环子树横向跨度（自列左缘 = 宿主卡右缘+56 起算）：自身列宽 与 各嵌套 BATCH
  1247→  //     子列（左缘含让位与兄弟列级联）及其子树跨度最右值取大。与 placeChain 的
  1248→  //     nestLeft 公式严格同构，供同层横向铺开时为循环子树让位，防止循环列压到
  1249→  //     同层右侧兄弟卡片
  1250→  const colW = new Map<string, number>()
  1251→  function colSubtreeW(id: string, stack: Set<string> = new Set()): number {
  1252→    if (colW.has(id)) return colW.get(id)!
  1253→    const chain = bodyChains.get(id)
  1254→    if (!chain?.length) return 0
  1255→    if (stack.has(id)) return 0
  1256→    stack.add(id)
  1257→    const colMaxW = Math.max(...chain.map((cid) => nodeSize(nodesById.get(cid)!).w))
  1258→    let extent = colMaxW
  1259→    let prevRight = -Infinity
  1260→    for (const cid of chain) {
  1261→      const cn = nodesById.get(cid)
  1262→      if (!cn || cn.data.nodeType !== 'BATCH') continue
  1263→      const nestLeftRel = Math.max(
  1264→        (colMaxW + nodeSize(cn).w) / 2 + BODY_COL_GAP_X,
  1265→        colMaxW + BODY_COL_NEST_MARGIN,
  1266→        prevRight + BODY_COL_GAP_X
  1267→      )
  1268→      prevRight = nestLeftRel + colSubtreeW(cid, stack)
  1269→      extent = Math.max(extent, prevRight)
  1270→    }
  1271→    stack.delete(id)
  1272→    colW.set(id, extent)
  1273→    return extent
  1274→  }
  1275→  mainIds.forEach((id) => colSubtreeW(id))
  1276→
  1277→  // 4) 分层布局：层内主流节点卡片共享「层中线」垂直居中（中线 = 层顶 + 最高卡高/2，
  1278→  //    高低卡片对齐同一水平视线）；层高按「中线 - 卡高/2 + 块高」最大值累计
  1279→  //    （BATCH 的循环列块底缘计入，保证下层不被循环列侵入）；同层按累计宽度横向铺开
  1280→  const byLayer = new Map<number, FlowNodeModel[]>()
  1281→  mainIds.forEach((id) => {
  1282→    const layer = depth.get(id) ?? 0
  1283→    if (!byLayer.has(layer)) byLayer.set(layer, [])
  1284→    byLayer.get(layer)!.push(nodesById.get(id)!)
  1285→  })
  1286→  const layerCenter = new Map<number, number>()
  1287→  let cursorY = topY
  1288→  const layers = [...byLayer.entries()].sort(([a], [b]) => a - b)
  1289→  layers.forEach(([layer, list]) => {
  1290→    const maxCardH = Math.max(...list.map((n) => nodeSize(n).h))
  1291→    const center = cursorY + maxCardH / 2
  1292→    layerCenter.set(layer, center)
  1293→    let bandBottom = center + maxCardH / 2
  1294→    list.forEach((n) => {
  1295→      bandBottom = Math.max(bandBottom, center - nodeSize(n).h / 2 + blockHeight(n.id))
  1296→    })
  1297→    cursorY = bandBottom + layerGapY
  1298→  })
  1299→  layers.forEach(([layer, list]) => {
  1300→    const center = layerCenter.get(layer)!
  1301→    list.sort((a, b) => a.position.x - b.position.x)
  1302→    let cursorX = leftX
  1303→    list.forEach((n) => {
  1304→      const size = nodeSize(n)
  1305→      n.position = { x: Math.round(cursorX), y: Math.round(center - size.h / 2) }
  1306→      // 循环子树横向让位：BATCH 的循环列悬于卡右侧，同层后续节点须为其整棵
  1307→      // 循环子树（含嵌套列级联）让出横向空间，防止列卡重叠
  1308→      const subtreeW = colSubtreeW(n.id)
  1309→      cursorX += size.w + (subtreeW > 0 ? BODY_COL_GAP_X + subtreeW + colGapX : colGapX)
  1310→    })
  1311→  })
  1312→
  1313→  // 5) 循环列落位：列悬于宿主 BATCH 右侧、按链序自上而下（连线自右侧 loop_start
  1314→  //    手柄出 → 列内竖直直线下行 → 末节点回 loop_end，顺时针闭环）；列内节点共享
  1315→  //    「列中轴」水平居中（x = 中轴 - 卡宽/2，in/out 手柄居中故链边为竖直直线），
  1316→  //    列首顶缘锚定 loop_start 线（卡高 30%）下方 BODY_COL_ANCHOR_DROP。嵌套 BATCH
  1317→  //    的循环列以宿主为锚继续右移下探：子列左缘 ≥ max(宿主卡右缘+56, 宿主列最宽
  1318→  //    成员右缘+24, 前序嵌套兄弟子树右缘+56)，嵌套层次从外到内自左向右展开且列间
  1319→  //    互不重叠；minLeftX 由宿主列逐层下传，placed 防交叉嵌套环。
  1320→  //    注：布局后坐标不再匹配 DSL 默认位公式，将按绝对坐标持久化（拖动位置同路径）
  1321→  const placed = new Set<string>()
  1322→  /** 落位某 BATCH 的循环列，返回整棵循环子树的最右缘（含嵌套列级联，供兄弟列让位） */
  1323→  function placeChain(batchId: string, minLeftX: number): number {
  1324→    if (placed.has(batchId)) return minLeftX
  1325→    placed.add(batchId)
  1326→    const chain = bodyChains.get(batchId)
  1327→    const batch = nodesById.get(batchId)
  1328→    if (!chain?.length || !batch) return minLeftX
  1329→    const bSize = nodeSize(batch)
  1330→    const colMaxW = Math.max(...chain.map((cid) => nodeSize(nodesById.get(cid)!).w))
  1331→    const axisX = minLeftX + colMaxW / 2
  1332→    let cursorY = batch.position.y + bSize.h * LOOP_START_ANCHOR + BODY_COL_ANCHOR_DROP
  1333→    let subtreeRight = axisX + colMaxW / 2
  1334→    let prevNestRight = -Infinity
  1335→    chain.forEach((cid) => {
  1336→      const cn = nodesById.get(cid)
  1337→      if (!cn) return
  1338→      const size = nodeSize(cn)
  1339→      cn.position = {
  1340→        x: Math.round(axisX - size.w / 2),
  1341→        y: Math.round(cursorY),
  1342→      }
  1343→      cursorY += size.h + BODY_COL_GAP_Y
  1344→      if (cn.data.nodeType === 'BATCH') {
  1345→        // 子列左缘 = max(子 BATCH 卡右缘+56, 宿主列最宽成员右缘+24, 前一嵌套
  1346→        // 兄弟子树右缘+56)：让出宿主卡与宿主列全部成员，且兄弟嵌套列纵向并行
  1347→        // 时依次向右级联，互不重叠
  1348→        const nestLeft = Math.max(
  1349→          cn.position.x + size.w + BODY_COL_GAP_X,
  1350→          axisX + colMaxW / 2 + BODY_COL_NEST_MARGIN,
  1351→          prevNestRight + BODY_COL_GAP_X
  1352→        )
  1353→        prevNestRight = placeChain(cid, nestLeft)
  1354→        subtreeRight = Math.max(subtreeRight, prevNestRight)
  1355→      }
  1356→    })
  1357→    return subtreeRight
  1358→  }
  1359→  for (const batchId of bodyChains.keys()) {
  1360→    if (bodyOwner.has(batchId)) continue
  1361→    const batch = nodesById.get(batchId)
  1362→    if (!batch) continue
  1363→    placeChain(batchId, batch.position.x + nodeSize(batch).w + BODY_COL_GAP_X)
  1364→  }
  1365→
  1366→  setTimeout(() => fitView({ padding: 0.15, maxZoom: 1, duration: 300 }), 50)
  1367→}

  1378→}

  1385→async function handleSave(): Promise<boolean> {

  1389→    await logicFlowApi.update(flowId, {

  1395→    savedName.value = flowName.value.trim() || defaultFlowName()
  1396→    if (!savingQuiet) ElMessage.success('保存成功')
  1397→    return true
  1398→  } catch {
  1399→    return false
  1400→  } finally {
  1401→    saving.value = false
  1402→  }

  1418→  }
  1419→  publishing.value = true

  1422→    await logicFlowApi.publish(flowId)
  1423→    // 刷新状态与版本号
  1424→    const res = await logicFlowApi.get(flowId)

  1426→    flowVersion.value = Number(res.data.version) || flowVersion.value + 1

  1428→  } catch {

  1430→  } finally {
  1431→    publishing.value = false
  1432→  }
